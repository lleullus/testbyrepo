# Ingress·Gateway 요청이 502·503을 반환하는 경우

## 입력값과 빠른 수집

실행 host는 승인된 kubeconfig가 있는 작업 PC 또는 bastion이며 모든 명령은 같은 Bash에서 실행한다. 연결 hop은 `외부 client → VIP/listener → Ingress·Gateway controller → route backendRef → Service port → EndpointSlice port/address → Pod process listener → application`이다. HTTP `502`·`503`, TCP 연결, EndpointSlice `ready`, Pod `Ready`는 각각 다른 단계의 증거다.

| lifecycle | 값 | 계약 |
|---|---|---|
| 최초 입력 | `CONTEXT`, `NS`, `ROUTE_KIND`, `ROUTE_NAME`, `TEST_HOST`, `TEST_PATH`, `EXPECTED_STATUS` | 승인된 context와 exact route, 비민감 host/path, 정상 status를 한 번 입력한다. |
| API 자동값 | `ROUTE_UID`, `ROUTE_RV`, `ROUTE_GENERATION`, class·parent·backend 관계 | 현재 route snapshot이 모두 성공했을 때만 등록한다. |
| 계산 후보 | `ROUTE_RE`, `OBSERVED_AT`, Event selector, PromQL | 특수문자를 escape하고 bounded 출력으로 확인한다. |
| 운영자 선택값 | `SERVICE`, `SERVICE_PORT`, `CTRL_NS`, `CTRL_POD`, `CTRL_CONTAINER`, `START_RFC3339`, `END_RFC3339` | route 출력과 실제 controller inventory를 보고 해당 분기 직전에 고른다. |

**최초 입력과 전체 파생 상태 초기화**

`reset_route_state(){ unset AUTO_READY AUTO_LATEST_READY RELATION_READY RELATION_LATEST_READY SELECTION_READY TIME_READY REQUEST_READY INCIDENT_READY REQUEST_OK LOG_READY QUERY_READY EVIDENCE_READY FUNCTION_READY ACTION_READY NORMAL_READY ROUTE_JSON ROUTE_UID ROUTE_RV ROUTE_GENERATION ROUTE_OWNER ROUTE_HOST BACKEND_RELATION SERVICE SERVICE_UID SERVICE_RV SERVICE_PORT SERVICE_JSON EPS_JSON ENDPOINT_UID POD_UID POD_RV POD_OWNER POD_NODE CONTAINER_ID CTRL_UID CTRL_RV CTRL_OWNER CTRL_NODE CTRL_CONTAINER_ID FUNCTION_EVIDENCE_ID OBSERVED_AT START_RFC3339 END_RFC3339 ROUTE_RE ARGV_RE; AUTO_READY=false; AUTO_LATEST_READY=false; RELATION_READY=false; RELATION_LATEST_READY=false; SELECTION_READY=false; TIME_READY=false; REQUEST_READY=false; INCIDENT_READY=false; REQUEST_OK=false; LOG_READY=false; QUERY_READY=false; EVIDENCE_READY=false; FUNCTION_READY=false; ACTION_READY=false; NORMAL_READY=false; }`

`reset_route_state; CONTEXT='PASTE_CONTEXT_HERE'; NS='PASTE_NAMESPACE_HERE'; ROUTE_KIND='PASTE_INGRESS_OR_HTTPROUTE_HERE'; ROUTE_NAME='PASTE_ROUTE_NAME_HERE'; TEST_HOST='PASTE_APPROVED_HOST_HERE'; TEST_PATH='PASTE_APPROVED_PATH_HERE'; EXPECTED_STATUS='PASTE_EXPECTED_HTTP_STATUS_HERE'; INPUT_READY=false; if [ "$TOOLS_READY" != true ]; then printf '%s\n' '중단: TOOLS_READY가 아니다.' >&2; elif [ "$ROUTE_KIND" != ingress ] && [ "$ROUTE_KIND" != httproute ]; then printf '%s\n' '중단: ROUTE_KIND는 ingress 또는 httproute다.' >&2; elif [ -z "$CONTEXT" ] || [ -z "$NS" ] || [ -z "$ROUTE_NAME" ] || [ -z "$TEST_HOST" ] || [ -z "$TEST_PATH" ] || ! [[ "$EXPECTED_STATUS" =~ ^[1-5][0-9][0-9]$ ]] || [ "$EXPECTED_STATUS" = 502 ] || [ "$EXPECTED_STATUS" = 503 ]; then printf '%s\n' '중단: 최초 입력 또는 기대 HTTP status가 잘못됐다.' >&2; elif kubectl --context "$CONTEXT" get "$ROUTE_KIND" "$ROUTE_NAME" -n "$NS" -o name >/dev/null 2>&1; then INPUT_READY=true; export CONTEXT NS ROUTE_KIND ROUTE_NAME TEST_HOST TEST_PATH EXPECTED_STATUS; printf 'INPUT_READY=true context=%s namespace=%s route=%s/%s expected_status=%s\n' "$CONTEXT" "$NS" "$ROUTE_KIND" "$ROUTE_NAME" "$EXPECTED_STATUS"; else printf '%s\n' '중단: exact route 조회 실패' >&2; fi`

각 `PASTE_..._HERE` 문자열은 문서 전체에 한 번만 나타난다. 재수집 전 `reset_route_state`가 UID·resourceVersion·owner·containerID·node·host·relation·time·argv/regex와 모든 downstream gate를 먼저 폐기한다. 조회 실패 뒤에는 이전 값이 남지 않는다.

**API 자동 수집·관계 출력**

`reset_route_state; if [ "$INPUT_READY" != true ]; then printf '%s\n' '중단: INPUT_READY가 아니다.' >&2; elif ROUTE_TMP="$(kubectl --context "$CONTEXT" get "$ROUTE_KIND" "$ROUTE_NAME" -n "$NS" -o json)" && UID_TMP="$(jq -r '.metadata.uid // empty' <<<"$ROUTE_TMP")" && RV_TMP="$(jq -r '.metadata.resourceVersion // empty' <<<"$ROUTE_TMP")" && [ -n "$UID_TMP" ] && [ -n "$RV_TMP" ]; then ROUTE_JSON="$ROUTE_TMP"; ROUTE_UID="$UID_TMP"; ROUTE_RV="$RV_TMP"; ROUTE_GENERATION="$(jq -r '.metadata.generation // 0' <<<"$ROUTE_JSON")"; ROUTE_OWNER="$(jq -r '[.metadata.ownerReferences[]? | "\(.kind)/\(.name)/\(.uid)"]|join(",")' <<<"$ROUTE_JSON")"; ROUTE_HOST="$(jq -r 'if .kind=="Ingress" then [.spec.rules[]?.host]|join(",") else [.spec.hostnames[]?]|join(",") end' <<<"$ROUTE_JSON")"; ROUTE_RE="^$(printf '%s' "$ROUTE_NAME" | sed 's/[][\\.^$*+?(){}|]/\\&/g')$"; OBSERVED_AT="$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; AUTO_READY=true; export ROUTE_UID ROUTE_RV ROUTE_GENERATION ROUTE_RE OBSERVED_AT; jq -r --arg observed "$OBSERVED_AT" '"=== Route ===\n종류: \(.kind)\n이름: \(.metadata.name)\nUID: \(.metadata.uid)\nResourceVersion: \(.metadata.resourceVersion)\nGeneration: \(.metadata.generation)\nHost: " + (if .kind=="Ingress" then ([.spec.rules[]?.host]|join(",")) else ([.spec.hostnames[]?]|join(",")) end) + "\nBackend 관계: " + (if .kind=="Ingress" then ([.spec.rules[]?.http.paths[]? | "host=\(.backend.service.name):\(.backend.service.port.name // .backend.service.port.number) path=\(.path)"]|join("; ")) else ([.spec.rules[]?.backendRefs[]? | "service=\(.name):\(.port) weight=\(.weight // 1)"]|join("; ")) end) + "\n관측 UTC: \($observed)"' <<<"$ROUTE_JSON"; printf 'AUTO_READY=true ROUTE_RE=%s\n' "$ROUTE_RE"; else reset_route_state; printf '%s\n' '중단: route API/identity 수집 실패; stale 값 없음' >&2; fi`

**모든 route 명령 직전 fresh gate**

`route_fresh(){ local j uid rv generation; AUTO_LATEST_READY=false; j="$(kubectl --context "$CONTEXT" get "$ROUTE_KIND" "$ROUTE_NAME" -n "$NS" -o json)" || { reset_route_state; return 1; }; uid="$(jq -r '.metadata.uid // empty' <<<"$j")"; rv="$(jq -r '.metadata.resourceVersion // empty' <<<"$j")"; generation="$(jq -r '.metadata.generation // 0' <<<"$j")"; if [ "$uid" = "$ROUTE_UID" ] && [ "$generation" = "$ROUTE_GENERATION" ] && [ -n "$rv" ]; then ROUTE_JSON="$j"; ROUTE_RV="$rv"; AUTO_LATEST_READY=true; printf 'AUTO_LATEST_READY=true uid=%s rv=%s generation=%s observed=%s\n' "$uid" "$rv" "$generation" "$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; return 0; else reset_route_state; printf '중단: route replacement/spec/API 변경 actual_uid=%s generation=%s\n' "${uid:--}" "${generation:--}" >&2; return 1; fi; }`

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | 승인된 host/path 요청에서 `502 Bad Gateway` 또는 `503 Service Unavailable`이 재현된다. |
| 적용 | 같은 route UID와 요청 UTC 시각으로 listener/controller/backend hop을 분리할 수 있다. |
| 제외 | `404`는 [[01. Ingress·Gateway 요청이 404를 반환하는 경우]], TLS·redirect는 [[03. Ingress·Gateway TLS handshake·인증서·redirect가 실패하는 경우]], 외부 VIP 미도달은 [[09. 외부 트래픽/01. LoadBalancer·VIP/02. VIP·External IP가 할당됐지만 외부에서 연결되지 않는 경우]]로 이동한다. |

검색 문자열: `502`, `503`, `upstream`, `no endpoints`, `connection refused`, `reset`, `timeout`, `targetPort`, `ResolvedRefs`.

## 최초 확인과 요청 status 분리

`REQUEST_READY=false; INCIDENT_READY=false; REQUEST_OK=false; RELATION_READY=false; SELECTION_READY=false; LOG_READY=false; QUERY_READY=false; EVIDENCE_READY=false; FUNCTION_READY=false; NORMAL_READY=false; unset SERVICE_UID SERVICE_RV SERVICE_JSON EPS_JSON BACKEND_RELATION CTRL_UID CTRL_RV CTRL_OWNER CTRL_NODE CTRL_CONTAINER_ID FUNCTION_EVIDENCE_ID; if [ "$AUTO_READY" != true ] || ! route_fresh; then printf '%s\n' '중단: latest route gate 실패' >&2; else BODY="$(mktemp)" || exit 1; CODE="$(curl -sS --connect-timeout 5 --max-time 10 -o "$BODY" -w '%{http_code}' "https://$TEST_HOST$TEST_PATH")"; CURL_RC=$?; BYTES="$(wc -c <"$BODY")"; rm -f "$BODY"; printf 'transport_rc=%s http_status=%s body_bytes=%s expected=%s observed=%s\n' "$CURL_RC" "${CODE:-000}" "$BYTES" "$EXPECTED_STATUS" "$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; if [ "$CURL_RC" -eq 0 ] && [[ "$CODE" =~ ^[1-5][0-9][0-9]$ ]]; then REQUEST_READY=true; if [ "$CODE" = 502 ] || [ "$CODE" = 503 ]; then INCIDENT_READY=true; fi; [ "$CODE" = "$EXPECTED_STATUS" ] && REQUEST_OK=true; fi; fi`

`curl` exit, HTTP status와 body 존재는 별도다. `curl` 실패·`000`은 HTTP 502/503 증거가 아니고, HTTP status는 어느 hop이 생성했는지 보장하지 않는다. body는 저장하지 않는다.

## 관측별 진단 경로

| 지금 보이는 것 | 직접 근거 | 다음 경로 |
|---|---|---|
| route condition이 거부됨 | Ingress class 또는 HTTPRoute `Accepted/ResolvedRefs` | class·attachment/controller |
| Service port가 route와 다름 | backendRef와 `spec.ports[].port/name` exact 관계 | Service owner |
| EndpointSlice가 없거나 ready endpoint 없음 | selector, targetRef UID, ready/serving | EndpointSlice·workload |
| endpoint는 ready지만 refused/reset | EndpointSlice port와 실제 process listener·controller log | workload/controller |
| timeout이 여러 backend에 공통 | 같은 controller·node·VIP 경계 | controller/network owner |

## Service·EndpointSlice·Pod 관계 등록

`SERVICE='PASTE_BACKEND_SERVICE_HERE'; SERVICE_PORT='PASTE_SERVICE_PORT_HERE'; unset SERVICE_UID SERVICE_RV SERVICE_JSON EPS_JSON BACKEND_RELATION FUNCTION_EVIDENCE_ID; RELATION_READY=false; SELECTION_READY=false; LOG_READY=false; FUNCTION_READY=false; NORMAL_READY=false; if [ "$REQUEST_READY" != true ] || ! route_fresh; then printf '%s\n' '중단: REQUEST_READY/latest route가 아니다.' >&2; elif ! jq -e --arg s "$SERVICE" --arg p "$SERVICE_PORT" 'if .kind=="Ingress" then any(.spec.rules[]?.http.paths[]?; .backend.service.name==$s and (((.backend.service.port.name // "")==$p) or ((.backend.service.port.number|tostring)==$p))) else any(.spec.rules[]?.backendRefs[]?; .name==$s and ((.port|tostring)==$p) and ((.group // "")=="") and ((.kind // "Service")=="Service")) end' <<<"$ROUTE_JSON" >/dev/null; then printf '%s\n' '중단: 선택값이 현재 route backendRef와 불일치' >&2; elif SVC_TMP="$(kubectl --context "$CONTEXT" get service "$SERVICE" -n "$NS" -o json)" && EPS_TMP="$(kubectl --context "$CONTEXT" get endpointslice -n "$NS" -l "kubernetes.io/service-name=$SERVICE" -o json)" && SVC_UID_TMP="$(jq -r '.metadata.uid // empty' <<<"$SVC_TMP")" && SVC_RV_TMP="$(jq -r '.metadata.resourceVersion // empty' <<<"$SVC_TMP")" && [ -n "$SVC_UID_TMP" ] && [ -n "$SVC_RV_TMP" ] && jq -e --arg p "$SERVICE_PORT" 'any(.spec.ports[]?; ((.name // "") == $p) or ((.port|tostring) == $p))' <<<"$SVC_TMP" >/dev/null && jq -e --arg s "$SERVICE" --arg u "$SVC_UID_TMP" 'all(.items[]?; any(.metadata.ownerReferences[]?; .kind=="Service" and .name==$s and .uid==$u))' <<<"$EPS_TMP" >/dev/null; then SERVICE_UID="$SVC_UID_TMP"; SERVICE_RV="$SVC_RV_TMP"; SERVICE_JSON="$SVC_TMP"; EPS_JSON="$EPS_TMP"; BACKEND_RELATION="$ROUTE_UID->$SERVICE_UID"; RELATION_READY=true; export SERVICE SERVICE_PORT SERVICE_UID SERVICE_RV; jq -r '"=== Service ===\nUID: \(.metadata.uid)\nSelector: \(.spec.selector)\nPort 관계: " + ([.spec.ports[]? | "service=\(.name // \"-\"):\(.port)/\(.protocol) target=\(.targetPort) nodePort=\(.nodePort // \"-\")"]|join("; "))' <<<"$SERVICE_JSON"; jq -r '.items[] | "=== EndpointSlice: \(.metadata.name) ===\nUID: \(.metadata.uid)\nPort: " + ([.ports[]? | "\(.name // \"-\"):\(.port)/\(.protocol)"]|join(",")) + "\nEndpoint 관계: " + ([.endpoints[]? | "pod=\(.targetRef.name // \"-\") uid=\(.targetRef.uid // \"-\") node=\(.nodeName // \"-\") ready=\(.conditions.ready // \"nil\") serving=\(.conditions.serving // \"nil\") terminating=\(.conditions.terminating // \"nil\")"]|join("; "))' <<<"$EPS_JSON"; printf 'RELATION_READY=true relation=%s observed=%s\n' "$BACKEND_RELATION" "$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; else unset SERVICE_UID SERVICE_RV SERVICE_JSON EPS_JSON BACKEND_RELATION; RELATION_READY=false; printf '%s\n' '중단: route→Service exact port 또는 EndpointSlice owner/API 실패' >&2; fi`

`backend_fresh(){ local s e u; RELATION_LATEST_READY=false; s="$(kubectl --context "$CONTEXT" get service "$SERVICE" -n "$NS" -o json)" || { RELATION_READY=false; return 1; }; e="$(kubectl --context "$CONTEXT" get endpointslice -n "$NS" -l "kubernetes.io/service-name=$SERVICE" -o json)" || { RELATION_READY=false; return 1; }; u="$(jq -r '.metadata.uid // empty' <<<"$s")"; if [ "$u" = "$SERVICE_UID" ] && jq -e --arg p "$SERVICE_PORT" 'any(.spec.ports[]?; ((.name // "")==$p) or ((.port|tostring)==$p))' <<<"$s" >/dev/null && jq -e --arg n "$SERVICE" --arg u "$SERVICE_UID" 'all(.items[]?; any(.metadata.ownerReferences[]?; .kind=="Service" and .name==$n and .uid==$u))' <<<"$e" >/dev/null; then SERVICE_JSON="$s"; EPS_JSON="$e"; SERVICE_RV="$(jq -r '.metadata.resourceVersion' <<<"$s")"; RELATION_LATEST_READY=true; return 0; else RELATION_READY=false; unset SERVICE_UID SERVICE_RV SERVICE_JSON EPS_JSON BACKEND_RELATION; return 1; fi; }`

ready endpoint가 없다는 결과는 selector 불일치·readiness 실패·API/RBAC 실패와 구분한다. `ready=true`는 해당 address/port에 process가 listen하거나 application이 정상임을 보장하지 않는다.

## controller 로그와 strict UTC

`CTRL_NS='PASTE_CONTROLLER_NAMESPACE_HERE'; CTRL_POD='PASTE_CONTROLLER_POD_HERE'; CTRL_CONTAINER='PASTE_CONTROLLER_CONTAINER_HERE'; START_RFC3339='PASTE_START_RFC3339_HERE'; END_RFC3339='PASTE_END_RFC3339_HERE'`

`TIME_READY=false; if [[ "$START_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && [[ "$END_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && S_E="$(date -u -d "$START_RFC3339" +%s 2>/dev/null)" && E_E="$(date -u -d "$END_RFC3339" +%s 2>/dev/null)" && [ "$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ')" = "$START_RFC3339" ] && [ "$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ')" = "$END_RFC3339" ] && [ "$S_E" -lt "$E_E" ]; then TIME_READY=true; printf 'TIME_READY=true start=%s end=%s\n' "$START_RFC3339" "$END_RFC3339"; else printf '%s\n' '중단: canonical UTC 또는 시간 순서 오류' >&2; fi`

`SELECTION_READY=false; LOG_READY=false; EVIDENCE_READY=false; NORMAL_READY=false; unset CTRL_UID CTRL_RV CTRL_OWNER CTRL_NODE CTRL_CONTAINER_ID; if [ "$RELATION_READY" != true ] || [ "$TIME_READY" != true ] || ! route_fresh || ! backend_fresh; then printf '%s\n' '중단: relation/time/latest route/backend가 아니다.' >&2; elif CTRL_JSON="$(kubectl --context "$CONTEXT" get pod "$CTRL_POD" -n "$CTRL_NS" -o json)" && CTRL_UID_TMP="$(jq -r '.metadata.uid // empty' <<<"$CTRL_JSON")" && CTRL_ID_TMP="$(jq -r --arg c "$CTRL_CONTAINER" '[.status.containerStatuses[]? | select(.name==$c and .state.running!=null) | .containerID][0] // empty' <<<"$CTRL_JSON")" && [ -n "$CTRL_UID_TMP" ] && [ -n "$CTRL_ID_TMP" ]; then CTRL_UID="$CTRL_UID_TMP"; CTRL_RV="$(jq -r '.metadata.resourceVersion' <<<"$CTRL_JSON")"; CTRL_OWNER="$(jq -r '[.metadata.ownerReferences[]? | "\(.kind)/\(.name)/\(.uid)"]|join(",")' <<<"$CTRL_JSON")"; CTRL_NODE="$(jq -r '.spec.nodeName // empty' <<<"$CTRL_JSON")"; CTRL_CONTAINER_ID="$CTRL_ID_TMP"; SELECTION_READY=true; printf 'SELECTION_READY=true controller_uid=%s container_id=%s node=%s owner=%s\n' "$CTRL_UID" "$CTRL_CONTAINER_ID" "$CTRL_NODE" "$CTRL_OWNER"; else printf '%s\n' '중단: controller exact running identity 실패' >&2; fi`

`ctrl_fresh(){ local j; CTRL_LATEST_READY=false; j="$(kubectl --context "$CONTEXT" get pod "$CTRL_POD" -n "$CTRL_NS" -o json)" || return 1; if jq -e --arg uid "$CTRL_UID" --arg c "$CTRL_CONTAINER" --arg id "$CTRL_CONTAINER_ID" '.metadata.uid==$uid and .metadata.deletionTimestamp==null and any(.status.containerStatuses[]?; .name==$c and .state.running!=null and .containerID==$id)' <<<"$j" >/dev/null; then CTRL_LATEST_READY=true; return 0; else SELECTION_READY=false; LOG_READY=false; unset CTRL_UID CTRL_RV CTRL_OWNER CTRL_NODE CTRL_CONTAINER_ID; return 1; fi; }`

`LOG_READY=false; if [ "$SELECTION_READY" != true ] || [ "$TIME_READY" != true ] || ! route_fresh || ! backend_fresh || ! ctrl_fresh; then printf '%s\n' '중단: latest route/backend/controller/time gate 실패' >&2; elif AUTH="$(kubectl --context "$CONTEXT" auth can-i get pods "$CTRL_POD" --subresource=log -n "$CTRL_NS" 2>&1)" && [ "$AUTH" = yes ]; then TMP="$(mktemp)" || exit 1; WINDOW="$(mktemp)" || { rm -f "$TMP"; exit 1; }; if kubectl --context "$CONTEXT" logs "$CTRL_POD" -n "$CTRL_NS" -c "$CTRL_CONTAINER" --timestamps --since-time="$START_RFC3339" --tail=300 >"$TMP"; then PRODUCER_RC=0; else PRODUCER_RC=$?; fi; if [ "$PRODUCER_RC" -eq 0 ]; then awk -v end="$END_RFC3339" 'substr($1,1,19) "Z" <= end' "$TMP" >"$WINDOW"; WINDOW_RC=$?; else WINDOW_RC=not-run; fi; if [ "$PRODUCER_RC" -eq 0 ] && [ "$WINDOW_RC" -eq 0 ]; then grep -E -i -C 3 "$ROUTE_UID|$SERVICE_UID|502|503|upstream|no endpoint|refused|reset|timeout" "$WINDOW"; FILTER_RC=$?; printf 'producer_rc=%s window_rc=%s filter_rc=%s controller_uid=%s container_id=%s\n' "$PRODUCER_RC" "$WINDOW_RC" "$FILTER_RC" "$CTRL_UID" "$CTRL_CONTAINER_ID"; [ "$FILTER_RC" -le 1 ] && LOG_READY=true; else printf 'producer_rc=%s window_rc=%s filter_rc=not-run\n' "$PRODUCER_RC" "$WINDOW_RC" >&2; fi; rm -f "$TMP" "$WINDOW"; else printf '중단: exact pods/log RBAC 결과=%s\n' "${AUTH:--}" >&2; fi`

producer 실패와 filter 무일치는 별도다. `filter_rc=1`은 오류 부재가 아니며 보존·시간·식별자·controller 선택을 재검토한다. 로그의 credential, header, body, cookie, token과 개인정보는 출력·저장하지 않는다. API audit가 필요하면 정확한 subject, `get pods/log`, namespace, Pod UID, `responseStatus.code`, UTC `stageTimestamp`만 승인된 audit owner에게 요청한다.

## 판정과 안전한 변경 경계

- route condition 실패는 listener/class/controller 계층이다. Service를 바꾸지 않는다.
- route backend port와 Service port 불일치는 선언 관계 오류다. EndpointSlice를 직접 수정하지 않는다.
- ready endpoint 부재는 selector/readiness 계층이다. controller 재시작으로 해결하지 않는다.
- 같은 UID·시각의 `connection refused`는 listener/port 후보, `timeout`은 path/policy/backend 지연 후보이며 단독 원인 확정이 아니다.
- backend가 요청을 받아 자체 503을 반환한 증거가 있으면 workload owner로 이관한다.

이 문서에는 mutation 명령을 두지 않는다. 변경은 별도 승인된 GitOps/관리 경로에서만 수행하며 fresh `ROUTE_UID/RV`, 직접 증거, 승인 ID, 변경 창, replica/PDB/quorum, 멱등성·외부 side effect, rollback revision과 중단 조건이 모두 기록돼야 한다. TLS 우회, EndpointSlice 직접 수정, Pod 삭제, 무조건 restart/scale은 허용하지 않는다.

## 변경 후 replacement 재등록과 실제 정상화

변경 뒤에는 이전 UID·RV·owner·relation·controller containerID·시각·regex와 gate를 전부 폐기하고 `reset_route_state`부터 다시 실행한다. 같은 이름의 route·Service·Pod가 교체되면 이전 증거를 재사용하지 않는다.

`FUNCTION_EVIDENCE_ID='PASTE_FUNCTION_EVIDENCE_ID_HERE'; FUNCTION_READY=false; if [ "$REQUEST_OK" != true ] || [ "$RELATION_READY" != true ] || [ -z "$FUNCTION_EVIDENCE_ID" ] || ! route_fresh || ! backend_fresh; then printf '%s\n' '중단: 승인된 실제 기능 evidence와 latest route/backend가 필요하다.' >&2; else FUNCTION_READY=true; printf 'FUNCTION_READY=true evidence=%s route_uid=%s service_uid=%s\n' "$FUNCTION_EVIDENCE_ID" "$ROUTE_UID" "$SERVICE_UID"; fi`

`NORMAL_READY=false; if [ "$REQUEST_READY" != true ] || [ "$REQUEST_OK" != true ] || [ "$FUNCTION_READY" != true ] || [ "$RELATION_READY" != true ] || ! route_fresh || ! backend_fresh; then printf '%s\n' '중단: 변경 후 전체 재등록, 기대 status, 실제 기능, latest route/backend gate가 필요하다.' >&2; elif jq -e '[.items[].endpoints[]? | select((.conditions.ready // true)==true and (.conditions.terminating // false)==false)] | length>0' <<<"$EPS_JSON" >/dev/null; then NORMAL_READY=true; printf 'NORMAL_READY=true expected_status=%s route_uid=%s service_uid=%s function_evidence=%s observed=%s\n' "$EXPECTED_STATUS" "$ROUTE_UID" "$SERVICE_UID" "$FUNCTION_EVIDENCE_ID" "$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; else printf '%s\n' '중단: current serving backend가 없다.' >&2; fi`

정상화는 실제 비민감 요청의 기대 status·핵심 기능, controller→Service→EndpointSlice→listener 관계, desired/Ready, 새 Warning·restart·502/503 재발 부재를 관찰 창에서 모두 확인한다. 한 번의 HTTP 200, Pod Running, Endpoint ready만으로 닫지 않는다.

## Prometheus 빠른 관측

`QUERY_READY=false; if [ "$AUTO_READY" != true ] || [ "$RELATION_READY" != true ] || ! route_fresh || ! backend_fresh; then printf '%s\n' '중단: latest identity/relation gate 실패' >&2; else QUERY_READY=true; printf 'sum by(status)(rate(nginx_ingress_controller_requests{namespace="%s",ingress=~"%s",status=~"502|503"}[5m]))\nsum(kube_endpoint_address{namespace="%s",endpoint="%s",ready="true"})\n' "$NS" "$ROUTE_RE" "$NS" "$SERVICE"; fi`

controller metric 이름·labels와 `kube_endpoint_address`가 실제 노출될 때만 사용한다. query transport/HTTP/API 오류, metric 미수집, label 불일치, 시간 범위 오류와 빈 결과를 구분한다. metric이 없으면 0으로 만들지 않고 API·controller log·승인된 요청 증거를 사용한다.

## 증거와 1.26 경계

UTC 시각, context·namespace, route UID/RV/generation, route→Service UID/port→EndpointSlice UID/targetRef UID→Pod UID/node 관계, controller UID/containerID, 요청 transport rc·HTTP status, producer/filter rc와 승인 기록만 보존한다. 내부 주소·response body·Secret·token·개인키·개인정보는 보존하지 않는다. Gateway API CRD version과 controller/provider 구현은 환경별 미검증이며 실제 1.26 클러스터에서 실행하지 않았다.
