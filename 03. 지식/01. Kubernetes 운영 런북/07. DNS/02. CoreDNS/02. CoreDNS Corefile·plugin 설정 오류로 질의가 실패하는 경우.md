# CoreDNS Corefile·plugin 설정 오류로 질의가 실패하는 경우

## 입력값과 빠른 수집

실행 경로는 `작업 PC/bastion의 kubectl → API server → Deployment volume → ConfigMap key → 선택한 CoreDNS Pod의 mounted content/reload → plugin chain → client DNS 응답`이다. ConfigMap API 변경, projected volume 반영, reload와 query 성공은 각각 다른 gate다.

| 분류 | 변수 | 출처와 gate |
|---|---|---|
| 최초 입력 | namespace·Deployment·ConfigMap·key | 승인된 선언 원본과 실제 volume 관계 |
| API 자동값 | Deployment/ConfigMap/ReplicaSet/Pod UID·RV, node UID/machineID, current containerID | 선택 Pod가 current owner와 volume 관계에 속할 때 등록 |
| 계산 후보 | `POD_RE` | 선택 Pod 이름을 regex escape한 값 |
| 운영자 선택 | CoreDNS Pod/regular container, client Pod/regular container, query, UTC 창 | 후보와 장애 증거를 확인한 뒤 선택 |

### 1. 등록 lifecycle

```bash
TOOLS_READY=false
MISSING=''
if [ -z "${BASH_VERSION:-}" ]; then printf '%s\n' '중단: Bash에서 실행한다.' >&2; else
  for tool in kubectl jq sed grep date; do command -v "$tool" >/dev/null 2>&1 || MISSING="$MISSING $tool"; done
  if [ -z "$MISSING" ]; then TOOLS_READY=true; else printf '중단: 없는 도구=%s\n' "$MISSING" >&2; fi
fi

unset INPUT_READY AUTO_READY FRESH_READY CONFIG_READY LOG_READY QUERY_READY
unset DEPLOYMENT_UID DEPLOYMENT_RV CONFIGMAP_UID CONFIGMAP_RV RS RS_UID POD_UID POD_RV NODE NODE_UID NODE_MACHINE_ID CONTAINER_ID POD_RE CLIENT_UID CLIENT_CONTAINER_ID
NS='PASTE_DNS_NAMESPACE_HERE'
DEPLOYMENT='PASTE_COREDNS_DEPLOYMENT_HERE'
CONFIGMAP='PASTE_COREFILE_CONFIGMAP_HERE'
CORE_KEY='PASTE_COREFILE_KEY_HERE'
POD='PASTE_COREDNS_POD_HERE'
CONTAINER='PASTE_COREDNS_REGULAR_CONTAINER_HERE'
CLIENT_NS='PASTE_CLIENT_NAMESPACE_HERE'
CLIENT_POD='PASTE_CLIENT_POD_HERE'
CLIENT_CONTAINER='PASTE_CLIENT_REGULAR_CONTAINER_HERE'
QUERY_NAME='PASTE_APPROVED_QUERY_HERE'
START_RFC3339='PASTE_START_UTC_HERE'
END_RFC3339='PASTE_END_UTC_HERE'
INPUT_READY=false
TIME_RE='^[0-9]{4}-(0[1-9]|1[0-2])-([0-2][0-9]|3[01])T([01][0-9]|2[0-3]):[0-5][0-9]:[0-5][0-9]Z$'
if [ "$TOOLS_READY" = true ] && [ -n "$NS" ] && [ -n "$DEPLOYMENT" ] && [ -n "$CONFIGMAP" ] && [ -n "$CORE_KEY" ] && [ -n "$POD" ] && [ -n "$CONTAINER" ] && [ -n "$QUERY_NAME" ] && [[ "$START_RFC3339" =~ $TIME_RE && "$END_RFC3339" =~ $TIME_RE ]] && S="$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" && E="$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" && [ "$S" = "$START_RFC3339" ] && [ "$E" = "$END_RFC3339" ] && [ "$(date -u -d "$S" +%s)" -lt "$(date -u -d "$E" +%s)" ]; then INPUT_READY=true; else printf '%s\n' '중단: 입력 또는 strict UTC start<end gate 실패' >&2; fi
unset S E

AUTO_READY=false
if [ "$INPUT_READY" = true ] && DJ="$(kubectl get deployment "$DEPLOYMENT" -n "$NS" -o json)" && MJ="$(kubectl get configmap "$CONFIGMAP" -n "$NS" -o json)" && PJ="$(kubectl get pod "$POD" -n "$NS" -o json)" && CJ="$(kubectl get pod "$CLIENT_POD" -n "$CLIENT_NS" -o json)"; then
  DEPLOYMENT_UID="$(jq -r '.metadata.uid//empty' <<<"$DJ")"; DEPLOYMENT_RV="$(jq -r '.metadata.resourceVersion//empty' <<<"$DJ")"
  CONFIGMAP_UID="$(jq -r '.metadata.uid//empty' <<<"$MJ")"; CONFIGMAP_RV="$(jq -r '.metadata.resourceVersion//empty' <<<"$MJ")"
  POD_UID="$(jq -r '.metadata.uid//empty' <<<"$PJ")"; POD_RV="$(jq -r '.metadata.resourceVersion//empty' <<<"$PJ")"; NODE="$(jq -r '.spec.nodeName//empty' <<<"$PJ")"
  CONTAINER_ID="$(jq -r --arg c "$CONTAINER" '[.status.containerStatuses[]?|select(.name==$c)][0].containerID//empty' <<<"$PJ")"
  RS="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true and .kind=="ReplicaSet")][0].name//empty' <<<"$PJ")"; RS_UID="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true and .kind=="ReplicaSet")][0].uid//empty' <<<"$PJ")"
  CLIENT_UID="$(jq -r '.metadata.uid//empty' <<<"$CJ")"; CLIENT_CONTAINER_ID="$(jq -r --arg c "$CLIENT_CONTAINER" '[.status.containerStatuses[]?|select(.name==$c)][0].containerID//empty' <<<"$CJ")"
  if RJ="$(kubectl get replicaset "$RS" -n "$NS" -o json)" && NJ="$(kubectl get node "$NODE" -o json)"; then
    NODE_UID="$(jq -r '.metadata.uid//empty' <<<"$NJ")"; NODE_MACHINE_ID="$(jq -r '.status.nodeInfo.machineID//empty' <<<"$NJ")"; POD_RE="^$(printf '%s' "$POD" | sed 's/[][\\.^$*+?(){}|]/\\&/g')$"
    if [ -n "$DEPLOYMENT_UID" ] && [ -n "$CONFIGMAP_UID" ] && jq -e --arg cm "$CONFIGMAP" --arg key "$CORE_KEY" 'any(.spec.template.spec.volumes[]?;.configMap.name==$cm and ((.configMap.items|not) or any(.configMap.items[]?;.key==$key)))' <<<"$DJ" >/dev/null && jq -e --arg key "$CORE_KEY" '.data|has($key)' <<<"$MJ" >/dev/null && jq -e --argjson sel "$(jq -c '.spec.selector.matchLabels' <<<"$DJ")" '.metadata.labels as $l|($sel|to_entries|all($l[.key]==.value))' <<<"$PJ" >/dev/null && [ "$(jq -r '.metadata.uid' <<<"$RJ")" = "$RS_UID" ] && [ "$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true and .kind=="Deployment")][0].uid//empty' <<<"$RJ")" = "$DEPLOYMENT_UID" ] && [ "$(jq -r '.status.phase' <<<"$PJ")" = Running ] && [ "$(jq -r '.status.phase' <<<"$CJ")" = Running ] && [[ "$CONTAINER_ID" =~ ^(containerd|cri-o|docker)://[^[:space:]]+$ ]] && [[ "$CLIENT_CONTAINER_ID" =~ ^(containerd|cri-o|docker)://[^[:space:]]+$ ]] && [ -n "$NODE_UID" ] && [ -n "$NODE_MACHINE_ID" ]; then
      AUTO_READY=true
      printf 'AUTO_READY=true deployment=%s/%s rv=%s configmap=%s/%s rv=%s key=%s rs=%s/%s pod=%s/%s node=%s/%s runtime=%s client=%s/%s runtime=%s podRE=%s\n' "$DEPLOYMENT" "$DEPLOYMENT_UID" "$DEPLOYMENT_RV" "$CONFIGMAP" "$CONFIGMAP_UID" "$CONFIGMAP_RV" "$CORE_KEY" "$RS" "$RS_UID" "$POD" "$POD_UID" "$NODE" "$NODE_UID" "${CONTAINER_ID%%://*}" "$CLIENT_POD" "$CLIENT_UID" "${CLIENT_CONTAINER_ID%%://*}" "$POD_RE"
    fi
  fi
fi
unset DJ MJ PJ CJ RJ NJ
[ "$AUTO_READY" = true ] || printf '%s\n' '중단: config-owner-Pod-node-client 자동 등록 실패' >&2

config_fresh() {
  FRESH_READY=false; CONFIG_READY=false; LOG_READY=false; QUERY_READY=false
  local d m p r n c cid ccid
  [ "$AUTO_READY" = true ] || return 1
  d="$(kubectl get deployment "$DEPLOYMENT" -n "$NS" -o json)" || return 1; m="$(kubectl get configmap "$CONFIGMAP" -n "$NS" -o json)" || return 1
  p="$(kubectl get pod "$POD" -n "$NS" -o json)" || return 1; r="$(kubectl get replicaset "$RS" -n "$NS" -o json)" || return 1; n="$(kubectl get node "$NODE" -o json)" || return 1
  c="$(kubectl get pod "$CLIENT_POD" -n "$CLIENT_NS" -o json)" || return 1
  cid="$(jq -r --arg x "$CONTAINER" '[.status.containerStatuses[]?|select(.name==$x)][0].containerID//empty' <<<"$p")"; ccid="$(jq -r --arg x "$CLIENT_CONTAINER" '[.status.containerStatuses[]?|select(.name==$x)][0].containerID//empty' <<<"$c")"
  [ "$(jq -r '.metadata.uid' <<<"$d")" = "$DEPLOYMENT_UID" ] && [ "$(jq -r '.metadata.uid' <<<"$m")" = "$CONFIGMAP_UID" ] && jq -e --arg key "$CORE_KEY" '.data|has($key)' <<<"$m" >/dev/null && jq -e --arg cm "$CONFIGMAP" --arg key "$CORE_KEY" 'any(.spec.template.spec.volumes[]?;.configMap.name==$cm and ((.configMap.items|not) or any(.configMap.items[]?;.key==$key)))' <<<"$d" >/dev/null && jq -e --argjson sel "$(jq -c '.spec.selector.matchLabels' <<<"$d")" '.metadata.labels as $l|($sel|to_entries|all($l[.key]==.value))' <<<"$p" >/dev/null && [ "$(jq -r '.metadata.uid' <<<"$p")" = "$POD_UID" ] && [ "$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid//empty' <<<"$p")" = "$RS_UID" ] && [ "$(jq -r '.metadata.uid' <<<"$r")" = "$RS_UID" ] && [ "$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid//empty' <<<"$r")" = "$DEPLOYMENT_UID" ] && [ "$(jq -r '.metadata.uid' <<<"$n")" = "$NODE_UID" ] && [ "$(jq -r '.status.nodeInfo.machineID' <<<"$n")" = "$NODE_MACHINE_ID" ] && [ "$cid" = "$CONTAINER_ID" ] && [ "$(jq -r '.metadata.uid' <<<"$c")" = "$CLIENT_UID" ] && [ "$ccid" = "$CLIENT_CONTAINER_ID" ] || return 1
  FRESH_READY=true
}

config_logs() {
  config_fresh || { printf '%s\n' '중단: fresh config identity gate 실패' >&2; return 1; }
  [ "$(kubectl auth can-i get pods/log -n "$NS")" = yes ] || { printf '%s\n' '중단: pods/log RBAC gate 실패' >&2; return 1; }
  local raw rc line ts epoch end_epoch window out filter_rc
  raw="$(kubectl logs "$POD" -n "$NS" -c "$CONTAINER" --timestamps --since-time="$START_RFC3339" --tail=500)"; rc=$?
  [ "$rc" -eq 0 ] || { printf '중단: log producer rc=%s\n' "$rc" >&2; return "$rc"; }
  end_epoch="$(date -u -d "$END_RFC3339" +%s)"; window=''
  while IFS= read -r line; do ts="${line%% *}"; epoch="$(date -u -d "$ts" +%s 2>/dev/null)" || return 2; [ "$epoch" -le "$end_epoch" ] && window+="$line"$'\n'; done <<<"$raw"
  [ -n "$window" ] || { printf '%s\n' '증거 창 empty; producer 성공과 로그 부재를 구분함' >&2; return 3; }
  if out="$(grep -E -i -C 3 'parse|unknown directive|wrong argument|reload|plugin/errors|servfail|refused|timeout|fatal|panic' <<<"$window")"; then LOG_READY=true; printf '%s\n' "$out"; else filter_rc=$?; printf 'filter no-match rc=%s; 오류 부재 아님\n' "$filter_rc" >&2; return "$filter_rc"; fi
}
```

복수 Ready Pod 중 첫 번째를 자동 선택하지 않는다. 선택한 `POD`가 current Deployment의 ReplicaSet에 속하지 않으면 등록이 실패한다.

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | `parse error`, `unknown directive`, reload 실패 또는 plugin chain과 연결된 `SERVFAIL`·`REFUSED`가 보이는 경우 |
| 적용 | current ConfigMap key→Deployment volume→Pod owner/reload→client query를 같은 UTC 창에 연결할 수 있는 경우 |
| 제외 | Corefile은 유효하지만 Ready/Service가 실패하면 [[01. CoreDNS Pod·Service가 Ready하지 않거나 질의에 응답하지 않는 경우]], upstream만 실패하면 [[07. DNS/03. 서비스·외부 이름 해석/02. 외부 도메인만 해석되지 않거나 upstream DNS 전달이 실패하는 경우]]로 이동한다. |

검색 문자열: `parse error`, `unknown directive`, `failed to reload`, `plugin/errors`, `SERVFAIL`, `REFUSED`.

## 관측별 진단 경로

| 지금 보이는 것 | 먼저 볼 것 | 그다음 어디를 볼지 | 상세 절 |
|---|---|---|---|
| parse·unknown directive | current ConfigMap UID/RV/key와 plugin 순서 | 승인 선언 원본 | [[#1. 현재 반영 경로와 plugin 구조|1]] |
| reload 실패 | mounted source와 같은 RV의 log | projection·reload owner | [[#2. bounded 로그와 실제 응답|2]] |
| SERVFAIL·REFUSED | plugin 이름과 같은 query/log 창 | kubernetes·forward plugin owner | [[#2. bounded 로그와 실제 응답|2]] |

### 1. 현재 반영 경로와 plugin 구조

`if config_fresh && [ "$FRESH_READY" = true ]; then kubectl get deployment "$DEPLOYMENT" -n "$NS" -o jsonpath='uid={.metadata.uid}{" rv="}{.metadata.resourceVersion}{" generation="}{.metadata.generation}{" observed="}{.status.observedGeneration}{" ready="}{.status.readyReplicas}{"\n"}{range .spec.template.spec.volumes[?(@.configMap)]}{.configMap.name}{" keys="}{.configMap.items[*].key}{"\n"}{end}' && kubectl get configmap "$CONFIGMAP" -n "$NS" -o go-template='uid={{.metadata.uid}} rv={{.metadata.resourceVersion}} keys={{range $k, $v := .data}}{{$k}} {{end}}{{"\n"}}' && kubectl get pod "$POD" -n "$NS" -o custom-columns='UID:.metadata.uid,RV:.metadata.resourceVersion,NODE:.spec.nodeName,PHASE:.status.phase,READY:.status.containerStatuses[*].ready,RESTARTS:.status.containerStatuses[*].restartCount'; else printf '%s\n' '중단: fresh config identity gate 실패' >&2; fi`

**허용된 plugin 이름·순서만 출력**

`if config_fresh && [ "$FRESH_READY" = true ]; then RAW="$(kubectl get configmap "$CONFIGMAP" -n "$NS" -o json | jq -r --arg k "$CORE_KEY" '.data[$k]//empty')"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -ne 0 ] || [ -z "$RAW" ]; then printf '중단: Corefile producer/empty rc=%s\n' "$PRODUCER_RC" >&2; elif PLUGINS="$(grep -E -i '^[[:space:]]*(errors|health|ready|prometheus|kubernetes|cache|forward|proxy|reload|loop|rewrite|template|hosts|autopath|loadbalance|fallthrough)([[:space:]]|\{|$)' <<<"$RAW")"; then CONFIG_READY=true; sed -E 's/^[[:space:]]*([^[:space:]{]+).*/\1/' <<<"$PLUGINS"; else FILTER_RC=$?; printf '중단: plugin filter no-match rc=%s\n' "$FILTER_RC" >&2; fi; unset RAW PLUGINS; else printf '%s\n' '중단: fresh config gate 실패' >&2; fi`

Corefile 원문, zone, forward 대상과 hosts 경로는 출력·저장하지 않는다. 허용 목록 밖 plugin을 정상으로 자동 판정하지 않는다.

### 2. bounded 로그와 실제 응답

`config_logs`

`if config_fresh && [ "$FRESH_READY" = true ] && [ "$(kubectl auth can-i create pods/exec -n "$CLIENT_NS")" = yes ]; then kubectl exec "$CLIENT_POD" -n "$CLIENT_NS" -c "$CLIENT_CONTAINER" -- sh -c 'if command -v dig >/dev/null 2>&1; then exec dig +time=2 +tries=1 "$1" A; elif command -v nslookup >/dev/null 2>&1; then exec nslookup "$1"; elif command -v getent >/dev/null 2>&1; then exec getent hosts "$1"; else exit 127; fi' sh "$QUERY_NAME"; RC=$?; printf 'clientUID=%s query_rc=%s observed=%s window=%s..%s\n' "$CLIENT_UID" "$RC" "$(date -u '+%Y-%m-%dT%H:%M:%SZ')" "$START_RFC3339" "$END_RFC3339"; else printf '%s\n' '중단: fresh config/client gate 실패' >&2; fi`

ConfigMap API의 새 RV만으로 mounted content나 reload 성공을 보장하지 않는다. log와 query의 관측 시각이 증거 창 밖이면 현재 사건에 결합하지 않는다.

## Prometheus 빠른 관측

CoreDNS series에 current Pod를 식별하는 `pod` label이 실제 있을 때만 사용한다.

`if config_fresh && [ "$FRESH_READY" = true ]; then QUERY_READY=true; WINDOW_SECONDS=$(( $(date -u -d "$END_RFC3339" +%s) - $(date -u -d "$START_RFC3339" +%s) )); printf 'increase(coredns_panics_total{pod=~"%s"}[%ss]) > 0\n' "$POD_RE" "$WINDOW_SECONDS"; printf 'sum by(rcode)(rate(coredns_dns_responses_total{pod=~"%s",rcode=~"SERVFAIL|REFUSED"}[5m]))\n' "$POD_RE"; unset WINDOW_SECONDS; else QUERY_READY=false; printf '%s\n' '중단: fresh PromQL gate 실패' >&2; fi`

빈 결과를 0으로 대체하지 않고 metric 지원, scrape, label, API/RBAC와 조회 범위를 구분한다.

## 변경 gate와 정상화

이 문서는 ConfigMap 수정·reload·restart 명령을 제공하지 않는다. 별도 mutation lane에는 fresh Deployment·ConfigMap·ReplicaSet·Pod UID/current RV, current selector/revision/image digest, 같은 RV의 parse/reload 증거, 승인 manifest와 `kubectl diff`, 승인 ID·변경 시각, replica/PDB/quorum, query 부하·멱등성, 검증된 rollback revision이 필요하다. 민감한 Corefile·upstream이 diff 증거에 노출되면 저장을 중단한다.

Pod 교체나 rollback 뒤 과거 UID/RV/node/containerID와 gate를 폐기하고 current controller UID·selector·revision·creationTimestamp로 단일 새 Pod를 다시 선택한다. 정상화는 `ConfigMap key/RV → Deployment volume/reload → 새 Pod process/Ready → Service endpoint → 같은 client response/latency → 실제 기능` 순이며 관찰 창 동안 재발이 없어야 한다.

## 증거 보존

UTC 창, Deployment·ConfigMap·ReplicaSet·Pod·node UID/current RV, key와 plugin 이름·순서, regular container/runtime prefix, 비식별 log 유형, query 상태, 승인·rollback ID만 보존한다. Corefile·label·domain·upstream·Secret/token·전체 log/audit body는 저장하지 않는다.
