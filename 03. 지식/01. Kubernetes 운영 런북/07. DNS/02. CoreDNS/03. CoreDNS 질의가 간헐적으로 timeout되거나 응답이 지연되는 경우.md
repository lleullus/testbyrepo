# CoreDNS 질의가 간헐적으로 timeout되거나 응답이 지연되는 경우

## 입력값과 빠른 수집

관측 hop은 `client Pod resolver → DNS Service UDP/TCP 53 → EndpointSlice → 선택 CoreDNS Pod/process → cache/plugin → upstream`이다. 작업 PC의 API/Prometheus 관측, client exec와 target node SSH는 실행 host가 다르므로 UTC 창과 identity로만 결합한다.

| 분류 | 변수 | 출처와 gate |
|---|---|---|
| 최초 입력 | DNS namespace·Service·Deployment | 승인된 exact 객체 |
| API 자동값 | Service/Deployment/ReplicaSet/Pod/node UID·RV, current containerID | selector와 owner chain 검증 뒤 등록 |
| 계산 후보 | `POD_RE`, 동일 길이 baseline 창 | regex escape와 UTC epoch 계산 뒤 출력 |
| 운영자 선택 | CoreDNS Pod/container, client identity/query, incident 창, 승인 SSH target | 실제 후보·증거·inventory에서 선택 |

### 1. 등록 lifecycle

```bash
MISSING=''
if [ -z "${BASH_VERSION:-}" ]; then printf '%s\n' '중단: Bash에서 실행한다.' >&2; else

fi

unset INPUT_READY AUTO_READY FRESH_READY QUERY_READY HOST_READY LOG_READY
unset SERVICE_UID SERVICE_RV DEPLOYMENT_UID DEPLOYMENT_RV RS RS_UID POD_UID POD_RV NODE NODE_UID NODE_MACHINE_ID CONTAINER_ID POD_RE
unset CLIENT_UID CLIENT_NODE CLIENT_CONTAINER_ID BASE_START_RFC3339 BASE_END_RFC3339
NS='PASTE_DNS_NAMESPACE_HERE'
SERVICE='PASTE_DNS_SERVICE_HERE'
DEPLOYMENT='PASTE_COREDNS_DEPLOYMENT_HERE'
POD='PASTE_COREDNS_POD_HERE'
CONTAINER='PASTE_COREDNS_REGULAR_CONTAINER_HERE'
CLIENT_NS='PASTE_CLIENT_NAMESPACE_HERE'
CLIENT_POD='PASTE_CLIENT_POD_HERE'
CLIENT_CONTAINER='PASTE_CLIENT_REGULAR_CONTAINER_HERE'
QUERY_NAME='PASTE_APPROVED_QUERY_HERE'
START_RFC3339='PASTE_INCIDENT_START_UTC_HERE'
END_RFC3339='PASTE_INCIDENT_END_UTC_HERE'
SSH_TARGET='PASTE_APPROVED_SSH_TARGET_HERE'
INPUT_READY=false
TIME_RE='^[0-9]{4}-(0[1-9]|1[0-2])-([0-2][0-9]|3[01])T([01][0-9]|2[0-3]):[0-5][0-9]:[0-5][0-9]Z$'
if [ "$TOOLS_READY" = true ] && [[ "$START_RFC3339" =~ $TIME_RE && "$END_RFC3339" =~ $TIME_RE ]] && S="$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" && E="$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" && [ "$S" = "$START_RFC3339" ] && [ "$E" = "$END_RFC3339" ] && START_EPOCH="$(date -u -d "$S" +%s)" && END_EPOCH="$(date -u -d "$E" +%s)" && [ "$START_EPOCH" -lt "$END_EPOCH" ]; then
  DURATION=$((END_EPOCH-START_EPOCH)); BASE_END_RFC3339="$(date -u -d "@$((START_EPOCH-1))" '+%Y-%m-%dT%H:%M:%SZ')"; BASE_START_RFC3339="$(date -u -d "@$((START_EPOCH-DURATION-1))" '+%Y-%m-%dT%H:%M:%SZ')"; INPUT_READY=true
  printf 'INPUT_READY=true baseline=%s..%s incident=%s..%s duration=%ss\n' "$BASE_START_RFC3339" "$BASE_END_RFC3339" "$START_RFC3339" "$END_RFC3339" "$DURATION"
else printf '%s\n' '중단: strict UTC start<end gate 실패' >&2; fi
unset S E START_EPOCH END_EPOCH DURATION

AUTO_READY=false
if [ "$INPUT_READY" = true ] && SJ="$(kubectl get service "$SERVICE" -n "$NS" -o json)" && DJ="$(kubectl get deployment "$DEPLOYMENT" -n "$NS" -o json)" && PJ="$(kubectl get pod "$POD" -n "$NS" -o json)" && CJ="$(kubectl get pod "$CLIENT_POD" -n "$CLIENT_NS" -o json)"; then
  SERVICE_UID="$(jq -r '.metadata.uid//empty' <<<"$SJ")"; SERVICE_RV="$(jq -r '.metadata.resourceVersion//empty' <<<"$SJ")"; DEPLOYMENT_UID="$(jq -r '.metadata.uid//empty' <<<"$DJ")"; DEPLOYMENT_RV="$(jq -r '.metadata.resourceVersion//empty' <<<"$DJ")"
  POD_UID="$(jq -r '.metadata.uid//empty' <<<"$PJ")"; POD_RV="$(jq -r '.metadata.resourceVersion//empty' <<<"$PJ")"; NODE="$(jq -r '.spec.nodeName//empty' <<<"$PJ")"; CONTAINER_ID="$(jq -r --arg c "$CONTAINER" '[.status.containerStatuses[]?|select(.name==$c)][0].containerID//empty' <<<"$PJ")"
  RS="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true and .kind=="ReplicaSet")][0].name//empty' <<<"$PJ")"; RS_UID="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true and .kind=="ReplicaSet")][0].uid//empty' <<<"$PJ")"
  CLIENT_UID="$(jq -r '.metadata.uid//empty' <<<"$CJ")"; CLIENT_NODE="$(jq -r '.spec.nodeName//empty' <<<"$CJ")"; CLIENT_CONTAINER_ID="$(jq -r --arg c "$CLIENT_CONTAINER" '[.status.containerStatuses[]?|select(.name==$c)][0].containerID//empty' <<<"$CJ")"
  if RJ="$(kubectl get replicaset "$RS" -n "$NS" -o json)" && NJ="$(kubectl get node "$NODE" -o json)"; then
    NODE_UID="$(jq -r '.metadata.uid//empty' <<<"$NJ")"; NODE_MACHINE_ID="$(jq -r '.status.nodeInfo.machineID//empty' <<<"$NJ")"; POD_RE="^$(printf '%s' "$POD" | sed 's/[][\\.^$*+?(){}|]/\\&/g')$"
    if [ -n "$SERVICE_UID" ] && [ -n "$DEPLOYMENT_UID" ] && jq -e --argjson sel "$(jq -c '.spec.selector' <<<"$SJ")" '.metadata.labels as $l|($sel|to_entries|all($l[.key]==.value))' <<<"$PJ" >/dev/null && jq -e --argjson sel "$(jq -c '.spec.selector.matchLabels' <<<"$DJ")" '.metadata.labels as $l|($sel|to_entries|all($l[.key]==.value))' <<<"$PJ" >/dev/null && [ "$(jq -r '.metadata.uid' <<<"$RJ")" = "$RS_UID" ] && [ "$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid//empty' <<<"$RJ")" = "$DEPLOYMENT_UID" ] && [ "$(jq -r '.status.phase' <<<"$PJ")" = Running ] && [ "$(jq -r '.status.phase' <<<"$CJ")" = Running ] && [[ "$CONTAINER_ID" =~ ^(containerd|cri-o|docker)://[^[:space:]]+$ ]] && [[ "$CLIENT_CONTAINER_ID" =~ ^(containerd|cri-o|docker)://[^[:space:]]+$ ]] && [ -n "$NODE_UID" ] && [ -n "$NODE_MACHINE_ID" ]; then AUTO_READY=true; printf 'AUTO_READY=true service=%s/%s deployment=%s/%s rs=%s/%s pod=%s/%s node=%s/%s machineID=%s runtime=%s client=%s/%s podRE=%s\n' "$SERVICE" "$SERVICE_UID" "$DEPLOYMENT" "$DEPLOYMENT_UID" "$RS" "$RS_UID" "$POD" "$POD_UID" "$NODE" "$NODE_UID" "$NODE_MACHINE_ID" "${CONTAINER_ID%%://*}" "$CLIENT_POD" "$CLIENT_UID" "$POD_RE"; fi
  fi
fi
unset SJ DJ PJ CJ RJ NJ
[ "$AUTO_READY" = true ] || printf '%s\n' '중단: Service-owner-Pod-node-client 자동 등록 실패' >&2

latency_fresh() {
  FRESH_READY=false; QUERY_READY=false; HOST_READY=false; LOG_READY=false
  local s d p r n c cid ccid
  [ "$AUTO_READY" = true ] || return 1
  s="$(kubectl get service "$SERVICE" -n "$NS" -o json)" || return 1; d="$(kubectl get deployment "$DEPLOYMENT" -n "$NS" -o json)" || return 1; p="$(kubectl get pod "$POD" -n "$NS" -o json)" || return 1
  r="$(kubectl get replicaset "$RS" -n "$NS" -o json)" || return 1; n="$(kubectl get node "$NODE" -o json)" || return 1; c="$(kubectl get pod "$CLIENT_POD" -n "$CLIENT_NS" -o json)" || return 1
  cid="$(jq -r --arg x "$CONTAINER" '[.status.containerStatuses[]?|select(.name==$x)][0].containerID//empty' <<<"$p")"; ccid="$(jq -r --arg x "$CLIENT_CONTAINER" '[.status.containerStatuses[]?|select(.name==$x)][0].containerID//empty' <<<"$c")"
  [ "$(jq -r '.metadata.uid' <<<"$s")" = "$SERVICE_UID" ] && jq -e --argjson sel "$(jq -c '.spec.selector' <<<"$s")" '.metadata.labels as $l|($sel|to_entries|all($l[.key]==.value))' <<<"$p" >/dev/null && [ "$(jq -r '.metadata.uid' <<<"$d")" = "$DEPLOYMENT_UID" ] && jq -e --argjson sel "$(jq -c '.spec.selector.matchLabels' <<<"$d")" '.metadata.labels as $l|($sel|to_entries|all($l[.key]==.value))' <<<"$p" >/dev/null && [ "$(jq -r '.metadata.uid' <<<"$p")" = "$POD_UID" ] && [ "$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid//empty' <<<"$p")" = "$RS_UID" ] && [ "$(jq -r '.metadata.uid' <<<"$r")" = "$RS_UID" ] && [ "$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid//empty' <<<"$r")" = "$DEPLOYMENT_UID" ] && [ "$(jq -r '.spec.nodeName' <<<"$p")" = "$NODE" ] && [ "$(jq -r '.metadata.uid' <<<"$n")" = "$NODE_UID" ] && [ "$(jq -r '.status.nodeInfo.machineID' <<<"$n")" = "$NODE_MACHINE_ID" ] && [ "$cid" = "$CONTAINER_ID" ] && [ "$(jq -r '.metadata.uid' <<<"$c")" = "$CLIENT_UID" ] && [ "$(jq -r '.spec.nodeName' <<<"$c")" = "$CLIENT_NODE" ] && [ "$ccid" = "$CLIENT_CONTAINER_ID" ] || return 1
  FRESH_READY=true
}

latency_logs() {
  latency_fresh || { printf '%s\n' '중단: fresh identity gate 실패' >&2; return 1; }
  [ "$(kubectl auth can-i get pods/log -n "$NS")" = yes ] || return 1
  local raw rc line ts epoch end_epoch window out frc
  raw="$(kubectl logs "$POD" -n "$NS" -c "$CONTAINER" --timestamps --since-time="$START_RFC3339" --tail=600)"; rc=$?; [ "$rc" -eq 0 ] || { printf 'producer rc=%s\n' "$rc" >&2; return "$rc"; }
  end_epoch="$(date -u -d "$END_RFC3339" +%s)"; window=''
  while IFS= read -r line; do ts="${line%% *}"; epoch="$(date -u -d "$ts" +%s 2>/dev/null)" || return 2; [ "$epoch" -le "$end_epoch" ] && window+="$line"$'\n'; done <<<"$raw"
  [ -n "$window" ] || { printf '%s\n' '증거 창 empty' >&2; return 3; }
  if out="$(grep -E -i -C 3 'timeout|servfail|refused|forward|healthcheck|ready|reload|plugin|error|fatal|panic' <<<"$window")"; then LOG_READY=true; printf '%s\n' "$out"; else frc=$?; printf 'filter no-match rc=%s; 오류 부재 아님\n' "$frc" >&2; return "$frc"; fi
}
```

복수 Pod에서 첫 후보를 자동 선택하지 않는다. 다른 replica를 비교할 때 전체 상태를 폐기하고 그 Pod를 다시 등록한다.

| 적용 | 같은 길이 baseline/incident 창과 current Service→Pod→node identity를 연결할 수 있는 경우 |

검색 문자열: `i/o timeout`, `context deadline exceeded`, `forward`, `healthcheck`, `SERVFAIL`, `no ready endpoints`.

## 관측별 진단 경로

| 지금 보이는 것 | 먼저 볼 것 | 그다음 어디를 볼지 | 상세 절 |
|---|---|---|---|
| QPS와 p99가 함께 증가 | selected Pod process/node saturation | capacity·plugin | [[#1. 객체·로그·client 재현|1]] |

### 1. 객체·로그·client 재현

`if latency_fresh && [ "$FRESH_READY" = true ]; then kubectl get deployment "$DEPLOYMENT" -n "$NS" -o custom-columns='UID:.metadata.uid,RV:.metadata.resourceVersion,DESIRED:.spec.replicas,READY:.status.readyReplicas,AVAILABLE:.status.availableReplicas,OBSERVED:.status.observedGeneration' && kubectl get pod "$POD" -n "$NS" -o custom-columns='UID:.metadata.uid,RV:.metadata.resourceVersion,NODE:.spec.nodeName,READY:.status.containerStatuses[*].ready,RESTARTS:.status.containerStatuses[*].restartCount,START:.status.startTime' && kubectl get service "$SERVICE" -n "$NS" -o custom-columns='UID:.metadata.uid,RV:.metadata.resourceVersion,PORTS:.spec.ports[*].port,PROTOCOLS:.spec.ports[*].protocol,TARGET-PORTS:.spec.ports[*].targetPort' && kubectl get endpointslice -n "$NS" -l "kubernetes.io/service-name=$SERVICE" -o custom-columns='UID:.metadata.uid,RV:.metadata.resourceVersion,TARGET-UID:.endpoints[*].targetRef.uid,READY:.endpoints[*].conditions.ready,SERVING:.endpoints[*].conditions.serving,TERMINATING:.endpoints[*].conditions.terminating,PORTS:.ports[*].port,PROTOCOLS:.ports[*].protocol'; else printf '%s\n' '중단: fresh identity gate 실패' >&2; fi`

EndpointSlice의 target UID·UDP/TCP port가 current Pod와 맞아야 한다. 빈 출력은 endpoint 0과 API/RBAC 실패를 구분한다.

`latency_logs`

`if latency_fresh && [ "$FRESH_READY" = true ] && [ "$(kubectl auth can-i create pods/exec -n "$CLIENT_NS")" = yes ]; then kubectl exec "$CLIENT_POD" -n "$CLIENT_NS" -c "$CLIENT_CONTAINER" -- sh -c 'start=$(date +%s); if command -v dig >/dev/null 2>&1; then dig +time=2 +tries=1 "$1" A >/dev/null; elif command -v nslookup >/dev/null 2>&1; then nslookup "$1" >/dev/null; elif command -v getent >/dev/null 2>&1; then getent hosts "$1" >/dev/null; else exit 127; fi; rc=$?; end=$(date +%s); printf "rc=%s elapsed_s=%s\n" "$rc" "$((end-start))"; exit "$rc"' sh "$QUERY_NAME"; RC=$?; printf 'clientUID=%s containerID=%s outer_rc=%s observed=%s\n' "$CLIENT_UID" "$CLIENT_CONTAINER_ID" "$RC" "$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; else printf '%s\n' '중단: fresh client/exec gate 실패' >&2; fi`

한 query는 재현 증거일 뿐 p99가 아니다. 반복 부하를 만들지 않는다.

### 2. 조건부 target node 관측

특정 Pod/node에서만 지연되고 node owner가 승인한 경우에만 실행한다. `SSH_TARGET`은 inventory상 current `NODE`에 대응하는 host여야 한다. 작업 PC 변수를 원격 shell에 삽입하지 않고 positional argument로 전달하며, 원격 `/etc/machine-id`를 Kubernetes Node의 current machineID와 다시 대조한다.

`if latency_fresh && [ "$FRESH_READY" = true ] && [[ "$SSH_TARGET" =~ ^[A-Za-z0-9._@:-]+$ ]]; then ssh "$SSH_TARGET" 'set -eu; expected_machine=$1; actual_machine=$(tr -d "\n" </etc/machine-id); [ "$actual_machine" = "$expected_machine" ] || { printf "machine-id-mismatch expected=%s actual=%s\n" "$expected_machine" "$actual_machine" >&2; exit 42; }; printf "host=%s machineID=%s observed=" "$(hostname -f)" "$actual_machine"; date -u "+%Y-%m-%dT%H:%M:%SZ"; uptime' sh "$NODE_MACHINE_ID"; RC=$?; if [ "$RC" -eq 0 ]; then HOST_READY=true; else HOST_READY=false; printf '중단: remote producer/identity rc=%s\n' "$RC" >&2; fi; else HOST_READY=false; printf '%s\n' '중단: fresh/approved SSH gate 실패' >&2; fi`

원격 identity와 `uptime` 성공은 DNS latency 원인을 증명하지 않는다. 확인된 systemd unit과 strict UTC journal 창이 필요하면 node/runtime 런북으로 이관한다. node 전용 도구는 이 분기 밖에서 요구하지 않는다.

## Prometheus 빠른 관측

CoreDNS series에 `pod` label이 실제 있을 때만 사용한다.

`if latency_fresh && [ "$FRESH_READY" = true ]; then QUERY_READY=true; printf 'sum(rate(coredns_dns_requests_total{pod=~"%s"}[5m]))\n' "$POD_RE"; printf 'histogram_quantile(0.99,sum by(le)(rate(coredns_dns_request_duration_seconds_bucket{pod=~"%s"}[5m])))\n' "$POD_RE"; printf 'sum(increase(coredns_proxy_healthcheck_failures_total{pod=~"%s",proxy_name="forward"}[5m]))\n' "$POD_RE"; else QUERY_READY=false; printf '%s\n' '중단: fresh PromQL gate 실패' >&2; fi`

완성 query를 baseline과 incident UTC 창에서 동일 step으로 각각 조회한다. metric/label 미지원 또는 빈 결과는 0이 아니며 scrape/API/권한/조회 범위를 분리한다.

## 변경·정상화

이 문서는 scale/restart/reload 명령을 제공하지 않는다. 별도 mutation lane에는 fresh UID/current RV, selector/revision/image digest, 같은 창의 QPS·histogram·endpoint·log·client 기능, 승인 ID·변경 시각, capacity/PDB/quorum, 비용·side effect 멱등성, 검증된 rollback revision이 필요하다.

교체와 rollback 뒤 모든 identity/gate를 폐기하고 current controller UID·selector·revision·creationTimestamp로 단일 새 Pod를 재등록한다. 정상화는 `Deployment/Pod identity → endpoint 분포 → process/node → plugin/upstream → 같은 client query → 실제 기능` 순으로 확인하며 동일 길이 정상 창에서 latency quantile과 failure counter가 baseline으로 돌아오고 재발이 없어야 한다.

## 증거 보존

UTC baseline/incident, 객체 UID/current RV, Pod-node-containerID, endpoint condition/port, query rc/elapsed, 비식별 log·metric, remote host machineID 검증과 승인·rollback ID만 보존한다. query/domain/address/label 원문, token, 전체 log/audit body는 저장하지 않는다.
