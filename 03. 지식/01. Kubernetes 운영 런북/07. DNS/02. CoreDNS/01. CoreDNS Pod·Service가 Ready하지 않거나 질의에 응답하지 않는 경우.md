# CoreDNS Pod·Service가 Ready하지 않거나 질의에 응답하지 않는 경우

## 입력값과 빠른 수집

연결 hop은 `승인된 client Pod resolver → DNS Service UDP/TCP 53 → EndpointSlice ready/serving endpoint → 선택한 CoreDNS Pod의 regular container → CoreDNS process`다. API 조회는 승인된 작업 PC/bastion에서, 질의는 client container에서 실행한다.

| 분류 | 변수 | 출처와 gate |
|---|---|---|
| 최초 입력 | DNS namespace·Service | 실제 DNS Service exact name |
| API 자동값 | Service UID/RV·selector, CoreDNS Pod→ReplicaSet→Deployment UID/RV, node UID/machineID, 두 current containerID | 현재 API 관계가 모두 맞을 때만 등록 |
| 계산 후보 | `DNS_POD_RE` | 선택 Pod 이름을 regex escape한 값 |
| 운영자 선택 | CoreDNS Pod/container, client Pod/container, 승인 query, UTC 창 | 후보 출력과 실제 재현 대상에서 선택 |

### 1. 등록 lifecycle

```bash
TOOLS_READY=false
MISSING=''
if [ -z "${BASH_VERSION:-}" ]; then
  printf '%s\n' '중단: Bash에서 실행한다.' >&2
else
  for tool in kubectl jq sed grep date; do command -v "$tool" >/dev/null 2>&1 || MISSING="$MISSING $tool"; done
  if [ -z "$MISSING" ]; then TOOLS_READY=true; else printf '중단: 없는 도구=%s\n' "$MISSING" >&2; fi
fi

unset INPUT_READY AUTO_READY FRESH_READY LOG_READY QUERY_READY
unset SERVICE_UID SERVICE_RV DNS_SELECTOR DNS_POD_UID DNS_POD_RV DNS_NODE DNS_NODE_UID DNS_MACHINE_ID DNS_CONTAINER_ID DNS_POD_RE
unset RS RS_UID DEPLOYMENT DEPLOYMENT_UID DEPLOYMENT_RV CLIENT_UID CLIENT_RV CLIENT_NODE CLIENT_CONTAINER_ID
DNS_NS='PASTE_DNS_NAMESPACE_HERE'
DNS_SERVICE='PASTE_DNS_SERVICE_HERE'
DNS_POD='PASTE_COREDNS_POD_HERE'
DNS_CONTAINER='PASTE_COREDNS_REGULAR_CONTAINER_HERE'
CLIENT_NS='PASTE_CLIENT_NAMESPACE_HERE'
CLIENT_POD='PASTE_CLIENT_POD_HERE'
CLIENT_CONTAINER='PASTE_CLIENT_REGULAR_CONTAINER_HERE'
QUERY_NAME='PASTE_APPROVED_QUERY_HERE'
START_RFC3339='PASTE_START_UTC_HERE'
END_RFC3339='PASTE_END_UTC_HERE'
INPUT_READY=false
TIME_RE='^[0-9]{4}-(0[1-9]|1[0-2])-([0-2][0-9]|3[01])T([01][0-9]|2[0-3]):[0-5][0-9]:[0-5][0-9]Z$'
if [ "$TOOLS_READY" = true ] && [ -n "$DNS_NS" ] && [ -n "$DNS_SERVICE" ] && [ -n "$DNS_POD" ] && [ -n "$DNS_CONTAINER" ] && [ -n "$CLIENT_NS" ] && [ -n "$CLIENT_POD" ] && [ -n "$CLIENT_CONTAINER" ] && [ -n "$QUERY_NAME" ] && [[ "$START_RFC3339" =~ $TIME_RE && "$END_RFC3339" =~ $TIME_RE ]] && S="$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" && E="$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" && [ "$S" = "$START_RFC3339" ] && [ "$E" = "$END_RFC3339" ] && [ "$(date -u -d "$S" +%s)" -lt "$(date -u -d "$E" +%s)" ]; then
  INPUT_READY=true
else
  printf '%s\n' '중단: 입력 또는 strict UTC start<end gate 실패' >&2
fi
unset S E

AUTO_READY=false
if [ "$INPUT_READY" = true ] && SJ="$(kubectl get service "$DNS_SERVICE" -n "$DNS_NS" -o json)" && DP="$(kubectl get pod "$DNS_POD" -n "$DNS_NS" -o json)" && CP="$(kubectl get pod "$CLIENT_POD" -n "$CLIENT_NS" -o json)"; then
  SERVICE_UID="$(jq -r '.metadata.uid//empty' <<<"$SJ")"; SERVICE_RV="$(jq -r '.metadata.resourceVersion//empty' <<<"$SJ")"
  DNS_SELECTOR="$(jq -r '.spec.selector|to_entries|sort_by(.key)|map("\(.key)=\(.value)")|join(",")' <<<"$SJ")"
  DNS_POD_UID="$(jq -r '.metadata.uid//empty' <<<"$DP")"; DNS_POD_RV="$(jq -r '.metadata.resourceVersion//empty' <<<"$DP")"; DNS_NODE="$(jq -r '.spec.nodeName//empty' <<<"$DP")"
  DNS_CONTAINER_ID="$(jq -r --arg c "$DNS_CONTAINER" '[.status.containerStatuses[]?|select(.name==$c)][0].containerID//empty' <<<"$DP")"
  RS="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true and .kind=="ReplicaSet")][0].name//empty' <<<"$DP")"
  CLIENT_UID="$(jq -r '.metadata.uid//empty' <<<"$CP")"; CLIENT_RV="$(jq -r '.metadata.resourceVersion//empty' <<<"$CP")"; CLIENT_NODE="$(jq -r '.spec.nodeName//empty' <<<"$CP")"
  CLIENT_CONTAINER_ID="$(jq -r --arg c "$CLIENT_CONTAINER" '[.status.containerStatuses[]?|select(.name==$c)][0].containerID//empty' <<<"$CP")"
  if RSJ="$(kubectl get replicaset "$RS" -n "$DNS_NS" -o json)" && NJ="$(kubectl get node "$DNS_NODE" -o json)"; then
    RS_UID="$(jq -r '.metadata.uid//empty' <<<"$RSJ")"; DEPLOYMENT="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true and .kind=="Deployment")][0].name//empty' <<<"$RSJ")"; DEPLOYMENT_UID="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true and .kind=="Deployment")][0].uid//empty' <<<"$RSJ")"
    DNS_NODE_UID="$(jq -r '.metadata.uid//empty' <<<"$NJ")"; DNS_MACHINE_ID="$(jq -r '.status.nodeInfo.machineID//empty' <<<"$NJ")"
    if DJ="$(kubectl get deployment "$DEPLOYMENT" -n "$DNS_NS" -o json)"; then
      DEPLOYMENT_RV="$(jq -r '.metadata.resourceVersion//empty' <<<"$DJ")"; DNS_POD_RE="^$(printf '%s' "$DNS_POD" | sed 's/[][\\.^$*+?(){}|]/\\&/g')$"
      if [ -n "$SERVICE_UID" ] && [ -n "$DNS_SELECTOR" ] && jq -e --argjson sel "$(jq -c '.spec.selector' <<<"$SJ")" '.metadata.labels as $l|($sel|to_entries|all($l[.key]==.value))' <<<"$DP" >/dev/null && [ "$(jq -r '.metadata.uid' <<<"$DJ")" = "$DEPLOYMENT_UID" ] && [ "$(jq -r '.status.phase' <<<"$DP")" = Running ] && [ "$(jq -r '.status.phase' <<<"$CP")" = Running ] && [[ "$DNS_CONTAINER_ID" =~ ^(containerd|cri-o|docker)://[^[:space:]]+$ ]] && [[ "$CLIENT_CONTAINER_ID" =~ ^(containerd|cri-o|docker)://[^[:space:]]+$ ]] && [ -n "$DNS_NODE_UID" ] && [ -n "$DNS_MACHINE_ID" ]; then
        AUTO_READY=true
        printf 'AUTO_READY=true service=%s/%s rv=%s selector=%s deployment=%s/%s rs=%s/%s dnsPod=%s/%s node=%s/%s runtime=%s client=%s/%s runtime=%s podRE=%s\n' "$DNS_SERVICE" "$SERVICE_UID" "$SERVICE_RV" "$DNS_SELECTOR" "$DEPLOYMENT" "$DEPLOYMENT_UID" "$RS" "$RS_UID" "$DNS_POD" "$DNS_POD_UID" "$DNS_NODE" "$DNS_NODE_UID" "${DNS_CONTAINER_ID%%://*}" "$CLIENT_POD" "$CLIENT_UID" "${CLIENT_CONTAINER_ID%%://*}" "$DNS_POD_RE"
      fi
    fi
  fi
fi
unset SJ DP CP RSJ NJ DJ
[ "$AUTO_READY" = true ] || printf '%s\n' '중단: Service-Pod-owner-node-client 자동 등록 실패' >&2

coredns_fresh() {
  FRESH_READY=false; LOG_READY=false; QUERY_READY=false
  local s p r d n c dns_cid client_cid
  [ "$AUTO_READY" = true ] || return 1
  s="$(kubectl get service "$DNS_SERVICE" -n "$DNS_NS" -o json)" || return 1
  p="$(kubectl get pod "$DNS_POD" -n "$DNS_NS" -o json)" || return 1
  r="$(kubectl get replicaset "$RS" -n "$DNS_NS" -o json)" || return 1
  d="$(kubectl get deployment "$DEPLOYMENT" -n "$DNS_NS" -o json)" || return 1
  n="$(kubectl get node "$DNS_NODE" -o json)" || return 1
  c="$(kubectl get pod "$CLIENT_POD" -n "$CLIENT_NS" -o json)" || return 1
  dns_cid="$(jq -r --arg x "$DNS_CONTAINER" '[.status.containerStatuses[]?|select(.name==$x)][0].containerID//empty' <<<"$p")"
  client_cid="$(jq -r --arg x "$CLIENT_CONTAINER" '[.status.containerStatuses[]?|select(.name==$x)][0].containerID//empty' <<<"$c")"
  [ "$(jq -r '.metadata.uid' <<<"$s")" = "$SERVICE_UID" ] && jq -e --argjson sel "$(jq -c '.spec.selector' <<<"$s")" '.metadata.labels as $l|($sel|to_entries|all($l[.key]==.value))' <<<"$p" >/dev/null && [ "$(jq -r '.metadata.uid' <<<"$p")" = "$DNS_POD_UID" ] && [ "$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid//empty' <<<"$p")" = "$RS_UID" ] && [ "$(jq -r '.metadata.uid' <<<"$r")" = "$RS_UID" ] && [ "$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid//empty' <<<"$r")" = "$DEPLOYMENT_UID" ] && [ "$(jq -r '.metadata.uid' <<<"$d")" = "$DEPLOYMENT_UID" ] && jq -e --argjson sel "$(jq -c '.spec.selector.matchLabels' <<<"$d")" '.metadata.labels as $l|($sel|to_entries|all($l[.key]==.value))' <<<"$p" >/dev/null && [ "$(jq -r '.spec.nodeName' <<<"$p")" = "$DNS_NODE" ] && [ "$(jq -r '.metadata.uid' <<<"$n")" = "$DNS_NODE_UID" ] && [ "$(jq -r '.status.nodeInfo.machineID' <<<"$n")" = "$DNS_MACHINE_ID" ] && [ "$dns_cid" = "$DNS_CONTAINER_ID" ] && [ "$(jq -r '.metadata.uid' <<<"$c")" = "$CLIENT_UID" ] && [ "$(jq -r '.spec.nodeName' <<<"$c")" = "$CLIENT_NODE" ] && [ "$client_cid" = "$CLIENT_CONTAINER_ID" ] || return 1
  FRESH_READY=true
}

bounded_coredns_log() {
  coredns_fresh || { printf '%s\n' '중단: fresh identity gate 실패' >&2; return 1; }
  [ "$(kubectl auth can-i get pods/log -n "$DNS_NS")" = yes ] || { printf '%s\n' '중단: pods/log RBAC gate 실패' >&2; return 1; }
  local raw rc line ts epoch end_epoch window matches filter_rc
  raw="$(kubectl logs "$DNS_POD" -n "$DNS_NS" -c "$DNS_CONTAINER" --timestamps --since-time="$START_RFC3339" --tail=500)"; rc=$?
  [ "$rc" -eq 0 ] || { printf '중단: log producer rc=%s\n' "$rc" >&2; return "$rc"; }
  end_epoch="$(date -u -d "$END_RFC3339" +%s)"; window=''
  while IFS= read -r line; do
    ts="${line%% *}"; epoch="$(date -u -d "$ts" +%s 2>/dev/null)" || { printf '중단: log timestamp parse 실패=%s\n' "$ts" >&2; return 2; }
    [ "$epoch" -le "$end_epoch" ] && window+="$line"$'\n'
  done <<<"$raw"
  [ -n "$window" ] || { printf '%s\n' '필터 전 증거 창이 비어 있음; 로그 없음과 producer 실패를 구분함' >&2; return 3; }
  if matches="$(grep -E -i -C 3 'ready|health|reload|plugin|parse|servfail|refused|timeout|upstream|error|fatal|panic' <<<"$window")"; then LOG_READY=true; printf '%s\n' "$matches"; else filter_rc=$?; printf 'filter no-match rc=%s; 오류 부재로 해석하지 않음\n' "$filter_rc" >&2; return "$filter_rc"; fi
}
```

`AUTO_READY=true`가 아니면 중단한다. 복수 CoreDNS Pod에서 첫 후보를 자동 선택하지 않는다. `DNS_POD`는 출력과 실제 증상 범위를 보고 선택하며, 다른 replica를 조사할 때는 전체 자동값을 폐기하고 다시 등록한다.

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | CoreDNS desired보다 Ready가 적거나 DNS Service 질의에 응답하지 않는 경우 |
| 적용 | Service→EndpointSlice→현재 CoreDNS Pod/owner와 current client identity를 연결할 수 있는 경우 |
| 제외 | Corefile parse/reload 오류는 [[02. CoreDNS Corefile·plugin 설정 오류로 질의가 실패하는 경우]], 지속 지연은 [[03. CoreDNS 질의가 간헐적으로 timeout되거나 응답이 지연되는 경우]]로 이동한다. |

검색 문자열: `Ready=False`, `no ready endpoints`, `SERVFAIL`, `REFUSED`, `i/o timeout`, `plugin/errors`.

## 관측별 진단 경로

| 지금 보이는 것 | 먼저 볼 것 | 그다음 어디를 볼지 | 상세 절 |
|---|---|---|---|
| desired보다 Ready가 적음 | observedGeneration, selected Pod restart/condition | process·probe·node | [[#1. 객체와 전달 관계|1]] |
| ready endpoint가 없음 | selector, target UID, endpoint port/condition | Service·EndpointSlice owner | [[#1. 객체와 전달 관계|1]] |
| endpoint는 ready지만 질의 실패 | UDP/TCP 53과 DNS response code | CoreDNS log·plugin | [[#2. 로그와 실제 질의|2]] |
| 한 client만 실패 | current client resolver와 containerID | Pod DNS policy | [[#2. 로그와 실제 질의|2]] |

### 1. 객체와 전달 관계

`if coredns_fresh && [ "$FRESH_READY" = true ]; then kubectl get deployment "$DEPLOYMENT" -n "$DNS_NS" -o custom-columns='UID:.metadata.uid,RV:.metadata.resourceVersion,DESIRED:.spec.replicas,UPDATED:.status.updatedReplicas,AVAILABLE:.status.availableReplicas,READY:.status.readyReplicas,GENERATION:.metadata.generation,OBSERVED:.status.observedGeneration' && kubectl get pod "$DNS_POD" -n "$DNS_NS" -o custom-columns='UID:.metadata.uid,RV:.metadata.resourceVersion,NODE:.spec.nodeName,PHASE:.status.phase,READY:.status.containerStatuses[*].ready,RESTARTS:.status.containerStatuses[*].restartCount,START:.status.startTime' && kubectl get service "$DNS_SERVICE" -n "$DNS_NS" -o custom-columns='UID:.metadata.uid,RV:.metadata.resourceVersion,PORTS:.spec.ports[*].port,PROTOCOLS:.spec.ports[*].protocol,TARGET-PORTS:.spec.ports[*].targetPort' && kubectl get endpointslice -n "$DNS_NS" -l "kubernetes.io/service-name=$DNS_SERVICE" -o custom-columns='UID:.metadata.uid,RV:.metadata.resourceVersion,TARGET-UID:.endpoints[*].targetRef.uid,READY:.endpoints[*].conditions.ready,SERVING:.endpoints[*].conditions.serving,TERMINATING:.endpoints[*].conditions.terminating,PORTS:.ports[*].port,PROTOCOLS:.ports[*].protocol'; else printf '%s\n' '중단: current identity gate 실패' >&2; fi`

Service에는 UDP와 TCP 53이 각각 있어야 하고, 각 `targetPort`가 EndpointSlice의 같은 protocol port로 materialize되며, 선택 endpoint의 target UID가 current CoreDNS Pod UID여야 한다. 빈 EndpointSlice 출력은 endpoint 0으로 간주하지 않고 API/RBAC 실패와 실제 빈 목록을 별도 조회로 구분한다. `ready=true`는 listener와 DNS application 응답을 보장하지 않는다.

### 2. 로그와 실제 질의

**strict UTC 창의 bounded 로그**

`bounded_coredns_log`

domain·upstream·token은 저장 전에 마스킹한다. producer 실패, timestamp 오류, 증거 창 empty와 filter no-match는 서로 다른 종료 상태다.

**current client의 단일 질의**

`if coredns_fresh && [ "$FRESH_READY" = true ] && [ "$(kubectl auth can-i create pods/exec -n "$CLIENT_NS")" = yes ]; then kubectl exec "$CLIENT_POD" -n "$CLIENT_NS" -c "$CLIENT_CONTAINER" -- sh -c 'if command -v dig >/dev/null 2>&1; then exec dig +time=2 +tries=1 "$1" A; elif command -v nslookup >/dev/null 2>&1; then exec nslookup "$1"; elif command -v getent >/dev/null 2>&1; then exec getent hosts "$1"; else exit 127; fi' sh "$QUERY_NAME"; RC=$?; printf 'clientUID=%s containerID=%s query_rc=%s observed=%s window=%s..%s\n' "$CLIENT_UID" "$CLIENT_CONTAINER_ID" "$RC" "$(date -u '+%Y-%m-%dT%H:%M:%SZ')" "$START_RFC3339" "$END_RFC3339"; else printf '%s\n' '중단: CoreDNS/client identity 또는 exec gate 실패' >&2; fi`

`NXDOMAIN`, `SERVFAIL`, timeout과 `getent` empty를 구분한다. query와 응답 주소는 저장하지 않는다. UDP 성공만으로 TCP fallback을 보장하지 않으므로 truncation 또는 TCP 증상이 있으면 승인된 단일 TCP query로 별도 재현한다.

## Prometheus 빠른 관측

CoreDNS scrape series에 `pod` label이 실제 있을 때만 사용한다.

`if coredns_fresh && [ "$FRESH_READY" = true ]; then QUERY_READY=true; printf 'sum by(rcode)(rate(coredns_dns_responses_total{pod=~"%s",rcode=~"SERVFAIL|REFUSED"}[5m]))\n' "$DNS_POD_RE"; printf 'kube_deployment_status_replicas_available{namespace="%s",deployment="%s"} < kube_deployment_spec_replicas{namespace="%s",deployment="%s"}\n' "$DNS_NS" "$DEPLOYMENT" "$DNS_NS" "$DEPLOYMENT"; else QUERY_READY=false; printf '%s\n' '중단: fresh PromQL gate 실패' >&2; fi`

완성 query를 입력 UTC 창에서 사용한다. 빈 결과는 0이 아니며 metric 지원, scrape target, label, 권한/API와 조회 범위를 구분한다.

## 변경·교체·정상화

이 문서는 restart, scale, ConfigMap·Service 변경 명령을 제공하지 않는다. 별도 mutation lane에는 fresh Service·Deployment·ReplicaSet·Pod UID/current RV와 selector/revision/image digest, 동일 UTC 증거, 승인 ID·변경 시각, PDB/replica/quorum, query 부하와 외부 side effect 멱등성, 검증된 rollback revision이 필요하다. 실행 직전 대상과 availability를 다시 확인하고 이미 원하는 상태면 실행하지 않는다.

교체나 rollback 뒤 이전 UID/node/containerID와 모든 gate를 폐기한다. current Deployment UID·selector·revision·creationTimestamp로 단일 새 Pod를 다시 선택한다. 정상화는 `Deployment 수렴 → 새 Pod identity/process → EndpointSlice target UID와 UDP/TCP port → 같은 client DNS 응답 → 실제 의존 기능` 순으로 확인하며 승인 관찰 창에 재발이 없어야 한다.

## 증거 보존

UTC 창, Service·Deployment·ReplicaSet·Pod·node·EndpointSlice UID/current RV, container kind/name/ID runtime prefix, port/protocol/condition, 비식별 오류와 query rc, 승인·rollback ID만 보존한다. 주소, domain, label 원문, Secret/token, 전체 log/audit body는 저장하지 않는다.
