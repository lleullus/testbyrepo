# Pod의 dnsPolicy·dnsConfig·검색 도메인이 기대와 다른 경우

## 입력값과 빠른 수집

실행 경로는 `승인된 작업 PC 또는 bastion의 같은 Bash → API server → 현재 Pod의 regular container network namespace → resolver → nameserver`다. `hostNetwork=true`이면 container가 node network namespace를 공유한다. 작업 host나 다른 Pod의 결과를 대상 Pod의 결과로 대체하지 않는다.

| 분류 | 변수 | 출처와 중단 조건 |
|---|---|---|
| 최초 입력 | `NS`, `POD` | 승인된 namespace와 exact Pod name. API 조회 실패 시 중단한다. |
| API 자동값 | Pod UID/RV, owner kind/name/UID, node UID/machineID, current regular-container ID, 현재 DNS spec | 같은 API snapshot에서 등록한다. 하나라도 비거나 관계가 바뀌면 폐기한다. |
| 계산 후보 | `POD_RE` | Pod 이름을 PromQL regex용으로 escape한다. 출력 확인 전에는 사용하지 않는다. |
| 운영자 선택 | `CONTAINER`, `EXPECTED_POLICY`, `QUERY_NAME`, UTC 시작·종료 | container 후보, 승인된 선언 원본, 실제 장애 질의와 증거 창에서 선택한다. |

### 1. 입력·자동값 등록

아래 스크립트는 새 Bash에서 실행한다. 각 `PASTE_..._HERE` 입력칸은 이 문서에서 한 번만 나온다. 재실행 시 모든 downstream gate와 자동값을 먼저 폐기한다.

```bash
TOOLS_READY=false
MISSING=''
if [ -z "${BASH_VERSION:-}" ]; then
  printf '%s\n' '중단: Bash에서 실행한다.' >&2
else
  for tool in kubectl jq sed grep date; do
    command -v "$tool" >/dev/null 2>&1 || MISSING="$MISSING $tool"
  done
  if [ -z "$MISSING" ]; then TOOLS_READY=true; else printf '중단: 없는 도구=%s\n' "$MISSING" >&2; fi
fi

unset INPUT_READY AUTO_READY SELECTION_READY FRESH_READY EXEC_READY QUERY_READY
unset POD_UID POD_RV OWNER_KIND OWNER_NAME OWNER_UID NODE NODE_UID NODE_MACHINE_ID
unset CONTAINER_ID DNS_POLICY HOST_NETWORK DNS_CONFIG POD_RE CURRENT_RV
NS='PASTE_NAMESPACE_HERE'
POD='PASTE_POD_HERE'
CONTAINER='PASTE_REGULAR_CONTAINER_HERE'
EXPECTED_POLICY='PASTE_EXPECTED_DNS_POLICY_HERE'
QUERY_NAME='PASTE_APPROVED_QUERY_NAME_HERE'
START_RFC3339='PASTE_START_UTC_HERE'
END_RFC3339='PASTE_END_UTC_HERE'
INPUT_READY=false
TIME_RE='^[0-9]{4}-(0[1-9]|1[0-2])-([0-2][0-9]|3[01])T([01][0-9]|2[0-3]):[0-5][0-9]:[0-5][0-9]Z$'
if [ "$TOOLS_READY" != true ] || [ -z "$NS" ] || [ -z "$POD" ] || [ -z "$CONTAINER" ] || [ -z "$QUERY_NAME" ]; then
  printf '%s\n' '중단: 도구와 입력값을 확인한다.' >&2
elif ! [[ "$START_RFC3339" =~ $TIME_RE && "$END_RFC3339" =~ $TIME_RE ]]; then
  printf '%s\n' '중단: 시간 형식은 UTC RFC3339 초 단위여야 한다.' >&2
elif ! START_CANON="$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" || ! END_CANON="$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" || [ "$START_CANON" != "$START_RFC3339" ] || [ "$END_CANON" != "$END_RFC3339" ] || [ "$(date -u -d "$START_RFC3339" +%s)" -ge "$(date -u -d "$END_RFC3339" +%s)" ]; then
  printf '%s\n' '중단: UTC round-trip 또는 start<end 검증 실패' >&2
else
  INPUT_READY=true
fi
unset START_CANON END_CANON

AUTO_READY=false
SELECTION_READY=false
if [ "$INPUT_READY" = true ] && POD_JSON="$(kubectl get pod "$POD" -n "$NS" -o json)"; then
  POD_UID="$(jq -r '.metadata.uid//empty' <<<"$POD_JSON")"
  POD_RV="$(jq -r '.metadata.resourceVersion//empty' <<<"$POD_JSON")"
  OWNER_KIND="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].kind//"Pod"' <<<"$POD_JSON")"
  OWNER_NAME="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].name//.metadata.name' <<<"$POD_JSON")"
  OWNER_UID="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid//.metadata.uid' <<<"$POD_JSON")"
  NODE="$(jq -r '.spec.nodeName//empty' <<<"$POD_JSON")"
  CONTAINER_ID="$(jq -r --arg c "$CONTAINER" '[.status.containerStatuses[]?|select(.name==$c)][0].containerID//empty' <<<"$POD_JSON")"
  DNS_POLICY="$(jq -r '.spec.dnsPolicy//"ClusterFirst"' <<<"$POD_JSON")"
  HOST_NETWORK="$(jq -r '.spec.hostNetwork//false' <<<"$POD_JSON")"
  DNS_CONFIG="$(jq -c '.spec.dnsConfig//null' <<<"$POD_JSON")"
  if NODE_JSON="$(kubectl get node "$NODE" -o json)" && OWNER_JSON="$(kubectl get "$OWNER_KIND" "$OWNER_NAME" -n "$NS" -o json)"; then
    NODE_UID="$(jq -r '.metadata.uid//empty' <<<"$NODE_JSON")"
    NODE_MACHINE_ID="$(jq -r '.status.nodeInfo.machineID//empty' <<<"$NODE_JSON")"
    POD_RE="^$(printf '%s' "$POD" | sed 's/[][\\.^$*+?(){}|]/\\&/g')$"
    if [ -n "$POD_UID" ] && [ -n "$POD_RV" ] && [ "$(jq -r '.metadata.uid//empty' <<<"$OWNER_JSON")" = "$OWNER_UID" ] && [ -n "$NODE_UID" ] && [ -n "$NODE_MACHINE_ID" ] && [[ "$CONTAINER_ID" =~ ^(containerd|cri-o|docker)://[^[:space:]]+$ ]] && [[ "$EXPECTED_POLICY" =~ ^(ClusterFirst|ClusterFirstWithHostNet|Default|None)$ ]]; then
      AUTO_READY=true
      SELECTION_READY=true
      printf 'AUTO_READY=true pod=%s uid=%s rv=%s owner=%s/%s/%s node=%s/%s runtime=%s policy=%s hostNetwork=%s podRE=%s\n' "$POD" "$POD_UID" "$POD_RV" "$OWNER_KIND" "$OWNER_NAME" "$OWNER_UID" "$NODE" "$NODE_UID" "${CONTAINER_ID%%://*}" "$DNS_POLICY" "$HOST_NETWORK" "$POD_RE"
    fi
  fi
fi
unset POD_JSON NODE_JSON OWNER_JSON
[ "$AUTO_READY" = true ] || printf '%s\n' '중단: Pod-owner-node-container 자동 등록 실패' >&2

pod_fresh() {
  FRESH_READY=false; EXEC_READY=false; QUERY_READY=false; unset CURRENT_RV
  [ "$AUTO_READY" = true ] && [ "$SELECTION_READY" = true ] || return 1
  local p n o cid owner_kind owner_name owner_uid
  p="$(kubectl get pod "$POD" -n "$NS" -o json)" || return 1
  n="$(kubectl get node "$NODE" -o json)" || return 1
  owner_kind="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].kind//"Pod"' <<<"$p")"
  owner_name="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].name//.metadata.name' <<<"$p")"
  owner_uid="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid//.metadata.uid' <<<"$p")"
  o="$(kubectl get "$owner_kind" "$owner_name" -n "$NS" -o json)" || return 1
  cid="$(jq -r --arg c "$CONTAINER" '[.status.containerStatuses[]?|select(.name==$c)][0].containerID//empty' <<<"$p")"
  CURRENT_RV="$(jq -r '.metadata.resourceVersion//empty' <<<"$p")"
  [ "$(jq -r '.metadata.uid//empty' <<<"$p")" = "$POD_UID" ] && [ "$(jq -r '.spec.nodeName//empty' <<<"$p")" = "$NODE" ] && [ "$(jq -r '.status.phase//empty' <<<"$p")" = Running ] && [ "$owner_kind/$owner_name/$owner_uid" = "$OWNER_KIND/$OWNER_NAME/$OWNER_UID" ] && [ "$(jq -r '.metadata.uid//empty' <<<"$o")" = "$OWNER_UID" ] && [ "$(jq -r '.metadata.uid//empty' <<<"$n")" = "$NODE_UID" ] && [ "$(jq -r '.status.nodeInfo.machineID//empty' <<<"$n")" = "$NODE_MACHINE_ID" ] && [ "$cid" = "$CONTAINER_ID" ] || { unset CURRENT_RV; return 1; }
  FRESH_READY=true
}
```

`AUTO_READY=true`와 bounded identity 출력이 없으면 후속 명령을 실행하지 않는다. `resourceVersion`은 최신 조회마다 바뀔 수 있으므로 `CURRENT_RV`로 다시 기록하되, UID·owner·node·containerID가 바뀌면 과거 증거와 결합하지 않는다.

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | 현재 Pod spec 또는 `/etc/resolv.conf`가 승인된 policy·search·`ndots`와 다른 경우 |
| 적용 | 현재 Pod-owner-node-container identity와 같은 UTC 창에서 spec·resolver·질의를 연결할 수 있는 경우 |
| 제외 | `SERVFAIL`·timeout은 [[01. Pod의 DNS 질의가 timeout·SERVFAIL로 실패하는 경우]], Service record 오류는 [[07. DNS/03. 서비스·외부 이름 해석/01. Kubernetes Service 이름이 NXDOMAIN이거나 잘못된 주소를 반환하는 경우]]로 이동한다. |

검색 문자열: `dnsPolicy`, `dnsConfig`, `ClusterFirstWithHostNet`, `nameserver`, `search`, `ndots`, `NXDOMAIN`.

## 관측별 진단 경로

| 지금 보이는 것 | 먼저 볼 것 | 그다음 어디를 볼지 | 상세 절 |
|---|---|---|---|
| policy·hostNetwork 조합이 승인 원본과 다름 | current UID/RV와 owner template | 선언 원본 owner | [[#1. 현재 spec과 resolver 적용값|1]] |
| spec은 맞지만 resolver 적용값이 다름 | current container의 `/etc/resolv.conf` | kubelet·admission·NodeLocal owner | [[#1. 현재 spec과 resolver 적용값|1]] |
| short name만 NXDOMAIN | search 순서·`ndots`와 승인 FQDN 비교 | Service record 경로 | [[#2. 승인된 단일 질의|2]] |
| SERVFAIL·timeout | 응답 code 또는 무응답 | Pod DNS 실패 경로 | [[#2. 승인된 단일 질의|2]] |

### 1. 현재 spec과 resolver 적용값

**현재 spec snapshot**

`if pod_fresh && [ "$FRESH_READY" = true ]; then kubectl get pod "$POD" -n "$NS" -o jsonpath='uid={.metadata.uid}{" rv="}{.metadata.resourceVersion}{" node="}{.spec.nodeName}{" phase="}{.status.phase}{" policy="}{.spec.dnsPolicy}{" hostNetwork="}{.spec.hostNetwork}{" dnsConfig="}{.spec.dnsConfig}{"\n"}'; else printf '%s\n' '중단: current Pod identity gate 실패' >&2; fi`

API 성공은 resolver 파일 적용이나 DNS 응답을 보장하지 않는다. 현재 `DNS_POLICY`가 `EXPECTED_POLICY`와 다르면 현재 owner UID의 template과 승인된 선언 원본을 대조한다. 직접 owner에 Pod template이 없으면 상위 controller를 이름으로 추측하지 말고 owner chain을 다시 등록한다.

**resolver producer와 filter 분리**

`if pod_fresh && [ "$FRESH_READY" = true ] && [ "$(kubectl auth can-i create pods/exec -n "$NS")" = yes ]; then RAW="$(kubectl exec "$POD" -n "$NS" -c "$CONTAINER" -- sh -c 'test -r /etc/resolv.conf && exec cat /etc/resolv.conf')"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -ne 0 ]; then EXEC_READY=false; printf '중단: resolver producer rc=%s\n' "$PRODUCER_RC" >&2; elif OUT="$(grep -E '^(nameserver|search|options)[[:space:]]' <<<"$RAW")"; then EXEC_READY=true; printf '%s\n' "$OUT"; else FILTER_RC=$?; EXEC_READY=false; printf '중단: resolver 항목 없음 filter_rc=%s\n' "$FILTER_RC" >&2; fi; unset RAW OUT; else EXEC_READY=false; printf '%s\n' '중단: fresh identity 또는 pods/exec RBAC gate 실패' >&2; fi`

resolver 원문, 내부 domain과 nameserver 주소는 화면에서만 대조하고 사건 기록에는 저장하지 않는다. `grep` no-match는 정상이나 `0`이 아니다.

### 2. 승인된 단일 질의

`if pod_fresh && [ "$FRESH_READY" = true ] && [ "$(kubectl auth can-i create pods/exec -n "$NS")" = yes ]; then kubectl exec "$POD" -n "$NS" -c "$CONTAINER" -- sh -c 'if command -v dig >/dev/null 2>&1; then exec dig +time=2 +tries=1 "$1" A; elif command -v nslookup >/dev/null 2>&1; then exec nslookup "$1"; elif command -v getent >/dev/null 2>&1; then exec getent hosts "$1"; else exit 127; fi' sh "$QUERY_NAME"; QUERY_RC=$?; printf 'query_rc=%s podUID=%s containerID=%s observed=%s evidence_window=%s..%s\n' "$QUERY_RC" "$POD_UID" "$CONTAINER_ID" "$(date -u '+%Y-%m-%dT%H:%M:%SZ')" "$START_RFC3339" "$END_RFC3339"; else printf '%s\n' '중단: current identity/exec gate 실패' >&2; fi`

`dig`·`nslookup`만 `NXDOMAIN`·`SERVFAIL`·timeout을 구분한다. `getent` 빈 결과는 response code를 증명하지 않는다. 질의 이름과 응답 주소는 저장하지 않는다. 관측 시각이 입력한 증거 창 밖이면 사건 증거로 결합하지 않는다.

## Prometheus 빠른 관측

Prometheus와 kube-state-metrics에 `namespace`, `pod` label이 실제 존재할 때만 사용한다.

`if pod_fresh && [ "$FRESH_READY" = true ]; then QUERY_READY=true; printf 'kube_pod_status_ready{namespace="%s",pod=~"%s",condition="true"} == 0\n' "$NS" "$POD_RE"; else QUERY_READY=false; printf '%s\n' '중단: fresh PromQL gate 실패' >&2; fi`

출력된 완성 query만 승인된 UI에서 입력 UTC 창으로 조회한다. 빈 결과는 `0`이 아니며 metric 미노출, scrape 실패, label 불일치, 권한·API 오류와 조회 범위를 구분한다.

## 변경 gate와 정상화

이 문서는 mutation 명령을 제공하지 않는다. template 변경·rollout·Pod 교체는 fresh Pod/owner UID와 current RV, 같은 창의 spec/resolver/query 증거, 승인 ID·변경 시각, 현재 revision·image digest, 가용 replica·PDB·quorum 영향, 외부 호출 멱등성, 검증된 rollback revision이 모두 있는 별도 변경 절차에서만 수행한다. 변경 직전에도 대상을 재조회하며 이미 원하는 값이면 실행하지 않는다.

교체나 rollback이 일어나면 기존 Pod 이름을 재사용해도 UID를 재사용하지 않는다. 모든 자동값과 gate를 폐기하고 current owner UID·selector·revision·creationTimestamp로 단일 새 Pod를 고른 뒤 1절부터 재등록한다. 정상화는 `owner 원본 → 새 Pod spec → resolver 적용값 → 같은 질의 response/latency → 실제 애플리케이션 기능` 순으로 확인하고 승인 관찰 창 동안 재발이 없어야 한다.

## 증거 보존

UTC 시작·종료, namespace, Pod·owner·node UID와 current RV, regular container 이름과 containerID runtime prefix, policy·hostNetwork, resolver 비민감 요약, response 종류·exit status, 승인·rollback ID를 보존한다. domain·nameserver·응답 주소, Secret·token, resolver 원문과 audit body는 저장하지 않는다.
