# Kubernetes Service 이름이 NXDOMAIN이거나 잘못된 주소를 반환하는 경우

## 입력값과 빠른 수집

실행 hop은 `current client Pod resolver/search → cluster DNS nameserver → CoreDNS kubernetes record source/cache → current Service UID/type → EndpointSlice target UID`다. 일반 ClusterIP, headless와 ExternalName은 record 계약이 다르므로 Service type을 먼저 고정한다.

| 분류 | 변수 | 출처와 gate |
|---|---|---|
| 최초 입력 | Service namespace/name | 요청 원본의 exact object |
| API 자동값 | Service UID/RV/type/clusterIP/IP family, client owner/node/container identity | current API에서 관계 검증 뒤 등록 |
| 계산 후보 | `SERVICE_RE` | Service 이름을 regex escape한 값 |
| 운영자 선택 | client Pod/regular container, 실제 query name, UTC 창 | 실제 호출 workload와 장애 증거 |

### 1. lifecycle 등록

```bash
TOOLS_READY=false
MISSING=''
if [ -z "${BASH_VERSION:-}" ]; then printf '%s\n' '중단: Bash에서 실행한다.' >&2; else
  for tool in kubectl jq sed grep date; do command -v "$tool" >/dev/null 2>&1 || MISSING="$MISSING $tool"; done
  if [ -z "$MISSING" ]; then TOOLS_READY=true; else printf '중단: 없는 도구=%s\n' "$MISSING" >&2; fi
fi

unset INPUT_READY AUTO_READY FRESH_READY CLIENT_READY QUERY_READY
unset SERVICE_UID SERVICE_RV SERVICE_TYPE CLUSTER_IP IP_FAMILIES SERVICE_RE CLIENT_UID CLIENT_RV CLIENT_NODE CLIENT_NODE_UID CLIENT_MACHINE_ID CLIENT_CONTAINER_ID OWNER_KIND OWNER_NAME OWNER_UID
NS='PASTE_SERVICE_NAMESPACE_HERE'
SERVICE='PASTE_SERVICE_NAME_HERE'
CLIENT_NS='PASTE_CLIENT_NAMESPACE_HERE'
CLIENT_POD='PASTE_CLIENT_POD_HERE'
CLIENT_CONTAINER='PASTE_CLIENT_REGULAR_CONTAINER_HERE'
QUERY_NAME='PASTE_ACTUAL_QUERY_NAME_HERE'
START_RFC3339='PASTE_START_UTC_HERE'
END_RFC3339='PASTE_END_UTC_HERE'
INPUT_READY=false
TIME_RE='^[0-9]{4}-(0[1-9]|1[0-2])-([0-2][0-9]|3[01])T([01][0-9]|2[0-3]):[0-5][0-9]:[0-5][0-9]Z$'
if [ "$TOOLS_READY" = true ] && [ -n "$NS" ] && [ -n "$SERVICE" ] && [ -n "$CLIENT_POD" ] && [ -n "$CLIENT_CONTAINER" ] && [ -n "$QUERY_NAME" ] && [[ "$START_RFC3339" =~ $TIME_RE && "$END_RFC3339" =~ $TIME_RE ]] && S="$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" && E="$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" && [ "$S" = "$START_RFC3339" ] && [ "$E" = "$END_RFC3339" ] && [ "$(date -u -d "$S" +%s)" -lt "$(date -u -d "$E" +%s)" ]; then INPUT_READY=true; else printf '%s\n' '중단: 입력 또는 strict UTC start<end gate 실패' >&2; fi
unset S E

AUTO_READY=false
if [ "$INPUT_READY" = true ] && SJ="$(kubectl get service "$SERVICE" -n "$NS" -o json)" && CJ="$(kubectl get pod "$CLIENT_POD" -n "$CLIENT_NS" -o json)"; then
  SERVICE_UID="$(jq -r '.metadata.uid//empty' <<<"$SJ")"; SERVICE_RV="$(jq -r '.metadata.resourceVersion//empty' <<<"$SJ")"; SERVICE_TYPE="$(jq -r '.spec.type//empty' <<<"$SJ")"; CLUSTER_IP="$(jq -r '.spec.clusterIP//empty' <<<"$SJ")"; IP_FAMILIES="$(jq -r '.spec.ipFamilies//[]|join(",")' <<<"$SJ")"
  CLIENT_UID="$(jq -r '.metadata.uid//empty' <<<"$CJ")"; CLIENT_RV="$(jq -r '.metadata.resourceVersion//empty' <<<"$CJ")"; CLIENT_NODE="$(jq -r '.spec.nodeName//empty' <<<"$CJ")"; CLIENT_CONTAINER_ID="$(jq -r --arg c "$CLIENT_CONTAINER" '[.status.containerStatuses[]?|select(.name==$c)][0].containerID//empty' <<<"$CJ")"
  OWNER_KIND="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].kind//"Pod"' <<<"$CJ")"; OWNER_NAME="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].name//.metadata.name' <<<"$CJ")"; OWNER_UID="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid//.metadata.uid' <<<"$CJ")"
  if NJ="$(kubectl get node "$CLIENT_NODE" -o json)" && OJ="$(kubectl get "$OWNER_KIND" "$OWNER_NAME" -n "$CLIENT_NS" -o json)"; then
    CLIENT_NODE_UID="$(jq -r '.metadata.uid//empty' <<<"$NJ")"; CLIENT_MACHINE_ID="$(jq -r '.status.nodeInfo.machineID//empty' <<<"$NJ")"; SERVICE_RE="^$(printf '%s' "$SERVICE" | sed 's/[][\\.^$*+?(){}|]/\\&/g')$"
    if [ -n "$SERVICE_UID" ] && [ -n "$SERVICE_RV" ] && [[ "$SERVICE_TYPE" =~ ^(ClusterIP|NodePort|LoadBalancer|ExternalName)$ ]] && [ "$(jq -r '.status.phase' <<<"$CJ")" = Running ] && [ "$(jq -r '.metadata.uid' <<<"$OJ")" = "$OWNER_UID" ] && [[ "$CLIENT_CONTAINER_ID" =~ ^(containerd|cri-o|docker)://[^[:space:]]+$ ]] && [ -n "$CLIENT_NODE_UID" ] && [ -n "$CLIENT_MACHINE_ID" ]; then AUTO_READY=true; printf 'AUTO_READY=true service=%s/%s rv=%s type=%s ipFamilies=%s headless=%s client=%s/%s rv=%s owner=%s/%s/%s node=%s/%s runtime=%s serviceRE=%s\n' "$SERVICE" "$SERVICE_UID" "$SERVICE_RV" "$SERVICE_TYPE" "$IP_FAMILIES" "$([ "$CLUSTER_IP" = None ] && printf true || printf false)" "$CLIENT_POD" "$CLIENT_UID" "$CLIENT_RV" "$OWNER_KIND" "$OWNER_NAME" "$OWNER_UID" "$CLIENT_NODE" "$CLIENT_NODE_UID" "${CLIENT_CONTAINER_ID%%://*}" "$SERVICE_RE"; fi
  fi
fi
unset SJ CJ NJ OJ
[ "$AUTO_READY" = true ] || printf '%s\n' '중단: Service-client-owner-node 자동 등록 실패' >&2

record_fresh() {
  FRESH_READY=false; CLIENT_READY=false; QUERY_READY=false
  local s c n o cid okind oname ouid
  [ "$AUTO_READY" = true ] || return 1
  s="$(kubectl get service "$SERVICE" -n "$NS" -o json)" || return 1; c="$(kubectl get pod "$CLIENT_POD" -n "$CLIENT_NS" -o json)" || return 1; n="$(kubectl get node "$CLIENT_NODE" -o json)" || return 1
  okind="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].kind//"Pod"' <<<"$c")"; oname="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].name//.metadata.name' <<<"$c")"; ouid="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid//.metadata.uid' <<<"$c")"; o="$(kubectl get "$okind" "$oname" -n "$CLIENT_NS" -o json)" || return 1
  cid="$(jq -r --arg x "$CLIENT_CONTAINER" '[.status.containerStatuses[]?|select(.name==$x)][0].containerID//empty' <<<"$c")"
  [ "$(jq -r '.metadata.uid' <<<"$s")" = "$SERVICE_UID" ] && [ "$(jq -r '.spec.type' <<<"$s")" = "$SERVICE_TYPE" ] && [ "$(jq -r '.spec.clusterIP//empty' <<<"$s")" = "$CLUSTER_IP" ] && [ "$(jq -r '.metadata.uid' <<<"$c")" = "$CLIENT_UID" ] && [ "$okind/$oname/$ouid" = "$OWNER_KIND/$OWNER_NAME/$OWNER_UID" ] && [ "$(jq -r '.metadata.uid' <<<"$o")" = "$OWNER_UID" ] && [ "$(jq -r '.spec.nodeName' <<<"$c")" = "$CLIENT_NODE" ] && [ "$(jq -r '.metadata.uid' <<<"$n")" = "$CLIENT_NODE_UID" ] && [ "$(jq -r '.status.nodeInfo.machineID' <<<"$n")" = "$CLIENT_MACHINE_ID" ] && [ "$cid" = "$CLIENT_CONTAINER_ID" ] || return 1
  FRESH_READY=true
}
```

`AUTO_READY=true`가 아니면 후속 명령을 실행하지 않는다. `resourceVersion`은 각 snapshot에서 current 값을 다시 기록하고 UID가 같더라도 spec 변경 전후 증거를 같은 snapshot처럼 섞지 않는다.

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | Kubernetes Service 이름이 `NXDOMAIN`, 예상과 다른 A/AAAA/CNAME 또는 오래된 주소를 반환하는 경우 |
| 적용 | current Service type과 client resolver, EndpointSlice record source를 같은 UTC 창에서 대조할 수 있는 경우 |
| 제외 | DNS Service 자체가 무응답이면 [[07. DNS/02. CoreDNS/01. CoreDNS Pod·Service가 Ready하지 않거나 질의에 응답하지 않는 경우]], endpoint가 stale이면 [[06. 클러스터 네트워크/02. Service·EndpointSlice/02. Service endpoint가 잘못되거나 오래된 Pod를 가리키는 경우]]로 이동한다. |

검색 문자열: `NXDOMAIN`, `NOERROR`, `SERVFAIL`, `clusterIP: None`, `ExternalName`, `publishNotReadyAddresses`.

## 관측별 진단 경로

| 지금 보이는 것 | 먼저 볼 것 | 그다음 어디를 볼지 | 상세 절 |
|---|---|---|---|
| NXDOMAIN | query namespace/suffix와 current Service UID/type | client search 또는 CoreDNS kubernetes plugin | [[#1. record source와 resolver|1]] |
| 일반 Service의 다른 A/AAAA | current ClusterIP/IP family와 TTL | cache·Service lifecycle | [[#2. response code와 주소 유형|2]] |
| headless 주소 불일치 | EndpointSlice target UID/condition | endpoint owner | [[#1. record source와 resolver|1]] |
| ExternalName CNAME 실패 | current externalName 존재와 response chain | external authoritative 경로 | [[#2. response code와 주소 유형|2]] |

### 1. record source와 resolver

`if record_fresh && [ "$FRESH_READY" = true ]; then kubectl get service "$SERVICE" -n "$NS" -o jsonpath='uid={.metadata.uid}{" rv="}{.metadata.resourceVersion}{" type="}{.spec.type}{" clusterIP="}{.spec.clusterIP}{" ipFamilies="}{.spec.ipFamilies}{" publishNotReady="}{.spec.publishNotReadyAddresses}{" externalNameSet="}{.spec.externalName}{" ports="}{.spec.ports[*].port}{" selector="}{.spec.selector}{"\n"}' && kubectl get endpointslice -n "$NS" -l "kubernetes.io/service-name=$SERVICE" -o custom-columns='SLICE-UID:.metadata.uid,RV:.metadata.resourceVersion,ADDRESS-TYPE:.addressType,TARGET-NAME:.endpoints[*].targetRef.name,TARGET-UID:.endpoints[*].targetRef.uid,READY:.endpoints[*].conditions.ready,SERVING:.endpoints[*].conditions.serving,TERMINATING:.endpoints[*].conditions.terminating,PORTS:.ports[*].port'; else printf '%s\n' '중단: fresh Service/client identity gate 실패' >&2; fi`

일반 Service는 current ClusterIP, headless는 EndpointSlice addresses/target UID, ExternalName은 current CNAME 계약을 적용한다. 빈 EndpointSlice 출력은 API/RBAC/selector 실패와 실제 0개를 구분한다. 주소와 ExternalName 원문은 화면에서만 대조한다.

`if record_fresh && [ "$FRESH_READY" = true ] && [ "$(kubectl auth can-i create pods/exec -n "$CLIENT_NS")" = yes ]; then RAW="$(kubectl exec "$CLIENT_POD" -n "$CLIENT_NS" -c "$CLIENT_CONTAINER" -- sh -c 'test -r /etc/resolv.conf && exec cat /etc/resolv.conf')"; RC=$?; if [ "$RC" -ne 0 ]; then printf '중단: resolver producer rc=%s\n' "$RC" >&2; elif OUT="$(grep -E '^(nameserver|search|options)[[:space:]]' <<<"$RAW")"; then CLIENT_READY=true; printf '%s\n' "$OUT"; else FRC=$?; printf '중단: resolver filter no-match rc=%s\n' "$FRC" >&2; fi; unset RAW OUT; else CLIENT_READY=false; printf '%s\n' '중단: fresh/exec RBAC gate 실패' >&2; fi`

resolver 원문은 저장하지 않는다. short name이 다른 namespace로 확장되면 Service를 만들거나 patch하지 않고 호출 원본을 교정한다.

### 2. response code와 주소 유형

`if record_fresh && [ "$FRESH_READY" = true ] && [ "$(kubectl auth can-i create pods/exec -n "$CLIENT_NS")" = yes ]; then kubectl exec "$CLIENT_POD" -n "$CLIENT_NS" -c "$CLIENT_CONTAINER" -- sh -c 'if command -v dig >/dev/null 2>&1; then exec dig +time=2 +tries=1 "$1" A; elif command -v nslookup >/dev/null 2>&1; then exec nslookup "$1"; elif command -v getent >/dev/null 2>&1; then exec getent ahosts "$1"; else exit 127; fi' sh "$QUERY_NAME"; RC=$?; printf 'query_rc=%s serviceUID=%s clientUID=%s observed=%s window=%s..%s\n' "$RC" "$SERVICE_UID" "$CLIENT_UID" "$(date -u '+%Y-%m-%dT%H:%M:%SZ')" "$START_RFC3339" "$END_RFC3339"; else printf '%s\n' '중단: fresh client query gate 실패' >&2; fi`

`dig`·`nslookup`만 NXDOMAIN/SERVFAIL/timeout을 구분한다. `getent` empty는 미분류다. TTL/cache가 확인되기 전에 Service를 재생성하지 않는다.

## Prometheus 빠른 관측

kube-state-metrics의 exact Service label이 실제 있을 때만 사용한다.

`if record_fresh && [ "$FRESH_READY" = true ]; then QUERY_READY=true; printf 'kube_service_info{namespace="%s",service="%s"}\n' "$NS" "$SERVICE"; else QUERY_READY=false; printf '%s\n' '중단: fresh PromQL gate 실패' >&2; fi`

완성 query를 입력 UTC 창에서 조회한다. 빈 결과는 객체 부재가 아니며 metric 지원, scrape, label, 권한/API와 시간 범위를 구분한다.

## 변경·정상화

이 문서는 Service·EndpointSlice·CoreDNS cache·client template 변경 명령을 제공하지 않는다. 별도 mutation lane에는 fresh Service/client/endpoint UID/current RV, selector·type·port·record source, 승인 manifest/ID/변경 시각, 트래픽/replica/quorum, 외부 호출 멱등성, 검증된 rollback 값이 필요하다. EndpointSlice와 CoreDNS를 동시에 바꾸지 않는다.

Service나 client Pod 교체 뒤 모든 자동값과 gate를 폐기한다. current owner UID·selector·revision·creationTimestamp로 단일 새 client를 다시 선택한다. 정상화는 `Service identity/type → EndpointSlice record source → CoreDNS 응답 → 같은 client resolver/query → protocol에 맞는 실제 기능` 순으로 확인하고 관찰 창 동안 재발이 없어야 한다.

## 증거 보존

UTC 창, Service/client owner/node/EndpointSlice UID와 current RV, type·IP family·port·condition, target UID, regular containerID runtime prefix, response code/exit status, 승인·rollback ID를 보존한다. ClusterIP·endpoint·domain·ExternalName·resolver 원문·Secret/token·audit body는 저장하지 않는다.
