# StatefulSet rolling update partition 또는 OnDelete 전략 때문에 일부 Pod가 이전 revision인 경우

## 입력값과 빠른 수집

실행 경로는 `작업 PC의 Bash·kubectl → API server → StatefulSet·ControllerRevision·Pod`다. 이 문서는 node·Pod shell에 접속하지 않는다. 최초 입력은 namespace와 StatefulSet 이름뿐이며 UID, resourceVersion, selector digest, strategy, revision과 Pod owner 관계는 API 자동값이다. partition 의도와 변경 승인은 운영자 판단값이다.

**도구 확인**

`TOOLS_READY=false; MISSING=''; for T in kubectl jq sha256sum cut date; do command -v "$T" >/dev/null 2>&1 || MISSING="$MISSING $T"; done; if [ -n "${BASH_VERSION:-}" ] && [ -z "$MISSING" ]; then TOOLS_READY=true; printf '%s\n' 'TOOLS_READY=true'; else printf '중단: Bash 또는 도구 없음:%s\n' "$MISSING" >&2; fi`

**최초 입력과 이전 lifecycle 폐기**

`unset INPUT_READY TARGET_READY PODS_READY CHANGE_READY NORMAL_READY STS_JSON STS_UID STS_RV STS_CREATED STS_OWNER_KIND STS_OWNER_NAME STS_OWNER_UID SELECTOR_DIGEST STRATEGY PARTITION POD_POLICY CURRENT_REV UPDATE_REV PODS_JSON CHANGE_RFC3339 APPROVAL_ID WINDOW_ID; INPUT_READY=false; TARGET_READY=false; PODS_READY=false; CHANGE_READY=false; NORMAL_READY=false; NS='PASTE_NAMESPACE_HERE'; STS='PASTE_STATEFULSET_HERE'; if [ "$TOOLS_READY" = true ] && [ -n "$NS" ] && [ -n "$STS" ]; then INPUT_READY=true; export NS STS; printf 'INPUT_READY=true NS=%s STS=%s\n' "$NS" "$STS"; else printf '%s\n' '중단: 도구 gate와 최초 입력을 확인한다.' >&2; fi`

각 `PASTE_..._HERE`는 한 번만 입력한다. 새 shell이나 재등록을 시작하면 위 명령부터 실행해 UID, resourceVersion, owner, 관계, 시간과 모든 downstream READY를 폐기한다.

**대상 identity 등록**

`unset STS_JSON STS_UID STS_RV STS_CREATED STS_OWNER_KIND STS_OWNER_NAME STS_OWNER_UID SELECTOR_DIGEST STRATEGY PARTITION POD_POLICY CURRENT_REV UPDATE_REV; TARGET_READY=false; PODS_READY=false; CHANGE_READY=false; NORMAL_READY=false; if [ "$INPUT_READY" != true ]; then printf '%s\n' '중단: INPUT_READY가 아니다.' >&2; elif DATA="$(kubectl get statefulset "$STS" -n "$NS" -o json)" && jq -e --arg n "$NS" --arg s "$STS" '.metadata.namespace==$n and .metadata.name==$s and (.metadata.uid|length>0) and (.metadata.resourceVersion|length>0)' <<<"$DATA" >/dev/null; then STS_JSON="$DATA"; STS_UID="$(jq -r '.metadata.uid' <<<"$DATA")"; STS_RV="$(jq -r '.metadata.resourceVersion' <<<"$DATA")"; STS_CREATED="$(jq -r '.metadata.creationTimestamp' <<<"$DATA")"; STS_OWNER_KIND="$(jq -r '[.metadata.ownerReferences[]? | select(.controller==true)][0].kind // "-"' <<<"$DATA")"; STS_OWNER_NAME="$(jq -r '[.metadata.ownerReferences[]? | select(.controller==true)][0].name // "-"' <<<"$DATA")"; STS_OWNER_UID="$(jq -r '[.metadata.ownerReferences[]? | select(.controller==true)][0].uid // "-"' <<<"$DATA")"; SELECTOR_DIGEST="$(jq -cS '.spec.selector' <<<"$DATA" | sha256sum | cut -d' ' -f1)"; STRATEGY="$(jq -r '.spec.updateStrategy.type' <<<"$DATA")"; PARTITION="$(jq -r '.spec.updateStrategy.rollingUpdate.partition // 0' <<<"$DATA")"; POD_POLICY="$(jq -r '.spec.podManagementPolicy' <<<"$DATA")"; CURRENT_REV="$(jq -r '.status.currentRevision // empty' <<<"$DATA")"; UPDATE_REV="$(jq -r '.status.updateRevision // empty' <<<"$DATA")"; TARGET_READY=true; export STS_UID STS_RV; printf 'TARGET_READY=true uid=%s rv=%s created=%s owner=%s/%s ownerUID=%s selectorDigest=%s strategy=%s partition=%s policy=%s current=%s update=%s\n' "$STS_UID" "$STS_RV" "$STS_CREATED" "$STS_OWNER_KIND" "$STS_OWNER_NAME" "$STS_OWNER_UID" "$SELECTOR_DIGEST" "$STRATEGY" "$PARTITION" "$POD_POLICY" "$CURRENT_REV" "$UPDATE_REV"; else printf '%s\n' '중단: exact StatefulSet 조회 실패. 과거 gate는 모두 false다.' >&2; fi`

**현재 lifecycle 재검증 함수**

`fresh_sts(){ local d uid selector strategy partition policy; d="$(kubectl get statefulset "$STS" -n "$NS" -o json)" || { TARGET_READY=false; PODS_READY=false; CHANGE_READY=false; NORMAL_READY=false; return 1; }; uid="$(jq -r '.metadata.uid // empty' <<<"$d")"; selector="$(jq -cS '.spec.selector' <<<"$d" | sha256sum | cut -d' ' -f1)"; strategy="$(jq -r '.spec.updateStrategy.type' <<<"$d")"; partition="$(jq -r '.spec.updateStrategy.rollingUpdate.partition // 0' <<<"$d")"; policy="$(jq -r '.spec.podManagementPolicy' <<<"$d")"; if [ "$TARGET_READY" != true ] || [ -z "$uid" ] || [ "$uid" != "$STS_UID" ] || [ "$selector" != "$SELECTOR_DIGEST" ] || [ "$strategy" != "$STRATEGY" ] || [ "$partition" != "$PARTITION" ] || [ "$policy" != "$POD_POLICY" ]; then unset STS_JSON STS_RV PODS_JSON; TARGET_READY=false; PODS_READY=false; CHANGE_READY=false; NORMAL_READY=false; printf '%s\n' '중단: StatefulSet lifecycle 또는 update spec이 바뀌었다. 최초 입력부터 재등록한다.' >&2; return 1; fi; STS_JSON="$d"; STS_RV="$(jq -r '.metadata.resourceVersion' <<<"$d")"; return 0; }`

**Pod·revision 관계 등록**

`unset PODS_JSON; PODS_READY=false; CHANGE_READY=false; NORMAL_READY=false; if ! fresh_sts; then :; elif ALL="$(kubectl get pods -n "$NS" -o json)" && REVS="$(kubectl get controllerrevisions.apps -n "$NS" -o json)" && PODS="$(jq -c --arg uid "$STS_UID" '[.items[] | select(any(.metadata.ownerReferences[]?; .controller==true and .kind=="StatefulSet" and .uid==$uid))] | sort_by(.metadata.name)' <<<"$ALL")" && jq -e 'length>0' <<<"$PODS" >/dev/null && jq -e --arg uid "$STS_UID" --arg c "$CURRENT_REV" --arg u "$UPDATE_REV" 'all(.items[] | select(.metadata.name==$c or .metadata.name==$u); any(.metadata.ownerReferences[]?; .controller==true and .kind=="StatefulSet" and .uid==$uid))' <<<"$REVS" >/dev/null; then PODS_JSON="$PODS"; PODS_READY=true; jq -r '.[] | "=== Pod: \(.metadata.name) ===\nUID: \(.metadata.uid)\n생성: \(.metadata.creationTimestamp)\nowner: \([.metadata.ownerReferences[]? | select(.controller==true) | "\(.kind)/\(.name) uid=\(.uid)"] | join(","))\nrevision: \(.metadata.labels["controller-revision-hash"] // "-")\nphase/ready/deletion: \(.status.phase // "-")/\([.status.containerStatuses[]?.ready] | all)/\(.metadata.deletionTimestamp // "-")"' <<<"$PODS_JSON"; printf 'PODS_READY=true count=%s stsRV=%s\n' "$(jq 'length' <<<"$PODS_JSON")" "$STS_RV"; else printf '%s\n' '중단: Pod 또는 ControllerRevision 조회·owner 검증 실패. 빈 결과는 정상 아님.' >&2; fi`

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | `RollingUpdate` partition 아래 ordinal만 이전 revision이거나 `OnDelete`에서 기존 Pod가 이전 revision인 경우 |
| 적용 | 현재 StatefulSet UID와 ControllerRevision owner UID, 각 Pod UID·owner UID·revision·creation time을 연결할 수 있는 경우 |
| 제외 | owner/hash 모순은 [[02. StatefulSet currentRevision과 updateRevision이 수렴하지 않는 경우]], PVC·quorum 영향은 [[05. StatefulSet quorum workload의 제한적 복구와 정상화 판단]]으로 이관한다. |

검색 문자열: `RollingUpdate`, `partition`, `OnDelete`, `OrderedReady`, `currentRevision`, `updateRevision`, `controller-revision-hash`, `deletionTimestamp`.

## 관측별 진단 경로

| 지금 보이는 것 | 먼저 볼 것 | 그다음 어디를 볼지 | 상세 절 |
|---|---|---|---|
| partition 미만 ordinal만 current revision | 실제 partition과 승인된 canary 범위 | 계획 상태로 기록하되 Ready·quorum 별도 검증 | [[#현재 분포 판정\|분포 판정]] |
| partition 이상에도 current revision | generation/observedGeneration, 낮은 ordinal Ready | controller 지연 또는 blocking Pod 이관 | [[#현재 분포 판정\|분포 판정]] |
| `OnDelete` 기존 Pod만 current revision | strategy와 Pod creation time | 자동 교체를 기대하지 말고 승인된 교체 계획 확인 | [[#상태 변경 경계\|변경 경계]] |
| revision은 맞지만 NotReady·deleting | 해당 Pod UID의 condition·Event | Pod 전용 런북으로 이동 | [[#현재 분포 판정\|Pod 증거]] |

**현재 분포 판정**

`if [ "$PODS_READY" != true ] || ! fresh_sts; then printf '%s\n' '중단: 최신 PODS_READY와 TARGET_READY가 필요하다.' >&2; elif ALL="$(kubectl get pods -n "$NS" -o json)" && NOW_PODS="$(jq -c --arg uid "$STS_UID" '[.items[] | select(any(.metadata.ownerReferences[]?; .controller==true and .kind=="StatefulSet" and .uid==$uid))] | sort_by(.metadata.name)' <<<"$ALL")" && [ "$(jq -c '[.[]|[.metadata.name,.metadata.uid]]' <<<"$NOW_PODS")" = "$(jq -c '[.[]|[.metadata.name,.metadata.uid]]' <<<"$PODS_JSON")" ]; then jq -r --arg c "$(jq -r '.status.currentRevision // empty' <<<"$STS_JSON")" --arg u "$(jq -r '.status.updateRevision // empty' <<<"$STS_JSON")" '.[] | (.metadata.name|capture("-(?<ordinal>[0-9]+)$").ordinal? // "unknown") as $o | "pod=\(.metadata.name) uid=\(.metadata.uid) ordinal=\($o) revision=\(.metadata.labels["controller-revision-hash"] // "-") class=" + (if .metadata.labels["controller-revision-hash"]==$u then "update" elif .metadata.labels["controller-revision-hash"]==$c then "current" else "unknown" end)' <<<"$NOW_PODS"; else PODS_READY=false; CHANGE_READY=false; NORMAL_READY=false; printf '%s\n' '중단: Pod UID 집합이 바뀌었다. Pod 관계를 재등록한다.' >&2; fi`

`RollingUpdate`에서 partition은 업데이트 가능한 최소 ordinal이다. `partition=0`이면 이전 revision 잔존을 설명하지 못한다. `OrderedReady`에서는 낮은 ordinal의 NotReady·deletion이 상위 ordinal 진행을 막을 수 있다. `OnDelete`는 template 변경만으로 기존 Pod를 교체하지 않는다. 어떤 경우에도 `updatedReplicas` 하나만으로 ordinal mapping을 증명하지 않는다.

**대상 Pod condition과 UID 제한 Event**

`POD='PASTE_TARGET_POD_HERE'; POD_READY=false; if [ "$PODS_READY" != true ] || ! fresh_sts; then printf '%s\n' '중단: 최신 gate가 없다.' >&2; elif P="$(kubectl get pod "$POD" -n "$NS" -o json)" && jq -e --arg uid "$STS_UID" 'any(.metadata.ownerReferences[]?; .controller==true and .kind=="StatefulSet" and .uid==$uid)' <<<"$P" >/dev/null; then POD_UID="$(jq -r '.metadata.uid' <<<"$P")"; POD_RV="$(jq -r '.metadata.resourceVersion' <<<"$P")"; POD_READY=true; jq -r '"uid=\(.metadata.uid) rv=\(.metadata.resourceVersion) created=\(.metadata.creationTimestamp) node=\(.spec.nodeName // "-") revision=\(.metadata.labels["controller-revision-hash"] // "-") deletion=\(.metadata.deletionTimestamp // "-")", (.status.conditions[]? | "condition=\(.type) status=\(.status) reason=\(.reason // "-") transition=\(.lastTransitionTime // "-")")' <<<"$P"; else printf '%s\n' '중단: Pod exact owner 조회 실패.' >&2; fi`

`if [ "$POD_READY" != true ] || ! fresh_sts; then printf '%s\n' '중단: 최신 POD_READY가 필요하다.' >&2; elif NOW="$(kubectl get pod "$POD" -n "$NS" -o json)" && jq -e --arg uid "$POD_UID" '.metadata.uid==$uid' <<<"$NOW" >/dev/null; then if ! kubectl get events -n "$NS" --field-selector "involvedObject.kind=Pod,involvedObject.name=$POD,involvedObject.uid=$POD_UID" --sort-by=.metadata.creationTimestamp -o custom-columns='CREATED:.metadata.creationTimestamp,EVENT_TIME:.eventTime,SERIES_LAST:.series.lastObservedTime,TYPE:.type,REASON:.reason,COUNT:.count,MESSAGE:.message'; then printf '%s\n' '중단: Event producer 실패. 빈 결과와 다르다.' >&2; fi; else POD_READY=false; printf '%s\n' '중단: Pod가 교체됐거나 조회 실패.' >&2; fi`

필터된 Event가 비면 정상으로 보지 않는다. Event TTL·권한·집계 여부를 확인한다. `Terminating`은 [[15. Pod가 Terminating에서 멈춘 경우]], `Running/NotReady`는 [[16. Pod가 Running이지만 Ready가 아닌 경우]], mount 오류는 [[25. Pod 이벤트에 FailedMount·FailedAttachVolume이 기록된 경우]]로 이동한다.

## 상태 변경 경계

이 문서는 Pod 삭제, partition patch, strategy 전환, scale 또는 rollout 명령을 제공하지 않는다. 별도 절차는 현재 StatefulSet UID·resourceVersion·selector digest·strategy·partition·revision, 대상 Pod UID·owner UID·PVC, 승인 ID·변경 창, quorum·availability·중단·rollback 절차를 실행 직전에 다시 검증해야 한다. 이 항목 중 하나라도 바뀌면 승인을 폐기한다.

승인된 외부 변경 뒤에는 기존 Pod·revision 선택을 폐기한다. 아래 명령은 현재 StatefulSet UID·spec을 다시 확인하고, 변경 시각 뒤 current owner UID의 단일 새 Pod만 replacement로 등록한다. 이름만 같은 Pod를 연결하지 않는다.

**replacement 단일 후보 등록**

`CHANGE_RFC3339='PASTE_CHANGE_RFC3339_HERE'; APPROVAL_ID='PASTE_APPROVAL_ID_HERE'; WINDOW_ID='PASTE_WINDOW_ID_HERE'; PODS_READY=false; POD_READY=false; CHANGE_READY=false; NORMAL_READY=false; unset PODS_JSON POD POD_UID POD_RV REPLACEMENT_NAME REPLACEMENT_UID REPLACEMENT_CREATED; if ! fresh_sts; then printf '%s\n' '중단: 최신 StatefulSet target gate가 없다.' >&2; elif ! [[ "$CHANGE_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] || [ "$(date -u -d "$CHANGE_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" != "$CHANGE_RFC3339" ] || [ -z "$APPROVAL_ID" ] || [ -z "$WINDOW_ID" ]; then printf '%s\n' '중단: canonical UTC 시각·승인·변경 창이 필요하다.' >&2; elif C="$(kubectl get pods -n "$NS" -o json | jq -c --arg uid "$STS_UID" --arg t "$CHANGE_RFC3339" '[.items[] | select(any(.metadata.ownerReferences[]?; .controller==true and .kind=="StatefulSet" and .uid==$uid)) | select(.metadata.creationTimestamp > $t)]')" && [ "$(jq 'length' <<<"$C")" -eq 1 ]; then REPLACEMENT_NAME="$(jq -r '.[0].metadata.name' <<<"$C")"; REPLACEMENT_UID="$(jq -r '.[0].metadata.uid' <<<"$C")"; REPLACEMENT_CREATED="$(jq -r '.[0].metadata.creationTimestamp' <<<"$C")"; CHANGE_READY=true; printf 'CHANGE_READY=true replacement=%s uid=%s created=%s approval=%s window=%s\n' "$REPLACEMENT_NAME" "$REPLACEMENT_UID" "$REPLACEMENT_CREATED" "$APPROVAL_ID" "$WINDOW_ID"; else printf '%s\n' '중단: 변경 뒤 replacement 후보가 정확히 하나가 아니다.' >&2; fi`

## 정상화와 증거

`NORMAL_READY=false; if [ "$CHANGE_READY" != true ] || ! fresh_sts; then printf '%s\n' '중단: 최신 CHANGE_READY가 필요하다.' >&2; elif P="$(kubectl get pod "$REPLACEMENT_NAME" -n "$NS" -o json)" && jq -e --arg p "$REPLACEMENT_UID" --arg s "$STS_UID" '.metadata.uid==$p and any(.metadata.ownerReferences[]?; .controller==true and .kind=="StatefulSet" and .uid==$s) and .metadata.deletionTimestamp==null' <<<"$P" >/dev/null; then jq -r '"replacement uid=\(.metadata.uid) revision=\(.metadata.labels["controller-revision-hash"] // "-") phase=\(.status.phase) ready=\([.status.containerStatuses[]?.ready] | all) restarts=\([.status.containerStatuses[]?.restartCount] | add // 0)"' <<<"$P"; NORMAL_READY=true; else printf '%s\n' '중단: replacement identity가 바뀌었다.' >&2; fi`

`NORMAL_READY=true`는 identity 재연결만 뜻한다. 승인된 partition/OnDelete 의도, desired·Ready·updated, ordinal별 revision, PVC identity, 새 Warning·restart 증가 중단, EndpointSlice와 실제 요청, quorum·leader·replication·데이터 기능을 관찰 창 동안 확인해야 정상화한다.

**PromQL 완성본 생성**

`if [ "$TARGET_READY" != true ] || ! fresh_sts; then printf '%s\n' '중단: 최신 TARGET_READY가 필요하다.' >&2; else printf 'kube_statefulset_replicas{namespace=%s,statefulset=%s}-kube_statefulset_status_replicas_updated{namespace=%s,statefulset=%s}\n' "$(jq -Rn --arg v "$NS" '$v')" "$(jq -Rn --arg v "$STS" '$v')" "$(jq -Rn --arg v "$NS" '$v')" "$(jq -Rn --arg v "$STS" '$v')"; fi`

metric·label 지원 여부와 현재 cluster scrape를 확인한다. 빈 결과는 정상이나 0이 아니다. PromQL은 UID, partition 경계와 Pod owner를 증명하지 못한다.

보존 대상은 StatefulSet UID·resourceVersion·selector digest·strategy·partition·revision, ControllerRevision UID·owner UID, Pod old/new UID·owner·creation/revision/Ready, Event의 채워진 시각, 승인·창·제품 검증이다. Secret·token·개인키·전체 YAML은 보존하지 않는다.
