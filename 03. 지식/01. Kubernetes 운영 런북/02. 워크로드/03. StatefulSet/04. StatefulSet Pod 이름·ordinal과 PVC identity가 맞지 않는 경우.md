# StatefulSet Pod 이름·ordinal과 PVC identity가 맞지 않는 경우

## 입력값과 빠른 수집

연결 경로는 `작업 PC의 Bash·kubectl → API server → StatefulSet → Pod UID·ordinal → claimName → PVC UID → PV UID·claimRef → 조건부 VolumeAttachment UID·node`다. 최초 입력은 namespace와 StatefulSet 이름, 운영자 선택은 문제 Pod와 조건부 VolumeAttachment 이름이다. 이름 규칙은 후보일 뿐 UID 관계를 대신하지 않는다.

**도구 확인**

`TOOLS_READY=false; MISSING=''; for T in kubectl jq date sha256sum cut; do command -v "$T" >/dev/null 2>&1 || MISSING="$MISSING $T"; done; if [ -n "${BASH_VERSION:-}" ] && [ -z "$MISSING" ]; then TOOLS_READY=true; printf '%s\n' 'TOOLS_READY=true'; else printf '중단: Bash 또는 도구 없음:%s\n' "$MISSING" >&2; fi`

**최초 입력·stale 상태 폐기**

`unset INPUT_READY TARGET_READY POD_READY STORAGE_READY ATTACH_READY CHANGE_READY REPLACEMENT_READY NORMAL_READY STS_JSON POD_JSON PVC_JSON PV_JSON VA_JSON STS_UID STS_RV STS_CREATED SELECTOR_DIGEST CLAIM_TEMPLATES POD POD_UID POD_RV POD_CREATED POD_NODE POD_OWNER_UID POD_REVISION PVC PVC_UID PVC_RV PV PV_UID PV_RV CSI_DRIVER VA VA_UID CHANGE_RFC3339 APPROVAL_ID; INPUT_READY=false; TARGET_READY=false; POD_READY=false; STORAGE_READY=false; ATTACH_READY=false; CHANGE_READY=false; REPLACEMENT_READY=false; NORMAL_READY=false; NS='PASTE_NAMESPACE_HERE'; STS='PASTE_STATEFULSET_HERE'; if [ "$TOOLS_READY" = true ] && [ -n "$NS" ] && [ -n "$STS" ]; then INPUT_READY=true; export NS STS; printf 'INPUT_READY=true NS=%s STS=%s\n' "$NS" "$STS"; else printf '%s\n' '중단: 최초 입력 실패.' >&2; fi`

**StatefulSet API 자동값 등록**

`TARGET_READY=false; POD_READY=false; STORAGE_READY=false; ATTACH_READY=false; CHANGE_READY=false; NORMAL_READY=false; unset STS_JSON STS_UID STS_RV STS_CREATED SELECTOR_DIGEST CLAIM_TEMPLATES; if [ "$INPUT_READY" != true ]; then printf '%s\n' '중단: INPUT_READY가 아니다.' >&2; elif D="$(kubectl get statefulset "$STS" -n "$NS" -o json)" && jq -e --arg n "$NS" --arg s "$STS" '.metadata.namespace==$n and .metadata.name==$s and (.metadata.uid|length>0)' <<<"$D" >/dev/null; then STS_JSON="$D"; STS_UID="$(jq -r '.metadata.uid' <<<"$D")"; STS_RV="$(jq -r '.metadata.resourceVersion' <<<"$D")"; STS_CREATED="$(jq -r '.metadata.creationTimestamp' <<<"$D")"; SELECTOR_DIGEST="$(jq -cS '.spec.selector' <<<"$D" | sha256sum | cut -d' ' -f1)"; CLAIM_TEMPLATES="$(jq -r '[.spec.volumeClaimTemplates[].metadata.name] | join(",")' <<<"$D")"; TARGET_READY=true; printf 'TARGET_READY=true uid=%s rv=%s created=%s selectorDigest=%s claimTemplates=%s replicas=%s current=%s update=%s\n' "$STS_UID" "$STS_RV" "$STS_CREATED" "$SELECTOR_DIGEST" "$CLAIM_TEMPLATES" "$(jq -r '.spec.replicas' <<<"$D")" "$(jq -r '.status.currentRevision // "-"' <<<"$D")" "$(jq -r '.status.updateRevision // "-"' <<<"$D")"; else printf '%s\n' '중단: StatefulSet exact 조회 실패.' >&2; fi`

**현재 lifecycle 재검증 함수**

`fresh_sts(){ local d u selector templates; d="$(kubectl get statefulset "$STS" -n "$NS" -o json)" || { TARGET_READY=false; POD_READY=false; STORAGE_READY=false; ATTACH_READY=false; CHANGE_READY=false; NORMAL_READY=false; return 1; }; u="$(jq -r '.metadata.uid // empty' <<<"$d")"; selector="$(jq -cS '.spec.selector' <<<"$d" | sha256sum | cut -d' ' -f1)"; templates="$(jq -r '[.spec.volumeClaimTemplates[].metadata.name] | join(",")' <<<"$d")"; if [ "$TARGET_READY" != true ] || [ "$u" != "$STS_UID" ] || [ "$selector" != "$SELECTOR_DIGEST" ] || [ "$templates" != "$CLAIM_TEMPLATES" ]; then unset STS_JSON POD_JSON PVC_JSON PV_JSON VA_JSON; TARGET_READY=false; POD_READY=false; STORAGE_READY=false; ATTACH_READY=false; CHANGE_READY=false; NORMAL_READY=false; printf '%s\n' '중단: StatefulSet lifecycle 또는 storage spec 변경. 처음부터 재등록한다.' >&2; return 1; fi; STS_JSON="$d"; STS_RV="$(jq -r '.metadata.resourceVersion' <<<"$d")"; return 0; }`

**현재 owner Pod 후보 출력**

`if ! fresh_sts; then :; elif P="$(kubectl get pods -n "$NS" -o json)"; then jq -r --arg uid "$STS_UID" '.items[] | select(any(.metadata.ownerReferences[]?; .controller==true and .kind=="StatefulSet" and .uid==$uid)) | "=== Pod: \(.metadata.name) ===\nUID: \(.metadata.uid) rv=\(.metadata.resourceVersion) created=\(.metadata.creationTimestamp) node=\(.spec.nodeName // "-")\nrevision=\(.metadata.labels["controller-revision-hash"] // "-") phase=\(.status.phase // "-") deletion=\(.metadata.deletionTimestamp // "-")\nclaims=" + ([.spec.volumes[]? | select(.persistentVolumeClaim!=null) | "\(.name)->\(.persistentVolumeClaim.claimName)"] | join(","))' <<<"$P"; else printf '%s\n' '중단: Pod API 조회 실패.' >&2; fi`

**문제 Pod 선택·identity 등록**

`POD='PASTE_TARGET_POD_HERE'; POD_READY=false; STORAGE_READY=false; ATTACH_READY=false; CHANGE_READY=false; NORMAL_READY=false; unset POD_JSON POD_UID POD_RV POD_CREATED POD_NODE POD_OWNER_UID POD_REVISION PVC; if ! fresh_sts; then :; elif D="$(kubectl get pod "$POD" -n "$NS" -o json)" && jq -e --arg uid "$STS_UID" 'any(.metadata.ownerReferences[]?; .controller==true and .kind=="StatefulSet" and .uid==$uid)' <<<"$D" >/dev/null; then POD_JSON="$D"; POD_UID="$(jq -r '.metadata.uid' <<<"$D")"; POD_RV="$(jq -r '.metadata.resourceVersion' <<<"$D")"; POD_CREATED="$(jq -r '.metadata.creationTimestamp' <<<"$D")"; POD_NODE="$(jq -r '.spec.nodeName // empty' <<<"$D")"; POD_OWNER_UID="$STS_UID"; POD_REVISION="$(jq -r '.metadata.labels["controller-revision-hash"] // empty' <<<"$D")"; CLAIM_COUNT="$(jq '[.spec.volumes[]? | select(.persistentVolumeClaim!=null)] | length' <<<"$D")"; if [ "$CLAIM_COUNT" -eq 1 ]; then PVC="$(jq -r '.spec.volumes[] | select(.persistentVolumeClaim!=null) | .persistentVolumeClaim.claimName' <<<"$D")"; POD_READY=true; printf 'POD_READY=true pod=%s uid=%s rv=%s created=%s node=%s ownerUID=%s revision=%s pvc=%s\n' "$POD" "$POD_UID" "$POD_RV" "$POD_CREATED" "$POD_NODE" "$POD_OWNER_UID" "$POD_REVISION" "$PVC"; else printf '중단: PVC claim 후보가 정확히 하나가 아님 count=%s\n' "$CLAIM_COUNT" >&2; fi; else printf '%s\n' '중단: Pod 조회 또는 owner UID 검증 실패.' >&2; fi`

PVC가 여러 개인 workload는 mount path와 claim template 관계를 확인한 뒤 이 자동 경로를 사용하지 말고 storage owner에게 정확한 claim을 요청한다. 첫 번째 항목을 임의 선택하지 않는다.

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | Pod ordinal과 claimName, PVC·PV claimRef 또는 VolumeAttachment relation이 맞지 않는 경우 |
| 적용 | 현재 StatefulSet·Pod·PVC·PV UID와 resourceVersion을 연결할 수 있는 경우 |
| 제외 | identity가 맞고 attach·mount만 실패하면 [[25. Pod 이벤트에 FailedMount·FailedAttachVolume이 기록된 경우]]로 이동한다. |

검색 문자열: `volumeClaimTemplates`, `claimName`, `claimRef`, `Bound`, `VolumeAttachment`, `Multi-Attach`, `controller-revision-hash`.

## PVC·PV·CSI 관계 등록

**PVC → PV 자동 등록**

`STORAGE_READY=false; ATTACH_READY=false; CHANGE_READY=false; NORMAL_READY=false; unset PVC_JSON PVC_UID PVC_RV PV PV_JSON PV_UID PV_RV CSI_DRIVER; if [ "$POD_READY" != true ] || ! fresh_sts; then printf '%s\n' '중단: 최신 POD_READY가 필요하다.' >&2; elif PNOW="$(kubectl get pod "$POD" -n "$NS" -o json)" && jq -e --arg u "$POD_UID" '.metadata.uid==$u' <<<"$PNOW" >/dev/null && C="$(kubectl get pvc "$PVC" -n "$NS" -o json)" && [ "$(jq -r '.status.phase' <<<"$C")" = Bound ] && PV_NAME="$(jq -r '.spec.volumeName // empty' <<<"$C")" && [ -n "$PV_NAME" ] && V="$(kubectl get pv "$PV_NAME" -o json)" && jq -e --arg n "$NS" --arg c "$PVC" --arg u "$(jq -r '.metadata.uid' <<<"$C")" '.spec.claimRef.namespace==$n and .spec.claimRef.name==$c and .spec.claimRef.uid==$u' <<<"$V" >/dev/null; then PVC_JSON="$C"; PVC_UID="$(jq -r '.metadata.uid' <<<"$C")"; PVC_RV="$(jq -r '.metadata.resourceVersion' <<<"$C")"; PV="$PV_NAME"; PV_JSON="$V"; PV_UID="$(jq -r '.metadata.uid' <<<"$V")"; PV_RV="$(jq -r '.metadata.resourceVersion' <<<"$V")"; CSI_DRIVER="$(jq -r '.spec.csi.driver // empty' <<<"$V")"; STORAGE_READY=true; printf 'STORAGE_READY=true pod=%s/%s pvc=%s uid=%s rv=%s pv=%s uid=%s rv=%s driver=%s node=%s\n' "$POD" "$POD_UID" "$PVC" "$PVC_UID" "$PVC_RV" "$PV" "$PV_UID" "$PV_RV" "${CSI_DRIVER:--}" "${POD_NODE:--}"; else printf '%s\n' '중단: Pod lifecycle, PVC Bound, PV 또는 claimRef exact UID 관계 검증 실패.' >&2; fi`

**현재 Pod→PVC→PV relation 재검증 함수**

`fresh_storage(){ local p c v; fresh_sts || return 1; p="$(kubectl get pod "$POD" -n "$NS" -o json)" || return 1; c="$(kubectl get pvc "$PVC" -n "$NS" -o json)" || return 1; v="$(kubectl get pv "$PV" -o json)" || return 1; if [ "$STORAGE_READY" != true ] || ! jq -e --arg p "$POD_UID" --arg s "$STS_UID" --arg c "$PVC" '.metadata.uid==$p and any(.metadata.ownerReferences[]?; .controller==true and .uid==$s) and any(.spec.volumes[]?; .persistentVolumeClaim.claimName==$c)' <<<"$p" >/dev/null || ! jq -e --arg u "$PVC_UID" --arg p "$PV" '.metadata.uid==$u and .status.phase=="Bound" and .spec.volumeName==$p' <<<"$c" >/dev/null || ! jq -e --arg u "$PV_UID" --arg n "$NS" --arg c "$PVC" --arg cu "$PVC_UID" '.metadata.uid==$u and .spec.claimRef.namespace==$n and .spec.claimRef.name==$c and .spec.claimRef.uid==$cu' <<<"$v" >/dev/null; then POD_READY=false; STORAGE_READY=false; ATTACH_READY=false; CHANGE_READY=false; NORMAL_READY=false; printf '%s\n' '중단: Pod/PVC/PV identity relation이 바뀌었다. storage 관계를 재등록한다.' >&2; return 1; fi; POD_JSON="$p"; PVC_JSON="$c"; PV_JSON="$v"; return 0; }`

PVC/PV 이름·Bound만으로 데이터 identity, mount 또는 attach 성공을 보장하지 않는다. StatefulSet 기본 claim 후보는 `claim-template-statefulset-ordinal`이지만 실제 Pod claimName과 UID 관계를 먼저 사용한다. `spec.ordinals.start`는 Kubernetes 1.26의 `StatefulSetStartOrdinal` alpha gate와 양쪽 control-plane component 적용이 확인된 경우에만 기대 ordinal 계산에 사용한다.

**VolumeAttachment 후보 출력**

`if ! fresh_storage; then printf '%s\n' '중단: 최신 STORAGE_READY가 필요하다.' >&2; elif [ -z "$CSI_DRIVER" ]; then printf '%s\n' '적용 안 함: CSI PV가 아니다.'; elif A="$(kubectl get volumeattachments.storage.k8s.io -o json)"; then jq -r --arg pv "$PV" '.items[] | select(.spec.source.persistentVolumeName==$pv) | "name=\(.metadata.name) uid=\(.metadata.uid) rv=\(.metadata.resourceVersion) pv=\(.spec.source.persistentVolumeName) node=\(.spec.nodeName) attacher=\(.spec.attacher) attached=\(.status.attached)"' <<<"$A"; else printf '%s\n' '중단: VolumeAttachment list 권한 또는 API 조회 실패.' >&2; fi`

빈 결과는 attach 불필요, 객체 없음, 권한·API 실패를 구분하지 못한다. exact 후보를 storage/CSI owner와 확인한다.

**VolumeAttachment 선택·검증**

`VA='PASTE_VOLUMEATTACHMENT_HERE'; ATTACH_READY=false; if ! fresh_storage; then printf '%s\n' '중단: 최신 STORAGE_READY가 필요하다.' >&2; elif [ -z "$CSI_DRIVER" ]; then printf '%s\n' '중단: 이 PV에는 VolumeAttachment 경로를 적용하지 않는다.' >&2; elif D="$(kubectl get volumeattachment "$VA" -o json)" && jq -e --arg pv "$PV" --arg node "$POD_NODE" '.spec.source.persistentVolumeName==$pv and .spec.nodeName==$node and (.metadata.uid|length>0)' <<<"$D" >/dev/null; then VA_JSON="$D"; VA_UID="$(jq -r '.metadata.uid' <<<"$D")"; VA_RV="$(jq -r '.metadata.resourceVersion' <<<"$D")"; ATTACH_READY=true; printf 'ATTACH_READY=true va=%s uid=%s rv=%s pv=%s node=%s attached=%s\n' "$VA" "$VA_UID" "$VA_RV" "$PV" "$POD_NODE" "$(jq -r '.status.attached' <<<"$D")"; else printf '%s\n' '중단: VolumeAttachment exact PV-node 관계 검증 실패.' >&2; fi`

## 관측별 진단 경로

| 지금 보이는 것 | 먼저 볼 것 | 그다음 어디를 볼지 | 상세 절 |
|---|---|---|---|
| Pod claimName이 template·ordinal 후보와 다름 | Pod UID·owner UID·revision | workload owner의 manifest·mount 관계 | [[#PVC·PV·CSI 관계 등록\|storage 관계]] |
| PVC UID는 맞지만 PV claimRef가 다름 | PVC UID와 PV claimRef namespace/name | 데이터 보존 상태로 storage owner 이관, patch·rebind 금지 | [[#PVC·PV·CSI 관계 등록\|storage 관계]] |
| VolumeAttachment PV·node가 다름 | attachment UID·PV·node | old/new Pod UID·node와 CSI owner 이관, 삭제 금지 | [[#PVC·PV·CSI 관계 등록\|attachment 관계]] |
| 모든 identity가 맞지만 mount Event 발생 | 같은 Pod UID의 Event 시각 | [[25. Pod 이벤트에 FailedMount·FailedAttachVolume이 기록된 경우]] | [[#상태 변경 경계\|변경 경계]] |

## 상태 변경 경계

PVC/PV 삭제, claimRef patch, rebind, 강제 detach, VolumeAttachment 삭제, Pod 강제 삭제, scale·rollout·filesystem 조작은 이 문서의 복구 명령이 아니다. 별도 승인 절차는 최신 UID·resourceVersion·claim mapping·PV reclaim·CSI attachment·consumer·quorum·backup, 승인 ID와 rollback을 실행 직전에 다시 검사해야 한다.

**외부 변경 전 승인 gate**

`CHANGE_RFC3339='PASTE_CHANGE_RFC3339_HERE'; APPROVAL_ID='PASTE_APPROVAL_ID_HERE'; CHANGE_READY=false; REPLACEMENT_READY=false; NORMAL_READY=false; if ! fresh_storage; then printf '%s\n' '중단: 최신 storage identity가 필요하다.' >&2; elif ! [[ "$CHANGE_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] || [ "$(date -u -d "$CHANGE_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" != "$CHANGE_RFC3339" ] || [ -z "$APPROVAL_ID" ]; then printf '%s\n' '중단: canonical UTC와 승인 ID가 필요하다.' >&2; else CHANGE_READY=true; printf 'CHANGE_READY=true pod=%s oldUID=%s pvc=%s/%s pv=%s/%s approval=%s change=%s\n' "$POD" "$POD_UID" "$PVC" "$PVC_UID" "$PV" "$PV_UID" "$APPROVAL_ID" "$CHANGE_RFC3339"; fi`

외부 승인 절차를 실행한 뒤에는 기존 `POD_READY`, `STORAGE_READY`, `ATTACH_READY`를 재사용하지 않는다. 아래 명령은 StatefulSet lifecycle만 다시 확인하고 변경 시각 뒤 같은 ordinal의 current owner Pod가 정확히 하나인지 등록한다.

**replacement 단일 후보 등록**

`REPLACEMENT_READY=false; NORMAL_READY=false; if [ "$CHANGE_READY" != true ] || ! fresh_sts; then printf '%s\n' '중단: 최신 CHANGE_READY와 StatefulSet identity가 필요하다.' >&2; elif C="$(kubectl get pods -n "$NS" -o json | jq -c --arg uid "$STS_UID" --arg t "$CHANGE_RFC3339" --arg name "$POD" '[.items[] | select(.metadata.name==$name and .metadata.creationTimestamp>$t and any(.metadata.ownerReferences[]?; .controller==true and .kind=="StatefulSet" and .uid==$uid))]')" && [ "$(jq 'length' <<<"$C")" -eq 1 ]; then REPLACEMENT_UID="$(jq -r '.[0].metadata.uid' <<<"$C")"; REPLACEMENT_CREATED="$(jq -r '.[0].metadata.creationTimestamp' <<<"$C")"; REPLACEMENT_READY=true; printf 'REPLACEMENT_READY=true pod=%s uid=%s created=%s approval=%s\n' "$POD" "$REPLACEMENT_UID" "$REPLACEMENT_CREATED" "$APPROVAL_ID"; else printf '%s\n' '중단: 동일 ordinal replacement가 단일 후보가 아니다.' >&2; fi`

변경 전에 사용한 Pod/PVC/PV/VolumeAttachment READY는 replacement 검증에 재사용하지 않는다. 최초 입력부터 자동값을 다시 수집한다.

## 정상화와 증거

`NORMAL_READY=false; if [ "$REPLACEMENT_READY" != true ] || ! fresh_sts; then printf '%s\n' '중단: 최신 REPLACEMENT_READY가 필요하다.' >&2; elif D="$(kubectl get pod "$POD" -n "$NS" -o json)" && jq -e --arg p "$REPLACEMENT_UID" --arg s "$STS_UID" '.metadata.uid==$p and any(.metadata.ownerReferences[]?; .controller==true and .uid==$s)' <<<"$D" >/dev/null; then jq -r '"pod=\(.metadata.name) uid=\(.metadata.uid) created=\(.metadata.creationTimestamp) node=\(.spec.nodeName // "-") revision=\(.metadata.labels["controller-revision-hash"] // "-") claims=" + ([.spec.volumes[]? | select(.persistentVolumeClaim!=null) | .persistentVolumeClaim.claimName] | join(","))' <<<"$D"; NORMAL_READY=true; else REPLACEMENT_READY=false; printf '%s\n' '중단: replacement identity가 바뀌었다.' >&2; fi`

정상화는 새 Pod UID·StatefulSet owner UID·revision·claimName, 새로 등록한 PVC UID/Bound·PV UID/claimRef, 적용되는 VolumeAttachment UID·PV·node·attached, Warning 중단, quorum·데이터 일관성과 실제 read/write를 모두 확인한다. Pod Running이나 PVC Bound만으로 정상화하지 않는다.

**PromQL 완성본 생성**

`if ! fresh_storage; then printf '%s\n' '중단: 최신 storage identity가 필요하다.' >&2; else printf 'kube_persistentvolumeclaim_status_phase{namespace=%s,persistentvolumeclaim=%s,phase="Bound"}==1\n' "$(jq -Rn --arg v "$NS" '$v')" "$(jq -Rn --arg v "$PVC" '$v')"; fi`

metric 미지원·빈 결과·label 불일치를 정상으로 해석하지 않는다. 이 쿼리는 PVC UID, PV claimRef, attach·mount·데이터 identity를 증명하지 못한다.

보존 대상은 StatefulSet·Pod·PVC·PV·VolumeAttachment의 이름·UID·resourceVersion·owner·creation time과 relation, CSI driver·node, Event 시각, 승인 ID, 데이터 검증 결과다. Secret·token·개인키·전체 YAML·데이터 원문은 보존하지 않는다.
