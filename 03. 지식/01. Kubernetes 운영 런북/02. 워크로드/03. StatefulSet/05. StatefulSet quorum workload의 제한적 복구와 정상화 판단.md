# StatefulSet quorum workload의 제한적 복구와 정상화 판단

## 입력값과 빠른 수집

실행 경로는 `작업 PC의 Bash·kubectl → API server → StatefulSet → ordinal별 Pod·PVC → Service·EndpointSlice`다. 제품 quorum·membership·leader·replication·backup은 Kubernetes Ready 수로 계산하지 않고 승인된 제품 절차의 판정만 사용한다. 이 문서는 제품 shell 명령이나 controller mutation 명령을 발명하지 않는다.

**도구 확인**

`TOOLS_READY=false; MISSING=''; for T in kubectl jq date sha256sum cut; do command -v "$T" >/dev/null 2>&1 || MISSING="$MISSING $T"; done; if [ -n "${BASH_VERSION:-}" ] && [ -z "$MISSING" ]; then TOOLS_READY=true; printf '%s\n' 'TOOLS_READY=true'; else printf '중단: Bash 또는 도구 없음:%s\n' "$MISSING" >&2; fi`

**최초 입력·모든 lifecycle 폐기**

`unset INPUT_READY TARGET_READY MEMBERS_READY PRODUCT_READY SERVICE_READY CHANGE_READY NORMAL_READY STS_JSON PODS_JSON STS_UID STS_RV STS_CREATED SELECTOR_DIGEST SERVICE SERVICE_UID SERVICE_RV ENDPOINTS_JSON QUORUM_PROCEDURE PRODUCT_EVIDENCE_ID PRODUCT_OBSERVED_RFC3339 PRODUCT_QUORUM PRODUCT_LEADER PRODUCT_REPLICATION PRODUCT_BACKUP APPROVAL_ID WINDOW_ID CHANGE_RFC3339; INPUT_READY=false; TARGET_READY=false; MEMBERS_READY=false; PRODUCT_READY=false; SERVICE_READY=false; CHANGE_READY=false; NORMAL_READY=false; NS='PASTE_NAMESPACE_HERE'; STS='PASTE_STATEFULSET_HERE'; if [ "$TOOLS_READY" = true ] && [ -n "$NS" ] && [ -n "$STS" ]; then INPUT_READY=true; export NS STS; printf 'INPUT_READY=true NS=%s STS=%s\n' "$NS" "$STS"; else printf '%s\n' '중단: 최초 입력 실패.' >&2; fi`

**StatefulSet identity 등록**

`TARGET_READY=false; MEMBERS_READY=false; PRODUCT_READY=false; SERVICE_READY=false; CHANGE_READY=false; NORMAL_READY=false; unset STS_JSON STS_UID STS_RV STS_CREATED SELECTOR_DIGEST SERVICE; if [ "$INPUT_READY" != true ]; then printf '%s\n' '중단: INPUT_READY가 아니다.' >&2; elif D="$(kubectl get statefulset "$STS" -n "$NS" -o json)" && jq -e --arg n "$NS" --arg s "$STS" '.metadata.namespace==$n and .metadata.name==$s and (.metadata.uid|length>0)' <<<"$D" >/dev/null; then STS_JSON="$D"; STS_UID="$(jq -r '.metadata.uid' <<<"$D")"; STS_RV="$(jq -r '.metadata.resourceVersion' <<<"$D")"; STS_CREATED="$(jq -r '.metadata.creationTimestamp' <<<"$D")"; SELECTOR_DIGEST="$(jq -cS '.spec.selector' <<<"$D" | sha256sum | cut -d' ' -f1)"; SERVICE="$(jq -r '.spec.serviceName' <<<"$D")"; TARGET_READY=true; printf 'TARGET_READY=true uid=%s rv=%s created=%s selectorDigest=%s service=%s desired=%s ready=%s current=%s update=%s\n' "$STS_UID" "$STS_RV" "$STS_CREATED" "$SELECTOR_DIGEST" "$SERVICE" "$(jq -r '.spec.replicas' <<<"$D")" "$(jq -r '.status.readyReplicas // 0' <<<"$D")" "$(jq -r '.status.currentRevision // "-"' <<<"$D")" "$(jq -r '.status.updateRevision // "-"' <<<"$D")"; else printf '%s\n' '중단: StatefulSet exact 조회 실패.' >&2; fi`

**현재 lifecycle 재검증 함수**

`fresh_sts(){ local d u selector service; d="$(kubectl get statefulset "$STS" -n "$NS" -o json)" || { TARGET_READY=false; MEMBERS_READY=false; PRODUCT_READY=false; SERVICE_READY=false; CHANGE_READY=false; NORMAL_READY=false; return 1; }; u="$(jq -r '.metadata.uid // empty' <<<"$d")"; selector="$(jq -cS '.spec.selector' <<<"$d" | sha256sum | cut -d' ' -f1)"; service="$(jq -r '.spec.serviceName' <<<"$d")"; if [ "$TARGET_READY" != true ] || [ "$u" != "$STS_UID" ] || [ "$selector" != "$SELECTOR_DIGEST" ] || [ "$service" != "$SERVICE" ]; then unset STS_JSON PODS_JSON ENDPOINTS_JSON; TARGET_READY=false; MEMBERS_READY=false; PRODUCT_READY=false; SERVICE_READY=false; CHANGE_READY=false; NORMAL_READY=false; printf '%s\n' '중단: StatefulSet lifecycle·selector·service relation 변경. 모든 자동값을 재등록한다.' >&2; return 1; fi; STS_JSON="$d"; STS_RV="$(jq -r '.metadata.resourceVersion' <<<"$d")"; return 0; }`

**ordinal member 자동 등록**

`MEMBERS_READY=false; PRODUCT_READY=false; SERVICE_READY=false; CHANGE_READY=false; NORMAL_READY=false; unset PODS_JSON MEMBERS_DIGEST; if ! fresh_sts; then :; elif A="$(kubectl get pods -n "$NS" -o json)" && P="$(jq -c --arg uid "$STS_UID" '[.items[] | select(any(.metadata.ownerReferences[]?; .controller==true and .kind=="StatefulSet" and .uid==$uid))] | sort_by(.metadata.name)' <<<"$A")" && [ "$(jq 'length' <<<"$P")" -gt 0 ]; then PODS_JSON="$P"; MEMBERS_DIGEST="$(jq -cS '[.[]|{name:.metadata.name,uid:.metadata.uid,node:.spec.nodeName,revision:.metadata.labels["controller-revision-hash"],containers:([.status.initContainerStatuses[]?,.status.containerStatuses[]?]|map({name,containerID})),claims:[.spec.volumes[]?|select(.persistentVolumeClaim!=null)|.persistentVolumeClaim.claimName]}]' <<<"$P" | sha256sum | cut -d' ' -f1)"; MEMBERS_READY=true; jq -r '.[] | "=== member \(.metadata.name) ===\nuid=\(.metadata.uid) rv=\(.metadata.resourceVersion) created=\(.metadata.creationTimestamp) node=\(.spec.nodeName // "-") revision=\(.metadata.labels["controller-revision-hash"] // "-") phase=\(.status.phase // "-") ready=\([.status.containerStatuses[]?.ready] | all) deletion=\(.metadata.deletionTimestamp // "-")\ncontainers=" + ([.status.initContainerStatuses[]? | "init/\(.name):\(.containerID // "-")"] + [.status.containerStatuses[]? | "regular/\(.name):\(.containerID // "-")"] | join(",")) + "\nclaims=" + ([.spec.volumes[]? | select(.persistentVolumeClaim!=null) | .persistentVolumeClaim.claimName] | join(","))' <<<"$PODS_JSON"; printf 'MEMBERS_READY=true count=%s digest=%s stsRV=%s\n' "$(jq 'length' <<<"$PODS_JSON")" "$MEMBERS_DIGEST" "$STS_RV"; else printf '%s\n' '중단: owner UID가 맞는 member Pod가 없거나 조회 실패.' >&2; fi`

**현재 member identity 재검증 함수**

`fresh_members(){ local a p digest; fresh_sts || return 1; a="$(kubectl get pods -n "$NS" -o json)" || return 1; p="$(jq -c --arg uid "$STS_UID" '[.items[] | select(any(.metadata.ownerReferences[]?; .controller==true and .kind=="StatefulSet" and .uid==$uid))] | sort_by(.metadata.name)' <<<"$a")"; digest="$(jq -cS '[.[]|{name:.metadata.name,uid:.metadata.uid,node:.spec.nodeName,revision:.metadata.labels["controller-revision-hash"],containers:([.status.initContainerStatuses[]?,.status.containerStatuses[]?]|map({name,containerID})),claims:[.spec.volumes[]?|select(.persistentVolumeClaim!=null)|.persistentVolumeClaim.claimName]}]' <<<"$p" | sha256sum | cut -d' ' -f1)"; if [ "$MEMBERS_READY" != true ] || [ "$digest" != "$MEMBERS_DIGEST" ]; then MEMBERS_READY=false; PRODUCT_READY=false; SERVICE_READY=false; CHANGE_READY=false; NORMAL_READY=false; printf '%s\n' '중단: member UID·node·revision·containerID·PVC 집합이 바뀌었다. 재등록한다.' >&2; return 1; fi; PODS_JSON="$p"; return 0; }`

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | ordinal 장애와 함께 제품 quorum·leader·replication·데이터 일관성 위험이 있는 경우 |
| 적용 | StatefulSet·Pod identity와 승인된 제품 증거를 같은 관측 시각에 연결할 수 있는 경우 |
| 제외 | 제품 절차가 없거나 quorum 판정이 불명확하면 상태 보존 후 product owner에게 이관한다. |

검색 문자열: `quorum`, `membership`, `leader`, `replication`, `split-brain`, `backup`, `controller-revision-hash`, `claimName`, `VolumeAttachment`.

## 제품 판정 등록

운영자 선택값은 승인된 절차 ID, 비민감 증거 ID, canonical 관측 시각과 그 절차가 반환한 네 상태다. Kubernetes 수치에서 값을 계산하지 않는다.

`QUORUM_PROCEDURE='PASTE_QUORUM_PROCEDURE_HERE'; PRODUCT_EVIDENCE_ID='PASTE_PRODUCT_EVIDENCE_ID_HERE'; PRODUCT_OBSERVED_RFC3339='PASTE_PRODUCT_OBSERVED_RFC3339_HERE'; PRODUCT_QUORUM='PASTE_QUORUM_STATE_HERE'; PRODUCT_LEADER='PASTE_LEADER_STATE_HERE'; PRODUCT_REPLICATION='PASTE_REPLICATION_STATE_HERE'; PRODUCT_BACKUP='PASTE_BACKUP_STATE_HERE'; PRODUCT_READY=false; CHANGE_READY=false; NORMAL_READY=false; if ! fresh_members; then printf '%s\n' '중단: 최신 MEMBERS_READY가 필요하다.' >&2; elif ! [[ "$PRODUCT_OBSERVED_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] || [ "$(date -u -d "$PRODUCT_OBSERVED_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" != "$PRODUCT_OBSERVED_RFC3339" ] || [ -z "$QUORUM_PROCEDURE" ] || [ -z "$PRODUCT_EVIDENCE_ID" ]; then printf '%s\n' '중단: 승인 절차·증거·canonical UTC가 필요하다.' >&2; elif [[ ! "$PRODUCT_QUORUM" =~ ^(healthy|degraded|lost|unknown)$ ]] || [[ ! "$PRODUCT_LEADER" =~ ^(elected|absent|split|unknown)$ ]] || [[ ! "$PRODUCT_REPLICATION" =~ ^(healthy|degraded|failed|unknown)$ ]] || [[ ! "$PRODUCT_BACKUP" =~ ^(verified|stale|missing|unknown)$ ]]; then printf '%s\n' '중단: 제품 상태 허용값을 확인한다.' >&2; else PRODUCT_READY=true; printf 'PRODUCT_READY=true procedure=%s evidence=%s observed=%s quorum=%s leader=%s replication=%s backup=%s\n' "$QUORUM_PROCEDURE" "$PRODUCT_EVIDENCE_ID" "$PRODUCT_OBSERVED_RFC3339" "$PRODUCT_QUORUM" "$PRODUCT_LEADER" "$PRODUCT_REPLICATION" "$PRODUCT_BACKUP"; fi`

`PRODUCT_READY=true`는 판정 등록일 뿐 mutation 승인 아니다. `unknown`, quorum 상실, leader 불명, replication 악화 또는 backup 미확인이면 추가 Pod 중단을 금지한다.

## Service·EndpointSlice relation

**Service port-targetPort와 endpoint identity 등록**

`SERVICE_READY=false; CHANGE_READY=false; NORMAL_READY=false; unset SERVICE_UID SERVICE_RV SERVICE_SPEC_DIGEST ENDPOINTS_DIGEST ENDPOINTS_JSON; if ! fresh_members; then printf '%s\n' '중단: 최신 MEMBERS_READY가 필요하다.' >&2; elif S="$(kubectl get service "$SERVICE" -n "$NS" -o json)" && E="$(kubectl get endpointslices.discovery.k8s.io -n "$NS" -l "kubernetes.io/service-name=$SERVICE" -o json)" && jq -e --arg n "$NS" --arg s "$SERVICE" '.metadata.namespace==$n and .metadata.name==$s and (.metadata.uid|length>0)' <<<"$S" >/dev/null && jq -e --argjson pods "$PODS_JSON" '([$pods[].metadata.uid] | unique) as $members | all(.items[].endpoints[]?.targetRef?; .kind=="Pod" and (.uid|length>0) and ((.uid as $u | $members | index($u)) != null))' <<<"$E" >/dev/null; then SERVICE_UID="$(jq -r '.metadata.uid' <<<"$S")"; SERVICE_RV="$(jq -r '.metadata.resourceVersion' <<<"$S")"; SERVICE_SPEC_DIGEST="$(jq -cS '.spec' <<<"$S" | sha256sum | cut -d' ' -f1)"; ENDPOINTS_DIGEST="$(jq -cS '[.items[]|{name:.metadata.name,uid:.metadata.uid,ports:.ports,targets:[.endpoints[]?|{uid:.targetRef.uid,ready:.conditions.ready,serving:.conditions.serving}]}]' <<<"$E" | sha256sum | cut -d' ' -f1)"; ENDPOINTS_JSON="$E"; SERVICE_READY=true; jq -r '"service uid=\(.metadata.uid) rv=\(.metadata.resourceVersion) clusterIP=\(.spec.clusterIP) ports=" + ([.spec.ports[] | "\(.name // "-"):\(.port)->\(.targetPort)" ] | join(","))' <<<"$S"; jq -r '.items[] | "slice=\(.metadata.name) uid=\(.metadata.uid) rv=\(.metadata.resourceVersion) ports=" + ([.ports[]? | "\(.name // "-"):\(.port // 0)" ] | join(",")) + " endpoints=" + ([.endpoints[]? | "pod=\(.targetRef.name // "-") uid=\(.targetRef.uid // "-") ready=\(.conditions.ready // "-") serving=\(.conditions.serving // "-")" ] | join(";"))' <<<"$E"; else printf '%s\n' '중단: Service 또는 EndpointSlice API 조회·member identity 검증 실패.' >&2; fi`

**현재 Service·EndpointSlice relation 재검증 함수**

`fresh_service(){ local s e sd ed; fresh_sts || return 1; s="$(kubectl get service "$SERVICE" -n "$NS" -o json)" || return 1; e="$(kubectl get endpointslices.discovery.k8s.io -n "$NS" -l "kubernetes.io/service-name=$SERVICE" -o json)" || return 1; sd="$(jq -cS '.spec' <<<"$s" | sha256sum | cut -d' ' -f1)"; ed="$(jq -cS '[.items[]|{name:.metadata.name,uid:.metadata.uid,ports:.ports,targets:[.endpoints[]?|{uid:.targetRef.uid,ready:.conditions.ready,serving:.conditions.serving}]}]' <<<"$e" | sha256sum | cut -d' ' -f1)"; if [ "$SERVICE_READY" != true ] || [ "$(jq -r '.metadata.uid // empty' <<<"$s")" != "$SERVICE_UID" ] || [ "$sd" != "$SERVICE_SPEC_DIGEST" ] || [ "$ed" != "$ENDPOINTS_DIGEST" ]; then SERVICE_READY=false; CHANGE_READY=false; NORMAL_READY=false; printf '%s\n' '중단: Service 또는 EndpointSlice relation이 바뀌었다. 재등록한다.' >&2; return 1; fi; return 0; }`

EndpointSlice가 비면 정상으로 해석하지 않는다. headless Service라도 Service port·targetPort, EndpointSlice port와 targetRef Pod UID를 연결한다. ready/serving은 leader·quorum·replication을 보장하지 않는다.

## 관측별 진단 경로

| 지금 보이는 것 | 먼저 볼 것 | 그다음 어디를 볼지 | 상세 절 |
|---|---|---|---|
| 한 ordinal NotReady, 제품 quorum healthy | 같은 시각의 제품 evidence와 member UID | Pod 원인은 전용 런북으로 이관하되 제품 승인 전 삭제 금지 | [[#제품 판정 등록\|제품 판정]] |
| 추가 중단 시 영향 unknown | quorum·leader·replication·backup의 unknown 항목 | 현재 상태 보존, product/storage owner escalation | [[#제한적 상태 변경 gate\|변경 gate]] |
| quorum lost·split-brain 가능 | 승인된 제품 절차의 직접 판정 | StatefulSet rollout 금지, data recovery 절차로 이동 | [[#제한적 상태 변경 gate\|변경 gate]] |
| Kubernetes Ready지만 제품 기능 실패 | EndpointSlice targetRef UID와 제품 read/write | controller 정상과 데이터 정상화를 분리 | [[#replacement 재등록과 정상화\|정상화]] |

`Running/NotReady`는 [[16. Pod가 Running이지만 Ready가 아닌 경우]], `Terminating`은 [[15. Pod가 Terminating에서 멈춘 경우]], mount·attach는 [[25. Pod 이벤트에 FailedMount·FailedAttachVolume이 기록된 경우]]로 이동한다. old/new Pod UID, containerID, PVC claim과 제품 증거 시각을 함께 넘긴다.

## 제한적 상태 변경 gate

이 문서는 `delete`, `scale`, `rollout`, `patch`, partition·strategy 변경, PVC/PV/VolumeAttachment 조작 명령을 제공하지 않는다. 승인된 제품 절차가 있더라도 실행 직전에 최신 StatefulSet UID·resourceVersion·selector digest·revision, 대상 ordinal Pod UID·containerID·PVC/PV/attachment, quorum·leader·replication·backup, availability, idempotency·중복 write, 승인 ID·창과 rollback target을 직접 재검사해야 한다.

**승인·변경 창 등록**

`APPROVAL_ID='PASTE_APPROVAL_ID_HERE'; WINDOW_ID='PASTE_WINDOW_ID_HERE'; CHANGE_RFC3339='PASTE_CHANGE_RFC3339_HERE'; CHANGE_READY=false; NORMAL_READY=false; if [ "$PRODUCT_READY" != true ] || ! fresh_members || ! fresh_service; then printf '%s\n' '중단: 최신 제품·member·서비스 gate가 필요하다.' >&2; elif [ "$PRODUCT_QUORUM" != healthy ] || [ "$PRODUCT_LEADER" != elected ] || [ "$PRODUCT_REPLICATION" != healthy ] || [ "$PRODUCT_BACKUP" != verified ]; then printf '%s\n' '중단: 제품 안전 상태가 확정되지 않았다.' >&2; elif ! [[ "$CHANGE_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] || [ "$(date -u -d "$CHANGE_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" != "$CHANGE_RFC3339" ] || [ "$(date -u -d "$PRODUCT_OBSERVED_RFC3339" +%s)" -gt "$(date -u -d "$CHANGE_RFC3339" +%s)" ] || [ -z "$APPROVAL_ID" ] || [ -z "$WINDOW_ID" ]; then printf '%s\n' '중단: 승인·창·canonical UTC와 evidence<=change 관계가 필요하다.' >&2; else CHANGE_READY=true; printf 'CHANGE_READY=true approval=%s window=%s change=%s stsUID=%s stsRV=%s productEvidence=%s\n' "$APPROVAL_ID" "$WINDOW_ID" "$CHANGE_RFC3339" "$STS_UID" "$STS_RV" "$PRODUCT_EVIDENCE_ID"; fi`

이 gate도 제품별 action을 승인하지 않는다. action 직전 값이 하나라도 바뀌면 `CHANGE_READY=false`로 두고 재승인한다.

## replacement 재등록과 정상화

외부 승인 절차가 Pod를 교체했다면 기존 member UID, node, containerID, PVC·제품·Service 관계를 폐기한다. 다음 명령은 current StatefulSet UID·spec만 재확인하고 변경 시각 뒤 current owner의 새 Pod가 정확히 하나일 때만 replacement를 등록한다.

`MEMBERS_READY=false; PRODUCT_READY=false; SERVICE_READY=false; NORMAL_READY=false; unset PODS_JSON MEMBERS_DIGEST ENDPOINTS_JSON SERVICE_UID SERVICE_RV SERVICE_SPEC_DIGEST ENDPOINTS_DIGEST REPLACEMENT_NAME REPLACEMENT_UID REPLACEMENT_CREATED; if [ "$CHANGE_READY" != true ] || ! fresh_sts; then printf '%s\n' '중단: 최신 CHANGE_READY가 필요하다.' >&2; elif C="$(kubectl get pods -n "$NS" -o json | jq -c --arg uid "$STS_UID" --arg t "$CHANGE_RFC3339" '[.items[] | select(.metadata.creationTimestamp>$t and any(.metadata.ownerReferences[]?; .controller==true and .kind=="StatefulSet" and .uid==$uid))]')" && [ "$(jq 'length' <<<"$C")" -eq 1 ]; then REPLACEMENT_NAME="$(jq -r '.[0].metadata.name' <<<"$C")"; REPLACEMENT_UID="$(jq -r '.[0].metadata.uid' <<<"$C")"; REPLACEMENT_CREATED="$(jq -r '.[0].metadata.creationTimestamp' <<<"$C")"; jq -r '.[0] | "replacement=\(.metadata.name) uid=\(.metadata.uid) created=\(.metadata.creationTimestamp) node=\(.spec.nodeName // "-") ownerUID=" + ([.metadata.ownerReferences[]? | select(.controller==true) | .uid] | join(",")) + " revision=\(.metadata.labels["controller-revision-hash"] // "-") claims=" + ([.spec.volumes[]? | select(.persistentVolumeClaim!=null) | .persistentVolumeClaim.claimName] | join(","))' <<<"$C"; NORMAL_READY=true; else printf '%s\n' '중단: replacement 후보가 정확히 하나가 아니다.' >&2; fi`

정상화는 controller 수렴과 데이터 정상화를 분리한다. 새 Pod UID·owner·revision·PVC/PV/attachment, Service/EndpointSlice targetRef UID, 새 Warning·restart 중단을 재수집하고, 새 `PRODUCT_EVIDENCE_ID`로 quorum·membership·leader·replication·backup·데이터 일관성과 실제 read/write를 관찰 창 동안 확인해야 한다.

**PromQL 완성본 생성**

`if [ "$TARGET_READY" != true ] || ! fresh_sts; then printf '%s\n' '중단: 최신 TARGET_READY가 필요하다.' >&2; else printf 'kube_statefulset_status_replicas_ready{namespace=%s,statefulset=%s}/clamp_min(kube_statefulset_replicas{namespace=%s,statefulset=%s},1)\n' "$(jq -Rn --arg v "$NS" '$v')" "$(jq -Rn --arg v "$STS" '$v')" "$(jq -Rn --arg v "$NS" '$v')" "$(jq -Rn --arg v "$STS" '$v')"; fi`

metric 미지원·빈 결과는 성공이 아니다. Ready 비율은 제품 quorum 증거가 아니다.

보존 대상은 StatefulSet·Pod·PVC/PV/VolumeAttachment·Service·EndpointSlice UID/resourceVersion/relation과 시각, 제품 절차·비민감 증거 ID·판정, 승인·창·중단 결과다. Secret·token·개인키·제품 데이터·전체 로그/YAML은 보존하지 않는다.
