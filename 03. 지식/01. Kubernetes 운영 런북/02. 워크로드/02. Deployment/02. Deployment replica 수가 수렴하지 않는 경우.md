# Deployment replica 수가 수렴하지 않는 경우

## 입력값과 빠른 수집

실행 host는 승인된 `kubectl` 작업 PC의 같은 Bash다. 경로는 `Deployment -> 직접 소유 ReplicaSet -> 직접 소유 Pod -> 조건부 HPA/PDB -> 실제 기능`이며 SSH·CRI·node 명령은 실행하지 않는다.

| 구분 | 값 | 출처 |
|---|---|---|
| 최초 입력 | `NS`, `DEP`, UTC `START_INPUT`·`END_INPUT` | 운영자 |
| API 자동값 | Deployment UID/resourceVersion/generation, selector, revision | exact-name API |
| 계산 후보 | selector 문자열, old/new 수와 strategy 예산 | API 값으로 계산 후 대조 |
| 운영자 선택값 | 영향 Pod, 적용 HPA/PDB | owner 출력 확인 뒤 선택 |

```bash
TOOLS_READY=false
missing=()
for t in kubectl jq date sha256sum sed; do command -v "$t" >/dev/null 2>&1 || missing+=("$t"); done
if [ -z "${BASH_VERSION:-}" ] || ((${#missing[@]})); then printf '중단: Bash/도구 확인 실패 missing=%s\n' "${missing[*]:--}" >&2; else TOOLS_READY=true; printf 'TOOLS_READY=true host=%s\n' "$(hostname)"; fi
```

```bash
unset INPUT_READY AUTO_READY TARGET_READY SNAPSHOT_READY POD_READY EVENT_READY NORMAL_READY PROMQL_READY DEP_UID DEP_RV DEP_GEN SELECTOR DEP_RE TEMPLATE_DIGEST REVISION OBS_START OBS_END SNAPSHOT_AT POD POD_UID POD_RV POD_OWNER_UID POD_NODE NODE_UID MACHINE_ID CONTAINER_ROWS HPA PDB
INPUT_READY=false; AUTO_READY=false; TARGET_READY=false; SNAPSHOT_READY=false; POD_READY=false; EVENT_READY=false; NORMAL_READY=false; PROMQL_READY=false
NS='PASTE_NAMESPACE_HERE'
DEP='PASTE_DEPLOYMENT_HERE'
START_INPUT='PASTE_START_RFC3339_HERE'
END_INPUT='PASTE_END_RFC3339_HERE'
if [ "$TOOLS_READY" != true ]; then printf '%s\n' '중단: TOOLS_READY가 아니다.' >&2
elif [[ ! "$START_INPUT" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] || [[ ! "$END_INPUT" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]]; then printf '%s\n' '중단: canonical UTC RFC3339 필요' >&2
elif [ "$(date -u -d "$START_INPUT" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" != "$START_INPUT" ] || [ "$(date -u -d "$END_INPUT" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" != "$END_INPUT" ] || (( $(date -u -d "$START_INPUT" +%s) >= $(date -u -d "$END_INPUT" +%s) )); then printf '%s\n' '중단: 시간 round-trip 또는 start<end 실패' >&2
elif kubectl get deployment "$DEP" -n "$NS" --request-timeout=15s >/dev/null 2>&1; then INPUT_READY=true; printf 'INPUT_READY=true %s/%s %s..%s\n' "$NS" "$DEP" "$START_INPUT" "$END_INPUT"; else printf '%s\n' '중단: Deployment 조회 실패' >&2; fi
```

재등록은 UID/resourceVersion/owner/containerID/node/관계/시간/digest/regex와 모든 downstream gate를 먼저 폐기한다.

```bash
unset AUTO_READY TARGET_READY SNAPSHOT_READY POD_READY EVENT_READY NORMAL_READY PROMQL_READY DEP_UID DEP_RV DEP_GEN SELECTOR DEP_RE TEMPLATE_DIGEST REVISION OBS_START OBS_END SNAPSHOT_AT POD POD_UID POD_RV POD_OWNER_UID POD_NODE NODE_UID MACHINE_ID CONTAINER_ROWS HPA PDB
AUTO_READY=false; TARGET_READY=false; SNAPSHOT_READY=false; POD_READY=false; EVENT_READY=false; NORMAL_READY=false; PROMQL_READY=false
if [ "$INPUT_READY" != true ]; then printf '%s\n' '중단: INPUT_READY가 아니다.' >&2
elif ! D="$(kubectl get deployment "$DEP" -n "$NS" -o json --request-timeout=15s)"; then printf '%s\n' '중단: API producer 실패, 과거 gate 폐기됨' >&2
elif ! S="$(jq -er '([.spec.selector.matchLabels|to_entries[]?|"\(.key)=\(.value)"]+[.spec.selector.matchExpressions[]?|if .operator=="In" then "\(.key) in (\(.values|join(",")))" elif .operator=="NotIn" then "\(.key) notin (\(.values|join(",")))" elif .operator=="Exists" then .key elif .operator=="DoesNotExist" then "!\(.key)" else error("operator") end])|join(",")|select(length>0)' <<<"$D")"; then printf '%s\n' '중단: selector filter 오류' >&2
elif ! DIG="$(jq -cS '.spec.template' <<<"$D" | sha256sum | sed 's/[[:space:]].*$//')" || [ -z "$DIG" ]; then printf '%s\n' '중단: digest 계산 실패' >&2
else DEP_UID="$(jq -r '.metadata.uid' <<<"$D")"; DEP_RV="$(jq -r '.metadata.resourceVersion' <<<"$D")"; DEP_GEN="$(jq -r '.metadata.generation' <<<"$D")"; SELECTOR="$S"; DEP_RE="$(printf '%s' "$DEP"|sed 's/[][\\.^$*+?(){}|]/\\&/g')"; TEMPLATE_DIGEST="$DIG"; REVISION="$(jq -r '.metadata.annotations["deployment.kubernetes.io/revision"]//"-"' <<<"$D")"; OBS_START="$START_INPUT"; OBS_END="$END_INPUT"; AUTO_READY=true; printf 'AUTO_READY=true uid=%s rv=%s generation=%s revision=%s\nselector=%s\ndigest=%s\n' "$DEP_UID" "$DEP_RV" "$DEP_GEN" "$REVISION" "$SELECTOR" "$TEMPLATE_DIGEST"; fi
```

```bash
fresh_deployment(){ TARGET_READY=false; unset D_LIVE S_LIVE DIG_LIVE; [ "$AUTO_READY" = true ] || return 1; D_LIVE="$(kubectl get deployment "$DEP" -n "$NS" -o json --request-timeout=15s)" || return 1; [ "$(jq -r '.metadata.uid' <<<"$D_LIVE")" = "$DEP_UID" ] || { printf '%s\n' '중단: Deployment UID 교체' >&2; return 1; }; S_LIVE="$(jq -er '([.spec.selector.matchLabels|to_entries[]?|"\(.key)=\(.value)"]+[.spec.selector.matchExpressions[]?|if .operator=="In" then "\(.key) in (\(.values|join(",")))" elif .operator=="NotIn" then "\(.key) notin (\(.values|join(",")))" elif .operator=="Exists" then .key elif .operator=="DoesNotExist" then "!\(.key)" else error("operator") end])|join(",")|select(length>0)' <<<"$D_LIVE")" || return 1; DIG_LIVE="$(jq -cS '.spec.template' <<<"$D_LIVE"|sha256sum|sed 's/[[:space:]].*$//')" || return 1; [ "$S_LIVE" = "$SELECTOR" ] && [ "$DIG_LIVE" = "$TEMPLATE_DIGEST" ] || { printf '%s\n' '중단: selector/template 변경, 자동값 재등록 필요' >&2; return 1; }; DEP_RV="$(jq -r '.metadata.resourceVersion' <<<"$D_LIVE")"; TARGET_READY=true; }
```

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | desired/current/updated/ready/available/unavailable 값이 수렴하지 않는다. |
| 적용 | HPA write, Deployment strategy, old/new RS, Pod availability를 같은 UID와 시각으로 분리한다. |
| 제외 | generation/ProgressDeadlineExceeded는 [[01. Deployment rollout이 진행되지 않거나 ProgressDeadlineExceeded인 경우]], owner/revision 모순은 [[03. Deployment의 새 ReplicaSet이 생성되지 않거나 revision·owner 관계가 맞지 않는 경우]]로 이동한다. |

검색 문자열: `updatedReplicas`, `availableReplicas`, `maxSurge`, `maxUnavailable`, `scaleTargetRef`, `disruptionsAllowed`.

## 최초 확인

```bash
SNAPSHOT_READY=false
if ! fresh_deployment; then printf '%s\n' '중단: 최신 TARGET_READY 실패' >&2
elif ! RSJ="$(kubectl get rs -n "$NS" -l "$SELECTOR" -o json --request-timeout=15s)"; then printf '%s\n' '중단: RS producer 실패' >&2
elif ! PJ="$(kubectl get pods -n "$NS" -l "$SELECTOR" -o json --request-timeout=15s)"; then printf '%s\n' '중단: Pod producer 실패' >&2
elif ! HJ="$(kubectl get hpa -n "$NS" -o json --request-timeout=15s)" || ! BJ="$(kubectl get pdb -n "$NS" -o json --request-timeout=15s)"; then printf '%s\n' '중단: HPA/PDB producer 실패' >&2
elif [ "$(jq --arg u "$DEP_UID" '[.items[]|select(any(.metadata.ownerReferences[]?;.controller==true and .kind=="Deployment" and .uid==$u))]|length' <<<"$RSJ")" -eq 0 ]; then printf '%s\n' '분기: 직접 소유 RS no-match' >&2
else SNAPSHOT_AT="$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; jq -r '"=== Deployment ===\nUID/RV: \(.metadata.uid)/\(.metadata.resourceVersion)\ngeneration/observed: \(.metadata.generation)/\(.status.observedGeneration//0)\ndesired/current/updated/ready/available/unavailable: \(.spec.replicas//1)/\(.status.replicas//0)/\(.status.updatedReplicas//0)/\(.status.readyReplicas//0)/\(.status.availableReplicas//0)/\(.status.unavailableReplicas//0)\nstrategy surge/unavailable: \(.spec.strategy.rollingUpdate.maxSurge//"-")/\(.spec.strategy.rollingUpdate.maxUnavailable//"-")"' <<<"$D_LIVE"; jq -r --arg u "$DEP_UID" '.items[]|select(any(.metadata.ownerReferences[]?;.controller==true and .uid==$u))|"\n=== RS \(.metadata.name) ===\nUID/RV: \(.metadata.uid)/\(.metadata.resourceVersion) ownerUID="+$u+"\nrevision/hash/created: \(.metadata.annotations["deployment.kubernetes.io/revision"]//"-")/\(.metadata.labels["pod-template-hash"]//"-")/\(.metadata.creationTimestamp)\ndesired/current/ready/available: \(.spec.replicas//0)/\(.status.replicas//0)/\(.status.readyReplicas//0)/\(.status.availableReplicas//0)"' <<<"$RSJ"; jq -r --argjson rs "$RSJ" --arg u "$DEP_UID" '.items[] as $p|($rs.items[]|select(.metadata.uid==([$p.metadata.ownerReferences[]?|select(.controller==true)][0].uid))|select(any(.metadata.ownerReferences[]?;.controller==true and .uid==$u))) as $r|"\nPod \($p.metadata.name) uid=\($p.metadata.uid) rv=\($p.metadata.resourceVersion) ownerRS=\($r.metadata.name)/\($r.metadata.uid) revision=\($r.metadata.annotations["deployment.kubernetes.io/revision"]//"-") node=\($p.spec.nodeName//"-") created=\($p.metadata.creationTimestamp) Ready=\([$p.status.conditions[]?|select(.type=="Ready")][0].status//"-") containers=\(([$p.status.initContainerStatuses[]?|"init/\(.name):\(.containerID//"-")"]+[$p.status.containerStatuses[]?|"regular/\(.name):\(.containerID//"-")"] )|join(";"))"' <<<"$PJ"; jq -r --arg d "$DEP" '.items[]|select(.spec.scaleTargetRef.kind=="Deployment" and .spec.scaleTargetRef.name==$d)|"HPA \(.metadata.name) uid=\(.metadata.uid) rv=\(.metadata.resourceVersion) min/max/current/desired=\(.spec.minReplicas//1)/\(.spec.maxReplicas)/\(.status.currentReplicas//0)/\(.status.desiredReplicas//0) conditions=\([.status.conditions[]?|"\(.type):\(.status)/\(.reason//"-")"]|join(";"))"' <<<"$HJ"; printf 'PDB 후보는 selector 교집합을 API로 별도 검증한다. snapshot=%s\n' "$SNAPSHOT_AT"; SNAPSHOT_READY=true; fi
```

판정은 다음 순서다.

1. HPA가 exact `scaleTargetRef=Deployment/$DEP`이면 desired source다. metric 실패와 외부 scaler write는 승인된 owner/audit 경로에서 확인하며 직접 scale하지 않는다.
2. updated는 늘고 available이 부족하면 새 RS owner Pod의 Ready/reason을 해당 Pod 런북으로 이관한다.
3. old/new 합계는 `RollingUpdate`에서 percent `maxSurge` 올림, `maxUnavailable` 내림으로 계산한다. `Recreate`에 이 산술을 적용하지 않는다.
4. `PDB disruptionsAllowed=0`은 Eviction API 여유다. Deployment rolling update를 직접 차단한다고 단정하거나 PDB를 완화하지 않는다.
5. owner UID·hash가 다르면 해당 객체를 replica에 세지 않는다.

## 관측별 진단 경로

| 지금 보이는 것 | 먼저 볼 것 | 그다음 어디를 볼지 | 상세 절 |
|---|---|---|---|
| desired가 반복 변경된다 | exact HPA scaleTargetRef와 current Deployment UID | HPA/direct writer 이관 | [[#최초 확인]] |
| updated는 늘고 available이 부족하다 | current owner Pod UID·Ready·containerID | 영향 Pod와 UID 제한 Event | [[#영향 Pod identity 등록과 UID 제한 Event]] |
| old/new 합계가 일시적으로 desired와 다르다 | strategy와 maxSurge/maxUnavailable 계산 | 정상 rollout 또는 old RS 정체 | [[04. Deployment의 이전 ReplicaSet이 축소되지 않는 경우]] |
| 수치는 수렴했다 | current owner Pod Ready와 새 identity | 실제 기능과 재발 관찰 | [[#상태 변경 경계와 정상화]] |

```bash
POD_READY=false; EVENT_READY=false
unset POD_UID POD_RV POD_OWNER_UID POD_NODE NODE_UID MACHINE_ID CONTAINER_ROWS
if [ "$SNAPSHOT_READY" != true ] || ! fresh_deployment; then printf '%s\n' '중단: 최신 snapshot/target 필요' >&2
elif ! P="$(kubectl get pod "$POD" -n "$NS" -o json --request-timeout=15s)"; then printf '%s\n' '중단: Pod 조회 실패' >&2
elif ! R="$(kubectl get rs "$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true and .kind=="ReplicaSet")][0].name//empty' <<<"$P")" -n "$NS" -o json --request-timeout=15s)"; then printf '%s\n' '중단: owner RS 조회 실패' >&2
elif ! jq -e --arg du "$DEP_UID" --arg ru "$(jq -r '.metadata.uid' <<<"$R")" 'any(.metadata.ownerReferences[]?;.controller==true and .uid==$ru)' <<<"$P" >/dev/null || ! jq -e --arg du "$DEP_UID" 'any(.metadata.ownerReferences[]?;.controller==true and .uid==$du)' <<<"$R" >/dev/null; then printf '%s\n' '중단: owner chain 불일치' >&2
elif [ -n "$(jq -r '.spec.nodeName//empty' <<<"$P")" ] && ! N="$(kubectl get node "$(jq -r '.spec.nodeName' <<<"$P")" -o json --request-timeout=15s)"; then printf '%s\n' '중단: node identity 조회 실패' >&2
else POD_UID="$(jq -r '.metadata.uid' <<<"$P")"; POD_RV="$(jq -r '.metadata.resourceVersion' <<<"$P")"; POD_OWNER_UID="$(jq -r '[.metadata.ownerReferences[]|select(.controller==true)][0].uid' <<<"$P")"; POD_NODE="$(jq -r '.spec.nodeName//"-"' <<<"$P")"; NODE_UID="$(jq -r '.metadata.uid//"-"' <<<"${N:-{}}")"; MACHINE_ID="$(jq -r '.status.nodeInfo.machineID//"-"' <<<"${N:-{}}")"; CONTAINER_ROWS="$(jq -c '([.status.initContainerStatuses[]?|{kind:"init",name,containerID:(.containerID//"-")}]+[.status.containerStatuses[]?|{kind:"regular",name,containerID:(.containerID//"-")}])' <<<"$P")"; POD_READY=true; printf 'POD_READY=true pod=%s uid=%s rv=%s ownerRSUID=%s node=%s nodeUID=%s machineID=%s containers=%s\n' "$POD" "$POD_UID" "$POD_RV" "$POD_OWNER_UID" "$POD_NODE" "$NODE_UID" "$MACHINE_ID" "$CONTAINER_ROWS"; fi
```

```bash
EVENT_READY=false
if [ "$POD_READY" != true ] || ! fresh_deployment; then printf '%s\n' '중단: 최신 POD_READY/TARGET_READY 필요' >&2
elif [ "$POD_UID" != "$(kubectl get pod "$POD" -n "$NS" -o jsonpath='{.metadata.uid}' --request-timeout=15s 2>/dev/null)" ]; then printf '%s\n' '중단: Pod UID 교체' >&2

ROWS="$(jq -er --arg u "$POD_UID" --arg s "$OBS_START" --arg e "$OBS_END" '[.items[]|select(.involvedObject.uid==$u)|(.eventTime//.series.lastObservedTime//.lastTimestamp//.firstTimestamp//.metadata.creationTimestamp) as $t|select($t>=$s and $t<=$e)|{uid:.metadata.uid,time:$t,reason,message}]' <<<"$E")"; then printf '%s\n' '중단: Event filter 오류' >&2

## 상태 변경 경계와 정상화

이 문서에는 mutation 명령이 없다. scale/HPA/PDB/strategy/Pod 변경은 direct writer, 현재 UID/resourceVersion·selector·revision·template digest, 승인 ID·UTC window·가용성·멱등성·중단 조건·검증된 rollback target을 승인 시스템에서 다시 검사한 절차로만 수행한다.

```bash
NORMAL_READY=false
if ! fresh_deployment; then printf '%s\n' '중단: 최신 target 실패' >&2
elif ! R2="$(kubectl get rs -n "$NS" -l "$SELECTOR" -o json --request-timeout=15s)" || ! P2="$(kubectl get pods -n "$NS" -l "$SELECTOR" -o json --request-timeout=15s)"; then printf '%s\n' '중단: 정상화 producer 실패' >&2
elif [ "$(jq -r '.metadata.generation==(.status.observedGeneration//0) and (.spec.replicas//1)==(.status.updatedReplicas//0) and (.spec.replicas//1)==(.status.readyReplicas//0) and (.spec.replicas//1)==(.status.availableReplicas//0)' <<<"$D_LIVE")" != true ]; then printf '%s\n' '미정상: Deployment 수치 미수렴' >&2
elif [ "$(jq --arg du "$DEP_UID" --argjson r "$R2" '[.items[] as $p|($r.items[]|select(.metadata.uid==([$p.metadata.ownerReferences[]?|select(.controller==true)][0].uid))|select(any(.metadata.ownerReferences[]?;.controller==true and .uid==$du)))|select(([$p.status.conditions[]?|select(.type=="Ready")][0].status//"False")!="True")]|length' <<<"$P2")" -ne 0 ]; then printf '%s\n' '미정상: current owner Pod Ready 불충족' >&2
else NORMAL_READY=true; printf 'NORMAL_READY=true deploymentUID=%s observed=%s\n' "$DEP_UID" "$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; fi
```

수치 수렴 뒤에도 새 Pod UID/owner/containerID 재등록, 새 Warning·restart 재발 없음, HPA/PDB 의도 상태, 승인된 실제 기능·오류율·지연을 관찰 창 동안 확인한다. Service를 사용하는 경우 port-targetPort-EndpointSlice port와 endpoint `targetRef.uid`가 현재 Pod UID 집합과 exact하게 연결돼야 하며 빈 endpoint를 정상으로 해석하지 않는다.

## Prometheus 빠른 관측

```bash
PROMQL_READY=false
if ! fresh_deployment; then printf '%s\n' '중단: 최신 target 실패' >&2; else printf 'kube_deployment_spec_replicas{namespace="%s",deployment="%s"}-kube_deployment_status_replicas_available{namespace="%s",deployment="%s"}\n' "$NS" "$DEP" "$NS" "$DEP"; printf 'kube_deployment_status_replicas_unavailable{namespace="%s",deployment="%s"}\n' "$NS" "$DEP"; PROMQL_READY=true; fi

출력만 외부 UI에 사용한다. metric/label 미지원, cluster 범위 오류, 빈 결과를 정상으로 처리하지 않는다. API UID·owner chain과 실제 기능을 대체하지 않는다.
