# Indexed Job의 completion index가 수렴하지 않는 경우

## 입력값과 빠른 수집

실행 경로는 `작업 PC Bash·kubectl → API server → Indexed Job UID → index annotation이 있는 attempt Pod UID → init/regular containerID → 외부 shard 결과`다. 최초 입력은 namespace와 Job 이름, API 자동값은 Job identity·spec·attempt 관계, 계산 후보는 index 범위, 운영자 선택값은 문제 index·attempt container·증거 시간 창·외부 결과다.

**도구 확인**

`TOOLS_READY=false; MISSING=''; for T in kubectl jq date grep awk sha256sum cut; do command -v "$T" >/dev/null 2>&1 || MISSING="$MISSING $T"; done; if [ -n "${BASH_VERSION:-}" ] && [ -z "$MISSING" ]; then TOOLS_READY=true; printf '%s\n' 'TOOLS_READY=true'; else printf '중단: Bash 또는 도구 없음:%s\n' "$MISSING" >&2; fi`

**최초 입력·모든 downstream 상태 폐기**

`unset INPUT_READY JOB_READY ATTEMPTS_READY INDEX_READY ATTEMPT_READY WINDOW_READY LOG_READY EVIDENCE_READY NORMAL_READY JOB_JSON PODS_JSON INDEX_PODS_JSON ATTEMPTS_DIGEST JOB_UID JOB_RV JOB_CREATED JOB_SPEC_DIGEST JOB_OWNER_KIND JOB_OWNER_NAME JOB_OWNER_UID COMPLETIONS PARALLELISM COMPLETED_INDEXES INDEX POD POD_UID POD_RV POD_CREATED CONTAINER_KIND CONTAINER CONTAINER_ID START_RFC3339 END_RFC3339 LOG_REGEX EXTERNAL_SHARD IDEMPOTENCY_KEY EXTERNAL_STATE EXTERNAL_CHECKSUM EXTERNAL_WATERMARK EVIDENCE_RFC3339; INPUT_READY=false; JOB_READY=false; ATTEMPTS_READY=false; INDEX_READY=false; ATTEMPT_READY=false; WINDOW_READY=false; LOG_READY=false; EVIDENCE_READY=false; NORMAL_READY=false; NS='PASTE_NAMESPACE_HERE'; JOB='PASTE_JOB_HERE'; if [ "$TOOLS_READY" = true ] && [ -n "$NS" ] && [ -n "$JOB" ]; then INPUT_READY=true; export NS JOB; printf 'INPUT_READY=true NS=%s JOB=%s\n' "$NS" "$JOB"; else printf '%s\n' '중단: 도구 gate와 최초 입력을 확인한다.' >&2; fi`

각 고유 `PASTE_..._HERE` 입력칸은 한 번만 사용한다. 재실행하거나 API 조회가 실패하면 위 명령부터 실행해 UID, owner, containerID, 시간·증거 선택과 모든 READY를 폐기한다.

**Job identity·Indexed 계약 등록**

`JOB_READY=false; ATTEMPTS_READY=false; INDEX_READY=false; ATTEMPT_READY=false; WINDOW_READY=false; LOG_READY=false; EVIDENCE_READY=false; NORMAL_READY=false; unset JOB_JSON JOB_UID JOB_RV JOB_CREATED JOB_SPEC_DIGEST JOB_OWNER_KIND JOB_OWNER_NAME JOB_OWNER_UID COMPLETIONS PARALLELISM COMPLETED_INDEXES; if [ "$INPUT_READY" != true ]; then printf '%s\n' '중단: INPUT_READY가 아니다.' >&2; elif D="$(kubectl get job "$JOB" -n "$NS" -o json)" && jq -e --arg n "$NS" --arg j "$JOB" '.metadata.namespace==$n and .metadata.name==$j and (.metadata.uid|length>0) and (.metadata.resourceVersion|length>0) and .spec.completionMode=="Indexed" and (.spec.completions|type=="number" and .>0)' <<<"$D" >/dev/null; then JOB_JSON="$D"; JOB_UID="$(jq -r '.metadata.uid' <<<"$D")"; JOB_RV="$(jq -r '.metadata.resourceVersion' <<<"$D")"; JOB_CREATED="$(jq -r '.metadata.creationTimestamp' <<<"$D")"; JOB_SPEC_DIGEST="$(jq -cS '.spec' <<<"$D" | sha256sum | cut -d' ' -f1)"; JOB_OWNER_KIND="$(jq -r '[.metadata.ownerReferences[]? | select(.controller==true)][0].kind // "-"' <<<"$D")"; JOB_OWNER_NAME="$(jq -r '[.metadata.ownerReferences[]? | select(.controller==true)][0].name // "-"' <<<"$D")"; JOB_OWNER_UID="$(jq -r '[.metadata.ownerReferences[]? | select(.controller==true)][0].uid // "-"' <<<"$D")"; COMPLETIONS="$(jq -r '.spec.completions' <<<"$D")"; PARALLELISM="$(jq -r '.spec.parallelism // 1' <<<"$D")"; COMPLETED_INDEXES="$(jq -r '.status.completedIndexes // ""' <<<"$D")"; JOB_READY=true; printf 'JOB_READY=true uid=%s rv=%s created=%s specDigest=%s owner=%s/%s ownerUID=%s completions=%s parallelism=%s completedIndexes=%s\n' "$JOB_UID" "$JOB_RV" "$JOB_CREATED" "$JOB_SPEC_DIGEST" "$JOB_OWNER_KIND" "$JOB_OWNER_NAME" "$JOB_OWNER_UID" "$COMPLETIONS" "$PARALLELISM" "${COMPLETED_INDEXES:--}"; else printf '%s\n' '중단: exact Indexed Job 조회·schema 검증 실패.' >&2; fi`

**현재 Job lifecycle 재검증 함수**

`fresh_job(){ local d u digest; d="$(kubectl get job "$JOB" -n "$NS" -o json)" || { JOB_READY=false; ATTEMPTS_READY=false; INDEX_READY=false; ATTEMPT_READY=false; WINDOW_READY=false; LOG_READY=false; EVIDENCE_READY=false; NORMAL_READY=false; return 1; }; u="$(jq -r '.metadata.uid // empty' <<<"$d")"; digest="$(jq -cS '.spec' <<<"$d" | sha256sum | cut -d' ' -f1)"; if [ "$JOB_READY" != true ] || [ "$u" != "$JOB_UID" ] || [ "$digest" != "$JOB_SPEC_DIGEST" ]; then unset JOB_JSON PODS_JSON INDEX_PODS_JSON POD_UID CONTAINER_ID; JOB_READY=false; ATTEMPTS_READY=false; INDEX_READY=false; ATTEMPT_READY=false; WINDOW_READY=false; LOG_READY=false; EVIDENCE_READY=false; NORMAL_READY=false; printf '%s\n' '중단: Job lifecycle 또는 spec이 바뀌었다. 처음부터 재등록한다.' >&2; return 1; fi; JOB_JSON="$d"; JOB_RV="$(jq -r '.metadata.resourceVersion' <<<"$d")"; return 0; }`

**현재 attempt 자동 등록**

`ATTEMPTS_READY=false; INDEX_READY=false; ATTEMPT_READY=false; WINDOW_READY=false; LOG_READY=false; EVIDENCE_READY=false; NORMAL_READY=false; unset PODS_JSON INDEX_PODS_JSON ATTEMPTS_DIGEST; if ! fresh_job; then :; elif A="$(kubectl get pods -n "$NS" -o json)" && P="$(jq -c --arg uid "$JOB_UID" '[.items[] | select(any(.metadata.ownerReferences[]?; .controller==true and .kind=="Job" and .uid==$uid))] | sort_by(.metadata.creationTimestamp)' <<<"$A")"; then PODS_JSON="$P"; ATTEMPTS_DIGEST="$(jq -cS '[.[]|{name:.metadata.name,uid:.metadata.uid,index:.metadata.annotations["batch.kubernetes.io/job-completion-index"],containers:([.status.initContainerStatuses[]?,.status.containerStatuses[]?]|map({name,containerID}))}]' <<<"$P" | sha256sum | cut -d' ' -f1)"; ATTEMPTS_READY=true; jq -r '.[] | "=== attempt \(.metadata.name) ===\nuid=\(.metadata.uid) rv=\(.metadata.resourceVersion) created=\(.metadata.creationTimestamp) index=\(.metadata.annotations["batch.kubernetes.io/job-completion-index"] // "-") phase=\(.status.phase // "-") deletion=\(.metadata.deletionTimestamp // "-")\nowner=" + ([.metadata.ownerReferences[]? | select(.controller==true) | "\(.kind)/\(.name) uid=\(.uid)"] | join(",")) + "\ncontainers=" + ([.status.initContainerStatuses[]? | "init/\(.name):\(.containerID // "-"):restart=\(.restartCount)"] + [.status.containerStatuses[]? | "regular/\(.name):\(.containerID // "-"):restart=\(.restartCount)"] | join(";"))' <<<"$PODS_JSON"; printf 'ATTEMPTS_READY=true count=%s digest=%s jobRV=%s\n' "$(jq 'length' <<<"$PODS_JSON")" "$ATTEMPTS_DIGEST" "$JOB_RV"; else printf '%s\n' '중단: Pod API 조회 실패.' >&2; fi`

**현재 attempt identity 재검증 함수**

`fresh_attempts(){ local a p digest; fresh_job || return 1; a="$(kubectl get pods -n "$NS" -o json)" || return 1; p="$(jq -c --arg uid "$JOB_UID" '[.items[] | select(any(.metadata.ownerReferences[]?; .controller==true and .kind=="Job" and .uid==$uid))] | sort_by(.metadata.creationTimestamp)' <<<"$a")"; digest="$(jq -cS '[.[]|{name:.metadata.name,uid:.metadata.uid,index:.metadata.annotations["batch.kubernetes.io/job-completion-index"],containers:([.status.initContainerStatuses[]?,.status.containerStatuses[]?]|map({name,containerID}))}]' <<<"$p" | sha256sum | cut -d' ' -f1)"; if [ "$ATTEMPTS_READY" != true ] || [ "$digest" != "$ATTEMPTS_DIGEST" ]; then ATTEMPTS_READY=false; INDEX_READY=false; ATTEMPT_READY=false; WINDOW_READY=false; LOG_READY=false; EVIDENCE_READY=false; NORMAL_READY=false; printf '%s\n' '중단: attempt UID·index·containerID 집합이 바뀌었다. 재등록한다.' >&2; return 1; fi; PODS_JSON="$p"; return 0; }`

빈 attempt 배열은 현재 Pod가 없다는 관측이지 completion 성공이 아니다. 이름·`job-name` label·Pod suffix로 index나 owner를 추측하지 않는다. 1.26에서는 `batch.kubernetes.io/job-completion-index` annotation을 사용하며 같은 이름의 label은 기본 계약으로 간주하지 않는다.

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | `completionMode=Indexed` Job의 `completedIndexes`가 `0..completions-1`로 수렴하지 않거나 외부 shard 결과와 다른 경우 |
| 적용 | 현재 Job UID와 annotation index가 있는 모든 current attempt owner UID를 연결할 수 있는 경우 |
| 제외 | 일반 Job은 [[01. Job completion이 수렴하지 않는 경우]], Failed condition 뒤 재실행은 [[02. BackoffLimitExceeded 또는 DeadlineExceeded 뒤 재실행 판단이 필요한 경우]]로 이동한다. |

검색 문자열: `completionMode=Indexed`, `batch.kubernetes.io/job-completion-index`, `completedIndexes`, `uncountedTerminatedPods`, `BackoffLimitExceeded`, `DeadlineExceeded`, `commit`, `checksum`, `watermark`.

## 관측별 진단 경로

| 지금 보이는 것 | 먼저 볼 것 | 그다음 어디를 볼지 | 상세 절 |
|---|---|---|---|
| 일부 index만 완료 | 문제 index의 모든 owner UID 일치 attempt | Pod 종료·controller accounting·외부 shard | [[#문제 index와 attempt 선택\|index 선택]] |
| 같은 index attempt가 반복됨 | Pod UID·containerID·종료 시각 | bounded 로그와 외부 side effect | [[#attempt container와 증거 창\|attempt 증거]] |
| Succeeded Pod와 status가 다름 | `uncountedTerminatedPods`·condition | controller accounting·finalizer 경로 | [[#status·Event와 외부 결과 대조\|status 대조]] |
| Kubernetes와 외부 shard가 다름 | idempotency·commit·checksum·watermark | 업무 owner, 재실행 보류 | [[#status·Event와 외부 결과 대조\|외부 대조]] |

## 문제 index와 attempt 선택

`INDEX='PASTE_COMPLETION_INDEX_HERE'; INDEX_READY=false; ATTEMPT_READY=false; WINDOW_READY=false; LOG_READY=false; EVIDENCE_READY=false; NORMAL_READY=false; unset INDEX_PODS_JSON; if ! fresh_attempts; then printf '%s\n' '중단: 최신 ATTEMPTS_READY가 필요하다.' >&2; elif ! [[ "$INDEX" =~ ^(0|[1-9][0-9]*)$ ]] || [ "$INDEX" -ge "$COMPLETIONS" ]; then printf '중단: index 범위는 0..%s다.\n' "$((COMPLETIONS-1))" >&2; elif P="$(jq -c --arg i "$INDEX" '[.[] | select(.metadata.annotations["batch.kubernetes.io/job-completion-index"]==$i)] | sort_by(.metadata.creationTimestamp)' <<<"$PODS_JSON")"; then INDEX_PODS_JSON="$P"; INDEX_READY=true; jq -r --arg i "$INDEX" 'if length==0 then "index=\($i) current attempt 없음" else .[] | "index=\($i) pod=\(.metadata.name) uid=\(.metadata.uid) created=\(.metadata.creationTimestamp) phase=\(.status.phase // "-") finished=" + ([.status.initContainerStatuses[]?.state.terminated.finishedAt,.status.containerStatuses[]?.state.terminated.finishedAt] | map(select(.!=null)) | join(",")) end' <<<"$INDEX_PODS_JSON"; printf 'INDEX_READY=true index=%s attempts=%s\n' "$INDEX" "$(jq 'length' <<<"$INDEX_PODS_JSON")"; else printf '%s\n' '중단: index 후보 계산 실패.' >&2; fi`

attempt가 없으면 Event·삭제 이력·controller status를 확인한다. 여러 attempt는 모두 보존하며 하나를 대표로 덮어쓰지 않는다. `completedIndexes` 압축 문자열은 API 실제값으로만 해석하고 `succeeded` aggregate로 대신하지 않는다.

## attempt container와 증거 창

로그가 필요한 경우에만 current attempt 하나와 container를 선택한다.

`POD='PASTE_ATTEMPT_POD_HERE'; CONTAINER_KIND='PASTE_INIT_OR_REGULAR_HERE'; CONTAINER='PASTE_CONTAINER_HERE'; ATTEMPT_READY=false; WINDOW_READY=false; LOG_READY=false; unset POD_UID POD_RV POD_CREATED CONTAINER_ID; if [ "$INDEX_READY" != true ] || ! fresh_job; then printf '%s\n' '중단: 최신 INDEX_READY가 필요하다.' >&2; elif [ "$CONTAINER_KIND" != init ] && [ "$CONTAINER_KIND" != regular ]; then printf '%s\n' '중단: container kind는 init 또는 regular다.' >&2; elif D="$(kubectl get pod "$POD" -n "$NS" -o json)" && jq -e --arg j "$JOB_UID" --arg i "$INDEX" 'any(.metadata.ownerReferences[]?; .controller==true and .kind=="Job" and .uid==$j) and .metadata.annotations["batch.kubernetes.io/job-completion-index"]==$i' <<<"$D" >/dev/null && { [ "$CONTAINER_KIND" = init ] && jq -e --arg c "$CONTAINER" 'any(.spec.initContainers[]?; .name==$c)' <<<"$D" >/dev/null || [ "$CONTAINER_KIND" = regular ] && jq -e --arg c "$CONTAINER" 'any(.spec.containers[]?; .name==$c)' <<<"$D" >/dev/null; }; then POD_UID="$(jq -r '.metadata.uid' <<<"$D")"; POD_RV="$(jq -r '.metadata.resourceVersion' <<<"$D")"; POD_CREATED="$(jq -r '.metadata.creationTimestamp' <<<"$D")"; if [ "$CONTAINER_KIND" = init ]; then CONTAINER_ID="$(jq -r --arg c "$CONTAINER" '.status.initContainerStatuses[]? | select(.name==$c) | .containerID // empty' <<<"$D")"; else CONTAINER_ID="$(jq -r --arg c "$CONTAINER" '.status.containerStatuses[]? | select(.name==$c) | .containerID // empty' <<<"$D")"; fi; ATTEMPT_READY=true; printf 'ATTEMPT_READY=true pod=%s uid=%s rv=%s created=%s index=%s kind=%s container=%s containerID=%s\n' "$POD" "$POD_UID" "$POD_RV" "$POD_CREATED" "$INDEX" "$CONTAINER_KIND" "$CONTAINER" "${CONTAINER_ID:--}"; else printf '%s\n' '중단: Pod·owner·index·container 관계 검증 실패.' >&2; fi`

**현재 attempt container instance 재검증 함수**

`fresh_attempt(){ local d cid; fresh_job || return 1; d="$(kubectl get pod "$POD" -n "$NS" -o json)" || return 1; if [ "$CONTAINER_KIND" = init ]; then cid="$(jq -r --arg c "$CONTAINER" '.status.initContainerStatuses[]? | select(.name==$c) | .containerID // empty' <<<"$d")"; else cid="$(jq -r --arg c "$CONTAINER" '.status.containerStatuses[]? | select(.name==$c) | .containerID // empty' <<<"$d")"; fi; if [ "$ATTEMPT_READY" != true ] || [ -z "$CONTAINER_ID" ] || [ "$cid" != "$CONTAINER_ID" ] || ! jq -e --arg p "$POD_UID" --arg j "$JOB_UID" --arg i "$INDEX" '.metadata.uid==$p and any(.metadata.ownerReferences[]?; .controller==true and .uid==$j) and .metadata.annotations["batch.kubernetes.io/job-completion-index"]==$i' <<<"$d" >/dev/null; then ATTEMPT_READY=false; WINDOW_READY=false; LOG_READY=false; printf '%s\n' '중단: attempt Pod 또는 current containerID가 바뀌었다. 재등록한다.' >&2; return 1; fi; return 0; }`

`START_RFC3339='PASTE_START_RFC3339_HERE'; END_RFC3339='PASTE_END_RFC3339_HERE'; LOG_REGEX='PASTE_LOG_REGEX_HERE'; WINDOW_READY=false; LOG_READY=false; if ! fresh_attempt; then printf '%s\n' '중단: 최신 ATTEMPT_READY가 필요하다.' >&2; elif ! [[ "$START_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ && "$END_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] || [ "$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" != "$START_RFC3339" ] || [ "$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" != "$END_RFC3339" ] || [ "$(date -u -d "$START_RFC3339" +%s)" -ge "$(date -u -d "$END_RFC3339" +%s)" ] || [ -z "$LOG_REGEX" ]; then printf '%s\n' '중단: canonical UTC start<end와 regex가 필요하다.' >&2; else WINDOW_READY=true; printf 'WINDOW_READY=true start=%s end=%s regex=%q\n' "$START_RFC3339" "$END_RFC3339" "$LOG_REGEX"; fi`

`LOG_READY=false; if [ "$WINDOW_READY" != true ] || ! fresh_attempt; then printf '%s\n' '중단: 최신 WINDOW_READY와 current containerID가 필요하다.' >&2; else RAW="$(kubectl logs "$POD" -n "$NS" -c "$CONTAINER" --timestamps --since-time="$START_RFC3339" --tail=400 2>&1)"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -ne 0 ]; then printf 'logs producer 실패 rc=%s\n' "$PRODUCER_RC" >&2; elif BOUNDED="$(awk -v s="$START_RFC3339" -v e="$END_RFC3339" '$1 >= s && $1 < e' <<<"$RAW")"; then if [ -z "$BOUNDED" ]; then printf '%s\n' '시간 필터 no-match: 창 안의 로그 행이 없음.'; elif MATCH="$(grep -E -i -C 3 -- "$LOG_REGEX" <<<"$BOUNDED")"; then printf '%s\n' "$MATCH"; LOG_READY=true; else FILTER_RC=$?; if [ "$FILTER_RC" -eq 1 ]; then printf '%s\n' 'regex no-match: 오류 부재가 아니므로 Event·종료 상태를 확인한다.'; else printf 'regex 오류 rc=%s\n' "$FILTER_RC" >&2; fi; fi; else printf '%s\n' '시간 필터 오류.' >&2; fi; unset RAW BOUNDED MATCH; fi`

`--previous`는 같은 Pod UID·같은 container의 이전 instance가 실제로 있고 containerID가 바뀐 경우에만 별도 등록한다. 다른 attempt Pod의 로그를 대신하지 않는다. 민감정보가 보이면 출력을 중단하고 오류 종류·시각만 보존한다.

## status·Event와 외부 결과 대조

`if [ "$INDEX_READY" != true ] || ! fresh_attempts; then printf '%s\n' '중단: 최신 INDEX_READY와 attempt identity가 필요하다.' >&2; else jq -r '"jobUID=\(.metadata.uid) rv=\(.metadata.resourceVersion) completedIndexes=\(.status.completedIndexes // "-") active=\(.status.active // 0) succeeded=\(.status.succeeded // 0) failed=\(.status.failed // 0) uncountedSucceeded=\([.status.uncountedTerminatedPods.succeeded[]?] | join(",")) uncountedFailed=\([.status.uncountedTerminatedPods.failed[]?] | join(","))", (.status.conditions[]? | "condition=\(.type) status=\(.status) reason=\(.reason // "-") transition=\(.lastTransitionTime // "-")")' <<<"$JOB_JSON"; fi`

`if [ "$INDEX_READY" != true ] || ! fresh_attempts; then printf '%s\n' '중단: 최신 INDEX_READY와 attempt identity가 필요하다.' >&2; elif ! kubectl get events -n "$NS" --field-selector "involvedObject.kind=Job,involvedObject.name=$JOB,involvedObject.uid=$JOB_UID" --sort-by=.metadata.creationTimestamp -o custom-columns='CREATED:.metadata.creationTimestamp,EVENT_TIME:.eventTime,SERIES_LAST:.series.lastObservedTime,FIRST:.firstTimestamp,LAST:.lastTimestamp,TYPE:.type,REASON:.reason,COUNT:.count,MESSAGE:.message'; then printf '%s\n' '중단: Event producer 실패. 빈 결과와 다르다.' >&2; fi`

`EXTERNAL_SHARD='PASTE_EXTERNAL_SHARD_HERE'; IDEMPOTENCY_KEY='PASTE_IDEMPOTENCY_KEY_HERE'; EXTERNAL_STATE='PASTE_MISSING_COMMITTED_PARTIAL_OR_UNKNOWN_HERE'; EXTERNAL_CHECKSUM='PASTE_EXTERNAL_CHECKSUM_HERE'; EXTERNAL_WATERMARK='PASTE_EXTERNAL_WATERMARK_HERE'; EVIDENCE_RFC3339='PASTE_EVIDENCE_RFC3339_HERE'; EVIDENCE_READY=false; NORMAL_READY=false; if [ "$INDEX_READY" != true ] || ! fresh_attempts; then printf '%s\n' '중단: 최신 INDEX_READY와 attempt identity가 필요하다.' >&2; elif [ -z "$EXTERNAL_SHARD" ] || [ -z "$IDEMPOTENCY_KEY" ] || [ -z "$EXTERNAL_CHECKSUM" ] || [ -z "$EXTERNAL_WATERMARK" ] || { [ "$EXTERNAL_STATE" != missing ] && [ "$EXTERNAL_STATE" != committed ] && [ "$EXTERNAL_STATE" != partial ] && [ "$EXTERNAL_STATE" != unknown ]; }; then printf '%s\n' '중단: 비민감 외부 증거와 허용 상태가 필요하다.' >&2; elif ! [[ "$EVIDENCE_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] || [ "$(date -u -d "$EVIDENCE_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" != "$EVIDENCE_RFC3339" ]; then printf '%s\n' '중단: evidence 시각은 canonical UTC여야 한다.' >&2; else EVIDENCE_READY=true; printf 'EVIDENCE_READY=true index=%s shard=%q idempotency=%q state=%s checksum=%q watermark=%q observed=%s\n' "$INDEX" "$EXTERNAL_SHARD" "$IDEMPOTENCY_KEY" "$EXTERNAL_STATE" "$EXTERNAL_CHECKSUM" "$EXTERNAL_WATERMARK" "$EVIDENCE_RFC3339"; fi`

`uncountedTerminatedPods`의 Pod UID는 controller accounting 중일 수 있다. Event가 비면 TTL·권한·집계를 열어 둔다. 외부 상태가 `partial`·`unknown`이거나 checksum·watermark가 맞지 않으면 해당 index를 다시 실행하지 않는다.

## 상태 변경 경계

이 문서는 Pod 삭제, Job 재생성, index 재실행, `parallelism`·`completionMode`·`podFailurePolicy` 변경 명령을 제공하지 않는다. 별도 변경 절차는 실행 직전에 현재 Job UID·resourceVersion·spec digest, index와 모든 attempt UID·containerID·종료 시각, 외부 idempotency·commit·checksum·watermark·부분 결과, 동시 실행 부재, 승인 ID·변경 창·rollback 또는 보상 절차를 직접 재검증해야 한다. 하나라도 불명확하면 실행하지 않는다.

새 Job을 만들었다면 기존 이름·UID·READY를 재사용하지 않고 새 Job exact name·UID·owner·creation time과 새 attempt owner UID를 처음부터 등록한다. Kubernetes 객체 삭제는 외부 side effect rollback이 아니다.

## 정상화와 역순 검증

`NORMAL_READY=false; if [ "$EVIDENCE_READY" != true ] || ! fresh_attempts; then printf '%s\n' '중단: 최신 Job·attempt·외부 evidence gate가 필요하다.' >&2; else jq -r '"uid=\(.metadata.uid) rv=\(.metadata.resourceVersion) completions=\(.spec.completions) completedIndexes=\(.status.completedIndexes // "-") active=\(.status.active // 0) succeeded=\(.status.succeeded // 0) failed=\(.status.failed // 0)", (.status.conditions[]? | "condition=\(.type) status=\(.status) reason=\(.reason // "-") transition=\(.lastTransitionTime // "-")")' <<<"$JOB_JSON"; NORMAL_READY=true; fi`

`NORMAL_READY=true`는 재조회 성공만 뜻한다. `completedIndexes`가 전체 `0..completions-1`을 포함하고 각 index에 현재 Job owner UID의 성공 attempt가 있으며, 반복 실패·restart·Warning이 관찰 창 동안 멈추고, 외부 shard별 commit·checksum·watermark와 중복 부재가 일치해야 정상화한다.

**PromQL 완성본 생성**

`if [ "$JOB_READY" != true ] || ! fresh_job; then printf '%s\n' '중단: 최신 JOB_READY가 필요하다.' >&2; else printf 'kube_job_spec_completions{namespace=%s,job_name=%s}-kube_job_status_succeeded{namespace=%s,job_name=%s}\n' "$(jq -Rn --arg v "$NS" '$v')" "$(jq -Rn --arg v "$JOB" '$v')" "$(jq -Rn --arg v "$NS" '$v')" "$(jq -Rn --arg v "$JOB" '$v')"; fi`

metric·label 미지원과 빈 결과는 정상이나 0이 아니다. aggregate PromQL은 index annotation, Job UID, Pod owner UID와 외부 shard 결과를 증명하지 못한다.

보존 대상은 Job·Pod UID/resourceVersion/owner/creation, spec digest, index annotation, init/regular containerID, status·condition·Event·로그 시간 관계와 비민감 외부 evidence다. Secret·token·개인키·전체 YAML·로그 및 shard 원문은 저장하지 않는다.
