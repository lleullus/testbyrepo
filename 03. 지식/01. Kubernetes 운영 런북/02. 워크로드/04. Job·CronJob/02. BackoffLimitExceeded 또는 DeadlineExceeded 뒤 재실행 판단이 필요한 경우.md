# BackoffLimitExceeded 또는 DeadlineExceeded 뒤 재실행 판단이 필요한 경우

## 입력값과 빠른 수집

기본 결론은 재실행 보류다. 실행 경로는 `작업 PC Bash·kubectl → API server → 원본 Job UID → attempt Pod UID → 외부 commit·ack·watermark 증거 → 승인된 CronJob template → 새 Job UID`다. Kubernetes 실패는 외부 업무 미반영을 증명하지 않는다.

**도구 확인**

`TOOLS_READY=false; MISSING=''; for T in kubectl jq date sha256sum cut; do command -v "$T" >/dev/null 2>&1 || MISSING="$MISSING $T"; done; if [ -n "${BASH_VERSION:-}" ] && [ -z "$MISSING" ]; then TOOLS_READY=true; printf '%s\n' 'TOOLS_READY=true'; else printf '중단: Bash 또는 도구 없음:%s\n' "$MISSING" >&2; fi`

**최초 입력과 stale 자동값 폐기**

`unset INPUT_READY JOB_READY ATTEMPTS_READY EVIDENCE_READY SOURCE_READY CHANGE_READY CREATED_READY NORMAL_READY JOB_JSON PODS_JSON SOURCE_JSON NEW_JOB_JSON JOB_UID JOB_RV JOB_CREATED JOB_SPEC_DIGEST JOB_OWNER_KIND JOB_OWNER_NAME JOB_OWNER_UID SOURCE SOURCE_UID SOURCE_RV SOURCE_TEMPLATE_DIGEST NEW_JOB NEW_JOB_UID NEW_JOB_RV NEW_JOB_CREATED APPROVAL_ID WINDOW_START_RFC3339 WINDOW_END_RFC3339 IDEMPOTENCY_KEY INPUT_WATERMARK EXTERNAL_CHECKSUM EXTERNAL_STATE; INPUT_READY=false; JOB_READY=false; ATTEMPTS_READY=false; EVIDENCE_READY=false; SOURCE_READY=false; CHANGE_READY=false; CREATED_READY=false; NORMAL_READY=false; NS='PASTE_NAMESPACE_HERE'; JOB='PASTE_JOB_HERE'; if [ "$TOOLS_READY" = true ] && [ -n "$NS" ] && [ -n "$JOB" ]; then INPUT_READY=true; export NS JOB; printf 'INPUT_READY=true NS=%s JOB=%s\n' "$NS" "$JOB"; else printf '%s\n' '중단: 최초 입력 실패.' >&2; fi`

**원본 Job identity·실패 condition 등록**

`JOB_READY=false; ATTEMPTS_READY=false; EVIDENCE_READY=false; SOURCE_READY=false; CHANGE_READY=false; CREATED_READY=false; NORMAL_READY=false; unset JOB_JSON JOB_UID JOB_RV JOB_CREATED JOB_SPEC_DIGEST JOB_OWNER_KIND JOB_OWNER_NAME JOB_OWNER_UID; if [ "$INPUT_READY" != true ]; then printf '%s\n' '중단: INPUT_READY가 아니다.' >&2; elif D="$(kubectl get job "$JOB" -n "$NS" -o json)" && jq -e --arg n "$NS" --arg j "$JOB" '.metadata.namespace==$n and .metadata.name==$j and any(.status.conditions[]?; .type=="Failed" and .status=="True" and (.reason=="BackoffLimitExceeded" or .reason=="DeadlineExceeded"))' <<<"$D" >/dev/null; then JOB_JSON="$D"; JOB_UID="$(jq -r '.metadata.uid' <<<"$D")"; JOB_RV="$(jq -r '.metadata.resourceVersion' <<<"$D")"; JOB_CREATED="$(jq -r '.metadata.creationTimestamp' <<<"$D")"; JOB_SPEC_DIGEST="$(jq -cS '.spec' <<<"$D" | sha256sum | cut -d' ' -f1)"; JOB_OWNER_KIND="$(jq -r '[.metadata.ownerReferences[]? | select(.controller==true)][0].kind // "-"' <<<"$D")"; JOB_OWNER_NAME="$(jq -r '[.metadata.ownerReferences[]? | select(.controller==true)][0].name // "-"' <<<"$D")"; JOB_OWNER_UID="$(jq -r '[.metadata.ownerReferences[]? | select(.controller==true)][0].uid // "-"' <<<"$D")"; JOB_READY=true; jq -r '"JOB_READY=true uid=\(.metadata.uid) rv=\(.metadata.resourceVersion) created=\(.metadata.creationTimestamp) owner=" + (([.metadata.ownerReferences[]? | select(.controller==true) | "\(.kind)/\(.name) uid=\(.uid)"] | first) // "-") + " start=\(.status.startTime // "-") active=\(.status.active // 0) succeeded=\(.status.succeeded // 0) failed=\(.status.failed // 0)", (.status.conditions[]? | select(.type=="Failed" and .status=="True") | "reason=\(.reason) transition=\(.lastTransitionTime) message=\(.message)")' <<<"$D"; else printf '%s\n' '중단: exact Job 조회 또는 대상 Failed condition 검증 실패.' >&2; fi`

**현재 원본 lifecycle 재검증 함수**

`fresh_job(){ local d u digest; d="$(kubectl get job "$JOB" -n "$NS" -o json)" || { JOB_READY=false; ATTEMPTS_READY=false; EVIDENCE_READY=false; SOURCE_READY=false; CHANGE_READY=false; CREATED_READY=false; NORMAL_READY=false; return 1; }; u="$(jq -r '.metadata.uid // empty' <<<"$d")"; digest="$(jq -cS '.spec' <<<"$d" | sha256sum | cut -d' ' -f1)"; if [ "$JOB_READY" != true ] || [ "$u" != "$JOB_UID" ] || [ "$digest" != "$JOB_SPEC_DIGEST" ]; then unset JOB_JSON PODS_JSON SOURCE_JSON NEW_JOB_JSON; JOB_READY=false; ATTEMPTS_READY=false; EVIDENCE_READY=false; SOURCE_READY=false; CHANGE_READY=false; CREATED_READY=false; NORMAL_READY=false; printf '%s\n' '중단: 원본 Job lifecycle 또는 spec 변경.' >&2; return 1; fi; JOB_JSON="$d"; JOB_RV="$(jq -r '.metadata.resourceVersion' <<<"$d")"; return 0; }`

**모든 current attempt 등록**

`ATTEMPTS_READY=false; EVIDENCE_READY=false; SOURCE_READY=false; CHANGE_READY=false; CREATED_READY=false; NORMAL_READY=false; if ! fresh_job; then :; elif A="$(kubectl get pods -n "$NS" -o json)" && P="$(jq -c --arg uid "$JOB_UID" '[.items[] | select(any(.metadata.ownerReferences[]?; .controller==true and .kind=="Job" and .uid==$uid))] | sort_by(.metadata.creationTimestamp)' <<<"$A")"; then PODS_JSON="$P"; ATTEMPTS_READY=true; jq -r '.[] | "pod=\(.metadata.name) uid=\(.metadata.uid) rv=\(.metadata.resourceVersion) created=\(.metadata.creationTimestamp) phase=\(.status.phase // "-") ownerUID=" + ([.metadata.ownerReferences[]? | select(.controller==true) | .uid] | join(",")) + " containers=" + ([.status.initContainerStatuses[]? | "init/\(.name):\(.containerID // "-"):restart=\(.restartCount):finished=\(.state.terminated.finishedAt // .lastState.terminated.finishedAt // "-")"] + [.status.containerStatuses[]? | "regular/\(.name):\(.containerID // "-"):restart=\(.restartCount):finished=\(.state.terminated.finishedAt // .lastState.terminated.finishedAt // "-")"] | join(";"))' <<<"$PODS_JSON"; printf 'ATTEMPTS_READY=true count=%s\n' "$(jq 'length' <<<"$PODS_JSON")"; else printf '%s\n' '중단: attempt 조회 실패.' >&2; fi`

삭제된 attempt는 current 목록에 없을 수 있다. Event·보존 로그와 결합하되 이름·`job-name` label만으로 owner를 연결하지 않는다.

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | 현재 Job UID의 `Failed=True` reason이 `BackoffLimitExceeded` 또는 `DeadlineExceeded`이고 재실행 여부를 판단해야 하는 경우 |
| 적용 | 원본 Job·attempt identity, 종료 시각과 외부 commit·ack·watermark 증거를 연결할 수 있는 경우 |
| 제외 | 아직 completion이 진행 중이면 [[01. Job completion이 수렴하지 않는 경우]], index별 수렴 문제면 [[03. Indexed Job의 completion index가 수렴하지 않는 경우]]로 이동한다. |

검색 문자열: `BackoffLimitExceeded`, `DeadlineExceeded`, `commit`, `ack`, `watermark`, `checksum`, `idempotency`, `duplicate`.

## 관측별 진단 경로

| 지금 보이는 것 | 먼저 볼 것 | 그다음 어디를 볼지 | 상세 절 |
|---|---|---|---|
| `BackoffLimitExceeded` | 모든 attempt reason·finishedAt | 외부 commit 상태와 멱등성 | [[#외부 결과와 멱등성 gate\|외부 gate]] |
| `DeadlineExceeded` | Job start+deadline과 condition 시각 | 외부 timeout·부분 결과 | [[#외부 결과와 멱등성 gate\|외부 gate]] |
| commit timeout·ack unknown | 비민감 evidence ID·watermark | 재실행 보류, 업무 owner 이관 | [[#외부 결과와 멱등성 gate\|외부 gate]] |
| 원인 해소·누락 확정 | source owner UID·template digest | idempotency·동시 실행·승인 뒤 새 Job 검토 | [[#mutation gate와 조건부 보완 실행\|mutation gate]] |

**Job UID 제한 Event**

`if [ "$ATTEMPTS_READY" != true ] || ! fresh_job; then printf '%s\n' '중단: 최신 ATTEMPTS_READY가 필요하다.' >&2; elif ! kubectl get events -n "$NS" --field-selector "involvedObject.kind=Job,involvedObject.name=$JOB,involvedObject.uid=$JOB_UID" --sort-by=.metadata.creationTimestamp -o custom-columns='CREATED:.metadata.creationTimestamp,EVENT_TIME:.eventTime,SERIES_LAST:.series.lastObservedTime,FIRST:.firstTimestamp,LAST:.lastTimestamp,TYPE:.type,REASON:.reason,COUNT:.count,MESSAGE:.message'; then printf '%s\n' '중단: Event producer 실패. 빈 결과와 다르다.' >&2; fi`

Event 조회 실패와 빈 결과를 분리한다. 채워진 Event/condition/Pod finishedAt가 `Job start + activeDeadlineSeconds` 또는 backoff 시도 순서와 맞는지 본다. Event `creationTimestamp` 하나만 사용하지 않는다.

## 외부 결과와 멱등성 gate

외부 값은 승인된 read-only 업무 절차에서 얻은 비민감 식별자다. 원문 SQL·token·개인정보는 넣지 않는다. `EXTERNAL_STATE` 허용값은 `missing`, `committed`, `partial`, `unknown`이다.

`IDEMPOTENCY_KEY='PASTE_IDEMPOTENCY_KEY_HERE'; INPUT_WATERMARK='PASTE_INPUT_WATERMARK_HERE'; EXTERNAL_CHECKSUM='PASTE_EXTERNAL_CHECKSUM_HERE'; EXTERNAL_STATE='PASTE_EXTERNAL_STATE_HERE'; EVIDENCE_READY=false; SOURCE_READY=false; CHANGE_READY=false; CREATED_READY=false; NORMAL_READY=false; if [ "$ATTEMPTS_READY" != true ] || ! fresh_job; then printf '%s\n' '중단: 최신 attempt gate가 필요하다.' >&2; elif [ -z "$IDEMPOTENCY_KEY" ] || [ -z "$INPUT_WATERMARK" ] || [ -z "$EXTERNAL_CHECKSUM" ]; then printf '%s\n' '중단: idempotency·입력 범위·checksum 근거가 필요하다.' >&2; elif [ "$EXTERNAL_STATE" != missing ] && [ "$EXTERNAL_STATE" != committed ] && [ "$EXTERNAL_STATE" != partial ] && [ "$EXTERNAL_STATE" != unknown ]; then printf '%s\n' '중단: 외부 상태 허용값 위반.' >&2; else EVIDENCE_READY=true; printf 'EVIDENCE_READY=true idempotency=%q watermark=%q checksum=%q state=%s\n' "$IDEMPOTENCY_KEY" "$INPUT_WATERMARK" "$EXTERNAL_CHECKSUM" "$EXTERNAL_STATE"; fi`

`committed`면 중복 위험 때문에 실행하지 않는다. `partial`·`unknown`이면 보상·resume 계약 없이는 실행하지 않는다. 아래 생성 경로는 외부 상태가 `missing`이고 동일 입력의 다른 active Job/CronJob이 없다는 별도 승인 증거가 있을 때만 적용한다.

## 승인된 CronJob source 등록

`SOURCE='PASTE_SOURCE_CRONJOB_HERE'; NEW_JOB='PASTE_NEW_JOB_HERE'; SOURCE_READY=false; CHANGE_READY=false; CREATED_READY=false; NORMAL_READY=false; unset SOURCE_JSON SOURCE_UID SOURCE_RV SOURCE_TEMPLATE_DIGEST; if [ "$EVIDENCE_READY" != true ] || ! fresh_job; then printf '%s\n' '중단: 최신 EVIDENCE_READY가 필요하다.' >&2; elif [ "$EXTERNAL_STATE" != missing ] || [ -z "$SOURCE" ] || [ -z "$NEW_JOB" ] || [ "$NEW_JOB" = "$JOB" ]; then printf '%s\n' '중단: 누락 확정과 고유 새 이름이 필요하다.' >&2; elif S="$(kubectl get cronjob "$SOURCE" -n "$NS" -o json)" && jq -e --arg n "$SOURCE" --arg u "$JOB_OWNER_UID" '.metadata.name==$n and .metadata.uid==$u and (.metadata.resourceVersion|length>0)' <<<"$S" >/dev/null && [ "$JOB_OWNER_KIND" = CronJob ] && [ "$JOB_OWNER_NAME" = "$SOURCE" ]; then SOURCE_JSON="$S"; SOURCE_UID="$(jq -r '.metadata.uid' <<<"$S")"; SOURCE_RV="$(jq -r '.metadata.resourceVersion' <<<"$S")"; SOURCE_TEMPLATE_DIGEST="$(jq -cS '.spec.jobTemplate.spec' <<<"$S" | sha256sum | cut -d' ' -f1)"; SOURCE_READY=true; printf 'SOURCE_READY=true source=%s uid=%s rv=%s templateDigest=%s suspend=%s policy=%s active=%s newJob=%s\n' "$SOURCE" "$SOURCE_UID" "$SOURCE_RV" "$SOURCE_TEMPLATE_DIGEST" "$(jq -r '.spec.suspend // false' <<<"$S")" "$(jq -r '.spec.concurrencyPolicy' <<<"$S")" "$(jq -r '[.status.active[]?] | length' <<<"$S")" "$NEW_JOB"; else printf '%s\n' '중단: 원본 owner와 CronJob source identity가 일치하지 않는다.' >&2; fi`

원본 Job의 controller owner가 CronJob이면 kind/name/UID가 source와 일치해야 한다. ad hoc Job이거나 source template이 승인된 원본과 다르면 `--from`을 사용하지 않는다.

## mutation gate와 조건부 보완 실행

`APPROVAL_ID='PASTE_APPROVAL_ID_HERE'; WINDOW_START_RFC3339='PASTE_WINDOW_START_RFC3339_HERE'; WINDOW_END_RFC3339='PASTE_WINDOW_END_RFC3339_HERE'; CHANGE_READY=false; CREATED_READY=false; NORMAL_READY=false; if [ "$EVIDENCE_READY" != true ] || [ "$SOURCE_READY" != true ] || [ "$EXTERNAL_STATE" != missing ] || [ -z "$IDEMPOTENCY_KEY" ] || ! fresh_job; then printf '%s\n' '중단: 최신 evidence·source·원본 Job gate가 필요하다.' >&2; elif ! [[ "$WINDOW_START_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ && "$WINDOW_END_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] || [ "$(date -u -d "$WINDOW_START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" != "$WINDOW_START_RFC3339" ] || [ "$(date -u -d "$WINDOW_END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" != "$WINDOW_END_RFC3339" ] || [ "$(date -u -d "$WINDOW_START_RFC3339" +%s)" -ge "$(date -u -d "$WINDOW_END_RFC3339" +%s)" ] || [ -z "$APPROVAL_ID" ]; then printf '%s\n' '중단: 승인과 canonical UTC start<end 창이 필요하다.' >&2; elif NOW="$(date -u +%s)" && { [ "$NOW" -lt "$(date -u -d "$WINDOW_START_RFC3339" +%s)" ] || [ "$NOW" -gt "$(date -u -d "$WINDOW_END_RFC3339" +%s)" ]; }; then printf '%s\n' '중단: 현재 시각이 변경 창 밖이다.' >&2; elif S="$(kubectl get cronjob "$SOURCE" -n "$NS" -o json)" && jq -e --arg u "$SOURCE_UID" --arg r "$SOURCE_RV" '.metadata.uid==$u and .metadata.resourceVersion==$r and (.spec.suspend // false)==false and ([.status.active[]?]|length)==0' <<<"$S" >/dev/null && [ "$(jq -cS '.spec.jobTemplate.spec' <<<"$S" | sha256sum | cut -d' ' -f1)" = "$SOURCE_TEMPLATE_DIGEST" ] && EXISTING="$(kubectl get job "$NEW_JOB" -n "$NS" --ignore-not-found -o name)" && [ -z "$EXISTING" ]; then CHANGE_READY=true; printf 'CHANGE_READY=true approval=%s window=%s..%s originalUID=%s sourceUID=%s sourceRV=%s newJob=%s\n' "$APPROVAL_ID" "$WINDOW_START_RFC3339" "$WINDOW_END_RFC3339" "$JOB_UID" "$SOURCE_UID" "$SOURCE_RV" "$NEW_JOB"; else printf '%s\n' '중단: source freshness·availability·template 또는 새 이름 부재 검증 실패.' >&2; fi`

**유일한 생성 명령**

`CREATED_READY=false; NORMAL_READY=false; if [ "$CHANGE_READY" != true ] || [ "$EVIDENCE_READY" != true ] || [ "$SOURCE_READY" != true ] || [ "$EXTERNAL_STATE" != missing ] || [ -z "$IDEMPOTENCY_KEY" ] || ! fresh_job; then printf '%s\n' '중단: 최신 change·evidence·source·원본 Job gate가 필요하다.' >&2; elif NOW="$(date -u +%s)" && [ "$NOW" -ge "$(date -u -d "$WINDOW_START_RFC3339" +%s)" ] && [ "$NOW" -le "$(date -u -d "$WINDOW_END_RFC3339" +%s)" ] && S="$(kubectl get cronjob "$SOURCE" -n "$NS" -o json)" && jq -e --arg u "$SOURCE_UID" --arg r "$SOURCE_RV" '.metadata.uid==$u and .metadata.resourceVersion==$r and (.spec.suspend // false)==false and ([.status.active[]?]|length)==0' <<<"$S" >/dev/null && [ "$(jq -cS '.spec.jobTemplate.spec' <<<"$S" | sha256sum | cut -d' ' -f1)" = "$SOURCE_TEMPLATE_DIGEST" ] && [ -z "$(kubectl get job "$NEW_JOB" -n "$NS" --ignore-not-found -o name)" ]; then if kubectl create job "$NEW_JOB" -n "$NS" --from="cronjob/$SOURCE"; then CREATED_READY=true; CHANGE_READY=false; printf '%s\n' 'CREATED_READY=true; 추가 생성 금지, 새 identity 등록으로 이동한다.'; else printf '%s\n' '생성 실패. 재시도 전 새 이름 존재와 외부 상태를 다시 확인한다.' >&2; fi; else CHANGE_READY=false; printf '%s\n' '중단: 실행 직전 gate가 stale이다.' >&2; fi`

같은 명령을 재실행하지 않는다. 생성 응답 손실 시 `kubectl create`를 반복하지 말고 exact `NEW_JOB` 존재와 UID를 조회한다. Kubernetes 객체 삭제는 외부 side effect rollback이 아니므로 자동 rollback 명령을 제공하지 않는다. 중단·보상은 승인된 업무 절차를 따른다.

## 새 identity 재등록과 정상화

`NORMAL_READY=false; unset NEW_JOB_JSON NEW_JOB_UID NEW_JOB_RV NEW_JOB_CREATED; if [ "$CREATED_READY" != true ]; then printf '%s\n' '중단: CREATED_READY가 아니다.' >&2; elif D="$(kubectl get job "$NEW_JOB" -n "$NS" -o json)" && jq -e --arg s "$SOURCE_UID" 'any(.metadata.ownerReferences[]?; .controller==true and .kind=="CronJob" and .uid==$s)' <<<"$D" >/dev/null; then NEW_JOB_JSON="$D"; NEW_JOB_UID="$(jq -r '.metadata.uid' <<<"$D")"; NEW_JOB_RV="$(jq -r '.metadata.resourceVersion' <<<"$D")"; NEW_JOB_CREATED="$(jq -r '.metadata.creationTimestamp' <<<"$D")"; NORMAL_READY=true; printf 'NORMAL_READY=true newJob=%s uid=%s rv=%s created=%s sourceUID=%s\n' "$NEW_JOB" "$NEW_JOB_UID" "$NEW_JOB_RV" "$NEW_JOB_CREATED" "$SOURCE_UID"; else printf '%s\n' '중단: 새 Job owner identity 검증 실패.' >&2; fi`

이후 Pod 후보는 새 Job UID owner이며 creationTime이 새 Job 이후인 단일 lifecycle만 등록한다. init/regular containerID와 모든 READY를 새로 수집한다. `Complete=True`, active 0, Warning·동일 실패 중단 뒤 승인된 외부 절차로 같은 idempotency key·입력 범위의 commit·ack·watermark·checksum과 중복 부재를 확인해야 정상화한다.

**PromQL 완성본 생성**

`if [ "$JOB_READY" != true ] || ! fresh_job; then printf '%s\n' '중단: 최신 JOB_READY가 필요하다.' >&2; else printf 'kube_job_failed{namespace=%s,job_name=%s,condition="true"}==1\n' "$(jq -Rn --arg v "$NS" '$v')" "$(jq -Rn --arg v "$JOB" '$v')"; fi`

metric 미지원·빈 결과를 성공으로 해석하지 않는다. PromQL은 Job UID·reason·외부 결과·승인을 증명하지 못한다.

보존 대상은 원본/새 Job·Pod UID/resourceVersion/owner/creation, spec/template digest, condition·attempt 시각, 외부 비민감 evidence, 승인·창·생성 응답이다. Secret·token·개인키·전체 YAML·외부 원문은 보존하지 않는다.
