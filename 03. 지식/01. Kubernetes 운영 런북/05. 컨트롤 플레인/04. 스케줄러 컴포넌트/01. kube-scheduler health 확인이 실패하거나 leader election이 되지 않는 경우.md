# kube-scheduler health 확인이 실패하거나 leader election이 되지 않는 경우

## 입력값과 빠른 수집

연결 경로는 `작업 PC/bastion Bash → kubectl/API server → 현재 scheduler Pod`, health는 `승인된 probe host → TLS/HTTP endpoint`, host 분기는 `SSH → machine-id가 일치하는 control-plane host → systemd/kubelet → CRI socket`이다. 개별 Pod의 `FailedScheduling`은 이 컴포넌트 health 경로의 증거가 아니다.

| 입력 종류 | 변수 | 출처와 gate |
|---|---|---|
| 최초 입력 | `NS`, `POD` | exact-name scheduler Pod API 조회 |
| API 확인 자동값 | Pod UID·RV·node·owner·containerID, 조건부 Lease UID·RV | 현재 API에서만 등록 |
| 계산 후보값 | escaped `POD_RE`, `ARGV_RE`, `HOLDER_RE` | bounded 출력 뒤 사용 |
| 운영자 선택값 | container, health URL, Lease namespace/name, audit subject, UTC 창, host/unit, Prometheus job | argv·inventory·audit에서 확인 |

**도구 확인**

`TOOLS_READY=false; MISSING=''; if [ -z "${BASH_VERSION:-}" ]; then printf '%s\n' '중단: Bash가 아니다.' >&2; else for T in kubectl jq curl grep sed date mktemp ssh; do command -v "$T" >/dev/null 2>&1 || MISSING="$MISSING $T"; done; [ -z "$MISSING" ] && TOOLS_READY=true; printf 'TOOLS_READY=%s missing=%s\n' "$TOOLS_READY" "${MISSING:--}"; fi`

**최초 입력과 stale 상태 전부 무효화**

`unset INPUT_READY AUTO_READY SELECTION_READY TIME_READY HEALTH_READY LEASE_READY SNAPSHOT_READY LOG_READY HOST_READY CRI_READY RBAC_READY NORMALIZE_READY POD_UID POD_RV NODE OWNER_REL CONTAINER_REL CONTAINER_ID POD_RE ARGV_RE HOLDER_RE LEASE_UID LEASE_RV HOLDER RENEW DURATION TRANSITIONS CONTAINER HEALTH_URL LEASE_NS LEASE_NAME CLIENT_SUBJECT START_RFC3339 END_RFC3339 START_EPOCH END_EPOCH SSH_HOST EXPECTED_MACHINE_ID UNIT PROM_JOB; NS='PASTE_SCHEDULER_NAMESPACE_HERE'; POD='PASTE_SCHEDULER_POD_HERE'; INPUT_READY=false; if [ "$TOOLS_READY" = true ] && J="$(kubectl get pod "$POD" -n "$NS" -o json)" && jq -e --arg n "$POD" --arg ns "$NS" '.metadata.name==$n and .metadata.namespace==$ns' <<<"$J" >/dev/null; then INPUT_READY=true; export NS POD; printf 'INPUT_READY=true ns=%s pod=%s\n' "$NS" "$POD"; else printf '%s\n' '중단: exact scheduler Pod 조회 실패.' >&2; fi`

**현재 scheduler identity·관계 등록**

`unset AUTO_READY SELECTION_READY TIME_READY HEALTH_READY LEASE_READY SNAPSHOT_READY LOG_READY HOST_READY CRI_READY RBAC_READY NORMALIZE_READY POD_UID POD_RV NODE OWNER_REL CONTAINER_REL CONTAINER_ID POD_RE ARGV_RE HOLDER_RE; AUTO_READY=false; if [ "$INPUT_READY" != true ]; then printf '%s\n' '중단: INPUT_READY가 아니다.' >&2; elif J="$(kubectl get pod "$POD" -n "$NS" -o json)" && jq -e --arg n "$POD" '.metadata.name==$n and (.metadata.uid|length>0) and (.metadata.resourceVersion|length>0) and (.spec.nodeName|length>0)' <<<"$J" >/dev/null; then POD_UID="$(jq -r '.metadata.uid' <<<"$J")"; POD_RV="$(jq -r '.metadata.resourceVersion' <<<"$J")"; NODE="$(jq -r '.spec.nodeName' <<<"$J")"; OWNER_REL="$(jq -r '[.metadata.ownerReferences[]?|(.kind+"/"+.name+"/"+.uid)]|join(",")' <<<"$J")"; [ -n "$OWNER_REL" ] || OWNER_REL='static-or-unowned'; CONTAINER_REL="$(jq -r '[.status.containerStatuses[]?|(.name+"="+(.containerID//"-")+" ready="+(.ready|tostring)+" restart="+(.restartCount|tostring))]|join("; ")' <<<"$J")"; POD_RE="$(printf '%s' "$POD"|sed 's/[][\\.^$*+?(){}|]/\\&/g')"; AUTO_READY=true; export POD_UID POD_RV NODE OWNER_REL CONTAINER_REL POD_RE; printf 'AUTO_READY=true\nUID/RV: %s/%s\nNode: %s\nOwner: %s\nContainer 관계: %s\nPOD_RE: ^%s$\n' "$POD_UID" "$POD_RV" "$NODE" "$OWNER_REL" "$CONTAINER_REL" "$POD_RE"; else printf '%s\n' '중단: scheduler 현재 identity 수집 실패.' >&2; fi`

**운영자 선택값**

`CONTAINER='PASTE_SCHEDULER_CONTAINER_HERE'; HEALTH_URL='PASTE_SCHEDULER_APPROVED_HEALTH_URL_HERE'; LEASE_NS='PASTE_SCHEDULER_LEASE_NAMESPACE_HERE'; LEASE_NAME='PASTE_SCHEDULER_LEASE_NAME_HERE'; CLIENT_SUBJECT='PASTE_SCHEDULER_AUDIT_SUBJECT_HERE'; START_RFC3339='PASTE_SCHEDULER_START_UTC_HERE'; END_RFC3339='PASTE_SCHEDULER_END_UTC_HERE'; SSH_HOST='PASTE_SCHEDULER_SSH_DESTINATION_HERE'; EXPECTED_MACHINE_ID='PASTE_SCHEDULER_EXPECTED_MACHINE_ID_HERE'; UNIT='PASTE_SCHEDULER_CONFIRMED_UNIT_HERE'; PROM_JOB='PASTE_SCHEDULER_PROMETHEUS_JOB_HERE'`

`SELECTION_READY=false; TIME_READY=false; unset HEALTH_READY LEASE_READY SNAPSHOT_READY LOG_READY HOST_READY CRI_READY RBAC_READY NORMALIZE_READY CONTAINER_ID ARGV_RE HOLDER_RE; if [ "$AUTO_READY" = true ] && J="$(kubectl get pod "$POD" -n "$NS" -o json)" && [ "$(jq -r '.metadata.uid' <<<"$J")" = "$POD_UID" ] && [ "$(jq -r '.metadata.resourceVersion' <<<"$J")" = "$POD_RV" ] && jq -e --arg c "$CONTAINER" 'any(.spec.containers[]?;.name==$c)' <<<"$J" >/dev/null; then CONTAINER_ID="$(jq -r --arg c "$CONTAINER" '.status.containerStatuses[]?|select(.name==$c)|.containerID//empty' <<<"$J")"; ARGV="$(jq -r --arg c "$CONTAINER" '.spec.containers[]|select(.name==$c)|([.command[]?,.args[]?]|join(" "))' <<<"$J")"; ARGV_RE="$(printf '%s' "$ARGV"|sed 's/[][\\.^$*+?(){}|]/\\&/g')"; [ -n "$CONTAINER_ID" ] && [ -n "$ARGV_RE" ] && SELECTION_READY=true; fi; if [ "$SELECTION_READY" = true ] && [[ "$START_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && [[ "$END_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]]; then S="$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)"; E="$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)"; START_EPOCH="$(date -u -d "$START_RFC3339" '+%s' 2>/dev/null)"; END_EPOCH="$(date -u -d "$END_RFC3339" '+%s' 2>/dev/null)"; [ "$S" = "$START_RFC3339" ] && [ "$E" = "$END_RFC3339" ] && [ "$START_EPOCH" -lt "$END_EPOCH" ] && TIME_READY=true; fi; printf 'SELECTION_READY=%s TIME_READY=%s containerID=%s argv-regex-bytes=%s\n' "$SELECTION_READY" "$TIME_READY" "${CONTAINER_ID:--}" "${#ARGV_RE}"`

canonical UTC RFC3339 roundtrip, `start < end`, evidence timestamp가 창 안이라는 세 조건을 모두 적용한다.

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | scheduler health non-2xx, process/container 비정상, 또는 확인된 scheduler Lease acquire/renew 실패가 직접 관측됨 |
| 적용 | scheduler identity/lifecycle, health, Lease, API/RBAC를 분리해 확인 가능 |
| 제외 | 특정 Pod의 placement 실패는 [[18. Pod 이벤트에 FailedScheduling이 기록된 경우]], 다수 Pod queue/bind 정체는 [[02. scheduler queue가 정체되거나 Pod binding이 진행되지 않는 경우]]로 이동 |

검색 문자열: `healthz`, `readyz`, `failed to acquire lease`, `failed to renew`, `leaderelection lost`, `Failed to list`, `Forbidden`, `x509`, `connection refused`.

## 관측별 진단 경로

| 지금 보이는 것 | 먼저 볼 것 | 그다음 어디를 볼지 | 상세 절 |
|---|---|---|---|
| health/TLS/timeout | 현재 UID와 승인 URL의 status | process·API dependency | [[#1. health·lifecycle]] |
| Lease renew/transition 이상 | exact Lease 두 snapshot | leader log·API/RBAC | [[#2. leader Lease와 로그]] |
| containerID/restart 변화 | node와 active kubelet CRI endpoint | runtime/provider | [[#3. host·CRI·RBAC]] |

### 1. health·lifecycle

`HEALTH_READY=false; unset NORMALIZE_READY; if [ "$SELECTION_READY" != true ] || [ "$TIME_READY" != true ]; then printf '%s\n' '중단: health gate 실패.' >&2; elif J="$(kubectl get pod "$POD" -n "$NS" -o json)" && [ "$(jq -r '.metadata.uid' <<<"$J")" = "$POD_UID" ] && [ "$(jq -r '.metadata.resourceVersion' <<<"$J")" = "$POD_RV" ] && [ "$(jq -r '.spec.nodeName' <<<"$J")" = "$NODE" ]; then B="$(mktemp)" || exit 1; C="$(curl -sS --max-time 10 -o "$B" -w '%{http_code}' "$HEALTH_URL")"; RC=$?; if [ "$RC" -eq 0 ] && [ "$C" -ge 200 ] 2>/dev/null && [ "$C" -lt 300 ] 2>/dev/null; then HEALTH_READY=true; printf 'HEALTH_READY=true http=%s bytes=%s uid=%s\n' "$C" "$(wc -c <"$B")" "$POD_UID"; else printf 'health 실패 curl_rc=%s http=%s\n' "$RC" "${C:-000}" >&2; fi; rm -f "$B"; else printf '%s\n' '중단: scheduler identity/relation 변경.' >&2; fi`

`CURL_CA_BUNDLE` 등 승인된 CA 경로만 사용하고 `-k`, token/key 명령행 입력, body 저장을 금지한다. 2xx는 endpoint 응답만 보장하며 leader·queue·binding 성공을 보장하지 않는다. `401/403`, `404`, `5xx`, `000`을 각각 auth, path, server/dependency, TCP/TLS로 분리한다.

### 2. leader Lease와 로그

Lease name/namespace가 argv 또는 승인 설정에서 확인된 경우만 등록한다.

`unset LEASE_READY SNAPSHOT_READY LOG_READY RBAC_READY NORMALIZE_READY LEASE_UID LEASE_RV HOLDER RENEW DURATION TRANSITIONS HOLDER_RE; LEASE_READY=false; if [ "$SELECTION_READY" != true ] || [ -z "$LEASE_NS" ] || [ -z "$LEASE_NAME" ]; then printf '%s\n' '중단: Lease selection 미확정.' >&2; elif PJ="$(kubectl get pod "$POD" -n "$NS" -o json)" && [ "$(jq -r '.metadata.uid' <<<"$PJ")" = "$POD_UID" ] && LJ="$(kubectl get lease "$LEASE_NAME" -n "$LEASE_NS" -o json)" && jq -e --arg n "$LEASE_NAME" '.metadata.name==$n and (.metadata.uid|length>0)' <<<"$LJ" >/dev/null; then LEASE_UID="$(jq -r '.metadata.uid' <<<"$LJ")"; LEASE_RV="$(jq -r '.metadata.resourceVersion' <<<"$LJ")"; HOLDER="$(jq -r '.spec.holderIdentity//empty' <<<"$LJ")"; RENEW="$(jq -r '.spec.renewTime//empty' <<<"$LJ")"; DURATION="$(jq -r '.spec.leaseDurationSeconds//empty' <<<"$LJ")"; TRANSITIONS="$(jq -r '.spec.leaseTransitions//0' <<<"$LJ")"; HOLDER_RE="$(printf '%s' "$HOLDER"|sed 's/[][\\.^$*+?(){}|]/\\&/g')"; LEASE_READY=true; printf 'LEASE_READY=true uid=%s rv=%s holder=%s renew=%s duration=%s transitions=%s holder-re=^%s$\n' "$LEASE_UID" "$LEASE_RV" "${HOLDER:--}" "${RENEW:--}" "${DURATION:--}" "$TRANSITIONS" "$HOLDER_RE"; else printf '%s\n' '중단: current Pod/Lease 조회 실패.' >&2; fi`

`SNAPSHOT_READY=false; unset NORMALIZE_READY; if [ "$LEASE_READY" != true ]; then printf '%s\n' '중단: LEASE_READY가 아니다.' >&2; elif LJ="$(kubectl get lease "$LEASE_NAME" -n "$LEASE_NS" -o json)" && [ "$(jq -r '.metadata.uid' <<<"$LJ")" = "$LEASE_UID" ]; then SNAPSHOT_READY=true; printf 'SNAPSHOT_READY=true observed=%s uid=%s old-rv=%s new-rv=%s old-holder=%s new-holder=%s old-renew=%s new-renew=%s old-transitions=%s new-transitions=%s\n' "$(date -u '+%Y-%m-%dT%H:%M:%SZ')" "$LEASE_UID" "$LEASE_RV" "$(jq -r '.metadata.resourceVersion' <<<"$LJ")" "${HOLDER:--}" "$(jq -r '.spec.holderIdentity//empty' <<<"$LJ")" "${RENEW:--}" "$(jq -r '.spec.renewTime//empty' <<<"$LJ")" "$TRANSITIONS" "$(jq -r '.spec.leaseTransitions//0' <<<"$LJ")"; else printf '%s\n' '중단: Lease 교체/조회 실패.' >&2; fi`

`LOG_READY=false; unset NORMALIZE_READY; TMP=''; if [ "$SELECTION_READY" != true ] || [ "$TIME_READY" != true ]; then printf '%s\n' '중단: log gate 실패.' >&2; elif PJ="$(kubectl get pod "$POD" -n "$NS" -o json)" && [ "$(jq -r '.metadata.uid' <<<"$PJ")" = "$POD_UID" ] && [ "$(jq -r --arg c "$CONTAINER" '.status.containerStatuses[]?|select(.name==$c)|.containerID' <<<"$PJ")" = "$CONTAINER_ID" ]; then LOG_READY=true; TMP="$(mktemp)" || exit 1; if kubectl logs "$POD" -n "$NS" -c "$CONTAINER" --timestamps --since-time="$START_RFC3339" --tail=400 >"$TMP"; then grep -Eic 'health|leader|acquire|renew|lease|Failed to list|Failed to watch|forbidden|unauthorized|x509|timeout' "$TMP" >/dev/null; FRC=$?; case "$FRC" in 0) printf 'LOG_READY=true filter=matched count=%s\n' "$(grep -Eic 'health|leader|acquire|renew|lease|Failed to list|Failed to watch|forbidden|unauthorized|x509|timeout' "$TMP")";; 1) printf '%s\n' 'LOG_READY=true filter=no-match; 오류 부재가 아니다.';; *) printf 'filter-error rc=%s\n' "$FRC" >&2; LOG_READY=false;; esac; else printf 'log-producer-failure rc=%s\n' "$?" >&2; LOG_READY=false; fi; rm -f "$TMP"; else printf '%s\n' '중단: current Pod/container 불일치.' >&2; fi`

### 3. host·CRI·RBAC

Pod lifecycle이 runtime 경계를 가리킬 때만 실행한다. SSH 인수를 positional로 전달하고 remote machine-id, active kubelet, 단일 `--container-runtime-endpoint`, exact containerID/Pod UID, socket listener/provider 관계를 검증한다. `eval`과 기본 socket 추측을 금지한다.

이 문서 안의 `CRI_READY`나 endpoint 후보는 사용하지 않는다. scheduler container lifecycle 오류가 직접 확인된 경우에만 [[04. RunPodSandbox가 runtime·OCI 오류로 실패하는 경우]] 또는 [[07. runtime 상태와 Pod container ID가 일치하지 않는 경우]]에서 current Pod/Node/containerID, 원격 machine-id, active kubelet argv 또는 엄격히 검증한 config의 exact endpoint, runtime MainPID와 socket provider를 전부 새로 등록한다. 그 런북의 direct current guard를 통과한 CRI 요약만 scheduler 증거에 연결한다.

endpoint가 0/복수이면 owner가 검증한 단일 kubelet 설정 source를 확인하기 전 `crictl`을 실행하지 않는다. socket listener process가 containerID provider와 맞지 않으면 관계 미검증으로 남긴다.

`RBAC_READY=false; unset NORMALIZE_READY; if [ "$LEASE_READY" != true ] || [ -z "$CLIENT_SUBJECT" ]; then printf '%s\n' '중단: Lease/audit subject 미확정.' >&2; elif LJ="$(kubectl get lease "$LEASE_NAME" -n "$LEASE_NS" -o json)" && [ "$(jq -r '.metadata.uid' <<<"$LJ")" = "$LEASE_UID" ]; then RBAC_READY=true; for V in get create update patch; do R="$(kubectl auth can-i "$V" leases.coordination.k8s.io -n "$LEASE_NS" --as="$CLIENT_SUBJECT" 2>&1)"; RC=$?; printf 'subject=%s verb=%s resource=leases.coordination.k8s.io namespace=%s name=%s result=%s rc=%s\n' "$CLIENT_SUBJECT" "$V" "$LEASE_NS" "$LEASE_NAME" "$R" "$RC"; if [ "$RC" -ne 0 ] || [ "$R" != yes ]; then RBAC_READY=false; fi; done; else printf '%s\n' '중단: current Lease 불일치.' >&2; fi; printf 'RBAC_READY=%s\n' "$RBAC_READY"`

subject는 audit `user.username`으로 확인한다. auditID·timestamp·verb·objectRef·response code만 보존하고 request/response body와 credential은 출력하지 않는다. `can-i=yes`는 actual scheduler client의 credential/network/TLS 성공을 보장하지 않는다.

## 변경 gate와 정상화

진단은 읽기 전용이다. restart, rollout, manifest/RBAC/certificate 변경, Lease edit/delete는 별도 변경 절차에서만 수행한다. current Pod/Lease UID·RV, holder/renew와 direct evidence, 승인 ID·실행자·window, 남은 scheduler/API capacity와 quorum, tenant 영향, 멱등성, stop/rollback을 실행 직전에 fresh 확인해야 한다. rollback도 별도 fresh gate가 필요하다.

변경 후 기존 READY를 모두 폐기하고 replacement scheduler와 Lease를 재등록한다. 새 UID·owner·node·containerID, health, stable holder/renew, API list/watch, 신규 Pod의 nodeName/PodScheduled, workload Ready·EndpointSlice·기능을 관찰 창에서 확인해야 `NORMALIZE_READY=true`다. scheduler Running, HTTP 200, holder 한 번 존재는 충분하지 않다.

## Prometheus 빠른 관측

`if [ "$AUTO_READY" != true ] || [ -z "$PROM_JOB" ]; then printf '%s\n' '중단: current scheduler identity 또는 Prometheus job 미확정.' >&2; elif J="$(kubectl get pod "$POD" -n "$NS" -o json)" && [ "$(jq -r '.metadata.uid' <<<"$J")" = "$POD_UID" ]; then PJ="${PROM_JOB//\\/\\\\}"; PJ="${PJ//\"/\\\"}"; PN="${POD//\\/\\\\}"; PN="${PN//\"/\\\"}"; NN="${NS//\\/\\\\}"; NN="${NN//\"/\\\"}"; printf 'up{job="%s",namespace="%s",pod="%s"}\nchanges(leader_election_master_status{job="%s",namespace="%s",pod="%s"}[15m])\nsum by(instance,name)(increase(leader_election_slowpath_total{job="%s",namespace="%s",pod="%s"}[15m]))\nsum by(instance,code)(rate(rest_client_requests_total{job="%s",namespace="%s",pod="%s",code=~"401|403|5.."}[5m]))\n' "$PJ" "$NN" "$PN" "$PJ" "$NN" "$PN" "$PJ" "$NN" "$PN" "$PJ" "$NN" "$PN"; else printf '%s\n' '중단: current scheduler UID 불일치.' >&2; fi`

backup leader 값 0은 정상일 수 있다. metric/label unsupported와 empty는 정상/0이 아니며 target, scrape/TLS/RBAC, time range와 label mismatch를 구분한다.

## 증거와 미검증

UTC 창, Pod UID·RV·owner·node·containerID, health status, Lease UID·RV·holder/renew/duration/transitions, CRI endpoint/provider/container relation, exact audit/RBAC tuple, 승인·rollback·새 identity·기능만 보존한다. raw body/log/audit/config와 credential은 보존하지 않는다.

실제 endpoint·Lease·unit/host·CRI·audit·metric은 환경별 미검증이다. 실제 인프라 실행과 상태 변경은 수행하지 않았다.
