# kube-controller-manager leader election이 반복 실패하는 경우

## 입력값과 빠른 수집

실행 경로는 `승인된 작업 PC/bastion Bash → kubectl → API server → coordination.k8s.io Lease`, 로그 경로는 `API server → 현재 controller-manager Pod/container` 또는 `SSH → 확인된 host → 확인된 systemd unit`이다. 같은 Bash를 유지하고 Lease 이름, holder와 Pod 이름이 같다고 가정하지 않는다.

| 입력 종류 | 변수 | 출처와 gate |
|---|---|---|
| 최초 입력 | `NS`, `POD` | 정확한 현재 controller-manager Pod를 exact-name 조회한다. |
| API 확인 자동값 | `POD_UID`, `POD_RV`, `NODE`, `OWNER_REL`, `CONTAINER_REL`, `LEASE_UID`, `LEASE_RV` | 현재 API 응답에서만 등록한다. |
| 계산 후보값 | `POD_RE`, `HOLDER_RE`, `ARGV_RE` | 특수문자를 escape한 뒤 bounded 출력한다. holder를 Pod와 자동 결합하지 않는다. |
| 운영자 선택값 | container, Lease namespace/name, 실제 client subject, UTC 창, host/unit, Prometheus job | component argv·inventory·audit를 확인한 뒤 입력한다. |

**도구 확인**

`TOOLS_READY=false; MISSING=''; if [ -z "${BASH_VERSION:-}" ]; then printf '%s\n' '중단: Bash가 아니다.' >&2; else for T in kubectl jq grep sed date mktemp ssh; do command -v "$T" >/dev/null 2>&1 || MISSING="$MISSING $T"; done; [ -z "$MISSING" ] && TOOLS_READY=true; printf 'TOOLS_READY=%s missing=%s\n' "$TOOLS_READY" "${MISSING:--}"; fi`

**최초 입력과 UID·RV·owner·containerID·node·host·relation·time·argv-regex 및 모든 downstream READY 무효화**

`unset INPUT_READY AUTO_READY SELECTION_READY TIME_READY LEASE_READY SNAPSHOT_READY LOG_READY HOST_READY RBAC_READY NORMALIZE_READY POD_UID POD_RV NODE OWNER_REL CONTAINER_REL CONTAINER_ID POD_RE HOLDER_RE ARGV_RE LEASE_UID LEASE_RV HOLDER RENEW ACQUIRE DURATION TRANSITIONS START_RFC3339 END_RFC3339 START_EPOCH END_EPOCH CONTAINER LEASE_NS LEASE_NAME CLIENT_SUBJECT SSH_HOST EXPECTED_MACHINE_ID UNIT PROM_JOB; NS='PASTE_LEADER_CONTROLLER_NAMESPACE_HERE'; POD='PASTE_LEADER_CONTROLLER_POD_HERE'; INPUT_READY=false; if [ "$TOOLS_READY" = true ] && J="$(kubectl get pod "$POD" -n "$NS" -o json)" && jq -e --arg n "$POD" --arg ns "$NS" '.metadata.name==$n and .metadata.namespace==$ns' <<<"$J" >/dev/null; then INPUT_READY=true; export NS POD; printf 'INPUT_READY=true ns=%s pod=%s\n' "$NS" "$POD"; else printf '%s\n' '중단: exact Pod 입력 또는 API 조회 실패.' >&2; fi`

**현재 Pod 자동값 등록**

`unset AUTO_READY SELECTION_READY TIME_READY LEASE_READY SNAPSHOT_READY LOG_READY HOST_READY RBAC_READY NORMALIZE_READY POD_UID POD_RV NODE OWNER_REL CONTAINER_REL CONTAINER_ID POD_RE HOLDER_RE ARGV_RE; AUTO_READY=false; if [ "$INPUT_READY" != true ]; then printf '%s\n' '중단: INPUT_READY가 아니다.' >&2; elif J="$(kubectl get pod "$POD" -n "$NS" -o json)" && jq -e --arg n "$POD" '.metadata.name==$n and (.metadata.uid|length>0) and (.metadata.resourceVersion|length>0) and (.spec.nodeName|length>0)' <<<"$J" >/dev/null; then POD_UID="$(jq -r '.metadata.uid' <<<"$J")"; POD_RV="$(jq -r '.metadata.resourceVersion' <<<"$J")"; NODE="$(jq -r '.spec.nodeName' <<<"$J")"; OWNER_REL="$(jq -r '[.metadata.ownerReferences[]?|(.kind+"/"+.name+"/"+.uid)]|join(",")' <<<"$J")"; [ -n "$OWNER_REL" ] || OWNER_REL='static-or-unowned'; CONTAINER_REL="$(jq -r '[.status.containerStatuses[]?|(.name+"="+(.containerID//"-")+" ready="+(.ready|tostring)+" restart="+(.restartCount|tostring))]|join("; ")' <<<"$J")"; POD_RE="$(printf '%s' "$POD"|sed 's/[][\\.^$*+?(){}|]/\\&/g')"; AUTO_READY=true; export POD_UID POD_RV NODE OWNER_REL CONTAINER_REL POD_RE; printf 'AUTO_READY=true\nUID: %s\nRV: %s\nNode: %s\nOwner: %s\nContainers: %s\nPOD_RE: ^%s$\n' "$POD_UID" "$POD_RV" "$NODE" "$OWNER_REL" "$CONTAINER_REL" "$POD_RE"; else printf '%s\n' '중단: 현재 Pod 관계 수집 실패.' >&2; fi`

**운영자 선택값**

`CONTAINER='PASTE_LEADER_CONTAINER_HERE'; LEASE_NS='PASTE_CONFIRMED_LEASE_NAMESPACE_HERE'; LEASE_NAME='PASTE_CONFIRMED_LEASE_NAME_HERE'; CLIENT_SUBJECT='PASTE_AUDIT_CONFIRMED_CLIENT_SUBJECT_HERE'; START_RFC3339='PASTE_LEADER_START_UTC_HERE'; END_RFC3339='PASTE_LEADER_END_UTC_HERE'; SSH_HOST='PASTE_LEADER_SSH_DESTINATION_HERE'; EXPECTED_MACHINE_ID='PASTE_LEADER_EXPECTED_MACHINE_ID_HERE'; UNIT='PASTE_LEADER_CONFIRMED_UNIT_HERE'; PROM_JOB='PASTE_LEADER_PROMETHEUS_JOB_HERE'`

`SELECTION_READY=false; TIME_READY=false; unset LEASE_READY SNAPSHOT_READY LOG_READY HOST_READY RBAC_READY NORMALIZE_READY CONTAINER_ID ARGV_RE HOLDER_RE; if [ "$AUTO_READY" != true ]; then printf '%s\n' '중단: AUTO_READY가 아니다.' >&2; elif J="$(kubectl get pod "$POD" -n "$NS" -o json)" && [ "$(jq -r '.metadata.uid' <<<"$J")" = "$POD_UID" ] && [ "$(jq -r '.metadata.resourceVersion' <<<"$J")" = "$POD_RV" ] && jq -e --arg c "$CONTAINER" 'any(.spec.containers[]?;.name==$c)' <<<"$J" >/dev/null && [ -n "$LEASE_NS" ] && [ -n "$LEASE_NAME" ]; then CONTAINER_ID="$(jq -r --arg c "$CONTAINER" '.status.containerStatuses[]?|select(.name==$c)|.containerID//empty' <<<"$J")"; ARGV="$(jq -r --arg c "$CONTAINER" '.spec.containers[]|select(.name==$c)|([.command[]?,.args[]?]|join(" "))' <<<"$J")"; ARGV_RE="$(printf '%s' "$ARGV"|sed 's/[][\\.^$*+?(){}|]/\\&/g')"; [ -n "$CONTAINER_ID" ] && [ -n "$ARGV_RE" ] && SELECTION_READY=true; fi; if [ "$SELECTION_READY" = true ] && [[ "$START_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && [[ "$END_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]]; then S="$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)"; E="$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)"; START_EPOCH="$(date -u -d "$START_RFC3339" '+%s' 2>/dev/null)"; END_EPOCH="$(date -u -d "$END_RFC3339" '+%s' 2>/dev/null)"; [ "$S" = "$START_RFC3339" ] && [ "$E" = "$END_RFC3339" ] && [ "$START_EPOCH" -lt "$END_EPOCH" ] && TIME_READY=true; fi; printf 'SELECTION_READY=%s TIME_READY=%s container=%s containerID=%s argv-regex-bytes=%s\n' "$SELECTION_READY" "$TIME_READY" "$CONTAINER" "${CONTAINER_ID:--}" "${#ARGV_RE}"`

시간은 canonical UTC RFC3339 roundtrip과 `start < end`를 만족해야 한다. Lease·Pod·log·audit evidence가 이 창에 포함되지 않으면 인과관계로 연결하지 않는다.

**Lease 현재 identity 등록**

`unset LEASE_READY SNAPSHOT_READY LOG_READY HOST_READY RBAC_READY NORMALIZE_READY LEASE_UID LEASE_RV HOLDER RENEW ACQUIRE DURATION TRANSITIONS HOLDER_RE; LEASE_READY=false; if [ "$SELECTION_READY" != true ] || [ "$TIME_READY" != true ]; then printf '%s\n' '중단: 선택·시간 gate 실패.' >&2; elif PJ="$(kubectl get pod "$POD" -n "$NS" -o json)" && [ "$(jq -r '.metadata.uid' <<<"$PJ")" = "$POD_UID" ] && LJ="$(kubectl get lease "$LEASE_NAME" -n "$LEASE_NS" -o json)" && jq -e --arg n "$LEASE_NAME" --arg ns "$LEASE_NS" '.metadata.name==$n and .metadata.namespace==$ns and (.metadata.uid|length>0) and (.metadata.resourceVersion|length>0)' <<<"$LJ" >/dev/null; then LEASE_UID="$(jq -r '.metadata.uid' <<<"$LJ")"; LEASE_RV="$(jq -r '.metadata.resourceVersion' <<<"$LJ")"; HOLDER="$(jq -r '.spec.holderIdentity//empty' <<<"$LJ")"; RENEW="$(jq -r '.spec.renewTime//empty' <<<"$LJ")"; ACQUIRE="$(jq -r '.spec.acquireTime//empty' <<<"$LJ")"; DURATION="$(jq -r '.spec.leaseDurationSeconds//empty' <<<"$LJ")"; TRANSITIONS="$(jq -r '.spec.leaseTransitions//0' <<<"$LJ")"; HOLDER_RE="$(printf '%s' "$HOLDER"|sed 's/[][\\.^$*+?(){}|]/\\&/g')"; LEASE_READY=true; export LEASE_UID LEASE_RV HOLDER RENEW ACQUIRE DURATION TRANSITIONS HOLDER_RE; printf 'LEASE_READY=true\nLease UID: %s\nRV: %s\nHolder: %s\nAcquire: %s\nRenew: %s\nDuration: %s\nTransitions: %s\nHOLDER_RE: ^%s$\n' "$LEASE_UID" "$LEASE_RV" "${HOLDER:--}" "${ACQUIRE:--}" "${RENEW:--}" "${DURATION:--}" "$TRANSITIONS" "$HOLDER_RE"; else printf '%s\n' '중단: 현재 Pod 또는 exact Lease 조회 실패. Lease 값은 무효다.' >&2; fi`

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | `failed to acquire lease`, `failed to renew`, `leaderelection lost` 또는 Lease renew 정지/transition 반복이 직접 관측된 경우 |
| 적용 | 정확한 Lease UID·RV·holder·renew와 실제 component instance lifecycle을 비교할 수 있는 경우 |
| 제외 | health/process만 실패하면 [[01. kube-controller-manager health 확인이 실패하거나 프로세스가 비정상인 경우]], 단일 workload 정체는 해당 workload 런북으로 이동 |

검색 문자열: `failed to acquire lease`, `failed to renew`, `leaderelection lost`, `holderIdentity`, `renewTime`, `leaseTransitions`, `Forbidden`, `Unauthorized`, `x509`, `clock skew`.

## 관측별 진단 경로

| 지금 보이는 것 | 먼저 볼 것 | 그다음 어디를 볼지 | 상세 절 |
|---|---|---|---|
| renewTime·RV 정지 | 같은 UID의 두 snapshot과 API readiness | component 로그·API 경로 | [[#1. 두 snapshot과 시간 관계]] |
| holder·transition 반복 변화 | holder와 Pod/process lifecycle | restart·duplicate instance·clock | [[#2. 로그와 host lifecycle]] |
| `Forbidden`·`Unauthorized` | audit subject·verb·resource | RBAC·credential/TLS | [[#3. 정확한 RBAC·audit 경계]] |

### 1. 두 snapshot과 시간 관계

관찰 간격은 `leaseDurationSeconds`보다 충분히 짧고 운영 정책 안이어야 한다. 다음 명령은 기존 snapshot을 덮기 전에 current UID를 다시 확인한다.

`SNAPSHOT_READY=false; unset NORMALIZE_READY; if [ "$LEASE_READY" != true ]; then printf '%s\n' '중단: LEASE_READY가 아니다.' >&2; elif LJ="$(kubectl get lease "$LEASE_NAME" -n "$LEASE_NS" -o json)" && [ "$(jq -r '.metadata.uid' <<<"$LJ")" = "$LEASE_UID" ]; then NOW="$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; NEW_RV="$(jq -r '.metadata.resourceVersion' <<<"$LJ")"; NEW_HOLDER="$(jq -r '.spec.holderIdentity//empty' <<<"$LJ")"; NEW_RENEW="$(jq -r '.spec.renewTime//empty' <<<"$LJ")"; NEW_TRANSITIONS="$(jq -r '.spec.leaseTransitions//0' <<<"$LJ")"; SNAPSHOT_READY=true; printf 'SNAPSHOT_READY=true observed=%s uid=%s old-rv=%s new-rv=%s old-holder=%s new-holder=%s old-renew=%s new-renew=%s old-transitions=%s new-transitions=%s\n' "$NOW" "$LEASE_UID" "$LEASE_RV" "$NEW_RV" "${HOLDER:--}" "${NEW_HOLDER:--}" "${RENEW:--}" "${NEW_RENEW:--}" "$TRANSITIONS" "$NEW_TRANSITIONS"; else printf '%s\n' '중단: Lease 교체/조회 실패. 이전 snapshot은 현재 판단에 사용하지 않는다.' >&2; fi`

RV와 renew가 진행하면 해당 관찰 순간의 API update는 있었다. 안정된 leader, 모든 controller 기능과 clock 정상은 보장하지 않는다. `renewTime`을 parse해 `[START_RFC3339, END_RFC3339]` 밖이면 이 incident 증거에서 제외한다. API 조회 실패는 renew 정지로 해석하지 않는다.

### 2. 로그와 host lifecycle

`LOG_READY=false; unset NORMALIZE_READY; TMP=''; if [ "$LEASE_READY" != true ] || [ "$TIME_READY" != true ]; then printf '%s\n' '중단: log gate 실패.' >&2; elif PJ="$(kubectl get pod "$POD" -n "$NS" -o json)" && [ "$(jq -r '.metadata.uid' <<<"$PJ")" = "$POD_UID" ] && [ "$(jq -r --arg c "$CONTAINER" '.status.containerStatuses[]?|select(.name==$c)|.containerID' <<<"$PJ")" = "$CONTAINER_ID" ]; then LOG_READY=true; TMP="$(mktemp)" || exit 1; if kubectl logs "$POD" -n "$NS" -c "$CONTAINER" --timestamps --since-time="$START_RFC3339" --tail=400 >"$TMP"; then grep -Eic 'acquire|renew|lease|leader|resource lock|forbidden|unauthorized|x509|timeout|deadline|clock' "$TMP" >/dev/null; FRC=$?; case "$FRC" in 0) printf 'LOG_READY=true filter=matched count=%s\n' "$(grep -Eic 'acquire|renew|lease|leader|resource lock|forbidden|unauthorized|x509|timeout|deadline|clock' "$TMP")";; 1) printf '%s\n' 'LOG_READY=true filter=no-match; 오류 없음으로 해석하지 않는다.';; *) printf 'filter-error rc=%s\n' "$FRC" >&2; LOG_READY=false;; esac; else printf 'log-producer-failure rc=%s\n' "$?" >&2; LOG_READY=false; fi; rm -f "$TMP"; else printf '%s\n' '중단: current Pod/container identity 불일치.' >&2; fi`

systemd 경로가 실제 배포 방식일 때만 아래를 사용한다. SSH destination과 machine-id를 대조하고 positional argument로 unit·시간·필터를 안전하게 전달한다. `eval`은 사용하지 않는다.

`HOST_READY=false; unset NORMALIZE_READY; if [ "$TIME_READY" != true ]; then printf '%s\n' '중단: TIME_READY가 아니다.' >&2; else ssh -- "$SSH_HOST" bash -c 'expected=$1; unit=$2; start=$3; end=$4; actual=$(cat /etc/machine-id 2>/dev/null) || exit 40; [ "$actual" = "$expected" ] || exit 41; [ "$(systemctl is-active "$unit")" = active ] || exit 42; out=$(journalctl -u "$unit" --since "$start" --until "$end" --no-pager -n 500); prc=$?; [ "$prc" -eq 0 ] || { printf "journal-producer-failure rc=%s\n" "$prc" >&2; exit 43; }; grep -Eic "acquire|renew|lease|leader|forbidden|unauthorized|x509|timeout|deadline|clock" <<<"$out" >/dev/null; frc=$?; case "$frc" in 0) printf "HOST_READY=true filter=matched count=%s\n" "$(grep -Eic "acquire|renew|lease|leader|forbidden|unauthorized|x509|timeout|deadline|clock" <<<"$out")";; 1) printf "%s\n" "HOST_READY=true filter=no-match";; *) printf "filter-error rc=%s\n" "$frc" >&2; exit 44;; esac' bash "$EXPECTED_MACHINE_ID" "$UNIT" "$START_RFC3339" "$END_RFC3339"; RC=$?; [ "$RC" -eq 0 ] && HOST_READY=true || printf 'SSH/journal 검증 실패 rc=%s\n' "$RC" >&2; fi`

### 3. 정확한 RBAC·audit 경계

실제 audit에서 `CLIENT_SUBJECT`가 component request의 subject이고 Lease namespace/name에 대한 verb가 확인된 경우에만 실행한다.

`RBAC_READY=false; unset NORMALIZE_READY; if [ "$LEASE_READY" != true ] || [ -z "$CLIENT_SUBJECT" ]; then printf '%s\n' '중단: Lease 또는 subject 미확정.' >&2; elif LJ="$(kubectl get lease "$LEASE_NAME" -n "$LEASE_NS" -o json)" && [ "$(jq -r '.metadata.uid' <<<"$LJ")" = "$LEASE_UID" ]; then RBAC_READY=true; for VERB in get create update patch; do RESULT="$(kubectl auth can-i "$VERB" leases.coordination.k8s.io -n "$LEASE_NS" --as="$CLIENT_SUBJECT" 2>&1)"; RC=$?; printf 'subject=%s verb=%s resource=leases.coordination.k8s.io namespace=%s name=%s result=%s rc=%s\n' "$CLIENT_SUBJECT" "$VERB" "$LEASE_NS" "$LEASE_NAME" "$RESULT" "$RC"; if [ "$RC" -ne 0 ] || [ "$RESULT" != yes ]; then RBAC_READY=false; fi; done; else printf '%s\n' '중단: current Lease identity 불일치.' >&2; fi; printf 'RBAC_READY=%s\n' "$RBAC_READY"`

`can-i=yes`는 impersonation 기준 RBAC이며 component credential·TLS·network와 실제 request 성공을 보장하지 않는다. audit에는 같은 창의 `auditID`, `stageTimestamp`, `user.username`, `verb`, `objectRef.apiGroup/resource/namespace/name`, `responseStatus.code`가 정확히 일치해야 한다. request/response body, token, certificate와 Secret은 출력·보존하지 않는다.

## 변경 gate와 정상화

Lease edit/delete/patch, leader 강제 전환과 임의 restart는 이 런북의 진단 명령이 아니다. 별도 변경 전 current Pod/Lease UID·RV·holder·renew, 직접 오류, 승인 ID·주체·창, HA instance와 quorum/capacity, 영향 workload, 멱등성, 중단 기준, 실행 가능한 rollback을 검증한다. 변경과 rollback은 각각 실행 직전에 identity/evidence/approval/window/quorum을 다시 읽어 별도 `MUTATE_READY` 또는 `ROLLBACK_READY`를 만들어야 하며 stale gate를 재사용하지 않는다.

변경 후 모든 변수를 무효화하고 새 Pod/process와 Lease를 다시 등록한다. 새 holder identity와 현재 instance 관계, renew/RV 진행, transition 안정, acquire/renew 오류 부재를 관찰 창에서 확인한 뒤 workload status와 실제 기능까지 회복해야 `NORMALIZE_READY=true`다. holder 존재나 renew 1회는 정상화가 아니다.

## Prometheus 빠른 관측

`if [ "$LEASE_READY" != true ] || [ -z "$PROM_JOB" ]; then printf '%s\n' '중단: current Lease 또는 Prometheus job 미확정.' >&2; elif LJ="$(kubectl get lease "$LEASE_NAME" -n "$LEASE_NS" -o json)" && [ "$(jq -r '.metadata.uid' <<<"$LJ")" = "$LEASE_UID" ]; then PJ="${PROM_JOB//\\/\\\\}"; PJ="${PJ//\"/\\\"}"; PN="${POD//\\/\\\\}"; PN="${PN//\"/\\\"}"; NN="${NS//\\/\\\\}"; NN="${NN//\"/\\\"}"; printf 'changes(leader_election_master_status{job="%s",namespace="%s",pod="%s"}[15m])\nsum by(instance,name)(increase(leader_election_slowpath_total{job="%s",namespace="%s",pod="%s"}[15m]))\nsum by(instance,code)(rate(rest_client_requests_total{job="%s",namespace="%s",pod="%s",code=~"401|403|5.."}[5m]))\n' "$PJ" "$NN" "$PN" "$PJ" "$NN" "$PN" "$PJ" "$NN" "$PN"; else printf '%s\n' '중단: current Lease UID 불일치.' >&2; fi`

backup instance의 leader 값 0은 정상일 수 있다. metric/label이 unsupported이거나 빈 결과면 0으로 바꾸지 말고 target·scrape·권한·time range·label과 build 지원 여부를 구분한다. Lease UID/holder/renew와 실제 instance 관계 없이 PromQL로 leader 장애를 확정하지 않는다.

## 증거와 미검증

Pod UID·RV·owner·node·containerID, Lease UID·RV·holder/acquire/renew/duration/transitions의 두 snapshot, canonical UTC 창, 정확한 subject·verb·resource·namespace·auditID, 분류 count, 승인·변경·rollback·새 identity·기능 결과만 보존한다. raw log/audit, token, kubeconfig, key와 개인정보는 보존하지 않는다.

실제 Lease 설정·holder 형식, 배포 방식, unit/host, clock source, audit와 metric은 환경별 미검증이다. 실제 인프라 실행과 상태 변경은 수행하지 않았다.
