# LoadBalancer Service의 EXTERNAL-IP가 Pending이거나 할당되지 않는 경우

## 입력값과 빠른 수집

실행 host는 승인된 context가 있는 작업 PC 또는 bastion의 같은 Bash다. hop은 `Service spec/generation → load-balancer controller reconcile → provider account/region resource allocation → address pool/quota → status.loadBalancer.ingress 기록`이다. `Pending`은 provider 요청 성공, VIP 광고, listener 또는 backend health를 보장하지 않는다.

| lifecycle | 값 | 계약 |
|---|---|---|
| 최초 입력 | `CONTEXT`, `NS`, `SERVICE` | exact Service를 한 번 입력한다. |
| API 자동값 | `SERVICE_UID/RV/generation`, ports, policy, selector, EndpointSlice 관계 | 전체 수집 성공 뒤에만 등록한다. |
| 계산 후보 | `SERVICE_RE`, UID Event selector, 관측 UTC | escape·bounded 확인 뒤 사용한다. |
| 운영자 선택값 | provider/account/region/resource ID, controller Pod/container, UTC window | inventory·Event와 같은 Service를 가리킬 때만 등록한다. |

`reset_lb_state; CONTEXT='PASTE_CONTEXT_HERE'; NS='PASTE_NAMESPACE_HERE'; SERVICE='PASTE_SERVICE_HERE'; INPUT_READY=false; if [ "$TOOLS_READY" != true ]; then printf '%s\n' '중단: TOOLS_READY가 아니다.' >&2; elif kubectl --context "$CONTEXT" get service "$SERVICE" -n "$NS" -o name >/dev/null 2>&1; then INPUT_READY=true; export CONTEXT NS SERVICE; printf 'INPUT_READY=true context=%s namespace=%s service=%s\n' "$CONTEXT" "$NS" "$SERVICE"; else printf '%s\n' '중단: exact Service 조회 실패' >&2; fi`

재수집 전에 `reset_lb_state`가 UID·RV·owner·containerID·node·host·provider relation·time·argv/regex와 downstream gate를 전부 무효화한다. 각 `PASTE_..._HERE`는 한 번만 나타나며 실패 뒤 stale 값은 없다.

`reset_lb_state; if [ "$INPUT_READY" != true ]; then printf '%s\n' '중단: INPUT_READY가 아니다.' >&2; elif S="$(kubectl --context "$CONTEXT" get service "$SERVICE" -n "$NS" -o json)" && E="$(kubectl --context "$CONTEXT" get endpointslice -n "$NS" -l "kubernetes.io/service-name=$SERVICE" -o json)" && U="$(jq -r '.metadata.uid // empty' <<<"$S")" && R="$(jq -r '.metadata.resourceVersion // empty' <<<"$S")" && [ -n "$U" ] && [ -n "$R" ] && jq -e '.spec.type=="LoadBalancer"' <<<"$S" >/dev/null && jq -e --arg n "$SERVICE" --arg u "$U" 'all(.items[]?; any(.metadata.ownerReferences[]?; .kind=="Service" and .name==$n and .uid==$u))' <<<"$E" >/dev/null; then SERVICE_JSON="$S"; EPS_JSON="$E"; SERVICE_UID="$U"; SERVICE_RV="$R"; SERVICE_GENERATION="$(jq -r '.metadata.generation // 0' <<<"$S")"; SERVICE_OWNER="$(jq -r '[.metadata.ownerReferences[]?|"\(.kind)/\(.name)/\(.uid)"]|join(",")' <<<"$S")"; SERVICE_RE="^$(printf '%s' "$SERVICE" | sed 's/[][\\.^$*+?(){}|]/\\&/g')$"; EVENT_SELECTOR="involvedObject.kind=Service,involvedObject.name=$SERVICE,involvedObject.uid=$SERVICE_UID"; OBSERVED_AT="$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; AUTO_READY=true; RELATION_READY=true; jq -r --arg observed "$OBSERVED_AT" '"=== Service ===\nUID: \(.metadata.uid)\nRV: \(.metadata.resourceVersion)\nGeneration: \(.metadata.generation)\nType: \(.spec.type)\nIngress: \([.status.loadBalancer.ingress[]?]|tostring)\nClass: \(.spec.loadBalancerClass // \"-\")\nNodePort 할당: \(.spec.allocateLoadBalancerNodePorts // true)\nPolicy: \(.spec.externalTrafficPolicy // \"-\")\nPort 관계: " + ([.spec.ports[]?|"service=\(.name // \"-\"):\(.port)/\(.protocol) target=\(.targetPort) nodePort=\(.nodePort // \"-\")"]|join("; ")) + "\n관측 UTC: \($observed)"' <<<"$SERVICE_JSON"; jq -r '.items[] | "=== EndpointSlice: \(.metadata.name) ===\nUID: \(.metadata.uid)\nEndpoint 관계: " + ([.endpoints[]?|"target=\(.targetRef.name // \"-\") uid=\(.targetRef.uid // \"-\") node=\(.nodeName // \"-\") ready=\(.conditions.ready // \"nil\") serving=\(.conditions.serving // \"nil\")"]|join("; "))' <<<"$EPS_JSON"; printf 'AUTO_READY=true RELATION_READY=true SERVICE_RE=%s\n' "$SERVICE_RE"; else reset_lb_state; printf '%s\n' '중단: LoadBalancer Service/EndpointSlice owner identity 수집 실패' >&2; fi`

`lb_fresh(){ local j u r g; AUTO_LATEST_READY=false; j="$(kubectl --context "$CONTEXT" get service "$SERVICE" -n "$NS" -o json)" || { reset_lb_state; return 1; }; u="$(jq -r '.metadata.uid // empty' <<<"$j")"; r="$(jq -r '.metadata.resourceVersion // empty' <<<"$j")"; g="$(jq -r '.metadata.generation // 0' <<<"$j")"; if [ "$u" = "$SERVICE_UID" ] && [ "$g" = "$SERVICE_GENERATION" ] && [ -n "$r" ] && jq -e '.spec.type=="LoadBalancer"' <<<"$j" >/dev/null; then SERVICE_JSON="$j"; SERVICE_RV="$r"; AUTO_LATEST_READY=true; printf 'AUTO_LATEST_READY=true uid=%s rv=%s generation=%s\n' "$u" "$r" "$g"; return 0; else reset_lb_state; return 1; fi; }`

`lb_relation_fresh(){ local e; RELATION_LATEST_READY=false; e="$(kubectl --context "$CONTEXT" get endpointslice -n "$NS" -l "kubernetes.io/service-name=$SERVICE" -o json)" || { RELATION_READY=false; return 1; }; if jq -e --arg n "$SERVICE" --arg u "$SERVICE_UID" 'all(.items[]?; any(.metadata.ownerReferences[]?; .kind=="Service" and .name==$n and .uid==$u))' <<<"$e" >/dev/null; then EPS_JSON="$e"; RELATION_LATEST_READY=true; return 0; else RELATION_READY=false; unset EPS_JSON; return 1; fi; }`

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | `type=LoadBalancer`이고 `status.loadBalancer.ingress`가 비어 있거나 UI가 Pending으로 표시한다. |
| 적용 | Service UID/generation과 provider reconcile resource를 정확히 연결할 수 있다. |
| 제외 | 주소가 기록됐지만 연결 실패면 [[02. VIP·External IP가 할당됐지만 외부에서 연결되지 않는 경우]], NodePort 자체 실패는 [[09. 외부 트래픽/02. NodePort·외부 트래픽 정책/01. NodePort에 연결할 수 없거나 일부 노드에서만 실패하는 경우]]다. |

검색 문자열: `EnsuringLoadBalancer`, `EnsuredLoadBalancer`, `SyncLoadBalancerFailed`, `allocation`, `quota`, `pool`, `status.loadBalancer.ingress`.

## UID 제한 Event와 allocation 판정

`EVENT_READY=false; if [ "$AUTO_READY" != true ] || [ "$RELATION_READY" != true ] || ! lb_fresh || ! lb_relation_fresh; then printf '%s\n' '중단: latest Service/relation gate 실패' >&2; elif EV="$(kubectl --context "$CONTEXT" get events -n "$NS" --field-selector "$EVENT_SELECTOR" --sort-by=.metadata.creationTimestamp -o json)"; then EVENT_READY=true; jq -r 'if (.items|length)==0 then "Service Event 없음: 정상 또는 오류 없음으로 해석하지 않음" else .items[]|"reason=\(.reason // \"-\") count=\(.count // 0) first=\(.firstTimestamp // \"-\") last=\(.lastTimestamp // .eventTime // .series.lastObservedTime // \"-\") message=\(.message // \"-\")" end' <<<"$EV"; printf 'EVENT_READY=true service_uid=%s observed=%s\n' "$SERVICE_UID" "$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; else printf '%s\n' '중단: Event producer 실패' >&2; fi`

`type`, class/annotations, `allocateLoadBalancerNodePorts`, port/nodePort와 endpoint readiness를 먼저 대조한다. endpoint 부재가 provider address allocation을 항상 막는다고 일반화하지 않으며 구현 계약을 확인한다. Event 빈 결과는 audit/event retention·RBAC·controller 외부 실행을 구분한다.

## provider와 controller exact identity

`PROVIDER='PASTE_PROVIDER_HERE'; ACCOUNT='PASTE_PROVIDER_ACCOUNT_HERE'; REGION='PASTE_PROVIDER_REGION_HERE'; LB_RESOURCE_ID='PASTE_LB_RESOURCE_ID_HERE'; PROVIDER_STATE='PASTE_PROVIDER_STATE_HERE'; PROVIDER_EVIDENCE_ID='PASTE_PROVIDER_EVIDENCE_ID_HERE'; CTRL_NS='PASTE_CONTROLLER_NAMESPACE_HERE'; CTRL_POD='PASTE_CONTROLLER_POD_HERE'; CTRL_CONTAINER='PASTE_CONTROLLER_CONTAINER_HERE'; START_RFC3339='PASTE_START_RFC3339_HERE'; END_RFC3339='PASTE_END_RFC3339_HERE'`

`TIME_READY=false; if [[ "$START_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && [[ "$END_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && A="$(date -u -d "$START_RFC3339" +%s 2>/dev/null)" && B="$(date -u -d "$END_RFC3339" +%s 2>/dev/null)" && [ "$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ')" = "$START_RFC3339" ] && [ "$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ')" = "$END_RFC3339" ] && [ "$A" -lt "$B" ]; then TIME_READY=true; else printf '%s\n' '중단: canonical UTC window 오류' >&2; fi`

`PROVIDER_READY=false; CTRL_READY=false; FUNCTION_READY=false; NORMAL_READY=false; unset CTRL_UID CTRL_RV CTRL_OWNER CTRL_NODE CTRL_CONTAINER_ID FUNCTION_EVIDENCE_ID; if [ "$EVENT_READY" != true ] || [ "$TIME_READY" != true ] || ! lb_fresh || ! lb_relation_fresh; then printf '%s\n' '중단: Event/time/latest Service/relation gate 실패' >&2; elif [ -z "$PROVIDER" ] || [ -z "$ACCOUNT" ] || [ -z "$REGION" ] || [ -z "$LB_RESOURCE_ID" ] || [ -z "$PROVIDER_STATE" ] || [ -z "$PROVIDER_EVIDENCE_ID" ]; then printf '%s\n' '중단: provider exact identity/state/evidence가 비었다.' >&2; elif C="$(kubectl --context "$CONTEXT" get pod "$CTRL_POD" -n "$CTRL_NS" -o json)" && CU="$(jq -r '.metadata.uid // empty' <<<"$C")" && CI="$(jq -r --arg c "$CTRL_CONTAINER" '[.status.containerStatuses[]?|select(.name==$c and .state.running!=null)|.containerID][0] // empty' <<<"$C")" && [ -n "$CU" ] && [ -n "$CI" ]; then CTRL_UID="$CU"; CTRL_RV="$(jq -r '.metadata.resourceVersion' <<<"$C")"; CTRL_OWNER="$(jq -r '[.metadata.ownerReferences[]?|"\(.kind)/\(.name)/\(.uid)"]|join(",")' <<<"$C")"; CTRL_NODE="$(jq -r '.spec.nodeName' <<<"$C")"; CTRL_CONTAINER_ID="$CI"; PROVIDER_READY=true; CTRL_READY=true; printf 'PROVIDER_READY=true provider=%s account=%s region=%s resource=%s state=%s evidence=%s service_uid=%s\nCTRL_READY=true uid=%s container_id=%s node=%s\n' "$PROVIDER" "$ACCOUNT" "$REGION" "$LB_RESOURCE_ID" "$PROVIDER_STATE" "$PROVIDER_EVIDENCE_ID" "$SERVICE_UID" "$CTRL_UID" "$CTRL_CONTAINER_ID" "$CTRL_NODE"; else printf '%s\n' '중단: controller exact identity 실패' >&2; fi`

provider/account/region/resource ID는 승인된 inventory·controller Event·provider record에서 같은 Service UID를 가리켜야 한다. 그 관계가 없으면 provider CLI/API, pool, quota를 조회하지 않고 owner에게 이관한다.

`ctrl_fresh(){ local j; CTRL_LATEST_READY=false; j="$(kubectl --context "$CONTEXT" get pod "$CTRL_POD" -n "$CTRL_NS" -o json)" || return 1; jq -e --arg u "$CTRL_UID" --arg c "$CTRL_CONTAINER" --arg i "$CTRL_CONTAINER_ID" '.metadata.uid==$u and .metadata.deletionTimestamp==null and any(.status.containerStatuses[]?;.name==$c and .state.running!=null and .containerID==$i)' <<<"$j" >/dev/null || { CTRL_READY=false; LOG_READY=false; unset CTRL_UID CTRL_RV CTRL_OWNER CTRL_NODE CTRL_CONTAINER_ID; return 1; }; CTRL_LATEST_READY=true; }`

`LOG_READY=false; if [ "$PROVIDER_READY" != true ] || [ "$CTRL_READY" != true ] || [ "$TIME_READY" != true ] || ! lb_fresh || ! lb_relation_fresh || ! ctrl_fresh; then printf '%s\n' '중단: provider/time/latest Service/relation/controller gate 실패' >&2; elif AUTH="$(kubectl --context "$CONTEXT" auth can-i get pods "$CTRL_POD" --subresource=log -n "$CTRL_NS" 2>&1)" && [ "$AUTH" = yes ]; then T="$(mktemp)" || exit 1; W="$(mktemp)" || { rm -f "$T"; exit 1; }; kubectl --context "$CONTEXT" logs "$CTRL_POD" -n "$CTRL_NS" -c "$CTRL_CONTAINER" --timestamps --since-time="$START_RFC3339" --tail=300 >"$T"; P_RC=$?; if [ "$P_RC" -eq 0 ]; then awk -v end="$END_RFC3339" 'substr($1,1,19) "Z" <= end' "$T" >"$W"; W_RC=$?; else W_RC=not-run; fi; if [ "$P_RC" -eq 0 ] && [ "$W_RC" -eq 0 ]; then grep -E -i -C 3 "$SERVICE_UID|$LB_RESOURCE_ID|allocat|address|pool|quota|health|failed|error" "$W"; F_RC=$?; printf 'producer_rc=%s window_rc=%s filter_rc=%s\n' "$P_RC" "$W_RC" "$F_RC"; [ "$F_RC" -le 1 ] && LOG_READY=true; else printf 'producer_rc=%s window_rc=%s filter_rc=not-run\n' "$P_RC" "$W_RC" >&2; fi; rm -f "$T" "$W"; else printf '중단: exact pods/log RBAC=%s\n' "${AUTH:--}" >&2; fi`

producer 실패와 filter 무일치는 별도다. `filter_rc=1`은 오류 부재가 아니다. 외부 controller면 로그 명령을 실행하지 않고 provider owner의 승인된 read-only API evidence를 요청한다. provider credential·token·전체 로그는 저장하지 않는다.

## 제한적 변경과 변경 후 정상화

이 문서에는 mutation 명령을 두지 않는다. pool/quota/class/annotation/provider resource 변경 또는 재조정은 fresh Service UID/RV/generation, 직접 allocation 증거, provider exact identity, 승인 ID, window, tenant·backend/quorum 영향, 멱등성·중복 VIP 위험, rollback과 중단 조건이 모두 있는 provider/GitOps 경로에서만 수행한다. Service 삭제·재생성, controller restart, 임의 annotation patch는 금지한다.

변경 뒤 `reset_lb_state`부터 Service·EndpointSlice·provider/controller를 완전 재등록한다. replacement Service면 이전 UID Event·provider relation을 재사용하지 않는다. 정상화는 새 status address 기록, provider resource ACTIVE/assigned, ports와 endpoint readiness 유지, 새 Warning 부재와 다음 런북의 승인된 외부 도달성까지 확인한다.

`FUNCTION_EVIDENCE_ID='PASTE_FUNCTION_EVIDENCE_ID_HERE'; FUNCTION_READY=false; if [ "$PROVIDER_READY" != true ] || [ -z "$FUNCTION_EVIDENCE_ID" ] || ! lb_fresh || ! lb_relation_fresh; then printf '%s\n' '중단: 승인된 외부 도달성·실제 기능 evidence와 latest Service relation이 필요하다.' >&2; else FUNCTION_READY=true; printf 'FUNCTION_READY=true evidence=%s service_uid=%s provider_evidence=%s\n' "$FUNCTION_EVIDENCE_ID" "$SERVICE_UID" "$PROVIDER_EVIDENCE_ID"; fi`

`NORMAL_READY=false; if [ "$AUTO_READY" != true ] || [ "$PROVIDER_READY" != true ] || [ "$FUNCTION_READY" != true ] || [ "$RELATION_READY" != true ] || ! lb_fresh || ! lb_relation_fresh; then printf '%s\n' '중단: 변경 후 전체 재등록/latest Service/relation/provider/실제 기능 gate 필요' >&2; elif [ "$PROVIDER_STATE" != ACTIVE ] && [ "$PROVIDER_STATE" != ASSIGNED ]; then printf '미정상: provider state=%s evidence=%s\n' "$PROVIDER_STATE" "$PROVIDER_EVIDENCE_ID" >&2; elif jq -e '(.status.loadBalancer.ingress|length)>0' <<<"$SERVICE_JSON" >/dev/null; then NORMAL_READY=true; jq -r --arg evidence "$PROVIDER_EVIDENCE_ID" --arg function "$FUNCTION_EVIDENCE_ID" '"NORMAL_READY=true ingress=" + ([.status.loadBalancer.ingress[]?|(.ip // .hostname)]|join(",")) + " generation=" + (.metadata.generation|tostring) + " provider_evidence=" + $evidence + " function_evidence=" + $function' <<<"$SERVICE_JSON"; else printf '%s\n' '미정상: status.loadBalancer.ingress가 비어 있음' >&2; fi`

## Prometheus 빠른 관측

`QUERY_READY=false; if [ "$AUTO_READY" != true ] || [ "$RELATION_READY" != true ] || ! lb_fresh || ! lb_relation_fresh; then printf '%s\n' '중단: latest Service/relation gate 실패' >&2; else QUERY_READY=true; printf 'kube_service_spec_type{namespace="%s",service=~"%s",type="LoadBalancer"}==1\nkube_service_status_load_balancer_ingress{namespace="%s",service=~"%s"}\ntime()-kube_service_created{namespace="%s",service=~"%s"}\n' "$NS" "$SERVICE_RE" "$NS" "$SERVICE_RE" "$NS" "$SERVICE_RE"; fi`

metric·labels가 실제 수집될 때만 사용한다. 빈 결과는 address 미할당과 metric 미수집·label 불일치·조회 범위·Prometheus transport/HTTP/API 오류를 구분한다. metric만으로 provider allocation 실패나 재생성을 승인하지 않는다.

## 증거와 1.26 경계

UTC 시각, context·namespace, Service UID/RV/generation, class·ports·policy, EndpointSlice UID/targetRef UID/readiness, UID 제한 Event, provider/account/region/resource ID, controller UID/containerID, producer/filter rc, 승인/rollback 결과만 보존한다. credential·token·내부 주소·전체 provider 로그는 보존하지 않는다. provider/VIP controller 구현은 환경별 미검증이며 인프라 명령을 실행하지 않았다.
