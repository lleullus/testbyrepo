# VIP·External IP가 할당됐지만 외부에서 연결되지 않는 경우

## 입력값과 빠른 수집

실행 host는 승인된 외부 probe 또는 작업 PC의 같은 Bash다. hop은 `외부 source → DNS/route/firewall → VIP advertisement → LB listener → health probe/backend membership → nodePort 또는 direct Pod backend → Service data plane → EndpointSlice → process listener → application`이다. address 할당, TCP, TLS/HTTP와 application 성공을 분리한다.

| lifecycle | 값 | 계약 |
|---|---|---|
| 최초 입력 | context·namespace·Service | exact Service를 한 번 입력한다. |
| API 자동값 | Service UID/RV/generation, ingress address, ports/policy, EndpointSlice 관계 | 원자적으로 등록한다. |
| 계산 후보 | Service regex, 외부 address 후보, UTC | host와 IP를 구분한다. |
| 운영자 선택값 | observation point, external port/protocol, provider exact identity, health ID | 승인된 inventory 뒤 등록한다. |

`reset_vip_state(){ unset AUTO_READY AUTO_LATEST_READY RELATION_READY RELATION_LATEST_READY PROBE_READY PROBE_OK PROVIDER_READY QUERY_READY EVIDENCE_READY FUNCTION_READY ACTION_READY NORMAL_READY SERVICE_JSON EPS_JSON SERVICE_UID SERVICE_RV SERVICE_GENERATION SERVICE_OWNER SERVICE_HOST SERVICE_RE EXTERNAL_IP EXTERNAL_HOSTNAME EXTERNAL_ADDRESS EXTERNAL_PORT PROTOCOL SERVICE_PORT TARGET_PORT NODE_PORT ENDPOINT_UID POD_UID POD_RV POD_OWNER POD_NODE CONTAINER_ID PROVIDER ACCOUNT REGION LB_RESOURCE_ID HEALTH_ID HEALTH_STATE PROVIDER_EVIDENCE_ID FUNCTION_EVIDENCE_ID OBSERVATION_POINT OBSERVED_AT START_RFC3339 END_RFC3339 ARGV_RE; AUTO_READY=false; AUTO_LATEST_READY=false; RELATION_READY=false; RELATION_LATEST_READY=false; PROBE_READY=false; PROBE_OK=false; PROVIDER_READY=false; QUERY_READY=false; EVIDENCE_READY=false; FUNCTION_READY=false; ACTION_READY=false; NORMAL_READY=false; }`

`reset_vip_state; CONTEXT='PASTE_CONTEXT_HERE'; NS='PASTE_NAMESPACE_HERE'; SERVICE='PASTE_SERVICE_HERE'; INPUT_READY=false; if [ "$TOOLS_READY" != true ]; then printf '%s\n' '중단: TOOLS_READY가 아니다.' >&2; elif kubectl --context "$CONTEXT" get service "$SERVICE" -n "$NS" -o name >/dev/null 2>&1; then INPUT_READY=true; export CONTEXT NS SERVICE; printf 'INPUT_READY=true service=%s/%s\n' "$NS" "$SERVICE"; else printf '%s\n' '중단: exact Service 조회 실패' >&2; fi`

재수집 전 reset이 UID·RV·owner·containerID·node·host·address/relation·time·argv/regex와 모든 gate를 폐기한다. 입력 문자열은 각각 한 번만 나타난다.

`reset_vip_state; if [ "$INPUT_READY" != true ]; then printf '%s\n' '중단: INPUT_READY가 아니다.' >&2; elif S="$(kubectl --context "$CONTEXT" get service "$SERVICE" -n "$NS" -o json)" && E="$(kubectl --context "$CONTEXT" get endpointslice -n "$NS" -l "kubernetes.io/service-name=$SERVICE" -o json)" && U="$(jq -r '.metadata.uid // empty' <<<"$S")" && R="$(jq -r '.metadata.resourceVersion // empty' <<<"$S")" && [ -n "$U" ] && [ -n "$R" ] && jq -e '.spec.type=="LoadBalancer" and (.status.loadBalancer.ingress|length)>0' <<<"$S" >/dev/null && jq -e --arg n "$SERVICE" --arg u "$U" 'all(.items[]?; any(.metadata.ownerReferences[]?; .kind=="Service" and .name==$n and .uid==$u))' <<<"$E" >/dev/null; then SERVICE_JSON="$S"; EPS_JSON="$E"; SERVICE_UID="$U"; SERVICE_RV="$R"; SERVICE_GENERATION="$(jq -r '.metadata.generation' <<<"$S")"; SERVICE_OWNER="$(jq -r '[.metadata.ownerReferences[]?|"\(.kind)/\(.name)/\(.uid)"]|join(",")' <<<"$S")"; EXTERNAL_IP="$(jq -r '[.status.loadBalancer.ingress[]?.ip][0] // empty' <<<"$S")"; EXTERNAL_HOSTNAME="$(jq -r '[.status.loadBalancer.ingress[]?.hostname][0] // empty' <<<"$S")"; SERVICE_RE="^$(printf '%s' "$SERVICE" | sed 's/[][\\.^$*+?(){}|]/\\&/g')$"; OBSERVED_AT="$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; AUTO_READY=true; RELATION_READY=true; jq -r --arg o "$OBSERVED_AT" '"=== Service ===\nUID: \(.metadata.uid)\nRV: \(.metadata.resourceVersion)\nIngress: \([.status.loadBalancer.ingress[]?|(.ip // .hostname)]|join(","))\nPolicy: \(.spec.externalTrafficPolicy // \"-\")\nPort 관계: " + ([.spec.ports[]?|"service=\(.name // \"-\"):\(.port)/\(.protocol) target=\(.targetPort) nodePort=\(.nodePort // \"-\")"]|join("; ")) + "\n관측 UTC: \($o)"' <<<"$S"; jq -r '.items[]|"=== Slice: \(.metadata.name) ===\nUID: \(.metadata.uid)\n관계: " + ([.endpoints[]?|"target=\(.targetRef.name // \"-\") uid=\(.targetRef.uid // \"-\") node=\(.nodeName // \"-\") ready=\(.conditions.ready // \"nil\") serving=\(.conditions.serving // \"nil\")"]|join("; "))' <<<"$E"; printf 'AUTO_READY=true RELATION_READY=true ip=%s hostname=%s\n' "${EXTERNAL_IP:--}" "${EXTERNAL_HOSTNAME:--}"; else reset_vip_state; printf '%s\n' '중단: assigned LoadBalancer/EndpointSlice owner identity API 실패' >&2; fi`

`vip_fresh(){ local j u r g ip host; AUTO_LATEST_READY=false; j="$(kubectl --context "$CONTEXT" get service "$SERVICE" -n "$NS" -o json)" || { reset_vip_state; return 1; }; u="$(jq -r '.metadata.uid // empty' <<<"$j")"; r="$(jq -r '.metadata.resourceVersion // empty' <<<"$j")"; g="$(jq -r '.metadata.generation // 0' <<<"$j")"; ip="$(jq -r '[.status.loadBalancer.ingress[]?.ip][0] // empty' <<<"$j")"; host="$(jq -r '[.status.loadBalancer.ingress[]?.hostname][0] // empty' <<<"$j")"; if [ "$u" = "$SERVICE_UID" ] && [ "$g" = "$SERVICE_GENERATION" ] && [ "$ip" = "$EXTERNAL_IP" ] && [ "$host" = "$EXTERNAL_HOSTNAME" ] && [ -n "$r" ] && jq -e '.spec.type=="LoadBalancer"' <<<"$j" >/dev/null && { [ -z "${EXTERNAL_PORT:-}" ] || jq -e --argjson p "$EXTERNAL_PORT" 'any(.spec.ports[]?; .port==$p and .protocol=="TCP")' <<<"$j" >/dev/null; }; then SERVICE_JSON="$j"; SERVICE_RV="$r"; AUTO_LATEST_READY=true; return 0; else reset_vip_state; return 1; fi; }`

`vip_relation_fresh(){ local e; RELATION_LATEST_READY=false; e="$(kubectl --context "$CONTEXT" get endpointslice -n "$NS" -l "kubernetes.io/service-name=$SERVICE" -o json)" || { RELATION_READY=false; return 1; }; if jq -e --arg n "$SERVICE" --arg u "$SERVICE_UID" 'all(.items[]?; any(.metadata.ownerReferences[]?; .kind=="Service" and .name==$n and .uid==$u))' <<<"$e" >/dev/null; then EPS_JSON="$e"; RELATION_LATEST_READY=true; return 0; else RELATION_READY=false; unset EPS_JSON; return 1; fi; }`

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | status ingress address가 있으나 승인된 외부 관측점에서 연결되지 않는다. |
| 적용 | Service/backend와 provider VIP resource를 같은 identity·시각으로 연결한다. |
| 제외 | 미할당은 [[01. LoadBalancer Service의 EXTERNAL-IP가 Pending이거나 할당되지 않는 경우]], NodePort는 [[09. 외부 트래픽/02. NodePort·외부 트래픽 정책/01. NodePort에 연결할 수 없거나 일부 노드에서만 실패하는 경우]], L7 502/503은 [[08. 인그레스·게이트웨이/03. L7 라우팅·Backend·TLS/02. Ingress·Gateway 요청이 502·503을 반환하는 경우]]다. |

검색 문자열: `timeout`, `refused`, `unhealthy`, `advertisement withdrawn`, `listener`, `backend membership`, `externalTrafficPolicy`.

## 승인된 외부 probe

`OBSERVATION_POINT='PASTE_OBSERVATION_POINT_HERE'; EXTERNAL_ADDRESS='PASTE_EXTERNAL_ADDRESS_HERE'; EXTERNAL_PORT='PASTE_EXTERNAL_PORT_HERE'; PROTOCOL='PASTE_TCP_PROTOCOL_HERE'`

`PROBE_READY=false; PROBE_OK=false; PROVIDER_READY=false; FUNCTION_READY=false; NORMAL_READY=false; unset HEALTH_STATE PROVIDER_EVIDENCE_ID FUNCTION_EVIDENCE_ID; if [ "$RELATION_READY" != true ] || ! vip_fresh || ! vip_relation_fresh; then printf '%s\n' '중단: latest Service/relation gate 실패' >&2; elif [ -z "$OBSERVATION_POINT" ] || [ "$PROTOCOL" != TCP ] || ! [[ "$EXTERNAL_PORT" =~ ^[0-9]+$ ]] || [ "$EXTERNAL_PORT" -lt 1 ] || [ "$EXTERNAL_PORT" -gt 65535 ] || ! jq -e --argjson p "$EXTERNAL_PORT" 'any(.spec.ports[]?; .port==$p and .protocol=="TCP")' <<<"$SERVICE_JSON" >/dev/null; then printf '%s\n' '중단: observation point 또는 Service TCP port 관계가 잘못됐다.' >&2; elif [ "$EXTERNAL_ADDRESS" != "$EXTERNAL_IP" ] && [ "$EXTERNAL_ADDRESS" != "$EXTERNAL_HOSTNAME" ]; then printf '%s\n' '중단: address가 현재 status ingress와 불일치' >&2; else nc -vz -w 5 "$EXTERNAL_ADDRESS" "$EXTERNAL_PORT"; NC_RC=$?; printf 'observation=%s transport_rc=%s address=%s port=%s observed=%s\n' "$OBSERVATION_POINT" "$NC_RC" "$EXTERNAL_ADDRESS" "$EXTERNAL_PORT" "$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; PROBE_READY=true; [ "$NC_RC" -eq 0 ] && PROBE_OK=true; fi`

`nc` 성공은 TCP transport만 보장한다. timeout은 route/firewall/listener/backend 지연을, refused는 listener/rule/backend를 단독 확정하지 않는다. UDP/SCTP는 해당 protocol owner의 승인 probe로 이관한다.

HTTP/HTTPS는 승인된 host/path가 있을 때만 별도 L7 런북에서 body·credential 없이 확인한다. IP와 hostname을 혼동하지 않고 TLS 검증을 우회하지 않는다.

## provider exact identity와 health 경계

`PROVIDER='PASTE_PROVIDER_HERE'; ACCOUNT='PASTE_PROVIDER_ACCOUNT_HERE'; REGION='PASTE_PROVIDER_REGION_HERE'; LB_RESOURCE_ID='PASTE_LB_RESOURCE_ID_HERE'; HEALTH_ID='PASTE_HEALTH_CHECK_ID_HERE'; HEALTH_STATE='PASTE_HEALTH_STATE_HERE'; PROVIDER_EVIDENCE_ID='PASTE_PROVIDER_EVIDENCE_ID_HERE'`

`PROVIDER_READY=false; FUNCTION_READY=false; NORMAL_READY=false; unset FUNCTION_EVIDENCE_ID; if [ "$PROBE_READY" != true ] || ! vip_fresh || ! vip_relation_fresh; then printf '%s\n' '중단: probe/latest Service/relation gate 실패' >&2; elif [ -n "$PROVIDER" ] && [ -n "$ACCOUNT" ] && [ -n "$REGION" ] && [ -n "$LB_RESOURCE_ID" ] && [ -n "$HEALTH_ID" ] && [ -n "$HEALTH_STATE" ] && [ -n "$PROVIDER_EVIDENCE_ID" ]; then PROVIDER_READY=true; printf 'PROVIDER_READY=true provider=%s account=%s region=%s resource=%s health=%s health_state=%s evidence=%s service_uid=%s\n' "$PROVIDER" "$ACCOUNT" "$REGION" "$LB_RESOURCE_ID" "$HEALTH_ID" "$HEALTH_STATE" "$PROVIDER_EVIDENCE_ID" "$SERVICE_UID"; else printf '%s\n' '중단: provider exact identity/health evidence가 비었다.' >&2; fi`

provider record에서 address, listener protocol/port, backend address type, health probe source/port/path, membership와 state를 read-only로 확인한다. provider/account/region/resource ID와 health ID가 같은 Service UID를 가리키지 않으면 CLI/API를 실행하지 않는다. `externalTrafficPolicy=Local`이면 각 member node의 serving endpoint를, Cluster면 cross-node forwarding을 별도 확인한다. health 실패는 application·route·firewall 중 하나를 자동 확정하지 않는다.

route/security owner에게 source·destination·port·protocol, forward/return route, NAT translation, policy/rule ID와 hit/drop UTC 시각을 요청한다. 전체 firewall ruleset, packet payload, credential을 수집하지 않는다.

## 제한적 변경과 정상화

이 문서에는 mutation 명령을 두지 않는다. VIP 광고/listener/health/security/Service 선언 변경은 fresh Service UID/RV, external probe와 provider 직접 증거, 승인 ID/window, tenant·backend/PDB/quorum 영향, 멱등성·중복 address 위험, rollback resource/revision과 중단 조건을 충족한 owner 경로에서만 한다. Service 재생성, health 완화, firewall 전면 허용, controller restart는 금지한다.

변경 뒤 `reset_vip_state`부터 Service·EndpointSlice·provider relation과 probe를 완전 재등록한다. replacement UID/address면 이전 결과를 폐기한다. 정상화는 provider listener/advertisement/health membership, 동일 observation point TCP, 조건부 TLS/HTTP status, EndpointSlice serving, 실제 핵심 기능과 관찰 창 재발 부재를 역순으로 확인한다.

`FUNCTION_EVIDENCE_ID='PASTE_FUNCTION_EVIDENCE_ID_HERE'; FUNCTION_READY=false; if [ "$PROBE_OK" != true ] || [ "$PROVIDER_READY" != true ] || [ -z "$FUNCTION_EVIDENCE_ID" ] || ! vip_fresh || ! vip_relation_fresh; then printf '%s\n' '중단: 승인된 protocol별 실제 기능 evidence와 latest Service relation이 필요하다.' >&2; else FUNCTION_READY=true; printf 'FUNCTION_READY=true evidence=%s service_uid=%s address=%s\n' "$FUNCTION_EVIDENCE_ID" "$SERVICE_UID" "$EXTERNAL_ADDRESS"; fi`

`NORMAL_READY=false; if [ "$PROBE_READY" != true ] || [ "$PROBE_OK" != true ] || [ "$PROVIDER_READY" != true ] || [ "$FUNCTION_READY" != true ] || [ "$RELATION_READY" != true ] || ! vip_fresh || ! vip_relation_fresh; then printf '%s\n' '중단: 변경 후 전체 재등록/latest Service/relation/성공 probe/provider/실제 기능 gate 필요' >&2; elif [ "$HEALTH_STATE" != healthy ] && [ "$HEALTH_STATE" != HEALTHY ]; then printf '미정상: provider health=%s evidence=%s\n' "$HEALTH_STATE" "$PROVIDER_EVIDENCE_ID" >&2; elif ! jq -e '[.items[].endpoints[]? | select((.conditions.ready // true)==true and (.conditions.terminating // false)==false)] | length>0' <<<"$EPS_JSON" >/dev/null; then printf '%s\n' '미정상: current serving endpoint가 없다.' >&2; else NORMAL_READY=true; printf 'NORMAL_READY=true service_uid=%s rv=%s address=%s provider_evidence=%s function_evidence=%s observed=%s\n' "$SERVICE_UID" "$SERVICE_RV" "$EXTERNAL_ADDRESS" "$PROVIDER_EVIDENCE_ID" "$FUNCTION_EVIDENCE_ID" "$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; fi`

## Prometheus 빠른 관측

`QUERY_READY=false; if [ "$AUTO_READY" != true ] || [ "$RELATION_READY" != true ] || ! vip_fresh || ! vip_relation_fresh; then printf '%s\n' '중단: latest Service/relation gate 실패' >&2; else QUERY_READY=true; printf 'kube_service_status_load_balancer_ingress{namespace="%s",service=~"%s"}\nsum(kube_endpoint_address{namespace="%s",endpoint="%s",ready="true"})\n' "$NS" "$SERVICE_RE" "$NS" "$SERVICE"; fi`

metric·labels가 실제 수집될 때만 사용한다. 빈 결과를 0/정상으로 해석하지 않고 metric 부재·label 불일치·시간 범위·Prometheus transport/HTTP/API 오류를 구분한다. provider advertisement, listener와 health membership은 metric만으로 확정하지 않는다.

## 증거와 1.26 경계

UTC 시각, observation point, Service UID/RV/generation, ingress address, port/targetPort/nodePort/policy, EndpointSlice UID/targetRef UID/node/readiness, TCP rc, provider/account/region/resource/health ID, route/NAT/security rule ID, 승인·rollback 결과만 보존한다. 내부 address 목록·credential·payload·전체 ruleset/log는 보존하지 않는다. provider 구현은 환경별 미검증이며 인프라를 실행하지 않았다.
