# NodePort에 연결할 수 없거나 일부 노드에서만 실패하는 경우

## 입력값과 빠른 수집

kubectl은 승인된 작업 PC/bastion의 같은 Bash에서, node 명령은 승인된 `NODE_HOST`의 원격 Bash에서 실행한다. hop은 `외부 client → route/NAT/firewall → node address:nodePort → kube-proxy data plane → EndpointSlice port/address → Pod listener → application → return path`다. `ss` listener, proxy rule, TCP 연결, HTTP status는 서로 다른 증거다.

| lifecycle | 값 | 계약 |
|---|---|---|
| 최초 입력 | `CONTEXT`, `NS`, `SERVICE` | exact Service를 한 번 입력한다. |
| API 자동값 | Service UID/RV, port 관계, EndpointSlice targetRef UID/node, node identity | 전체 성공 뒤 등록한다. |
| 계산 후보 | Service/node regex, endpoint 후보, UTC | escape 후 bounded 확인한다. |
| 운영자 선택값 | failed/working node, node address, node SSH host, kube-proxy Pod/container/mode, 시간 | node별 증거와 inventory 뒤 선택한다. |

`reset_np_state; CONTEXT='PASTE_CONTEXT_HERE'; NS='PASTE_NAMESPACE_HERE'; SERVICE='PASTE_SERVICE_HERE'; INPUT_READY=false; if [ "$TOOLS_READY" != true ]; then printf '%s\n' '중단: TOOLS_READY가 아니다.' >&2; elif kubectl --context "$CONTEXT" get service "$SERVICE" -n "$NS" -o name >/dev/null 2>&1; then INPUT_READY=true; export CONTEXT NS SERVICE; printf 'INPUT_READY=true service=%s/%s\n' "$NS" "$SERVICE"; else printf '%s\n' '중단: exact Service 조회 실패' >&2; fi`

재수집 전 reset이 UID·RV·owner·containerID·node·host·port/relation·time·argv/regex와 downstream gate를 모두 폐기한다. 각 입력 문자열은 한 번만 나타나며 producer 실패 후 stale 값은 없다.

`reset_np_state; if [ "$INPUT_READY" != true ]; then printf '%s\n' '중단: INPUT_READY가 아니다.' >&2; elif S="$(kubectl --context "$CONTEXT" get service "$SERVICE" -n "$NS" -o json)" && E="$(kubectl --context "$CONTEXT" get endpointslice -n "$NS" -l "kubernetes.io/service-name=$SERVICE" -o json)" && U="$(jq -r '.metadata.uid // empty' <<<"$S")" && R="$(jq -r '.metadata.resourceVersion // empty' <<<"$S")" && [ -n "$U" ] && [ -n "$R" ] && jq -e '.spec.type=="NodePort" or .spec.type=="LoadBalancer"' <<<"$S" >/dev/null && jq -e --arg n "$SERVICE" --arg u "$U" 'all(.items[]?; any(.metadata.ownerReferences[]?; .kind=="Service" and .name==$n and .uid==$u))' <<<"$E" >/dev/null; then SERVICE_JSON="$S"; EPS_JSON="$E"; SERVICE_UID="$U"; SERVICE_RV="$R"; SERVICE_GENERATION="$(jq -r '.metadata.generation' <<<"$S")"; SERVICE_OWNER="$(jq -r '[.metadata.ownerReferences[]?|"\(.kind)/\(.name)/\(.uid)"]|join(",")' <<<"$S")"; SERVICE_RE="^$(printf '%s' "$SERVICE" | sed 's/[][\\.^$*+?(){}|]/\\&/g')$"; OBSERVED_AT="$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; AUTO_READY=true; RELATION_READY=true; jq -r --arg o "$OBSERVED_AT" '"=== Service ===\nUID: \(.metadata.uid)\nRV: \(.metadata.resourceVersion)\nType: \(.spec.type)\nPolicy: \(.spec.externalTrafficPolicy // \"-\")\nPort 관계: " + ([.spec.ports[]?|"service=\(.name // \"-\"):\(.port)/\(.protocol) target=\(.targetPort) nodePort=\(.nodePort // \"-\")"]|join("; ")) + "\n관측 UTC: \($o)"' <<<"$S"; jq -r '.items[]|"=== Slice: \(.metadata.name) ===\nUID: \(.metadata.uid)\nPort: " + ([.ports[]?|"\(.name // \"-\"):\(.port)/\(.protocol)"]|join(",")) + "\n관계: " + ([.endpoints[]?|"target=\(.targetRef.name // \"-\") uid=\(.targetRef.uid // \"-\") node=\(.nodeName // \"-\") ready=\(.conditions.ready // \"nil\") serving=\(.conditions.serving // \"nil\") terminating=\(.conditions.terminating // \"nil\")"]|join("; "))' <<<"$E"; printf 'AUTO_READY=true RELATION_READY=true SERVICE_RE=%s\n' "$SERVICE_RE"; else reset_np_state; printf '%s\n' '중단: NodePort Service/EndpointSlice owner 수집 실패' >&2; fi`

`np_fresh(){ local j u r g; AUTO_LATEST_READY=false; j="$(kubectl --context "$CONTEXT" get service "$SERVICE" -n "$NS" -o json)" || { reset_np_state; return 1; }; u="$(jq -r '.metadata.uid // empty' <<<"$j")"; r="$(jq -r '.metadata.resourceVersion // empty' <<<"$j")"; g="$(jq -r '.metadata.generation // 0' <<<"$j")"; if [ "$u" = "$SERVICE_UID" ] && [ "$g" = "$SERVICE_GENERATION" ] && [ -n "$r" ] && jq -e '.spec.type=="NodePort" or .spec.type=="LoadBalancer"' <<<"$j" >/dev/null && { [ -z "${NODE_PORT:-}" ] || jq -e --arg sp "$SERVICE_PORT" --argjson np "$NODE_PORT" 'any(.spec.ports[]?; (((.name // "")==$sp) or ((.port|tostring)==$sp)) and .nodePort==$np and .protocol=="TCP")' <<<"$j" >/dev/null; }; then SERVICE_JSON="$j"; SERVICE_RV="$r"; AUTO_LATEST_READY=true; return 0; else reset_np_state; return 1; fi; }`

`np_relation_fresh(){ local e; RELATION_LATEST_READY=false; e="$(kubectl --context "$CONTEXT" get endpointslice -n "$NS" -l "kubernetes.io/service-name=$SERVICE" -o json)" || { RELATION_READY=false; return 1; }; if jq -e --arg n "$SERVICE" --arg u "$SERVICE_UID" 'all(.items[]?; any(.metadata.ownerReferences[]?; .kind=="Service" and .name==$n and .uid==$u))' <<<"$e" >/dev/null; then EPS_JSON="$e"; RELATION_LATEST_READY=true; return 0; else RELATION_READY=false; unset EPS_JSON; return 1; fi; }`

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | 승인된 node address:nodePort가 모두 또는 일부 node에서 timeout/refused다. |
| 적용 | TCP Service의 node별 endpoint locality와 proxy data plane을 확인한다. |
| 제외 | Local/source IP는 [[02. externalTrafficPolicy 때문에 트래픽이 실패하거나 source IP가 보존되지 않는 경우]], VIP는 [[09. 외부 트래픽/01. LoadBalancer·VIP/02. VIP·External IP가 할당됐지만 외부에서 연결되지 않는 경우]]다. |

검색 문자열: `NodePort`, `timeout`, `refused`, `syncProxyRules`, `externalTrafficPolicy`, `healthCheckNodePort`, `EndpointSlice`.

## port와 node 선택·probe

`SERVICE_PORT='PASTE_SERVICE_PORT_HERE'; NODE_PORT='PASTE_NODE_PORT_HERE'; PROTOCOL='PASTE_PROTOCOL_HERE'; FAILED_NODE='PASTE_FAILED_NODE_HERE'; WORKING_NODE='PASTE_WORKING_NODE_HERE'; NODE_ADDRESS='PASTE_NODE_ADDRESS_HERE'; WORKING_NODE_ADDRESS='PASTE_WORKING_NODE_ADDRESS_HERE'; NODE_HOST='PASTE_NODE_SSH_HOST_HERE'`

`NODE_READY=false; PROBE_READY=false; PROBE_OK=false; PROXY_READY=false; TIME_READY=false; REMOTE_READY=false; LOG_READY=false; FUNCTION_READY=false; NORMAL_READY=false; unset NODE_UID NODE_MACHINE_ID WORKING_NODE_UID WORKING_NODE_MACHINE_ID KPROXY_UID KPROXY_RV KPROXY_OWNER KPROXY_NODE KPROXY_CONTAINER_ID FUNCTION_EVIDENCE_ID; if [ "$RELATION_READY" != true ] || ! np_fresh || ! np_relation_fresh; then printf '%s\n' '중단: latest Service/relation gate 실패' >&2; elif [ "$PROTOCOL" != TCP ] || ! [[ "$NODE_PORT" =~ ^[0-9]+$ ]] || ! jq -e --arg sp "$SERVICE_PORT" --argjson np "$NODE_PORT" 'any(.spec.ports[]?; (((.name // "")==$sp) or ((.port|tostring)==$sp)) and .nodePort==$np and .protocol=="TCP")' <<<"$SERVICE_JSON" >/dev/null; then printf '%s\n' '중단: Service port→TCP nodePort exact 관계 실패' >&2; elif N="$(kubectl --context "$CONTEXT" get node "$FAILED_NODE" -o json)" && W="$(kubectl --context "$CONTEXT" get node "$WORKING_NODE" -o json)" && NU="$(jq -r '.metadata.uid // empty' <<<"$N")" && NM="$(jq -r '.status.nodeInfo.machineID // empty' <<<"$N")" && WU="$(jq -r '.metadata.uid // empty' <<<"$W")" && WM="$(jq -r '.status.nodeInfo.machineID // empty' <<<"$W")" && jq -e --arg h "$NODE_HOST" --arg a "$NODE_ADDRESS" '([.metadata.name]+[.status.addresses[]?.address]|any(.==$h)) and ([.status.addresses[]?.address]|any(.==$a))' <<<"$N" >/dev/null && jq -e --arg a "$WORKING_NODE_ADDRESS" '[.status.addresses[]?.address]|any(.==$a)' <<<"$W" >/dev/null && [ -n "$NU" ] && [ -n "$NM" ] && [ -n "$WU" ] && [ -n "$WM" ] && [ "$NU" != "$WU" ]; then NODE_JSON="$N"; NODE_UID="$NU"; NODE_MACHINE_ID="$NM"; WORKING_NODE_UID="$WU"; WORKING_NODE_MACHINE_ID="$WM"; NODE_RE="^$(printf '%s' "$FAILED_NODE" | sed 's/[][\\.^$*+?(){}|]/\\&/g')$"; NODE_READY=true; printf 'NODE_READY=true failed=%s uid=%s machine=%s address=%s ssh_host=%s working=%s working_uid=%s working_machine=%s working_address=%s\n' "$FAILED_NODE" "$NODE_UID" "$NODE_MACHINE_ID" "$NODE_ADDRESS" "$NODE_HOST" "$WORKING_NODE" "$WORKING_NODE_UID" "$WORKING_NODE_MACHINE_ID" "$WORKING_NODE_ADDRESS"; else printf '%s\n' '중단: failed/working node API address/SSH host exact identity 실패' >&2; fi`

`node_fresh(){ local j w; NODE_LATEST_READY=false; j="$(kubectl --context "$CONTEXT" get node "$FAILED_NODE" -o json)" || return 1; w="$(kubectl --context "$CONTEXT" get node "$WORKING_NODE" -o json)" || return 1; jq -e --arg u "$NODE_UID" --arg m "$NODE_MACHINE_ID" --arg a "$NODE_ADDRESS" '.metadata.uid==$u and .status.nodeInfo.machineID==$m and ([.status.addresses[]?.address]|any(.==$a))' <<<"$j" >/dev/null && jq -e --arg u "$WORKING_NODE_UID" --arg m "$WORKING_NODE_MACHINE_ID" --arg a "$WORKING_NODE_ADDRESS" '.metadata.uid==$u and .status.nodeInfo.machineID==$m and ([.status.addresses[]?.address]|any(.==$a))' <<<"$w" >/dev/null || { NODE_READY=false; REMOTE_READY=false; PROXY_READY=false; unset NODE_UID NODE_MACHINE_ID WORKING_NODE_UID WORKING_NODE_MACHINE_ID; return 1; }; NODE_JSON="$j"; NODE_LATEST_READY=true; }`

`PROBE_READY=false; PROBE_OK=false; FUNCTION_READY=false; NORMAL_READY=false; unset FUNCTION_EVIDENCE_ID; if [ "$NODE_READY" != true ] || ! np_fresh || ! np_relation_fresh || ! node_fresh; then printf '%s\n' '중단: latest Service/relation/nodes gate 실패' >&2; else nc -vz -w 5 "$NODE_ADDRESS" "$NODE_PORT"; FAILED_RC=$?; nc -vz -w 5 "$WORKING_NODE_ADDRESS" "$NODE_PORT"; WORKING_RC=$?; printf 'failed_node=%s failed_rc=%s working_node=%s working_rc=%s node_port=%s observed=%s\n' "$FAILED_NODE" "$FAILED_RC" "$WORKING_NODE" "$WORKING_RC" "$NODE_PORT" "$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; PROBE_READY=true; [ "$FAILED_RC" -eq 0 ] && [ "$WORKING_RC" -eq 0 ] && PROBE_OK=true; fi`

TCP result는 transport만 보장한다. 같은 observation point에서 failed/working node를 동일 조건으로 비교하며 private address가 도달 불가능한 위치면 실행하지 않는다. UDP/SCTP는 별도 owner probe를 쓴다.

## kube-proxy identity·시간·로그

`KPROXY_NS='PASTE_KUBEPROXY_NAMESPACE_HERE'; KPROXY_POD='PASTE_KUBEPROXY_POD_HERE'; KPROXY_CONTAINER='PASTE_KUBEPROXY_CONTAINER_HERE'; KPROXY_MODE='PASTE_KUBEPROXY_MODE_HERE'; START_RFC3339='PASTE_START_RFC3339_HERE'; END_RFC3339='PASTE_END_RFC3339_HERE'`

`TIME_READY=false; if [[ "$START_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && [[ "$END_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && A="$(date -u -d "$START_RFC3339" +%s 2>/dev/null)" && B="$(date -u -d "$END_RFC3339" +%s 2>/dev/null)" && [ "$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ')" = "$START_RFC3339" ] && [ "$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ')" = "$END_RFC3339" ] && [ "$A" -lt "$B" ]; then TIME_READY=true; else printf '%s\n' '중단: canonical UTC window 오류' >&2; fi`

`PROXY_READY=false; REMOTE_READY=false; LOG_READY=false; FUNCTION_READY=false; NORMAL_READY=false; unset KPROXY_UID KPROXY_RV KPROXY_OWNER KPROXY_NODE KPROXY_CONTAINER_ID FUNCTION_EVIDENCE_ID; if [ "$NODE_READY" != true ] || [ "$TIME_READY" != true ] || ! np_fresh || ! np_relation_fresh || ! node_fresh; then printf '%s\n' '중단: latest Service/relation/node/time gate 실패' >&2; elif [ "$KPROXY_MODE" != iptables ] && [ "$KPROXY_MODE" != ipvs ] && [ "$KPROXY_MODE" != nftables ]; then printf '%s\n' '중단: 확인된 kube-proxy mode가 아니다.' >&2; elif P="$(kubectl --context "$CONTEXT" get pod "$KPROXY_POD" -n "$KPROXY_NS" -o json)" && PU="$(jq -r '.metadata.uid // empty' <<<"$P")" && PI="$(jq -r --arg c "$KPROXY_CONTAINER" '[.status.containerStatuses[]?|select(.name==$c and .state.running!=null)|.containerID][0] // empty' <<<"$P")" && PN="$(jq -r '.spec.nodeName // empty' <<<"$P")" && [ "$PN" = "$FAILED_NODE" ] && [ -n "$PU" ] && [ -n "$PI" ]; then KPROXY_UID="$PU"; KPROXY_RV="$(jq -r '.metadata.resourceVersion' <<<"$P")"; KPROXY_OWNER="$(jq -r '[.metadata.ownerReferences[]?|"\(.kind)/\(.name)/\(.uid)"]|join(",")' <<<"$P")"; KPROXY_NODE="$PN"; KPROXY_CONTAINER_ID="$PI"; PROXY_READY=true; printf 'PROXY_READY=true uid=%s container_id=%s node=%s mode=%s owner=%s\n' "$KPROXY_UID" "$KPROXY_CONTAINER_ID" "$KPROXY_NODE" "$KPROXY_MODE" "$KPROXY_OWNER"; else printf '%s\n' '중단: kube-proxy Pod/container/node exact identity 실패' >&2; fi`

`proxy_fresh(){ local j; PROXY_LATEST_READY=false; j="$(kubectl --context "$CONTEXT" get pod "$KPROXY_POD" -n "$KPROXY_NS" -o json)" || return 1; jq -e --arg u "$KPROXY_UID" --arg c "$KPROXY_CONTAINER" --arg i "$KPROXY_CONTAINER_ID" --arg n "$FAILED_NODE" '.metadata.uid==$u and .metadata.deletionTimestamp==null and .spec.nodeName==$n and any(.status.containerStatuses[]?;.name==$c and .state.running!=null and .containerID==$i)' <<<"$j" >/dev/null || { PROXY_READY=false; LOG_READY=false; unset KPROXY_UID KPROXY_RV KPROXY_OWNER KPROXY_NODE KPROXY_CONTAINER_ID; return 1; }; PROXY_LATEST_READY=true; }`

`LOG_READY=false; if [ "$PROXY_READY" != true ] || [ "$TIME_READY" != true ] || ! np_fresh || ! np_relation_fresh || ! node_fresh || ! proxy_fresh; then printf '%s\n' '중단: latest Service/relation/node/proxy/time gate 실패' >&2; elif AUTH="$(kubectl --context "$CONTEXT" auth can-i get pods "$KPROXY_POD" --subresource=log -n "$KPROXY_NS" 2>&1)" && [ "$AUTH" = yes ]; then T="$(mktemp)" || exit 1; W="$(mktemp)" || { rm -f "$T"; exit 1; }; kubectl --context "$CONTEXT" logs "$KPROXY_POD" -n "$KPROXY_NS" -c "$KPROXY_CONTAINER" --timestamps --since-time="$START_RFC3339" --tail=300 >"$T"; P_RC=$?; if [ "$P_RC" -eq 0 ]; then awk -v end="$END_RFC3339" 'substr($1,1,19) "Z" <= end' "$T" >"$W"; W_RC=$?; else W_RC=not-run; fi; if [ "$P_RC" -eq 0 ] && [ "$W_RC" -eq 0 ]; then grep -E -i -C 3 "$SERVICE_UID|$NODE_PORT|syncProxyRules|iptables|ipvs|nft|error|failed|timeout" "$W"; F_RC=$?; printf 'producer_rc=%s window_rc=%s filter_rc=%s\n' "$P_RC" "$W_RC" "$F_RC"; [ "$F_RC" -le 1 ] && LOG_READY=true; else printf 'producer_rc=%s window_rc=%s filter_rc=not-run\n' "$P_RC" "$W_RC" >&2; fi; rm -f "$T" "$W"; else printf '중단: exact pods/log RBAC=%s\n' "${AUTH:--}" >&2; fi`

## 안전한 원격 node 확인

원격 인수는 `ssh ... bash -s --`의 positional argument로만 전달하고 `eval`, 문자열 command 조립, local 변수 interpolation을 사용하지 않는다. API node UID/machineID와 remote `/etc/machine-id`, hostname/address를 먼저 대조한다.

`REMOTE_READY=false; if [ "$NODE_READY" != true ] || [ "$PROXY_READY" != true ] || ! np_fresh || ! np_relation_fresh || ! node_fresh || ! proxy_fresh; then printf '%s\n' '중단: latest Service/relation/node/proxy gate 실패' >&2; elif ssh -- "$NODE_HOST" bash -s -- "$FAILED_NODE" "$NODE_MACHINE_ID" "$NODE_PORT" "$KPROXY_MODE" <<< 'set -u; expected_node="$1"; expected_machine="$2"; node_port="$3"; mode="$4"; actual_host="$(hostname)"; actual_machine="$(cat /etc/machine-id)"; [ "$actual_machine" = "$expected_machine" ] || { printf "중단: remote machine mismatch expected_node=%s host=%s machine=%s\n" "$expected_node" "$actual_host" "$actual_machine" >&2; exit 1; }; printf "REMOTE_READY=true expected_node=%s host=%s machine=%s node_port=%s mode=%s\n" "$expected_node" "$actual_host" "$actual_machine" "$node_port" "$mode"; ss -ltn "sport = :$node_port"; ss_rc=$?; printf "ss_rc=%s (socket 부재는 iptables/ipvs/nft data plane 부재를 뜻하지 않음)\n" "$ss_rc"'; then REMOTE_READY=true; else printf '%s\n' '중단: remote preflight/read 실패' >&2; fi`

확인된 mode에 맞는 rule 조회는 node/platform owner의 승인된 read-only 절차 하나만 사용한다. iptables/ipvs/nft를 모두 실행하거나 ruleset 전체를 저장하지 않는다. CRI/kubelet endpoint와 `crictl`은 NodePort data-plane 판정에 필요하지 않으므로 실행하지 않는다. host firewall은 rule/policy ID와 hit/drop UTC 시각만 owner에게 요청한다.

## 판정·변경·정상화

- nodePort 불일치는 Service 선언 문제다.
- Local에서 해당 node serving endpoint가 없으면 locality 경로다.
- Cluster이고 한 node만 실패하면 node proxy/programming/route/firewall 후보다.
- `ss`가 비어도 socket-less proxy mode는 정상일 수 있다.
- proxy log/rule 성공은 외부 firewall·backend listener·application 정상의 증거가 아니다.

이 문서에는 mutation 명령을 두지 않는다. Service policy/port, proxy, node firewall/route 변경은 fresh Service/node/proxy identity, 직접 증거, 승인 ID/window, PDB/quorum·tenant 영향, 멱등성, rollback과 중단 조건을 갖춘 owner 경로에서만 한다. Service 재생성, kube-proxy restart, ruleset flush, firewall 전면 허용은 금지한다.

변경 후 `reset_np_state`부터 Service·EndpointSlice·node·proxy를 완전 재등록하고 failed/working node probe를 반복한다. replacement UID/containerID면 이전 증거를 폐기한다. 정상화는 node별 TCP, proxy programming, EndpointSlice serving, return path, 조건부 HTTP/TLS와 실제 기능, 관찰 창 재발 부재를 확인한다.

`FUNCTION_EVIDENCE_ID='PASTE_FUNCTION_EVIDENCE_ID_HERE'; FUNCTION_READY=false; if [ "$PROBE_OK" != true ] || [ "$PROXY_READY" != true ] || [ "$REMOTE_READY" != true ] || [ -z "$FUNCTION_EVIDENCE_ID" ] || ! np_fresh || ! np_relation_fresh || ! node_fresh || ! proxy_fresh; then printf '%s\n' '중단: 승인된 protocol별 실제 기능 evidence와 latest Service/node/proxy가 필요하다.' >&2; else FUNCTION_READY=true; printf 'FUNCTION_READY=true evidence=%s service_uid=%s failed_node_uid=%s working_node_uid=%s\n' "$FUNCTION_EVIDENCE_ID" "$SERVICE_UID" "$NODE_UID" "$WORKING_NODE_UID"; fi`

`NORMAL_READY=false; if [ "$PROBE_READY" != true ] || [ "$PROBE_OK" != true ] || [ "$PROXY_READY" != true ] || [ "$REMOTE_READY" != true ] || [ "$FUNCTION_READY" != true ] || ! np_fresh || ! np_relation_fresh || ! node_fresh || ! proxy_fresh; then printf '%s\n' '중단: 변경 후 전체 재등록/latest relation, 양쪽 node 성공 probe와 실제 기능이 필요' >&2; elif ! jq -e '[.items[].endpoints[]? | select((.conditions.ready // true)==true and (.conditions.terminating // false)==false)] | length>0' <<<"$EPS_JSON" >/dev/null; then printf '%s\n' '미정상: current serving endpoint가 없다.' >&2; else NORMAL_READY=true; printf 'NORMAL_READY=true service_uid=%s failed_node_uid=%s working_node_uid=%s proxy_uid=%s function_evidence=%s observed=%s\n' "$SERVICE_UID" "$NODE_UID" "$WORKING_NODE_UID" "$KPROXY_UID" "$FUNCTION_EVIDENCE_ID" "$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; fi`

## Prometheus 빠른 관측

`QUERY_READY=false; if [ "$AUTO_READY" != true ] || [ "$RELATION_READY" != true ] || [ "$NODE_READY" != true ] || ! np_fresh || ! np_relation_fresh || ! node_fresh; then printf '%s\n' '중단: latest Service/relation/nodes gate 실패' >&2; else QUERY_READY=true; printf 'kube_service_spec_type{namespace="%s",service=~"%s",type=~"NodePort|LoadBalancer"}==1\nsum(kube_endpoint_address{namespace="%s",endpoint="%s",ready="true"})\nup{node=~"%s",job=~".*kube-proxy.*"}==0\n' "$NS" "$SERVICE_RE" "$NS" "$SERVICE" "$NODE_RE"; fi`

metric/label 존재를 먼저 확인한다. 빈 결과는 정상·0이 아니며 미수집·label 불일치·시간 범위·Prometheus transport/HTTP/API 오류를 구분한다. `up=0`은 proxy process 장애와 scrape/TLS/RBAC 실패를 구분하지 않는다.

## 증거와 1.26 경계

UTC 시각, Service UID/RV·port 관계·policy, EndpointSlice UID/targetRef UID/node, failed/working node UID/machineID/address, probe rc, kube-proxy UID/containerID/mode와 producer/filter rc, remote machine gate, rule ID/hit, 승인·rollback 결과만 보존한다. credential·payload·전체 ruleset/log는 보존하지 않는다. kube-proxy mode와 node firewall은 환경별 미검증이며 인프라를 실행하지 않았다.
