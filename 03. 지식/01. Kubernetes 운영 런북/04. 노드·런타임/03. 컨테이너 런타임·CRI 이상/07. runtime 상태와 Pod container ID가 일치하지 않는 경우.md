# runtime 상태와 Pod container ID가 일치하지 않는 경우

## 입력값과 빠른 수집

실행 hop은 `승인된 작업 PC/Bastion의 Bash → Kubernetes API → 확인된 Node의 SSH → kubelet이 실제 사용하는 CRI socket → runtime`이다. 한 hop의 성공은 다음 hop을 보장하지 않는다. 같은 Bash를 유지하며 `eval`은 사용하지 않는다.

| 종류 | 값 | 확정 방법 |
|---|---|---|
| 최초 입력 | `NS`, `POD` | 승인된 namespace와 Pod exact name을 한 번 입력 |
| API 확인 자동값 | `POD_UID`, `POD_RV`, `NODE`, `OWNER_*`, `RAW_CONTAINER_ID`, `RUNTIME_PROVIDER` | 매 명령 직전 API에서 재조회 |
| 계산 후보값 | `CONTAINER_ID`, `CRI_ENDPOINT`, `RUNTIME_UNIT` | 허용 scheme 제거 또는 kubelet argv/config에서 추출하되 대조 전 미확정 |
| 운영자 선택값 | `CONTAINER`, `START_RFC3339`, `END_RFC3339`, `SSH_HOST` | 후보 출력과 사건 시각·승인된 host mapping을 보고 선택 |

**Bash·도구·현재 주체 확인**

`TOOLS_READY=false; MISSING=''; for T in kubectl jq grep sed date ssh; do command -v "$T" >/dev/null 2>&1 || MISSING="$MISSING $T"; done; if [ -z "${BASH_VERSION:-}" ]; then printf '%s\n' '중단: Bash에서 실행한다.' >&2; elif [ -n "$MISSING" ]; then printf '중단: 없는 도구=%s\n' "$MISSING" >&2; elif kubectl auth whoami >/dev/null 2>&1 && kubectl auth can-i get pods --all-namespaces | grep -Fxq yes && kubectl auth can-i get nodes | grep -Fxq yes; then TOOLS_READY=true; kubectl auth whoami; else printf '%s\n' '중단: 현재 주체 확인 또는 get pods/nodes 권한 실패' >&2; fi`

`kubectl auth can-i` 성공은 이후 실제 요청 성공을 보장하지 않는다. audit 조회는 별도 승인된 audit backend에서 사건 ID, 현재 주체, verb/resource, namespace/name, response code만 요청하며 token·인증서·요청 본문은 수집하지 않는다.

**최초 입력·이전 상태 완전 초기화**

`unset INPUT_READY AUTO_READY SELECTION_READY TIME_READY HOST_READY CRI_READY EVIDENCE_READY MUTATION_READY REPLACEMENT_READY NORMALIZATION_READY POD_UID POD_RV NODE NODE_UID NODE_RV OWNER_KIND OWNER_NAME OWNER_UID CONTAINER RAW_CONTAINER_ID CONTAINER_ID RUNTIME_VERSION RUNTIME_PROVIDER CRI_ENDPOINT CRI_SOCKET RUNTIME_UNIT RUNTIME_MAIN_PID KUBELET_UNIT HOST_IDENTITY SSH_HOST START_RFC3339 END_RFC3339 START_EPOCH END_EPOCH START_CANON END_CANON ARGV_RE MUTATION_STARTED_AT MUTATION_STARTED_EPOCH DOWNSTREAM_READY; SSH_ARGS=(); NS='PASTE_NAMESPACE_HERE'; POD='PASTE_POD_HERE'; INPUT_READY=false; if [ "$TOOLS_READY" != true ]; then printf '%s\n' '중단: TOOLS_READY가 아니다.' >&2; elif J="$(kubectl get pod "$POD" -n "$NS" -o json)" && [ "$(jq -r '.metadata.name' <<<"$J")" = "$POD" ] && [ "$(jq -r '.metadata.namespace' <<<"$J")" = "$NS" ]; then INPUT_READY=true; printf 'INPUT_READY=true namespace=%s pod=%s\n' "$NS" "$POD"; else printf '%s\n' '중단: Pod exact-name 조회 실패' >&2; fi`

**현재 identity 자동 등록**

printf 'AUTO_READY=true\nPod UID/RV: %s %s\nOwner: %s/%s UID=%s\nNode UID/RV: %s %s %s\nRuntime: %s\n' "$POD_UID" "$POD_RV" "$OWNER_KIND" "$OWNER_NAME" "$OWNER_UID" "$NODE" "$NODE_UID" "$NODE_RV" "$RUNTIME_VERSION"; else printf '%s\n' '중단: 필수 identity/owner/runtime provider가 비었음' >&2; fi; else printf '%s\n' '중단: Pod 또는 Node API 조회 실패' >&2; fi`

재등록 전에는 위 명령처럼 UID/RV/owner/containerID/node/host/relation/time/argv-regex와 모든 downstream `READY`를 먼저 무효화한다. 실패하면 이전 값은 사용하지 않는다.

**fresh API gate 함수 등록**

`fresh_api(){ local p n okind oname ouid; unset CURRENT_POD_RV CURRENT_NODE_RV; p="$(kubectl get pod "$POD" -n "$NS" -o json)" || return 1; n="$(kubectl get node "$NODE" -o json)" || return 1; okind="$(jq -r '.metadata.ownerReferences[0].kind // "None"' <<<"$p")"; oname="$(jq -r '.metadata.ownerReferences[0].name // "-"' <<<"$p")"; ouid="$(jq -r '.metadata.ownerReferences[0].uid // "-"' <<<"$p")"; [ "$(jq -r '.metadata.uid' <<<"$p")" = "$POD_UID" ] || return 1; [ "$(jq -r '.spec.nodeName' <<<"$p")" = "$NODE" ] || return 1; [ "$okind/$oname/$ouid" = "$OWNER_KIND/$OWNER_NAME/$OWNER_UID" ] || return 1; [ "$(jq -r '.metadata.uid' <<<"$n")" = "$NODE_UID" ] || return 1; CURRENT_POD_RV="$(jq -r '.metadata.resourceVersion' <<<"$p")"; CURRENT_NODE_RV="$(jq -r '.metadata.resourceVersion' <<<"$n")"; [ -n "$CURRENT_POD_RV" ] && [ -n "$CURRENT_NODE_RV" ]; }`

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | 현재 Pod의 container ID와 CRI list/inspect의 ID·Pod UID·시각이 다르거나 현재 ID가 `NotFound` |
| 적용 | Pod UID/RV, owner, node UID, container 이름, runtime provider와 실제 endpoint를 같은 사건 창에서 대조 가능 |
| 제외 | 종료 뒤 GC된 과거 ID, 새 UID의 replacement, `ContainerStatusUnknown`, 단순 logs/exec 실패는 각각 해당 lifecycle 런북으로 이관 |

검색 문자열: `containerID`, `NotFound`, `failed to update status`, `PLEG`, `garbage collect`, `runtime restart`, `pod UID`, `rpc error`.

## 최초 확인과 container 선택

`fresh_api && kubectl get pod "$POD" -n "$NS" -o json | jq -r '"=== Pod ===\nUID/RV: \(.metadata.uid) \(.metadata.resourceVersion)\nNode: \(.spec.nodeName)\nOwner: \(.metadata.ownerReferences[0].kind)/\(.metadata.ownerReferences[0].name) UID=\(.metadata.ownerReferences[0].uid)\n" + ([.status.containerStatuses[]? | "Container: \(.name)\nCurrent ID: \(.containerID // "-")\nReady: \(.ready)\nState: \(.state|keys[0])\nStarted: \(.state.running.startedAt // "-")\nFinished: \(.state.terminated.finishedAt // .lastState.terminated.finishedAt // "-")"] | join("\n"))'`

`CONTAINER='PASTE_CONTAINER_HERE'; unset SELECTION_READY RAW_CONTAINER_ID CONTAINER_ID ARGV_RE DOWNSTREAM_READY; SELECTION_READY=false; if fresh_api && PJ="$(kubectl get pod "$POD" -n "$NS" -o json)" && RAW_CONTAINER_ID="$(jq -r --arg c "$CONTAINER" '.status.containerStatuses[] | select(.name==$c) | .containerID // empty' <<<"$PJ")"; then case "$RAW_CONTAINER_ID" in containerd://*) [ "$RUNTIME_PROVIDER" = containerd ] && CONTAINER_ID="${RAW_CONTAINER_ID#containerd://}";; cri-o://*) [ "$RUNTIME_PROVIDER" = cri-o ] && CONTAINER_ID="${RAW_CONTAINER_ID#cri-o://}";; esac; if [[ "$CONTAINER_ID" =~ ^[a-fA-F0-9]{12,128}$ ]]; then ARGV_RE="$(printf '%s' "$CONTAINER_ID" | sed 's/[][\\.^$*+?(){}|]/\\&/g')"; SELECTION_READY=true; printf 'SELECTION_READY=true container=%s raw=%s normalized=%s regex=^%s$\n' "$CONTAINER" "$RAW_CONTAINER_ID" "$CONTAINER_ID" "$ARGV_RE"; else unset RAW_CONTAINER_ID CONTAINER_ID ARGV_RE; printf '%s\n' '중단: provider와 ID scheme/형식 exact match 실패' >&2; fi; else unset RAW_CONTAINER_ID CONTAINER_ID ARGV_RE; printf '%s\n' '중단: fresh Pod/container 조회 실패' >&2; fi`

`fresh_target(){ local p current owner; [ "$SELECTION_READY" = true ] || return 1; fresh_api || return 1; p="$(kubectl get pod "$POD" -n "$NS" -o json)" || return 1; current="$(jq -r --arg c "$CONTAINER" '.status.containerStatuses[]? | select(.name==$c) | .containerID // empty' <<<"$p")"; [ "$current" = "$RAW_CONTAINER_ID" ] || return 1; if [ "$OWNER_UID" != '-' ]; then owner="$(kubectl get "$OWNER_KIND" "$OWNER_NAME" -n "$NS" -o json)" || return 1; [ "$(jq -r '.metadata.uid' <<<"$owner")" = "$OWNER_UID" ] || return 1; fi; }`

ID 정규화는 허용된 provider scheme을 정확히 한 번 제거한다. suffix·부분 ID는 증거로 채택하지 않고 CRI 결과의 `status.id`와 exact match한다.

## 시간·host·CRI gate

`START_RFC3339='PASTE_START_RFC3339_HERE'; END_RFC3339='PASTE_END_RFC3339_HERE'; unset TIME_READY CRI_READY EVIDENCE_READY START_EPOCH END_EPOCH START_CANON END_CANON DOWNSTREAM_READY; TIME_READY=false; EVIDENCE_READY=false; if [[ "$START_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && [[ "$END_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && START_EPOCH="$(date -u -d "$START_RFC3339" +%s 2>/dev/null)" && END_EPOCH="$(date -u -d "$END_RFC3339" +%s 2>/dev/null)"; then START_CANON="$(date -u -d "@$START_EPOCH" '+%Y-%m-%dT%H:%M:%SZ')"; END_CANON="$(date -u -d "@$END_EPOCH" '+%Y-%m-%dT%H:%M:%SZ')"; if [ "$START_RFC3339" = "$START_CANON" ] && [ "$END_RFC3339" = "$END_CANON" ] && [ "$START_EPOCH" -lt "$END_EPOCH" ]; then TIME_READY=true; printf 'TIME_READY=true start=%s end=%s\n' "$START_CANON" "$END_CANON"; else printf '%s\n' '중단: UTC RFC3339 canonical round-trip 또는 start<end 실패' >&2; fi; else printf '%s\n' '중단: strict UTC RFC3339 regex/parse 실패' >&2; fi`

`SSH_HOST='PASTE_APPROVED_SSH_HOST_HERE'; unset HOST_READY CRI_READY HOST_IDENTITY DOWNSTREAM_READY; SSH_ARGS=(); HOST_READY=false; if fresh_api && [[ "$SSH_HOST" =~ ^[A-Za-z0-9_.@:-]+$ ]] && [[ "$SSH_HOST" != -* ]]; then SSH_ARGS=(-o BatchMode=yes -o StrictHostKeyChecking=yes -o ClearAllForwardings=yes -o ConnectTimeout=10 "$SSH_HOST"); REMOTE_ID="$(ssh "${SSH_ARGS[@]}" -- 'printf "hostname=%s machine_id=%s boot_id=%s\n" "$(hostname -f 2>/dev/null || hostname)" "$(cat /etc/machine-id 2>/dev/null)" "$(cat /proc/sys/kernel/random/boot_id 2>/dev/null)"')"; SSH_RC=$?; printf '%s\n' "$REMOTE_ID"; read -r -p 'CMDB에서 이 host가 대상 Node임을 확인했으면 Node 이름 입력: ' HOST_CONFIRM; if [ "$SSH_RC" -eq 0 ] && [ "$HOST_CONFIRM" = "$NODE" ] && [[ "$REMOTE_ID" =~ ^hostname=[^[:space:]]+[[:space:]]machine_id=[0-9a-fA-F-]+[[:space:]]boot_id=[0-9a-fA-F-]+$ ]]; then HOST_IDENTITY="$REMOTE_ID"; HOST_READY=true; printf 'HOST_READY=true host=%s node=%s uid=%s\n' "$SSH_HOST" "$NODE" "$NODE_UID"; else printf '%s\n' '중단: 원격 host identity/CMDB 승인 실패' >&2; fi; else printf '%s\n' '중단: fresh API 또는 승인 SSH host 형식 실패' >&2; fi`

`host_guard(){ local now; HOST_READY=false; fresh_api || return 1; now="$(ssh "${SSH_ARGS[@]}" -- 'printf "hostname=%s machine_id=%s boot_id=%s\n" "$(hostname -f 2>/dev/null || hostname)" "$(cat /etc/machine-id 2>/dev/null)" "$(cat /proc/sys/kernel/random/boot_id 2>/dev/null)"')" || return 1; [ "$now" = "$HOST_IDENTITY" ] || return 1; HOST_READY=true; }`

각 원격 명령은 작업 PC의 `host_guard`로 현재 Pod/owner/container/node UID와 원격 machine-id·boot-id를 다시 확인한 뒤 실행한다. 작업 PC 변수가 원격 shell에 자동 전달된다고 가정하지 않으며, 검증된 값만 SSH positional argument로 넘긴다.

`host_guard || exit 1; UNIT_OUTPUT="$(ssh "${SSH_ARGS[@]}" -- systemctl list-units --type=service --all --no-legend)"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ]; then mapfile -t KU < <(grep -E '(^|[[:space:]])kubelet\.service([[:space:]]|$)' <<<"$UNIT_OUTPUT" | sed -E 's/^[[:space:]]*([^[:space:]]+).*/\1/'); FILTER_RC=$?; else FILTER_RC=99; fi; if [ "$PRODUCER_RC" -eq 0 ] && [ "$FILTER_RC" -eq 0 ] && [ "${#KU[@]}" -eq 1 ]; then KUBELET_UNIT="${KU[0]}"; printf 'KUBELET_READY=true unit=%s\n' "$KUBELET_UNIT"; else printf '중단: kubelet unit producer=%s filter=%s count=%s\n' "$PRODUCER_RC" "$FILTER_RC" "${#KU[@]}" >&2; exit 1; fi`

`unset CRI_READY CRI_ENDPOINT CRI_SOCKET RUNTIME_UNIT RUNTIME_MAIN_PID DOWNSTREAM_READY; CRI_READY=false; host_guard || exit 1; KUBELET_ARGV="$(ssh "${SSH_ARGS[@]}" -- systemctl show "$KUBELET_UNIT" -p ExecStart --value)"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ]; then ENDPOINT_MATCHES="$(grep -Eo -- '--container-runtime-endpoint(=| )[[:space:]]*[^[:space:];]+' <<<"$KUBELET_ARGV")"; FILTER_RC=$?; EPS=(); [ "$FILTER_RC" -eq 0 ] && mapfile -t EPS < <(sed -E 's/^--container-runtime-endpoint(=| )[[:space:]]*//' <<<"$ENDPOINT_MATCHES"); else FILTER_RC=99; EPS=(); fi; if [ "$PRODUCER_RC" -ne 0 ] || [ "$FILTER_RC" -ne 0 ] || [ "${#EPS[@]}" -ne 1 ] || [[ ! "${EPS[0]}" =~ ^unix:///[-A-Za-z0-9._/]+\.sock$ ]]; then printf '중단: active kubelet argv endpoint producer=%s filter=%s count=%s; 기본 socket 금지, 실제 --config 절대경로의 regular/readable 파일에서 top-level containerRuntimeEndpoint exact-one을 별도 검증\n' "$PRODUCER_RC" "$FILTER_RC" "${#EPS[@]}" >&2; else CRI_ENDPOINT="${EPS[0]}"; CRI_SOCKET="${CRI_ENDPOINT#unix://}"; case "$RUNTIME_PROVIDER" in containerd) RUNTIME_UNIT=containerd.service;; cri-o) RUNTIME_UNIT=crio.service;; *) RUNTIME_UNIT='';; esac; RUNTIME_STATE="$(ssh "${SSH_ARGS[@]}" -- systemctl show "$RUNTIME_UNIT" -p ActiveState -p MainPID)"; STATE_RC=$?; SOCKET_STATE="$(ssh "${SSH_ARGS[@]}" -- sudo ss -lxnp)"; SOCKET_RC=$?; RUNTIME_MAIN_PID="$(sed -n 's/^MainPID=//p' <<<"$RUNTIME_STATE")"; SOCKET_LINE="$(grep -F "$CRI_SOCKET" <<<"$SOCKET_STATE")"; SOCKET_FILTER_RC=$?; if [ "$SOCKET_FILTER_RC" -eq 0 ]; then grep -Fq "pid=$RUNTIME_MAIN_PID," <<<"$SOCKET_LINE"; PROVIDER_FILTER_RC=$?; else PROVIDER_FILTER_RC=99; fi; if [ "$STATE_RC" -eq 0 ] && [ "$SOCKET_RC" -eq 0 ] && grep -Fxq 'ActiveState=active' <<<"$RUNTIME_STATE" && [[ "$RUNTIME_MAIN_PID" =~ ^[1-9][0-9]*$ ]] && [ "$SOCKET_FILTER_RC" -eq 0 ] && [ "$PROVIDER_FILTER_RC" -eq 0 ]; then CRI_READY=true; printf 'CRI_READY=true provider=%s endpoint=%s socket=%s unit=%s pid=%s\n' "$RUNTIME_PROVIDER" "$CRI_ENDPOINT" "$CRI_SOCKET" "$RUNTIME_UNIT" "$RUNTIME_MAIN_PID"; else unset CRI_ENDPOINT CRI_SOCKET RUNTIME_UNIT RUNTIME_MAIN_PID; printf '중단: runtime/socket 관계 state=%s socket_producer=%s socket_filter=%s provider_filter=%s\n' "$STATE_RC" "$SOCKET_RC" "$SOCKET_FILTER_RC" "$PROVIDER_FILTER_RC" >&2; fi; fi`

argv에 endpoint가 정확히 하나 없으면 `--config` 후보도 exact-one이어야 한다. fallback 파일은 절대경로, regular readable file, top-level `containerRuntimeEndpoint` key 정확히 하나, nonempty `unix:///...sock` 형식과 active argv가 가리키는 현재 파일임을 확인한 뒤 위 provider/MainPID/socket 검증을 반복한다. 이 검증 전에는 `crictl`을 실행하지 않는다.

`cri_guard(){ local state socket_state pid line; CRI_READY=false; fresh_target || return 1; host_guard || return 1; [ "$TIME_READY" = true ] || return 1; state="$(ssh "${SSH_ARGS[@]}" -- systemctl show "$RUNTIME_UNIT" -p ActiveState -p MainPID)" || return 1; socket_state="$(ssh "${SSH_ARGS[@]}" -- sudo ss -lxnp)" || return 1; pid="$(sed -n 's/^MainPID=//p' <<<"$state")"; line="$(grep -F "$CRI_SOCKET" <<<"$socket_state")" || return 1; [ "$(sed -n 's/^ActiveState=//p' <<<"$state")" = active ] && [ "$pid" = "$RUNTIME_MAIN_PID" ] && grep -Fq "pid=$pid," <<<"$line" || return 1; CRI_READY=true; }`

## 관측별 진단 경로

| 지금 보이는 것 | 먼저 볼 것 | 그다음 어디를 볼지 | 상세 절 |
|---|---|---|---|
| current ID가 CRI `NotFound` | exact ID와 현재 Pod UID | GC/restart와 kubelet status 보고 | [[#CRI와 journal 증거]] |
| 같은 이름의 다른 ID | CRI Pod UID·sandbox ID·createdAt | replacement 또는 stale API 분리 | [[#CRI와 journal 증거]] |
| 여러 Pod에서 동시 불일치 | runtime service start와 filesystem | node/runtime owner | [[#CRI와 journal 증거]] |

## CRI와 journal 증거

원격 Node에서 아래 명령은 각각 실행 직전에 `CRI_READY=true`, socket 존재, provider unit active를 다시 확인한다. `inspect` 전체는 저장하지 않는다.

`EVIDENCE_READY=false; cri_guard || exit 1; INSPECT="$(ssh "${SSH_ARGS[@]}" -- sudo crictl --runtime-endpoint="$CRI_ENDPOINT" inspect "$CONTAINER_ID")"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ] && jq -e --arg id "$CONTAINER_ID" --arg uid "$POD_UID" --arg ns "$NS" --arg pod "$POD" --arg c "$CONTAINER" '.status.id==$id and .status.labels["io.kubernetes.pod.uid"]==$uid and .status.labels["io.kubernetes.pod.namespace"]==$ns and .status.labels["io.kubernetes.pod.name"]==$pod and .status.labels["io.kubernetes.container.name"]==$c' <<<"$INSPECT" >/dev/null; then EVIDENCE_READY=true; printf 'EVIDENCE_READY=true id=%s pod_uid=%s\n' "$CONTAINER_ID" "$POD_UID"; else EVIDENCE_READY=false; printf '중단: CRI inspect producer=%s 또는 exact identity filter 실패\n' "$PRODUCER_RC" >&2; fi`

`[ "$EVIDENCE_READY" = true ] && jq -c '{id:.status.id,state:.status.state,createdAt:.status.createdAt,startedAt:.status.startedAt,finishedAt:.status.finishedAt,exitCode:.status.exitCode,reason:.status.reason,podSandboxId:.status.podSandboxId,labels:{pod_uid:.status.labels["io.kubernetes.pod.uid"],pod_name:.status.labels["io.kubernetes.pod.name"],namespace:.status.labels["io.kubernetes.pod.namespace"],container:.status.labels["io.kubernetes.container.name"]}}' <<<"$INSPECT"`

CRI label의 Pod UID·namespace·name·container와 API identity가 모두 exact match해야 같은 인스턴스다. `NotFound`의 명령 실패와 `jq` filter 실패를 성공/빈 결과로 바꾸지 않는다.

원격 journal용 시각은 canonical RFC3339 값을 인수로 사용한다. producer와 filter 종료코드를 분리한다.

`fresh_target || exit 1; host_guard || exit 1; [ "$TIME_READY" = true ] && [ "$EVIDENCE_READY" = true ] || exit 1; TMP="$(mktemp)" || exit 1; ssh "${SSH_ARGS[@]}" -- sudo journalctl -u "$KUBELET_UNIT" --since "$START_CANON" --until "$END_CANON" --no-pager >"$TMP"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ]; then grep -E -i -C 3 -- "$POD_UID|$ARGV_RE|failed to update status|PLEG|NotFound|runtime|timeout|sandbox" "$TMP"; FILTER_RC=$?; else FILTER_RC=99; fi; rm -f "$TMP"; printf 'producer_rc=%s filter_rc=%s\n' "$PRODUCER_RC" "$FILTER_RC"; [ "$PRODUCER_RC" -eq 0 ] && { [ "$FILTER_RC" -eq 0 ] || [ "$FILTER_RC" -eq 1 ]; }`

`cri_guard || exit 1; [ "$EVIDENCE_READY" = true ] || exit 1; TMP="$(mktemp)" || exit 1; ssh "${SSH_ARGS[@]}" -- sudo journalctl -u "$RUNTIME_UNIT" --since "$START_CANON" --until "$END_CANON" --no-pager >"$TMP"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ]; then grep -E -i -C 3 -- "$POD_UID|$ARGV_RE|gc|garbage|restart|start|stop|NotFound|shim|rpc|filesystem|error" "$TMP"; FILTER_RC=$?; else FILTER_RC=99; fi; rm -f "$TMP"; printf 'producer_rc=%s filter_rc=%s\n' "$PRODUCER_RC" "$FILTER_RC"; [ "$PRODUCER_RC" -eq 0 ] && { [ "$FILTER_RC" -eq 0 ] || [ "$FILTER_RC" -eq 1 ]; }`

`filter_rc=1`은 일치 행 없음이지 오류 부재가 아니다. `producer_rc!=0` 또는 `filter_rc>1`이면 증거 수집 실패다. API/CRI/journal의 시각은 UTC canonical 값으로 환산하고 `API startedAt <= CRI startedAt < finishedAt <= 사건 end`처럼 기대 관계를 명시해 대조한다. clock skew·정밀도 허용치는 확인된 NTP/플랫폼 계약이 없으면 미확정이다.

## 복구 전 gate와 replacement

진단 중 `crictl stop/rm`, Pod 삭제, kubelet/runtime restart, cache/metadata 삭제, cordon/drain을 실행하지 않는다. 변경은 fresh API·CRI·journal 증거, 승인 ID, 변경 window, PDB/quorum, 멱등성·외부 side effect, rollback/중단 조건이 모두 기록되고 `MUTATION_READY=true`로 별도 승인된 경우만 가능하다.

Controller Pod replacement가 승인됐더라도 실행 직전에 `fresh_api`, 현재 UID·owner UID, desired/Ready, PDB와 서비스 여유를 다시 확인한다. 삭제는 비가역이므로 rollback으로 쓰지 않는다.

`MUTATION_READY=false; unset MUTATION_STARTED_AT MUTATION_STARTED_EPOCH; if fresh_target && cri_guard && [ "$EVIDENCE_READY" = true ] && [ -n "${APPROVAL_ID:-}" ] && [ "${WINDOW_OPEN:-false}" = true ] && [ "${QUORUM_SAFE:-false}" = true ] && [ "${IDEMPOTENCE_SAFE:-false}" = true ] && [ "${ROLLBACK_READY:-false}" = true ] && MUTATION_STARTED_AT="$(date -u '+%Y-%m-%dT%H:%M:%SZ')" && MUTATION_STARTED_EPOCH="$(date -u -d "$MUTATION_STARTED_AT" +%s)"; then MUTATION_READY=true; printf 'MUTATION_READY=true approval=%s pod_uid=%s owner_uid=%s at=%s\n' "$APPROVAL_ID" "$POD_UID" "$OWNER_UID" "$MUTATION_STARTED_AT"; else printf '%s\n' '중단: fresh target/CRI/evidence/approval/window/quorum/idempotence/rollback gate 실패' >&2; fi`

`[ "$MUTATION_READY" = true ] && fresh_target && cri_guard && [ "$EVIDENCE_READY" = true ] && [ -n "$APPROVAL_ID" ] && [ "$WINDOW_OPEN" = true ] && [ "$QUORUM_SAFE" = true ] && [ "$IDEMPOTENCE_SAFE" = true ] && [ "$ROLLBACK_READY" = true ] && kubectl delete pod "$POD" -n "$NS" --wait=true --timeout=120s`

재등록은 기존 이름을 재사용하지 않는다. owner가 만든 현재 Pod를 owner UID로 조회해 후보가 정확히 하나일 때 새 이름·UID/RV/node/container ID를 모두 다시 등록한다. 후보가 0개 또는 여러 개면 `REPLACEMENT_READY=false`로 중단한다.

`unset REPLACEMENT_READY NEW_POD NEW_POD_UID NEW_POD_RV NEW_NODE NEW_CONTAINER_ID; REPLACEMENT_READY=false; if [ "$MUTATION_READY" = true ] && [ "$OWNER_UID" != '-' ] && OWNER_NOW="$(kubectl get "$OWNER_KIND" "$OWNER_NAME" -n "$NS" -o json)" && [ "$(jq -r '.metadata.uid' <<<"$OWNER_NOW")" = "$OWNER_UID" ] && LIST="$(kubectl get pods -n "$NS" -o json)"; then mapfile -t CAND < <(jq -r --arg u "$OWNER_UID" --arg c "$CONTAINER" --arg old_uid "$POD_UID" --arg old_name "$POD" --argjson started "$MUTATION_STARTED_EPOCH" '.items[] | select(any(.metadata.ownerReferences[]?; .uid==$u)) | select(.metadata.uid!=$old_uid and .metadata.name!=$old_name and .metadata.deletionTimestamp==null) | select((.metadata.creationTimestamp|fromdateiso8601) >= $started) | [.metadata.name,.metadata.uid,.metadata.resourceVersion,.spec.nodeName,([.status.containerStatuses[]? | select(.name==$c) | .containerID][0] // "")] | @tsv' <<<"$LIST"); if [ "${#CAND[@]}" -eq 1 ]; then IFS=$'\t' read -r NEW_POD NEW_POD_UID NEW_POD_RV NEW_NODE NEW_CONTAINER_ID <<<"${CAND[0]}"; if [ -n "$NEW_NODE" ] && [ -n "$NEW_CONTAINER_ID" ]; then REPLACEMENT_READY=true; printf 'REPLACEMENT_READY=true pod=%s uid=%s rv=%s node=%s id=%s\n' "$NEW_POD" "$NEW_POD_UID" "$NEW_POD_RV" "$NEW_NODE" "$NEW_CONTAINER_ID"; fi; else printf '중단: replacement 후보 count=%s\n' "${#CAND[@]}" >&2; fi; else printf '%s\n' '중단: current owner UID 또는 replacement 재등록 API 실패' >&2; fi`

## 정상화 검증

`NORMALIZATION_READY=false; if [ "$REPLACEMENT_READY" = true ] && J="$(kubectl get pod "$NEW_POD" -n "$NS" -o json)" && [ "$(jq -r '.metadata.uid' <<<"$J")" = "$NEW_POD_UID" ] && [ "$(jq -r '.metadata.resourceVersion' <<<"$J")" = "$NEW_POD_RV" ] && [ "$(jq -r '.spec.nodeName' <<<"$J")" = "$NEW_NODE" ] && [ "$(jq -r '.metadata.ownerReferences[0].uid // empty' <<<"$J")" = "$OWNER_UID" ] && jq -e '.status.containerStatuses|length>0 and all(.[]; .ready==true and (.containerID|length>0))' <<<"$J" >/dev/null; then NORMALIZATION_READY=true; jq -r '"Pod=\(.metadata.name) UID=\(.metadata.uid) RV=\(.metadata.resourceVersion) Node=\(.spec.nodeName) Ready=\([.status.containerStatuses[].ready]|join(",")) IDs=\([.status.containerStatuses[].containerID]|join(","))"' <<<"$J"; else printf '%s\n' '중단: replacement current owner/identity/Ready 정상화 실패' >&2; fi`

실제 정상화는 새 UID의 API container ID와 새 node의 확인된 CRI exact ID·Pod UID가 일치하고, kubelet/runtime에서 같은 오류가 관찰 window 동안 재발하지 않으며, owner desired=Ready, EndpointSlice `ready=true`, 실제 서비스 요청 또는 Job 산출물이 정상일 때만 선언한다. 어떤 단계든 실패하면 추가 mutation 없이 owner에게 이관한다.

## Prometheus 빠른 관측

Prometheus가 실제 수집되고 label을 확인한 경우에만 완성 쿼리를 생성한다. 이름은 regex escape하고 빈 결과를 정상으로 보지 않는다.

`fresh_target || exit 1; NODE_RE="$(printf '%s' "$NODE" | sed 's/[][\\.^$*+?(){}|]/\\&/g')"; printf 'sum by(operation_type)(rate(kubelet_runtime_operations_errors_total{node=~"^%s$",operation_type=~"container_status|pod_status|status"}[5m]))\n' "$NODE_RE"`

## 증거·환경 미검증

Pod/owner/node UID·RV, raw/normalized container ID, runtime provider, endpoint source, socket/unit, 원격 hostname/machine-id, canonical 시간과 producer/filter RC, CRI 요약, 승인·mutation·replacement·정상화 결과만 보존한다. Secret, token, 개인키, env/args/config 원문과 전체 journal은 저장하지 않는다.

Kubernetes 1.26의 CRI 출력, kubelet endpoint 설정 위치, runtime unit, SSH/CMDB mapping, NTP 정밀도, audit backend, Prometheus metric·label은 환경별 미검증이다. 이 문서는 작성 검토만 수행했으며 실제 인프라에는 실행하지 않았다.
