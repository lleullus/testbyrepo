# 같은 Node에서 container create·start와 shim 오류가 반복되는 경우

## 입력값과 빠른 수집

실행 hop은 `작업 PC/bastion → 현재 Pod UID·owner → 현재 Node UID/machineID → 원격 machine-id → active kubelet의 exact CRI endpoint → socket provider PID → sandbox/container → OCI shim → PID start time·mount/cgroup namespace`다. image·command·mount·securityContext처럼 한 Pod에만 있는 원인을 먼저 배제한다.

| 종류 | 값 | gate |
|---|---|---|
| 최초 입력 | namespace, 대표 Pod | exact API name |
| API 자동값 | Pod UID/RV/created, owner UID, node UID/RV/machineID, runtime provider, regular container 상태/ID | API 호출과 relation 모두 성공 |
| 계산 후보 | escaped Pod UID/container ID regex, CRI socket | exact endpoint/provider 확인 전 미확정 |
| 운영자 선택값 | regular container, SSH host, units, CRI endpoint, UTC 창, 조건부 shim PID·argv regex | 현재 출력·CMDB·journal 직접 증거 |

```bash
TOOLS_READY=false; MISSING=''
for tool in kubectl jq ssh date grep sed mktemp; do command -v "$tool" >/dev/null 2>&1 || MISSING="$MISSING $tool"; done
if [ -n "${BASH_VERSION:-}" ] && [ -z "$MISSING" ]; then TOOLS_READY=true; else printf '중단: Bash/도구 없음:%s\n' "$MISSING" >&2; fi

invalidate(){
  unset API_READY SELECTION_READY TIME_READY HOST_READY ENDPOINT_READY EVIDENCE_READY CRI_READY SHIM_READY NORMALIZATION_READY POD_JSON POD_UID POD_RV POD_CREATED OWNER_KIND OWNER_NAME OWNER_UID NODE NODE_UID NODE_RV NODE_MACHINE_ID RUNTIME_PROVIDER CONTAINER RAW_CONTAINER_ID CONTAINER_ID POD_RE ID_RE SSH_HOST REMOTE_MACHINE_ID KUBELET_UNIT RUNTIME_UNIT CRI_ENDPOINT CRI_SOCKET RUNTIME_MAIN_PID START_RFC3339 END_RFC3339 START_EPOCH END_EPOCH SHIM_PID SHIM_ARGV_RE SHIM_START SHIM_MNT_NS SHIM_CGROUP_NS SHIM_NET_NS
  API_READY=false; SELECTION_READY=false; TIME_READY=false; HOST_READY=false; ENDPOINT_READY=false; EVIDENCE_READY=false; CRI_READY=false; SHIM_READY=false; NORMALIZATION_READY=false
}
invalidate
NS='PASTE_NAMESPACE_HERE'
POD='PASTE_POD_HERE'
INPUT_READY=false
if [ "$TOOLS_READY" = true ] && kubectl get pod "$POD" -n "$NS" >/dev/null 2>&1; then INPUT_READY=true; else printf '%s\n' '중단: exact Pod 입력/API 실패' >&2; fi

refresh(){
  invalidate
  [ "$INPUT_READY" = true ] || return 1
  local p n count owner
  p="$(kubectl get pod "$POD" -n "$NS" -o json)" || return 1
  NODE="$(jq -er '.spec.nodeName|select(length>0)' <<<"$p")" || return 1
  n="$(kubectl get node "$NODE" -o json)" || return 1
  count="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)]|length' <<<"$p")" || return 1
  [ "$count" -eq 1 ] || return 1
  POD_JSON="$p"; POD_UID="$(jq -er '.metadata.uid' <<<"$p")"; POD_RV="$(jq -er '.metadata.resourceVersion' <<<"$p")"; POD_CREATED="$(jq -er '.metadata.creationTimestamp' <<<"$p")"
  OWNER_KIND="$(jq -er '[.metadata.ownerReferences[]|select(.controller==true)][0].kind' <<<"$p")"; OWNER_NAME="$(jq -er '[.metadata.ownerReferences[]|select(.controller==true)][0].name' <<<"$p")"; OWNER_UID="$(jq -er '[.metadata.ownerReferences[]|select(.controller==true)][0].uid' <<<"$p")"
  owner="$(kubectl get "$OWNER_KIND" "$OWNER_NAME" -n "$NS" -o json)" || return 1; [ "$(jq -r '.metadata.uid' <<<"$owner")" = "$OWNER_UID" ] || return 1
  NODE_UID="$(jq -er '.metadata.uid' <<<"$n")"; NODE_RV="$(jq -er '.metadata.resourceVersion' <<<"$n")"; NODE_MACHINE_ID="$(jq -er '.status.nodeInfo.machineID|select(length>0)' <<<"$n")"
  case "$(jq -r '.status.nodeInfo.containerRuntimeVersion // empty' <<<"$n")" in containerd://*) RUNTIME_PROVIDER=containerd;; cri-o://*) RUNTIME_PROVIDER=cri-o;; *) return 1;; esac
  POD_RE="$(printf '%s' "$POD_UID" | sed 's/[][\\.^$*+?(){}|]/\\&/g')"
  API_READY=true
  printf 'API_READY=true\nPod=%s/%s UID/RV=%s/%s created=%s\nOwner=%s/%s UID=%s\nNode=%s UID/RV=%s/%s machineID=%s runtime=%s\n' "$NS" "$POD" "$POD_UID" "$POD_RV" "$POD_CREATED" "$OWNER_KIND" "$OWNER_NAME" "$OWNER_UID" "$NODE" "$NODE_UID" "$NODE_RV" "$NODE_MACHINE_ID" "$RUNTIME_PROVIDER"
  jq -r '(.spec.initContainers[]? | "init/\(.name)"),(.spec.containers[]? as $c | (.status.containerStatuses[]?|select(.name==$c.name)) as $s | "regular/\($c.name) id=\($s.containerID // "-") waiting=\($s.state.waiting.reason // "-") message=\($s.state.waiting.message // "-")")' <<<"$p"
}
refresh || { invalidate; printf '%s\n' '중단: Pod/owner/node relation 등록 실패' >&2; }

CONTAINER='PASTE_REGULAR_CONTAINER_HERE'
SELECTION_READY=false
if [ "$API_READY" = true ] && jq -e --arg c "$CONTAINER" '([.spec.containers[]|select(.name==$c)]|length)==1' <<<"$POD_JSON" >/dev/null; then
  RAW_CONTAINER_ID="$(jq -r --arg c "$CONTAINER" '.status.containerStatuses[]?|select(.name==$c)|.containerID // empty' <<<"$POD_JSON")"
  CONTAINER_ID=''
  case "$RAW_CONTAINER_ID" in containerd://*) [ "$RUNTIME_PROVIDER" = containerd ] && CONTAINER_ID="${RAW_CONTAINER_ID#containerd://}";; cri-o://*) [ "$RUNTIME_PROVIDER" = cri-o ] && CONTAINER_ID="${RAW_CONTAINER_ID#cri-o://}";; '') :;; esac
  if [ -z "$CONTAINER_ID" ] || [[ "$CONTAINER_ID" =~ ^[a-fA-F0-9]{12,128}$ ]]; then ID_RE="$(printf '%s' "$CONTAINER_ID" | sed 's/[][\\.^$*+?(){}|]/\\&/g')"; SELECTION_READY=true; printf 'SELECTION_READY=true container=%s currentID=%s\n' "$CONTAINER" "${RAW_CONTAINER_ID:--}"; fi
fi

fresh_target(){
  local p n o
  [ "$SELECTION_READY" = true ] || return 1
  p="$(kubectl get pod "$POD" -n "$NS" -o json)" || return 1; n="$(kubectl get node "$NODE" -o json)" || return 1; o="$(kubectl get "$OWNER_KIND" "$OWNER_NAME" -n "$NS" -o json)" || return 1
  [ "$(jq -r '.metadata.uid' <<<"$p")" = "$POD_UID" ] && [ "$(jq -r '.spec.nodeName' <<<"$p")" = "$NODE" ] && [ "$(jq -r '.metadata.uid' <<<"$n")" = "$NODE_UID" ] && [ "$(jq -r '.status.nodeInfo.machineID' <<<"$n")" = "$NODE_MACHINE_ID" ] && [ "$(jq -r '.metadata.uid' <<<"$o")" = "$OWNER_UID" ] || return 1
  [ "$(jq -r --arg c "$CONTAINER" '.status.containerStatuses[]?|select(.name==$c)|.containerID // empty' <<<"$p")" = "$RAW_CONTAINER_ID" ]
}

SSH_HOST='PASTE_APPROVED_SSH_HOST_HERE'
KUBELET_UNIT='PASTE_KUBELET_UNIT_HERE'
RUNTIME_UNIT='PASTE_RUNTIME_UNIT_HERE'
CRI_ENDPOINT='PASTE_EXACT_CRI_ENDPOINT_HERE'
START_RFC3339='PASTE_START_RFC3339_HERE'
END_RFC3339='PASTE_END_RFC3339_HERE'

TIME_READY=false
if [[ "$START_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && [[ "$END_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && [ "$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" = "$START_RFC3339" ] && [ "$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" = "$END_RFC3339" ]; then START_EPOCH="$(date -u -d "$START_RFC3339" +%s)"; END_EPOCH="$(date -u -d "$END_RFC3339" +%s)"; [ "$START_EPOCH" -lt "$END_EPOCH" ] && TIME_READY=true; fi

host_gate(){ HOST_READY=false; fresh_target || return 1; [[ "$SSH_HOST" =~ ^[A-Za-z0-9_.@:-]+$ ]] || return 1; REMOTE_MACHINE_ID="$(ssh -o BatchMode=yes -o StrictHostKeyChecking=yes -- "$SSH_HOST" 'test -r /etc/machine-id && tr -d "\n" </etc/machine-id')" || return 1; [ "$REMOTE_MACHINE_ID" = "$NODE_MACHINE_ID" ] || return 1; HOST_READY=true; }

endpoint_gate(){
  ENDPOINT_READY=false; host_gate || return 1
  ssh -o BatchMode=yes -o StrictHostKeyChecking=yes -- "$SSH_HOST" bash -s -- "$KUBELET_UNIT" "$RUNTIME_UNIT" "$CRI_ENDPOINT" <<'REMOTE'
set -eu
ku=$1; ru=$2; expected=$3; [ "$(systemctl is-active "$ku")" = active ]; kp="$(systemctl show "$ku" -p MainPID --value)"; [ "$kp" -gt 1 ]; mapfile -d '' -t av <"/proc/$kp/cmdline"
vals=(); cfg=''; for ((i=0;i<${#av[@]};i++)); do case "${av[i]}" in --container-runtime-endpoint=*) vals+=("${av[i]#*=}");; --container-runtime-endpoint) ((i++)); vals+=("${av[i]:-}");; --config=*) cfg="${av[i]#*=}";; --config) ((i++)); cfg="${av[i]:-}";; esac; done
if [ -n "$cfg" ]; then [[ "$cfg" = /* ]] && [ -f "$cfg" ] && [ -r "$cfg" ]; mapfile -t cv < <(sed -n 's/^[[:space:]]*containerRuntimeEndpoint:[[:space:]]*["'"']\{0,1\}\([^"'"'[:space:]]*\)["'"']\{0,1\}[[:space:]]*$/\1/p' "$cfg"); [ "${#cv[@]}" -eq 1 ]; vals+=("${cv[0]}"); fi
mapfile -t u < <(printf '%s\n' "${vals[@]}"|sed '/^$/d'|sort -u); [ "${#u[@]}" -eq 1 ] && [ "${u[0]}" = "$expected" ] && [[ "$expected" == unix:///* ]]; sock=${expected#unix://}; [ -S "$sock" ]; [ "$(systemctl is-active "$ru")" = active ]; rp="$(systemctl show "$ru" -p MainPID --value)"; [ "$rp" -gt 1 ]; line="$(sudo ss -xlpn|grep -F -- "$sock")"; grep -Fq -- "pid=$rp," <<<"$line"; printf 'socket=%s providerPID=%s\n' "$sock" "$rp"
REMOTE
  local rc=$?; [ "$rc" -eq 0 ] || return 1; CRI_SOCKET="${CRI_ENDPOINT#unix://}"; RUNTIME_MAIN_PID="$(ssh -o BatchMode=yes -o StrictHostKeyChecking=yes -- "$SSH_HOST" systemctl show "$RUNTIME_UNIT" -p MainPID --value)"; ENDPOINT_READY=true
}
endpoint_gate || printf '%s\n' '중단: host/active kubelet endpoint/provider 실패' >&2
```

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | 서로 다른 owner의 현재 Pod UID가 같은 Node와 UTC 창에서 create/start·shim 오류를 반복 |
| 적용 | image/command/args/securityContext/mount/CNI 차이를 배제하고 kubelet→CRI→OCI relation을 확인 가능 |
| 제외 | 개별 Pod 오류는 [[10. Pod STATUS가 CreateContainerError·RunContainerError인 경우]], sandbox/CNI는 [[24. Pod 이벤트에 FailedCreatePodSandBox가 기록된 경우]] |

검색 문자열: `CreateContainerError`, `RunContainerError`, `failed to create shim task`, `OCI runtime create failed`, `failed to start container`, `context deadline exceeded`, `rpc error`.

## 관측별 진단 경로

| 지금 보이는 것 | 직접 증거 | 다음 경계 | 상세 절 |
|---|---|---|---|
| 한 Pod만 실패 | image digest·command·mount·securityContext | Pod별 원인 | [[10. Pod STATUS가 CreateContainerError·RunContainerError인 경우]] |
| 여러 owner가 같은 Node에서 실패 | UID 제한 Event와 kubelet/runtime journal | CRI/OCI | [[#bounded Event·journal·CRI]] |

## bounded Event·journal·CRI

```bash
if fresh_target && [ "$TIME_READY" = true ]; then kubectl get events -n "$NS" --field-selector "involvedObject.kind=Pod,involvedObject.name=$POD,involvedObject.uid=$POD_UID" --sort-by=.metadata.creationTimestamp -o custom-columns='CREATED:.metadata.creationTimestamp,EVENT-TIME:.eventTime,SERIES-LAST:.series.lastObservedTime,REASON:.reason,COUNT:.count,MESSAGE:.message'; else printf '%s\n' '중단: fresh target/time 실패' >&2; fi

EVIDENCE_READY=false
if [ "$TIME_READY" = true ] && endpoint_gate && fresh_target; then
  for unit in "$KUBELET_UNIT" "$RUNTIME_UNIT"; do
    TMP="$(mktemp)" || exit 1; ssh -o BatchMode=yes -o StrictHostKeyChecking=yes -- "$SSH_HOST" sudo journalctl -u "$unit" --since "$START_RFC3339" --until "$END_RFC3339" -o short-iso --no-pager >"$TMP"; PRODUCER_RC=$?
    if [ "$PRODUCER_RC" -eq 0 ]; then grep -E -i -C 3 -- "$POD_RE|${ID_RE:-no-current-id}|createcontainer|runcontainer|create|start|shim|OCI|runc|rpc|deadline|timeout|CNI|IPAM|mount|permission|error" "$TMP"; FILTER_RC=$?; else FILTER_RC=99; fi
    rm -f "$TMP"; printf 'unit=%s producer_rc=%s filter_rc=%s\n' "$unit" "$PRODUCER_RC" "$FILTER_RC"; [ "$PRODUCER_RC" -eq 0 ] && { [ "$FILTER_RC" -eq 0 ] || [ "$FILTER_RC" -eq 1 ]; } || exit 1
  done
  EVIDENCE_READY=true
fi

CRI_READY=false
if [ "$EVIDENCE_READY" = true ] && endpoint_gate && fresh_target; then
  IDS="$(ssh -o BatchMode=yes -o StrictHostKeyChecking=yes -- "$SSH_HOST" sudo crictl --runtime-endpoint="$CRI_ENDPOINT" pods --name "$POD" --namespace "$NS" -q)"; PRODUCER_RC=$?
  MATCHES=()
  if [ "$PRODUCER_RC" -eq 0 ]; then while IFS= read -r id; do [ -n "$id" ] || continue; uid="$(ssh -o BatchMode=yes -o StrictHostKeyChecking=yes -- "$SSH_HOST" sudo crictl --runtime-endpoint="$CRI_ENDPOINT" inspectp "$id" | jq -r '.status.metadata.uid // empty')" || exit 1; [ "$uid" = "$POD_UID" ] && MATCHES+=("$id"); done <<<"$IDS"; fi
  if [ "$PRODUCER_RC" -eq 0 ] && [ "${#MATCHES[@]}" -le 1 ]; then CRI_READY=true; printf 'CRI_READY=true exactSandboxCount=%s sandbox=%s\n' "${#MATCHES[@]}" "${MATCHES[0]:--}"; else printf '중단: CRI producer=%s exact sandbox count=%s\n' "$PRODUCER_RC" "${#MATCHES[@]}" >&2; fi
fi
```

Event의 첫 create/start 실패, kubelet RPC, runtime OCI/shim 오류가 같은 UID와 창에 있어야 공통 runtime 경로다. `filter_rc=1`은 오류 부재가 아니다. sandbox 0개는 생성 전 실패일 수 있지만 CRI producer 실패와 구분한다.

## shim process identity

shim PID는 journal에 실제로 표시됐을 때만 선택한다. PID를 이름 검색의 첫 행으로 자동 선택하지 않는다.

```bash
SHIM_PID='PASTE_SHIM_PID_HERE'
SHIM_ARGV_RE='PASTE_SHIM_ARGV_REGEX_HERE'
SHIM_READY=false
if [ "$EVIDENCE_READY" = true ] && endpoint_gate && [[ "$SHIM_PID" =~ ^[1-9][0-9]*$ ]] && { grep -Eq -- "$SHIM_ARGV_RE" </dev/null 2>/dev/null || [ "$?" -eq 1 ]; }; then
  SHIM_ID="$(ssh -o BatchMode=yes -o StrictHostKeyChecking=yes -- "$SSH_HOST" bash -s -- "$SHIM_PID" "$SHIM_ARGV_RE" <<'REMOTE'
set -eu
p=$1; re=$2; s="$(cat "/proc/$p/stat")"; rest=${s##*) }; set -- $rest; start=${20}; tr '\0' '\n' <"/proc/$p/cmdline"|grep -Eq -- "$re"; printf '%s %s %s %s\n' "$start" "$(readlink "/proc/$p/ns/mnt")" "$(readlink "/proc/$p/ns/cgroup")" "$(readlink "/proc/$p/ns/net")"
REMOTE
)" || SHIM_ID=''
  if read -r SHIM_START SHIM_MNT_NS SHIM_CGROUP_NS SHIM_NET_NS <<<"$SHIM_ID" && [ -n "$SHIM_NET_NS" ]; then SHIM_READY=true; printf 'SHIM_READY=true pid=%s start=%s mnt=%s cgroup=%s net=%s\n' "$SHIM_PID" "$SHIM_START" "$SHIM_MNT_NS" "$SHIM_CGROUP_NS" "$SHIM_NET_NS"; fi
fi
```

`D` state, I/O error, read-only filesystem, cgroup/namespace 오류가 같은 PID start time과 창에 연결되면 Linux owner로 이관한다. `kill`, task·namespace·mount 수동 제거를 실행하지 않는다.

## 변경 경계·정상화

이 문서는 runtime/kubelet restart, Pod 삭제, shim kill, CRI stop/remove를 제공하지 않는다. 별도 변경은 fresh Pod/owner/Node UID, machineID, exact endpoint/provider, evidence, approval ID/window, PDB/quorum·RWO storage, Job·외부 side effect 멱등성, rollback 원본을 명령 직전에 다시 검증해야 한다.

변경 뒤 기존 UID·containerID·sandbox·shim PID와 모든 READY를 폐기한다. controller current UID와 selector, 변경 이후 creation time, owner chain이 맞는 단일 replacement를 선택한 뒤 `refresh`부터 다시 수행한다.

```bash
NORMALIZATION_READY=false
if [ "$TIME_READY" = true ] && endpoint_gate && fresh_target && [ "$EVIDENCE_READY" = true ]; then
  NOW="$(kubectl get pod "$POD" -n "$NS" -o json)" || exit 1
  if [ "$(jq -r '.metadata.uid' <<<"$NOW")" = "$POD_UID" ] && jq -e --arg c "$CONTAINER" '.status.containerStatuses[]?|select(.name==$c and .ready==true and (.containerID|length)>0)' <<<"$NOW" >/dev/null; then NORMALIZATION_READY=true; printf 'NORMALIZATION_READY=true podUID=%s\n' "$POD_UID"; fi
fi
```

실제 정상화는 같은 Node의 여러 owner에서 create/start 성공, 같은 오류 재발 없음, owner desired/Ready, EndpointSlice와 실제 요청 또는 Job 산출물·중복 없음까지 필요하다.

## Prometheus 빠른 관측

```bash
if fresh_target; then NODE_JSON="$(jq -Rn --arg v "$NODE" '$v')"; printf 'sum by(operation_type)(rate(kubelet_runtime_operations_errors_total{node=%s,operation_type=~"create_container|start_container"}[5m]))\n' "$NODE_JSON"; else printf '%s\n' '중단: fresh target 실패' >&2; fi
```

metric의 `node` label과 `operation_type` 값은 환경에서 실제 존재해야 한다. 빈 결과를 성공으로 보지 않고 label/metric 미지원이면 Event·journal·CRI로 판정한다.

## 증거·미검증

Pod/owner/Node UID·RV, machineID, current containerID, canonical 창, Event time, producer/filter rc, exact endpoint/provider PID, sandbox UID relation, shim PID start/namespace만 보존한다. 전체 argv·inspect·journal, Secret·token·private key는 저장하지 않는다. runtime, OCI, systemd, SSH, CNI, metric은 환경별 미검증이다.
