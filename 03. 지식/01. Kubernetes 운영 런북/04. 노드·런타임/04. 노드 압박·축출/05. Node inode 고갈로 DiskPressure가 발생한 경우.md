# Node inode 고갈로 DiskPressure가 발생한 경우

## 입력값과 빠른 수집

실행 hop은 `승인된 작업 PC/Bastion → Kubernetes API → 승인된 SSH host → kubelet inode signal → 확인된 오류 path → 실제 mount/source → 제한된 inode 소비 경로`다. 같은 Bash를 유지하고 `eval`은 사용하지 않는다.

| 종류 | 값 | 확정 방법 |
|---|---|---|
| 최초 입력 | `NODE` | 승인된 Node exact name |
| API 확인 자동값 | `NODE_UID`, `NODE_RV`, runtime provider, 영향 Pod UID/owner | fresh API |
| 계산 후보값 | path→mount relation, escaped node/mount regex | exact relation/escape 검증 전 미확정 |
| 운영자 선택값 | `START_RFC3339`, `END_RFC3339`, `SSH_HOST`, `ERROR_PATH`, `CONFIRMED_ROOT` | signal·오류·CMDB·owner 근거로 선택 |

`TOOLS_READY=false; MISSING=''; for T in kubectl jq grep sed date ssh; do command -v "$T" >/dev/null 2>&1 || MISSING="$MISSING $T"; done; if [ -z "${BASH_VERSION:-}" ] || [ -n "$MISSING" ]; then printf '중단: Bash/도구 gate 실패%s\n' "$MISSING" >&2; elif kubectl auth whoami >/dev/null 2>&1 && kubectl auth can-i get nodes | grep -Fxq yes && kubectl auth can-i list pods --all-namespaces | grep -Fxq yes; then TOOLS_READY=true; kubectl auth whoami; else printf '%s\n' '중단: 현재 주체/RBAC 실패' >&2; fi`

`unset INPUT_READY AUTO_READY HOST_READY TIME_READY KUBELET_READY STORAGE_READY PATH_READY EVIDENCE_READY MUTATION_READY REPLACEMENT_READY NORMALIZATION_READY DOWNSTREAM_READY NODE_UID NODE_RV POD_UID OWNER_KIND OWNER_NAME OWNER_UID CONTAINER_ID RUNTIME_VERSION RUNTIME_PROVIDER SSH_HOST HOST_IDENTITY KUBELET_UNIT RUNTIME_UNIT ERROR_PATH CONFIRMED_ROOT FS_TARGET FS_SOURCE FS_FSTYPE START_RFC3339 END_RFC3339 START_EPOCH END_EPOCH START_CANON END_CANON ARGV_RE NODE_RE MOUNT_RE; SSH_ARGS=(); NODE='PASTE_NODE_HERE'; INPUT_READY=false; if [ "$TOOLS_READY" = true ] && J="$(kubectl get node "$NODE" -o json)" && [ "$(jq -r '.metadata.name' <<<"$J")" = "$NODE" ]; then INPUT_READY=true; printf 'INPUT_READY=true node=%s\n' "$NODE"; else printf '%s\n' '중단: Node exact-name 조회 실패' >&2; fi`

`unset AUTO_READY HOST_READY TIME_READY KUBELET_READY STORAGE_READY PATH_READY EVIDENCE_READY MUTATION_READY REPLACEMENT_READY NORMALIZATION_READY NODE_UID NODE_RV RUNTIME_VERSION RUNTIME_PROVIDER SSH_HOST HOST_IDENTITY KUBELET_UNIT RUNTIME_UNIT ERROR_PATH CONFIRMED_ROOT FS_TARGET FS_SOURCE FS_FSTYPE START_RFC3339 END_RFC3339 START_EPOCH END_EPOCH START_CANON END_CANON ARGV_RE NODE_RE MOUNT_RE DOWNSTREAM_READY; SSH_ARGS=(); AUTO_READY=false; if [ "$INPUT_READY" = true ] && J="$(kubectl get node "$NODE" -o json)"; then NODE_UID="$(jq -r '.metadata.uid // empty' <<<"$J")"; NODE_RV="$(jq -r '.metadata.resourceVersion // empty' <<<"$J")"; RUNTIME_VERSION="$(jq -r '.status.nodeInfo.containerRuntimeVersion // empty' <<<"$J")"; case "$RUNTIME_VERSION" in containerd://*) RUNTIME_PROVIDER=containerd;; cri-o://*) RUNTIME_PROVIDER=cri-o;; *) RUNTIME_PROVIDER='';; esac; if [ -n "$NODE_UID" ] && [ -n "$NODE_RV" ] && [ -n "$RUNTIME_PROVIDER" ]; then AUTO_READY=true; jq -r '"Node/UID/RV: \(.metadata.name) \(.metadata.uid) \(.metadata.resourceVersion)\nRuntime: \(.status.nodeInfo.containerRuntimeVersion)\n" + ([.status.conditions[] | "\(.type)=\(.status) reason=\(.reason) transition=\(.lastTransitionTime) message=\(.message)"]|join("\n"))' <<<"$J"; else printf '%s\n' '중단: identity/runtime provider 등록 실패' >&2; fi; else printf '%s\n' '중단: fresh Node API 실패' >&2; fi`

`fresh_node(){ local j; unset CURRENT_NODE_RV; j="$(kubectl get node "$NODE" -o json)" || return 1; [ "$(jq -r '.metadata.uid' <<<"$j")" = "$NODE_UID" ] || return 1; CURRENT_NODE_RV="$(jq -r '.metadata.resourceVersion // empty' <<<"$j")"; [ -n "$CURRENT_NODE_RV" ]; }`

재등록 전에 UID/RV/owner/containerID/node/host/path/mount relation/time/argv-regex와 downstream `READY`를 unset/false로 만든다. 실패 뒤 stale 경로를 `du` 대상으로 사용하지 않는다.

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | `nodefs.inodesFree`·`imagefs.inodesFree` 또는 같은 mount의 inode 고갈과 no-space 오류가 같은 창에 있음 |
| 적용 | signal family, 오류 path, 실제 mount/source, bytes와 inode를 분리 가능 |
| 제외 | bytes-only nodefs/imagefs, registry 오류, PIDPressure, PVC/CSI는 해당 담당으로 이관 |

검색 문자열: `DiskPressure`, `nodefs.inodesFree`, `imagefs.inodesFree`, `inodesFree`, `no space left on device`, `EvictionThresholdMet`, `pid.available`.

## 최초 확인·영향

`fresh_node && kubectl get pods -A --field-selector "spec.nodeName=$NODE" -o json | jq -r '.items[] | "\n=== \(.metadata.namespace)/\(.metadata.name) ===\nUID: \(.metadata.uid)\nOwner: \(.metadata.ownerReferences[0].kind // "-")/\(.metadata.ownerReferences[0].name // "-") UID=\(.metadata.ownerReferences[0].uid // "-")\nPhase: \(.status.phase) Ready=\([.status.containerStatuses[]?.ready]|join(",")) Restarts=\([.status.containerStatuses[]?.restartCount]|join(","))"'`

빈 목록을 정상으로 보지 않는다. Pod 이름 대신 UID/owner UID로 영향과 replacement를 분리한다.

## 시간·host gate

`START_RFC3339='PASTE_START_RFC3339_HERE'; END_RFC3339='PASTE_END_RFC3339_HERE'; unset TIME_READY KUBELET_READY STORAGE_READY PATH_READY EVIDENCE_READY START_EPOCH END_EPOCH START_CANON END_CANON DOWNSTREAM_READY; TIME_READY=false; if [[ "$START_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && [[ "$END_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && START_EPOCH="$(date -u -d "$START_RFC3339" +%s 2>/dev/null)" && END_EPOCH="$(date -u -d "$END_RFC3339" +%s 2>/dev/null)"; then START_CANON="$(date -u -d "@$START_EPOCH" '+%Y-%m-%dT%H:%M:%SZ')"; END_CANON="$(date -u -d "@$END_EPOCH" '+%Y-%m-%dT%H:%M:%SZ')"; if [ "$START_RFC3339" = "$START_CANON" ] && [ "$END_RFC3339" = "$END_CANON" ] && [ "$START_EPOCH" -lt "$END_EPOCH" ]; then TIME_READY=true; printf 'TIME_READY=true %s < %s\n' "$START_CANON" "$END_CANON"; else printf '%s\n' '중단: canonical round-trip/start<end 실패' >&2; fi; else printf '%s\n' '중단: strict UTC RFC3339 regex/parse 실패' >&2; fi`

`SSH_HOST='PASTE_APPROVED_SSH_HOST_HERE'; unset HOST_READY KUBELET_READY STORAGE_READY PATH_READY EVIDENCE_READY HOST_IDENTITY DOWNSTREAM_READY; SSH_ARGS=(); HOST_READY=false; if fresh_node && [[ "$SSH_HOST" =~ ^[A-Za-z0-9_.@:-]+$ ]] && [[ "$SSH_HOST" != -* ]]; then SSH_ARGS=(-o BatchMode=yes -o StrictHostKeyChecking=yes -o ClearAllForwardings=yes -o ConnectTimeout=10 "$SSH_HOST"); REMOTE_ID="$(ssh "${SSH_ARGS[@]}" -- 'printf "hostname=%s machine_id=%s boot_id=%s\n" "$(hostname -f 2>/dev/null || hostname)" "$(cat /etc/machine-id 2>/dev/null)" "$(cat /proc/sys/kernel/random/boot_id 2>/dev/null)"')"; SSH_RC=$?; printf '%s\n' "$REMOTE_ID"; read -r -p 'CMDB에서 이 host가 대상 Node임을 확인했으면 Node 이름 입력: ' HOST_CONFIRM; if [ "$SSH_RC" -eq 0 ] && [ "$HOST_CONFIRM" = "$NODE" ] && [[ "$REMOTE_ID" =~ ^hostname=[^[:space:]]+[[:space:]]machine_id=[0-9a-fA-F-]+[[:space:]]boot_id=[0-9a-fA-F-]+$ ]]; then HOST_IDENTITY="$REMOTE_ID"; HOST_READY=true; printf 'HOST_READY=true host=%s node=%s uid=%s rv=%s\n' "$SSH_HOST" "$NODE" "$NODE_UID" "$CURRENT_NODE_RV"; else printf '%s\n' '중단: 원격 host identity/CMDB 승인 실패' >&2; fi; else printf '%s\n' '중단: fresh Node/SSH host 형식 실패' >&2; fi`

`host_guard(){ local now; HOST_READY=false; fresh_node || return 1; now="$(ssh "${SSH_ARGS[@]}" -- 'printf "hostname=%s machine_id=%s boot_id=%s\n" "$(hostname -f 2>/dev/null || hostname)" "$(cat /etc/machine-id 2>/dev/null)" "$(cat /proc/sys/kernel/random/boot_id 2>/dev/null)"')" || return 1; [ "$now" = "$HOST_IDENTITY" ] || return 1; HOST_READY=true; }`

모든 node 명령은 작업 PC에서 `host_guard`를 직접 호출하고 검증된 절대경로를 SSH positional argument로 전달한다. 작업 PC 변수가 원격 shell에 자동 전달된다고 가정하지 않는다.

## 관측별 진단 경로

| 지금 보이는 것 | 먼저 볼 것 | 그다음 어디를 볼지 | 상세 절 |
|---|---|---|---|
| `nodefs.inodesFree` | nodefs 오류 path의 mount | 제한된 inode owner | [[#signal과 mount의 직접 관계]] |
| `imagefs.inodesFree` | runtime image path의 mount | runtime/imagefs owner | [[#signal과 mount의 직접 관계]] |
| bytes도 100% | bytes·inode 각각 | nodefs/imagefs bytes 런북 | [[#signal과 mount의 직접 관계]] |
| `pid.available` | PIDPressure condition | PID 런북 | [[#kubelet·runtime 증거]] |

## signal과 mount의 직접 관계

`ERROR_PATH='PASTE_CONFIRMED_ERROR_PATH_HERE'; unset STORAGE_READY PATH_READY FS_TARGET FS_SOURCE FS_FSTYPE DOWNSTREAM_READY; STORAGE_READY=false; if host_guard && [[ "$ERROR_PATH" =~ ^/[-A-Za-z0-9._/]+$ ]]; then MAP="$(ssh "${SSH_ARGS[@]}" -- findmnt -T "$ERROR_PATH" -rn -o TARGET,SOURCE,FSTYPE)"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ] && [ "$(wc -l <<<"$MAP")" -eq 1 ]; then read -r FS_TARGET FS_SOURCE FS_FSTYPE <<<"$MAP"; fi; if [ -n "$FS_TARGET" ] && [ -n "$FS_SOURCE" ] && [ -n "$FS_FSTYPE" ]; then STORAGE_READY=true; printf 'STORAGE_READY=true path=%s target=%s source=%s fstype=%s\n' "$ERROR_PATH" "$FS_TARGET" "$FS_SOURCE" "$FS_FSTYPE"; else printf '중단: findmnt producer=%s 또는 exact-one relation 실패\n' "$PRODUCER_RC" >&2; fi; else printf '%s\n' '중단: fresh host/안전한 error path 실패' >&2; fi`

`host_guard || exit 1; CURRENT_MAP="$(ssh "${SSH_ARGS[@]}" -- findmnt -T "$ERROR_PATH" -rn -o TARGET,SOURCE,FSTYPE)"; [ "$STORAGE_READY" = true ] && [ "$CURRENT_MAP" = "$MAP" ] && ssh "${SSH_ARGS[@]}" -- df -hT -- "$ERROR_PATH"`

`host_guard || exit 1; CURRENT_MAP="$(ssh "${SSH_ARGS[@]}" -- findmnt -T "$ERROR_PATH" -rn -o TARGET,SOURCE,FSTYPE)"; [ "$STORAGE_READY" = true ] && [ "$CURRENT_MAP" = "$MAP" ] && ssh "${SSH_ARGS[@]}" -- df -iT -- "$ERROR_PATH"`

`IFree=0` 또는 실제 threshold 이하와 inode signal·no-space 시각이 맞아야 inode 원인이다. bytes 고갈도 있으면 별도 원인 축으로 기록한다. 현재 snapshot만으로 과거 고갈을 확정하지 않는다.

`CONFIRMED_ROOT='PASTE_CONFIRMED_INODE_ROOT_HERE'; unset PATH_READY DOWNSTREAM_READY; PATH_READY=false; if host_guard && [ "$STORAGE_READY" = true ] && [[ "$CONFIRMED_ROOT" =~ ^/[-A-Za-z0-9._/]+$ ]] && [ "$(ssh "${SSH_ARGS[@]}" -- findmnt -T "$CONFIRMED_ROOT" -rn -o TARGET)" = "$FS_TARGET" ]; then case "$CONFIRMED_ROOT" in /|/var|/var/lib) printf '%s\n' '중단: root 범위가 너무 넓음' >&2;; *) PATH_READY=true; printf 'PATH_READY=true root=%s mount=%s\n' "$CONFIRMED_ROOT" "$FS_TARGET";; esac; else printf '%s\n' '중단: confirmed root/remote mount exact relation 실패' >&2; fi`

`host_guard || exit 1; ROOT_MAP="$(ssh "${SSH_ARGS[@]}" -- findmnt -T "$CONFIRMED_ROOT" -rn -o TARGET,SOURCE,FSTYPE)"; [ "$PATH_READY" = true ] && [ "$ROOT_MAP" = "$MAP" ] && ssh "${SSH_ARGS[@]}" -- sudo du -x --inodes -d 1 -- "$CONFIRMED_ROOT"`

한 단계 집계만 수행한다. 결과가 큰 하위 경로는 owner 승인을 받고 동일한 fresh mount gate로 한 단계씩 좁힌다. 광범위 `find`, 파일 목록/내용 수집, symlink 추적, 삭제는 금지한다.

## kubelet·runtime 증거

`unset KUBELET_READY KUBELET_UNIT EVIDENCE_READY DOWNSTREAM_READY; KUBELET_READY=false; host_guard || exit 1; UNIT_OUTPUT="$(ssh "${SSH_ARGS[@]}" -- systemctl list-units --type=service --all --no-legend)"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ]; then mapfile -t KU < <(grep -E '(^|[[:space:]])kubelet\.service([[:space:]]|$)' <<<"$UNIT_OUTPUT" | sed -E 's/^[[:space:]]*([^[:space:]]+).*/\1/'); FILTER_RC=$?; else FILTER_RC=99; fi; if [ "$PRODUCER_RC" -eq 0 ] && [ "$FILTER_RC" -eq 0 ] && [ "${#KU[@]}" -eq 1 ] && ssh "${SSH_ARGS[@]}" -- sudo systemctl is-active --quiet "${KU[0]}"; then KUBELET_UNIT="${KU[0]}"; KUBELET_READY=true; printf 'KUBELET_READY=true unit=%s\n' "$KUBELET_UNIT"; else printf '중단: kubelet producer=%s filter=%s candidates=%s\n' "$PRODUCER_RC" "$FILTER_RC" "${#KU[@]}" >&2; fi`

`host_guard || exit 1; [ "$KUBELET_READY" = true ] && [ "$TIME_READY" = true ] || exit 1; TMP="$(mktemp)" || exit 1; ssh "${SSH_ARGS[@]}" -- sudo journalctl -u "$KUBELET_UNIT" --since "$START_CANON" --until "$END_CANON" --no-pager >"$TMP"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ]; then grep -E -i -C 3 'nodefs\.inodesFree|imagefs\.inodesFree|nodefs\.available|imagefs\.available|pid\.available|no space left on device|DiskPressure|PIDPressure|evict|write|failed' "$TMP"; FILTER_RC=$?; else FILTER_RC=99; fi; rm -f "$TMP"; printf 'producer_rc=%s filter_rc=%s\n' "$PRODUCER_RC" "$FILTER_RC"; [ "$PRODUCER_RC" -eq 0 ] && { [ "$FILTER_RC" -eq 0 ] || [ "$FILTER_RC" -eq 1 ]; }`

runtime journal은 imagefs 또는 runtime write가 직접 보이고, [[07. runtime 상태와 Pod container ID가 일치하지 않는 경우]]의 active endpoint/socket/provider/unit gate를 새로 통과한 경우만 조회한다. unit 이름을 runtimeVersion에서 단독 추측하지 않는다.

producer/filter RC를 분리하고 `start <= signal/error <= end`를 확인한다. `filter_rc=1`은 오류 부재가 아니다. `pid.available`은 inode가 아니라 PIDPressure다.

## 보안·변경·정상화

정확한 RBAC/audit은 사용 verb/subresource와 최소 필드로 확인한다. 파일명·내용, credential, token, key, 개인정보와 전체 journal을 저장하지 않는다.

이 문서는 rm/find-delete/prune/log truncate, threshold 변경, Pod 삭제, restart/reboot/cordon/drain 명령을 제공하지 않는다. 별도 변경은 fresh evidence, 파일 owner·재생성/열림/runtime metadata 여부, 승인/window, PDB/quorum, idempotence, rollback 또는 비가역 중단 계획을 요구한다.

이 문서 안에서는 `MUTATION_READY=false`를 유지한다. 별도 변경 절차가 실제 명령을 제공할 때 그 명령 자체가 fresh Node UID/RV·host identity·path→mount/source/fstype·file owner/open/runtime metadata·evidence window·승인 ID·PDB/quorum·멱등성·rollback 또는 비가역 중단 조건을 다시 검사해야 한다.

실제 변경은 승인된 filesystem/runtime/workload owner 절차만 사용한다. 삭제는 일반 rollback이 아니며, replacement는 새 UID/RV·owner UID·node·container ID를 재등록하고 Ready 전 downstream을 금지한다.

`NORMALIZATION_READY=false; CURRENT_MAP=''; if fresh_node && host_guard && J="$(kubectl get node "$NODE" -o json)" && CURRENT_MAP="$(ssh "${SSH_ARGS[@]}" -- findmnt -T "$ERROR_PATH" -rn -o TARGET,SOURCE,FSTYPE)" && [ "$(jq -r '[.status.conditions[]|select(.type=="DiskPressure")|.status][0]' <<<"$J")" = False ] && [ "$STORAGE_READY" = true ] && [ "$CURRENT_MAP" = "$MAP" ]; then NORMALIZATION_READY=true; printf 'NORMALIZATION_READY=true node_uid=%s rv=%s mount=%s\n' "$NODE_UID" "$CURRENT_NODE_RV" "$FS_TARGET"; else printf '%s\n' '중단: condition/remote identity/mount 정상화 gate 실패' >&2; fi`

정상화는 inode와 bytes 안정, 새 inode/write warning 부재, DiskPressure/taint 해소, PIDPressure 별도 정상, replacement Ready/restart 안정, owner desired=Ready, EndpointSlice `ready=true`, 실제 요청까지 확인한다.

## Prometheus 빠른 관측

`fresh_node || exit 1; host_guard || exit 1; CURRENT_MAP="$(ssh "${SSH_ARGS[@]}" -- findmnt -T "$ERROR_PATH" -rn -o TARGET,SOURCE,FSTYPE)"; [ "$STORAGE_READY" = true ] && [ "$CURRENT_MAP" = "$MAP" ] || exit 1; NODE_RE="$(printf '%s' "$NODE" | sed 's/[][\\.^$*+?(){}|]/\\&/g')"; MOUNT_RE="$(printf '%s' "$FS_TARGET" | sed 's/[][\\.^$*+?(){}|]/\\&/g')"; printf '100*(1-node_filesystem_files_free{nodename=~"^%s$",mountpoint=~"^%s$"}/node_filesystem_files{nodename=~"^%s$",mountpoint=~"^%s$"})\n' "$NODE_RE" "$MOUNT_RE" "$NODE_RE" "$MOUNT_RE"`

완성 쿼리만 UI에 붙이고 metric/label/mount relabel 미확인과 빈 결과를 정상으로 보지 않는다.

## 증거·환경 미검증

Node UID/RV, Pod UID/owner, remote identity, canonical 시간, signal family, path→mount/source relation, bytes/inode, 제한 집계, producer/filter RC, 승인·replacement·정상화를 보존한다. filesystem inode 동작, threshold, runtime path/unit, SSH/NTP/audit/metrics는 환경별 미검증이다. 실제 인프라에는 실행하지 않았다.
