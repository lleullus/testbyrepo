# 여러 Node가 동시에 NotReady·Unknown이 된 경우

## 입력값과 빠른 수집

Node 개수만으로 공통 장애를 선언하지 않는다. `작업 PC → API → 각 Node/Lease`, `승인된 monitoring/audit → 동시성`, `선택한 대표 Node의 승인된 SSH host → kubelet journal`을 분리해 같은 UTC 창에서 대조한다.

| 구분 | 값 | 출처·gate |
|---|---|---|
| 최초 입력 | `NODE_CSV` | incident가 승인한 exact Node 이름의 comma 목록, 2개 이상 |
| API 확인 자동값 | 각 Node UID/RV/machineID/condition/fault-domain, 각 Lease UID/RV/renew | 모든 API 성공과 목록 exact match 뒤 일괄 등록 |
| 계산 후보값 | `NODE_RE`, `NODE_LABEL_RE`, `WINDOW_SECONDS` | 각 이름 regex escape와 UTC 시간 검증 뒤 생성 |
| 운영자 선택값 | 관측 창·공통 evidence, 대표 `TARGET_NODE`와 SSH host/unit | 동시성 증거와 승인된 host mapping에서 선택 |

### 1. 도구·목록 입력

재등록 전에 모든 Node/Lease/Pod owner identity·resourceVersion, containerID, node/host/fault-domain 관계, 시간·argv/regex와 downstream `READY`를 무효화한다. 한 API라도 실패하면 부분 목록을 사용하지 않는다.

### 2. Node·Lease 일괄 자동 등록

재수집 명령 직전에는 이전 일괄 JSON과 downstream 상태를 먼저 폐기한다.

`unset NODES_JSON LEASES_JSON; AUTO_READY=false; API_READY=false; QUERY_READY=false; COMMON_READY=false; IMPACT_READY=false; HOST_INPUT_READY=false; HOST_READY=false; NORMAL_READY=false`

`unset AUTO_READY API_READY TIME_READY QUERY_READY COMMON_READY IMPACT_READY HOST_INPUT_READY HOST_READY NODE_RE NODE_LABEL_RE WINDOW_SECONDS START_RFC3339 END_RFC3339 COMMON_EVIDENCE_REF AFFECTED_COUNT TARGET_NODE TARGET_NODE_UID SSH_HOST REMOTE_HOSTNAME KUBELET_UNIT OWNER_UID CONTAINER_ID CRI_ENDPOINT RUNTIME_UNIT ARGV_RE; AUTO_READY=false; if [ "$INPUT_READY" != true ]; then printf '%s\n' '중단: INPUT_READY가 아니다.' >&2; elif NODES_JSON="$(kubectl get nodes "${NODES[@]}" -o json)"; then GOT_COUNT="$(jq '.items|length' <<<"$NODES_JSON")"; EXPECTED_JSON="$(printf '%s\n' "${NODES[@]}" | jq -Rsc 'split("\n")|map(select(length>0))|sort')"; GOT_JSON="$(jq -c '[.items[].metadata.name]|sort' <<<"$NODES_JSON")"; if [ "$GOT_COUNT" -ne "${#NODES[@]}" ] || [ "$GOT_JSON" != "$EXPECTED_JSON" ]; then printf '%s\n' '중단: Node API 목록 exact match 실패' >&2; else LEASES_JSON='[]'; LEASE_OK=true; for N in "${NODES[@]}"; do if L="$(kubectl get lease "$N" -n kube-node-lease -o json)" && [ "$(jq -r '.spec.holderIdentity' <<<"$L")" = "$N" ]; then LEASES_JSON="$(jq -c --argjson item "$L" '. + [$item]' <<<"$LEASES_JSON")" || LEASE_OK=false; else LEASE_OK=false; break; fi; done; if [ "$LEASE_OK" = true ] && [ "$(jq 'length' <<<"$LEASES_JSON")" -eq "${#NODES[@]}" ]; then NODE_RE=''; NODE_LABEL_RE=''; for N in "${NODES[@]}"; do E="$(printf '%s' "$N" | sed 's/[][\\.^$*+?(){}|]/\\&/g')"; [ -z "$NODE_RE" ] && NODE_RE="$E" || NODE_RE="$NODE_RE|$E"; done; NODE_RE="^($NODE_RE)$"; NODE_LABEL_RE="$(printf '%s' "$NODE_RE" | sed 's/\\/\\\\/g;s/"/\\"/g')"; AUTO_READY=true; export NODE_RE NODE_LABEL_RE; jq -r '.items[] | "=== Node: \(.metadata.name) ===\nUID/RV: \(.metadata.uid) / \(.metadata.resourceVersion)\nReady: \([.status.conditions[]? | select(.type=="Ready") | "\(.status) reason=\(.reason // "-") heartbeat=\(.lastHeartbeatTime // "-") transition=\(.lastTransitionTime // "-")"] | join(""))\nPressure: \([.status.conditions[]? | select(.type=="MemoryPressure" or .type=="DiskPressure" or .type=="PIDPressure" or .type=="NetworkUnavailable") | "\(.type)=\(.status)"] | join(","))\nZone/Region: \(.metadata.labels["topology.kubernetes.io/zone"] // "-") / \(.metadata.labels["topology.kubernetes.io/region"] // "-")\nRuntime: \(.status.nodeInfo.containerRuntimeVersion // "-")"' <<<"$NODES_JSON"; jq -r '.[] | "Lease: \(.metadata.name) uid/rv=\(.metadata.uid)/\(.metadata.resourceVersion) renew=\(.spec.renewTime) holder=\(.spec.holderIdentity)"' <<<"$LEASES_JSON"; printf 'AUTO_READY=true NODE_RE=%s\n' "$NODE_RE"; else LEASES_JSON='[]'; printf '%s\n' '중단: Lease producer/identity 실패; 전체 자동값 무효' >&2; fi; fi; else printf '%s\n' '중단: Nodes API producer 실패' >&2; fi`

fault-domain label 부재는 서로 다른 domain이라는 증거가 아니다. runtime 문자열은 unit/socket을 확정하지 않는다.

### 3. 관측 창·동시성 gate

모든 후속 조회는 다음 guard로 목록 exact match, 각 Node UID·machineID와 Lease UID·holder를 다시 검증한다. 어느 producer든 실패하면 과거 공통 증거와 host·정상화 `READY`를 폐기한다.

`invalidate_group(){ GROUP_CURRENT_READY=false; QUERY_READY=false; COMMON_READY=false; IMPACT_READY=false; HOST_INPUT_READY=false; HOST_READY=false; NORMAL_READY=false; }; current_group_guard(){ GROUP_CURRENT_READY=false; local current_nodes current_leases l expected_nodes actual_nodes expected_leases actual_leases; [ "$AUTO_READY" = true ] || { invalidate_group; return 1; }; current_nodes="$(kubectl get nodes "${NODES[@]}" -o json)" || { invalidate_group; return 1; }; [ "$(jq '.items|length' <<<"$current_nodes")" -eq "${#NODES[@]}" ] || { invalidate_group; return 1; }; current_leases='[]'; for N in "${NODES[@]}"; do l="$(kubectl get lease "$N" -n kube-node-lease -o json)" || { invalidate_group; return 1; }; current_leases="$(jq -c --argjson item "$l" '. + [$item]' <<<"$current_leases")" || { invalidate_group; return 1; }; done; expected_nodes="$(jq -Sc '[.items[]|{name:.metadata.name,uid:.metadata.uid,machineID:.status.nodeInfo.machineID}]|sort_by(.name)' <<<"$NODES_JSON")" && actual_nodes="$(jq -Sc '[.items[]|{name:.metadata.name,uid:.metadata.uid,machineID:.status.nodeInfo.machineID}]|sort_by(.name)' <<<"$current_nodes")" && expected_leases="$(jq -Sc '[.[]|{name:.metadata.name,uid:.metadata.uid,holder:.spec.holderIdentity}]|sort_by(.name)' <<<"$LEASES_JSON")" && actual_leases="$(jq -Sc '[.[]|{name:.metadata.name,uid:.metadata.uid,holder:.spec.holderIdentity}]|sort_by(.name)' <<<"$current_leases")" || { invalidate_group; return 1; }; if [ "$expected_nodes" != "$actual_nodes" ] || [ "$expected_leases" != "$actual_leases" ]; then invalidate_group; printf '%s\n' '중단: current Node/Lease identity 목록 변경' >&2; return 1; fi; NODES_JSON="$current_nodes"; LEASES_JSON="$current_leases"; GROUP_CURRENT_READY=true; }`

`START_RFC3339='PASTE_START_RFC3339_HERE'; END_RFC3339='PASTE_END_RFC3339_HERE'; TIME_READY=false; QUERY_READY=false; COMMON_READY=false; IMPACT_READY=false; HOST_INPUT_READY=false; HOST_READY=false; NORMAL_READY=false; unset COMMON_EVIDENCE_REF AFFECTED_COUNT TARGET_NODE TARGET_NODE_UID TARGET_MACHINE_ID SSH_HOST REMOTE_HOSTNAME KUBELET_UNIT FUNCTION_EVIDENCE_REF; if [ "$AUTO_READY" != true ]; then printf '%s\n' '중단: AUTO_READY가 아니다.' >&2; elif [[ ! "$START_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] || [[ ! "$END_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]]; then printf '%s\n' '중단: UTC RFC3339 형식 오류' >&2; elif S="$(date -u -d "$START_RFC3339" +%s)" && E="$(date -u -d "$END_RFC3339" +%s)" && [ "$(date -u -d "@$S" +%Y-%m-%dT%H:%M:%SZ)" = "$START_RFC3339" ] && [ "$(date -u -d "@$E" +%Y-%m-%dT%H:%M:%SZ)" = "$END_RFC3339" ] && [ "$S" -lt "$E" ]; then WINDOW_SECONDS="$((E-S))"; TIME_READY=true; export START_RFC3339 END_RFC3339 WINDOW_SECONDS; printf 'TIME_READY=true window=%ss\n' "$WINDOW_SECONDS"; else printf '%s\n' '중단: date round-trip 또는 start<end 실패' >&2; fi`

`QUERY_READY=false; if ! current_group_guard || [ "$TIME_READY" != true ]; then printf '%s\n' '중단: current identity/time gate가 필요하다.' >&2; else QUERY_READY=true; printf 'sum by(status)(max_over_time(kube_node_status_condition{node=~"%s",condition="Ready",status=~"false|unknown"}[%ss]))\n' "$NODE_LABEL_RE" "$WINDOW_SECONDS"; printf 'max by(lease)(time()-kube_lease_renew_time{namespace="kube-node-lease",lease=~"%s"})\n' "$NODE_LABEL_RE"; fi`

`COMMON_EVIDENCE_REF='PASTE_APPROVED_COMMON_EVIDENCE_HERE'; AFFECTED_COUNT='PASTE_INTEGER_AFFECTED_COUNT_HERE'; COMMON_READY=false; if ! current_group_guard || [ "$TIME_READY" != true ] || [ -z "$COMMON_EVIDENCE_REF" ] || [[ ! "$AFFECTED_COUNT" =~ ^[0-9]+$ ]] || [ "$AFFECTED_COUNT" -lt 2 ] || [ "$AFFECTED_COUNT" -gt "${#NODES[@]}" ]; then printf '%s\n' '중단: 같은 창의 2개 이상 current UID evidence가 필요하다.' >&2; else COMMON_READY=true; export COMMON_EVIDENCE_REF AFFECTED_COUNT; printf 'COMMON_READY=true affected=%s evidence=%s\n' "$AFFECTED_COUNT" "$COMMON_EVIDENCE_REF"; fi`

외부 query 빈 결과는 정상/0이 아니라 metric 미수집·label/시간 범위 오류 후보이고 HTTP/API/query 실패와 분리한다. metric은 UID를 보장하지 않으므로 승인 evidence가 각 current Node UID와 전환/heartbeat/Lease 시간을 연결해야 한다.

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | current UID 2개 이상이 같은 승인 창에서 False/Unknown 또는 Lease 정체 |
| 적용 | API/control plane, 공통 network/fault domain, credential/time, rollout, 개별 Node를 분리 |
| 제외 | 공통성이 사라지고 한 Node만 남으면 [[03. Node Ready 상태가 반복 전환되는 경우]] |

검색 문자열: `context deadline exceeded`, `connection refused`, `x509`, `unauthorized`, `clock skew`, `PLEG`, `runtime restart`, `NotReady`, `Unknown`.

## 관측별 진단 경로

| 공통 관측 | 먼저 가를 것 | 다음 owner |
|---|---|---|
| readyz와 관리 API도 실패 | client RBAC/네트워크 vs API server | control plane |
| 특정 zone/segment Lease 동시 정체 | 실제 platform topology·route | network/platform |
| x509/unauthorized/clock 공통 | credential rotation·host time | security/time |
| rollout/change와 전환 일치 | DaemonSet/config revision·digest | lifecycle/change owner |
| pressure type·시각이 제각각 | 개별 Node/workload | 단일 Node 런북 |

### API와 workload blast radius

`API_READY=false; if ! current_group_guard || [ "$COMMON_READY" != true ]; then printf '%s\n' '중단: 최신 current group과 COMMON_READY가 필요하다.' >&2; elif READYZ="$(kubectl get --raw='/readyz?verbose')"; then API_READY=true; printf 'API_READY=true\n%s\n' "$READYZ"; else printf '%s\n' 'producer 오류: readyz 실패; Node 손실과 분리' >&2; fi`

`IMPACT_READY=false; if ! current_group_guard || [ "$COMMON_READY" != true ]; then printf '%s\n' '중단: current common identity gate가 필요하다.' >&2; else ALL_PODS='[]'; OK=true; for N in "${NODES[@]}"; do if P="$(kubectl get pods -A --field-selector "spec.nodeName=$N" -o json)"; then ALL_PODS="$(jq -c --arg node "$N" --argjson p "$P" '. + [$p.items[] | {node:$node,namespace:.metadata.namespace,name:.metadata.name,uid:.metadata.uid,rv:.metadata.resourceVersion,owners:.metadata.ownerReferences,phase:.status.phase,containers:[.status.containerStatuses[]?|{name,ready,restartCount,containerID}]}]' <<<"$ALL_PODS")" || OK=false; else OK=false; break; fi; done; if [ "$OK" = true ]; then IMPACT_READY=true; jq -r '.[] | "=== \(.node) -> \(.namespace)/\(.name) ===\nUID/RV: \(.uid)/\(.rv)\nOwner: \([.owners[]? | "\(.kind)/\(.name) uid=\(.uid)"]|join(","))\nPhase: \(.phase)\nContainers: \([.containers[] | "\(.name) ready=\(.ready) restart=\(.restartCount) id=\((.containerID // "-")|split("://")|last)"]|join("; "))"' <<<"$ALL_PODS"; else ALL_PODS='[]'; printf '%s\n' 'producer 오류: 어느 Node의 Pod API든 실패; partial blast radius 금지' >&2; fi; fi`

Pod/owner/containerID/node 관계는 impact 확인용이다. Secret·env·전체 spec은 수집하지 않는다. PDB, StatefulSet/quorum, Job 외부 write는 변경 전 별도 owner 확인이 필요하다.

## 대표 Node host 상관

각 Node를 한 SSH loop로 돌리지 않는다. API·fault-domain 증거에서 대표 한 Node를 고르고 host identity를 다시 검증한다.

`TARGET_NODE='PASTE_TARGET_NODE_FROM_LIST_HERE'; SSH_HOST='PASTE_APPROVED_SSH_HOST_HERE'; REMOTE_HOSTNAME='PASTE_REMOTE_HOSTNAME_HERE'; KUBELET_UNIT='PASTE_CONFIRMED_KUBELET_UNIT_HERE'; HOST_INPUT_READY=false; unset TARGET_NODE_UID TARGET_MACHINE_ID; if ! current_group_guard || [ "$COMMON_READY" != true ] || ! printf '%s\n' "${NODES[@]}" | grep -Fxq "$TARGET_NODE" || [[ "$SSH_HOST" == -* ]] || [[ ! "$KUBELET_UNIT" =~ ^[A-Za-z0-9_.@-]+\.service$ ]]; then printf '%s\n' '중단: 대표 Node/list/host/unit gate 실패' >&2; elif TARGET_NODE_UID="$(jq -er --arg n "$TARGET_NODE" '.items[]|select(.metadata.name==$n)|.metadata.uid' <<<"$NODES_JSON")" && TARGET_MACHINE_ID="$(jq -er --arg n "$TARGET_NODE" '.items[]|select(.metadata.name==$n)|.status.nodeInfo.machineID' <<<"$NODES_JSON")"; then HOST_INPUT_READY=true; export TARGET_NODE TARGET_NODE_UID TARGET_MACHINE_ID SSH_HOST REMOTE_HOSTNAME KUBELET_UNIT; printf 'HOST_INPUT_READY=true node=%s uid=%s machineID=%s host=%s\n' "$TARGET_NODE" "$TARGET_NODE_UID" "$TARGET_MACHINE_ID" "$REMOTE_HOSTNAME"; else printf '%s\n' '중단: 대표 Node current identity 등록 실패' >&2; fi`

```bash
if ! current_group_guard || [ "$COMMON_READY" != true ] || [ "$TIME_READY" != true ] || [ "$HOST_INPUT_READY" != true ]; then printf '%s\n' '중단: current common/time/host gate가 필요하다.' >&2; else HOST_READY=false; if ssh -- "$SSH_HOST" bash -s -- "$REMOTE_HOSTNAME" "$TARGET_MACHINE_ID" "$KUBELET_UNIT" "$START_RFC3339" "$END_RFC3339" <<'REMOTE'
set -o nounset
expected=$1
expected_machine=$2
unit=$3
start=$4
end=$5
actual=$(hostname -f 2>/dev/null || hostname)
[ "$actual" = "$expected" ] || { printf '중단: host identity 불일치 expected=%s actual=%s\n' "$expected" "$actual" >&2; exit 60; }
[ -f /etc/machine-id ] && [ -r /etc/machine-id ] || { printf '%s\n' '중단: regular readable /etc/machine-id가 없음' >&2; exit 61; }
IFS= read -r actual_machine </etc/machine-id
[ "$actual_machine" = "$expected_machine" ] || { printf '중단: machine-id 불일치 expected=%s actual=%s\n' "$expected_machine" "$actual_machine" >&2; exit 62; }
[ "$(ps -p 1 -o comm=)" = systemd ] || exit 63
systemctl show "$unit" -p Id -p LoadState -p ActiveState -p SubState -p ExecMainStartTimestamp -p NRestarts --no-pager || exit 64
tmp=$(mktemp) || exit 65
trap 'rm -f "$tmp"' EXIT
if ! journalctl -u "$unit" --since "$start" --until "$end" --no-pager >"$tmp"; then printf '%s\n' 'producer 오류: journalctl 실패' >&2; exit 66; fi
if grep -E -i -C 3 'lease|node status|timeout|context deadline exceeded|connection refused|x509|unauthorized|forbidden|clock|PLEG|runtime|pressure|oom|read-only|I/O error|stopped|started' "$tmp"; then exit 0; else rc=$?; [ "$rc" -eq 1 ] && { printf '%s\n' 'filter no-match: 공통 오류 부재로 해석 금지' >&2; exit 67; }; printf 'filter 오류: grep rc=%s\n' "$rc" >&2; exit 68; fi
REMOTE
then HOST_READY=true; printf 'HOST_READY=true node=%s uid=%s host=%s\n' "$TARGET_NODE" "$TARGET_NODE_UID" "$REMOTE_HOSTNAME"; else printf '%s\n' '중단: representative host pipeline 실패' >&2; fi; fi
```

SSH 인수는 positional로 전달하고 `eval`·host key 검증 우회를 금지한다. 대표 Node 결과를 다른 Node 전체에 일반화하지 않는다. CRI가 필요하면 각 host에서 active kubelet 설정의 단일 endpoint, runtime socket provider exact match와 current containerID를 재검증한 전용 절차를 사용한다.

## 변경·replacement·정상화

이 런북은 일괄 restart/reboot/cordon/drain, credential/network 변경 명령을 제공하지 않는다. 별도 mutation은 fresh 영향 Node/Lease UID·RV 목록, 공통 config spec/revision/digest·change record, workload owner/selector, PDB·quorum·대체 capacity, evidence·승인·window, 한 번에 한 Node, 멱등성, stop/rollback을 매 Node 직전에 직접 gate한다. API 불안정·두 번째 Node 재현·quorum 감소 시 중단한다.

replacement마다 새 Node UID/RV/providerID와 Lease, Pod owner/containerID/node 관계를 재등록한다. 정상화는 `API 안정 → 공통 network/security/time/change 원인 해소 → 모든 intended Node의 Lease 연속 갱신·Ready=True → 새 restart/pressure 없음 → workload desired/Ready·Endpoint → 실제 요청/error/queue/Job 결과`를 원래 창 이상 관찰한다. 일부 Ready 또는 rollback 명령 성공만으로 닫지 않는다.

`FUNCTION_EVIDENCE_REF='PASTE_APPROVED_FUNCTION_EVIDENCE_HERE'; NORMAL_READY=false; if ! current_group_guard || [ "$TIME_READY" != true ] || [ "$HOST_INPUT_READY" != true ] || [ -z "$FUNCTION_EVIDENCE_REF" ]; then printf '%s\n' '중단: current group/time/host와 실제 기능 evidence가 필요하다.' >&2; else REMOTE_HOST_NOW="$(ssh -- "$SSH_HOST" hostname -f 2>/dev/null || ssh -- "$SSH_HOST" hostname)" && REMOTE_MACHINE_NOW="$(ssh -- "$SSH_HOST" cat /etc/machine-id)" && REMOTE_UNIT_NOW="$(ssh -- "$SSH_HOST" systemctl show "$KUBELET_UNIT" -p Id -p LoadState)" || { printf '%s\n' '중단: 대표 host fresh identity 조회 실패' >&2; exit 1; }; if [ "$REMOTE_HOST_NOW" != "$REMOTE_HOSTNAME" ] || [ "$REMOTE_MACHINE_NOW" != "$TARGET_MACHINE_ID" ] || ! grep -q '^LoadState=loaded$' <<<"$REMOTE_UNIT_NOW"; then printf '%s\n' '중단: 대표 host machineID/unit identity 불일치' >&2; exit 1; fi; BEFORE="$(jq -Sc '[.[]|{name:.metadata.name,renew:.spec.renewTime}]|sort_by(.name)' <<<"$LEASES_JSON")"; sleep 60; if current_group_guard && AFTER="$(jq -Sc '[.[]|{name:.metadata.name,renew:.spec.renewTime}]|sort_by(.name)' <<<"$LEASES_JSON")" && jq -e --argjson before "$BEFORE" 'all(.[]; . as $after | any($before[]; .name==$after.name and .renew<$after.renew))' <<<"$AFTER" >/dev/null && jq -e 'all(.items[]; any(.status.conditions[]; .type=="Ready" and .status=="True") and all(.status.conditions[]; select(.type=="MemoryPressure" or .type=="DiskPressure" or .type=="PIDPressure") | .status=="False"))' <<<"$NODES_JSON" >/dev/null; then NORMAL_READY=true; printf 'NORMAL_READY=true nodes=%s function=%s\n' "${NODES[*]}" "$FUNCTION_EVIDENCE_REF"; else printf '%s\n' '중단: 모든 Lease/Ready/pressure 정상화 실패' >&2; fi; fi`

RBAC은 필요한 `nodes/leases/pods get,list`로 제한한다. audit는 별도 권한과 보존 정책의 owner가 current UID 목록과 UTC 창으로 제한한다. Secret·token·인증서 키·개인정보·전체 argv/journal은 저장하지 않는다.
