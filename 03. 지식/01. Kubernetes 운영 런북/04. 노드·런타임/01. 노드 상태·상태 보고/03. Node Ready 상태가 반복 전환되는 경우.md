# Node Ready 상태가 반복 전환되는 경우

## 입력값과 빠른 수집

현재 `Ready` 한 번은 flap 증거가 아니다. `작업 PC → API의 현재 Node/Lease identity`, `승인된 monitoring/audit의 과거 전환`, `조건부 SSH → 같은 host의 kubelet/runtime 시각`을 같은 UTC 창에서 결합한다.

| 구분 | 값 | 출처·gate |
|---|---|---|
| 최초 입력 | `NODE` | 단일 exact Node 이름 |
| API 확인 자동값 | Node UID/RV/machineID, current conditions, Lease UID/RV/renewTime | 두 API 성공과 holder identity 일치 |
| 계산 후보값 | escaped Node label/regex, `WINDOW_SECONDS` | RFC3339 round-trip과 start<end 뒤 생성 |
| 운영자 선택값 | 관측 창, `EVIDENCE_REF`, `OBSERVED_TRANSITIONS`, SSH host/unit | 승인된 monitoring/audit와 host mapping에서 선택 |

`unset INPUT_READY AUTO_READY TIME_READY QUERY_READY FLAP_READY HOST_READY IMPACT_READY NORMAL_READY NODE_UID NODE_RV NODE_MACHINE_ID LEASE_UID LEASE_RV LEASE_RENEW NODE_RE NODE_LABEL WINDOW_SECONDS START_RFC3339 END_RFC3339 EVIDENCE_REF OBSERVED_TRANSITIONS SSH_HOST REMOTE_HOSTNAME KUBELET_UNIT RUNTIME_UNIT OWNER_UID CONTAINER_ID CRI_ENDPOINT ARGV_RE FUNCTION_EVIDENCE_REF; NODE='PASTE_NODE_HERE'; INPUT_READY=false; if [ "$TOOLS_READY" = true ] && [[ "$NODE" =~ ^[a-z0-9]([-a-z0-9.]*[a-z0-9])?$ ]] && NODE_JSON="$(kubectl get node "$NODE" -o json)" && [ "$(jq -r '.metadata.name' <<<"$NODE_JSON")" = "$NODE" ]; then INPUT_READY=true; export NODE; printf 'INPUT_READY=true node=%s\n' "$NODE"; else printf '%s\n' '중단: 도구 또는 exact Node 조회 gate 실패' >&2; fi`

재등록은 object/controller/container/node/runtime/service/Linux host identity, resourceVersion, owner/containerID/host 관계, time·argv/regex와 모든 downstream `READY`를 먼저 지운다. API 실패 뒤 이전 전환·Lease 값은 사용하지 않는다.

`unset AUTO_READY TIME_READY QUERY_READY FLAP_READY HOST_READY IMPACT_READY NORMAL_READY NODE_UID NODE_RV NODE_MACHINE_ID LEASE_UID LEASE_RV LEASE_RENEW NODE_RE NODE_LABEL WINDOW_SECONDS START_RFC3339 END_RFC3339 EVIDENCE_REF OBSERVED_TRANSITIONS SSH_HOST REMOTE_HOSTNAME KUBELET_UNIT RUNTIME_UNIT OWNER_UID CONTAINER_ID CRI_ENDPOINT ARGV_RE FUNCTION_EVIDENCE_REF; then NODE_UID="$(jq -er '.metadata.uid' <<<"$NODE_JSON")" && NODE_RV="$(jq -er '.metadata.resourceVersion' <<<"$NODE_JSON")" && NODE_MACHINE_ID="$(jq -er '.status.nodeInfo.machineID' <<<"$NODE_JSON")" && LEASE_UID="$(jq -er '.metadata.uid' <<<"$LEASE_JSON")" && LEASE_RV="$(jq -er '.metadata.resourceVersion' <<<"$LEASE_JSON")" && LEASE_RENEW="$(jq -er '.spec.renewTime' <<<"$LEASE_JSON")" && [ "$(jq -r '.metadata.name' <<<"$NODE_JSON")" = "$NODE" ] && [ "$(jq -r '.spec.holderIdentity' <<<"$LEASE_JSON")" = "$NODE" ] && NODE_RE="$(printf '%s' "$NODE" | sed 's/[][\\.^$*+?(){}|]/\\&/g')" && NODE_LABEL="$(printf '%s' "$NODE" | sed 's/\\/\\\\/g;s/"/\\"/g')" && AUTO_READY=true && export NODE_UID NODE_RV NODE_MACHINE_ID LEASE_UID LEASE_RV LEASE_RENEW NODE_RE NODE_LABEL; if [ "$AUTO_READY" = true ]; then jq -r '"=== Node ===\n이름: \(.metadata.name)\nUID/RV: \(.metadata.uid) / \(.metadata.resourceVersion)\nMachine ID: \(.status.nodeInfo.machineID)\nRuntime: \(.status.nodeInfo.containerRuntimeVersion // "-")\n" + ([.status.conditions[] | "\(.type): \(.status) reason=\(.reason // "-") heartbeat=\(.lastHeartbeatTime // "-") transition=\(.lastTransitionTime // "-")"] | join("\n"))' <<<"$NODE_JSON"; printf 'Lease UID/RV/renew: %s / %s / %s\nAUTO_READY=true\n' "$LEASE_UID" "$LEASE_RV" "$LEASE_RENEW"; else printf '%s\n' '중단: identity 필드 누락' >&2; fi; else printf '%s\n' '중단: Node/Lease API producer 실패' >&2; fi`

모든 후속 조회는 다음 guard로 같은 Node UID·machineID와 Lease UID·holder를 다시 확인한다. 실패하면 과거 `READY`를 전부 폐기한다.

`current_api_guard(){ API_CURRENT_READY=false; local n l; n="$(kubectl get node "$NODE" -o json)" && l="$(kubectl get lease "$NODE" -n kube-node-lease -o json)" || { AUTO_READY=false; QUERY_READY=false; FLAP_READY=false; HOST_READY=false; IMPACT_READY=false; NORMAL_READY=false; return 1; }; if [ "$(jq -r '.metadata.uid' <<<"$n")" != "$NODE_UID" ] || [ "$(jq -r '.status.nodeInfo.machineID' <<<"$n")" != "$NODE_MACHINE_ID" ] || [ "$(jq -r '.metadata.uid' <<<"$l")" != "$LEASE_UID" ] || [ "$(jq -r '.spec.holderIdentity // empty' <<<"$l")" != "$NODE" ]; then AUTO_READY=false; QUERY_READY=false; FLAP_READY=false; HOST_READY=false; IMPACT_READY=false; NORMAL_READY=false; printf '%s\n' '중단: current Node/Lease identity 변경' >&2; return 1; fi; NODE_RV="$(jq -r '.metadata.resourceVersion' <<<"$n")"; LEASE_RV="$(jq -r '.metadata.resourceVersion' <<<"$l")"; LEASE_RENEW="$(jq -r '.spec.renewTime // empty' <<<"$l")"; if [ -z "$NODE_RV" ] || [ -z "$LEASE_RV" ] || [ -z "$LEASE_RENEW" ]; then AUTO_READY=false; QUERY_READY=false; FLAP_READY=false; HOST_READY=false; IMPACT_READY=false; NORMAL_READY=false; return 1; fi; API_CURRENT_READY=true; }`

### 2. 관측 창과 반복 증거

`START_RFC3339='PASTE_START_RFC3339_HERE'; END_RFC3339='PASTE_END_RFC3339_HERE'; TIME_READY=false; QUERY_READY=false; FLAP_READY=false; HOST_READY=false; IMPACT_READY=false; NORMAL_READY=false; unset EVIDENCE_REF OBSERVED_TRANSITIONS SSH_HOST REMOTE_HOSTNAME KUBELET_UNIT RUNTIME_UNIT FUNCTION_EVIDENCE_REF; if [ "$AUTO_READY" != true ]; then printf '%s\n' '중단: AUTO_READY가 아니다.' >&2; elif [[ ! "$START_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] || [[ ! "$END_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]]; then printf '%s\n' '중단: UTC RFC3339 형식 오류' >&2; elif S="$(date -u -d "$START_RFC3339" +%s)" && E="$(date -u -d "$END_RFC3339" +%s)" && [ "$(date -u -d "@$S" +%Y-%m-%dT%H:%M:%SZ)" = "$START_RFC3339" ] && [ "$(date -u -d "@$E" +%Y-%m-%dT%H:%M:%SZ)" = "$END_RFC3339" ] && [ "$S" -lt "$E" ]; then WINDOW_SECONDS="$((E-S))"; TIME_READY=true; export START_RFC3339 END_RFC3339 WINDOW_SECONDS; printf 'TIME_READY=true window=%ss\n' "$WINDOW_SECONDS"; else printf '%s\n' '중단: date round-trip 또는 start<end 실패' >&2; fi`

`QUERY_READY=false; if ! current_api_guard || [ "$TIME_READY" != true ]; then printf '%s\n' '중단: current identity/time gate가 필요하다.' >&2; else QUERY_READY=true; printf 'changes(kube_node_status_condition{node="%s",condition="Ready",status="true"}[%ss])\n' "$NODE_LABEL" "$WINDOW_SECONDS"; printf 'changes(kube_node_status_condition{node="%s",condition="Ready",status=~"false|unknown"}[%ss])\n' "$NODE_LABEL" "$WINDOW_SECONDS"; printf 'max_over_time(kube_node_status_condition{node="%s",condition=~"MemoryPressure|DiskPressure|PIDPressure|NetworkUnavailable",status="true"}[%ss])\n' "$NODE_LABEL" "$WINDOW_SECONDS"; fi`

출력 완성본을 현재 cluster와 정확한 UTC 창에서 조회한다. 빈 결과는 정상이나 전환 0회가 아니라 metric 미수집·label/범위 오류 후보이며 query/API 오류와 분리한다. scrape 누락·Prometheus restart도 전환처럼 보일 수 있어 승인된 monitoring/audit와 API identity를 함께 확인한다.

`EVIDENCE_REF='PASTE_APPROVED_EVIDENCE_REFERENCE_HERE'; OBSERVED_TRANSITIONS='PASTE_INTEGER_TRANSITION_COUNT_HERE'; FLAP_READY=false; if ! current_api_guard || [ "$TIME_READY" != true ] || [ -z "$EVIDENCE_REF" ] || [[ ! "$OBSERVED_TRANSITIONS" =~ ^[0-9]+$ ]] || [ "$OBSERVED_TRANSITIONS" -lt 2 ]; then printf '%s\n' '중단: 동일 UID·창에서 두 번 이상 전환한 승인 evidence가 필요하다.' >&2; else FLAP_READY=true; export EVIDENCE_REF OBSERVED_TRANSITIONS; printf 'FLAP_READY=true uid=%s transitions=%s evidence=%s\n' "$NODE_UID" "$OBSERVED_TRANSITIONS" "$EVIDENCE_REF"; fi`

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | 동일 Node UID가 승인 창에서 Ready 상태를 두 번 이상 전환 |
| 적용 | pressure, Lease/API 단절, kubelet/runtime restart, host/network/time과 전환 시각을 대조 |
| 제외 | 여러 Node 동시 전환은 [[04. 여러 Node가 동시에 NotReady·Unknown이 된 경우]], current False/Unknown 단일 상태는 각 전용 런북 |

검색 문자열: `Ready`, `NotReady`, `Unknown`, `PLEG`, `runtime`, `pressure`, `failed to update node status`, `x509`, `clock`.

## 관측별 진단 경로

| 같은 창의 상관 | 직접 검증 | 다음 owner |
|---|---|---|
| pressure/taint와 전환 | condition 시각·Node 범위 Pod | Linux/resource |
| Unknown과 Lease gap | API readyz·renewTime·kubelet API 오류 | API/network/kubelet |
| Lease 최신, False↔True | PLEG/runtime message와 Pod 공통성 | runtime/CRI |
| service restart와 전환 | 확인된 unit start/restart 시각 | service/host |

### 영향 범위

`IMPACT_READY=false; if ! current_api_guard || [ "$FLAP_READY" != true ]; then printf '%s\n' '중단: 최신 current identity와 FLAP_READY가 필요하다.' >&2; elif PODS_JSON="$(kubectl get pods -A --field-selector "spec.nodeName=$NODE" -o json)"; then IMPACT_READY=true; jq -r '.items[] | "=== \(.metadata.namespace)/\(.metadata.name) ===\nUID/RV: \(.metadata.uid) / \(.metadata.resourceVersion)\nOwner: \([.metadata.ownerReferences[]? | "\(.kind)/\(.name) uid=\(.uid)"] | join(","))\nPhase: \(.status.phase // "-")\nContainers: \([.status.containerStatuses[]? | "\(.name) ready=\(.ready) restart=\(.restartCount) id=\((.containerID // "-")|split("://")|last)"] | join("; "))"' <<<"$PODS_JSON"; else printf '%s\n' 'producer 오류: Pod API 실패' >&2; fi`

Pod 하나만 흔들리면 Node flap 원인으로 확대하지 않는다. 출력은 owner/containerID/node lifecycle 관계만 보존하며 Secret·env·전체 spec을 포함하지 않는다.

## 조건부 host 상관

`SSH_HOST='PASTE_APPROVED_SSH_HOST_HERE'; REMOTE_HOSTNAME='PASTE_REMOTE_HOSTNAME_HERE'; KUBELET_UNIT='PASTE_CONFIRMED_KUBELET_UNIT_HERE'; RUNTIME_UNIT='PASTE_CONFIRMED_RUNTIME_UNIT_HERE'; HOST_READY=false; if ! current_api_guard || [ "$FLAP_READY" != true ] || [[ "$SSH_HOST" == -* ]] || [[ ! "$KUBELET_UNIT" =~ ^[A-Za-z0-9_.@-]+\.service$ ]] || [[ ! "$RUNTIME_UNIT" =~ ^[A-Za-z0-9_.@-]+\.service$ ]]; then printf '%s\n' '중단: current flap identity와 확인된 host/unit이 필요하다.' >&2; else export SSH_HOST REMOTE_HOSTNAME KUBELET_UNIT RUNTIME_UNIT; printf 'HOST_INPUT_READY=true host=%s kubelet=%s runtime=%s\n' "$REMOTE_HOSTNAME" "$KUBELET_UNIT" "$RUNTIME_UNIT"; fi`

```bash
if ! current_api_guard || [ "$FLAP_READY" != true ] || [ "$TIME_READY" != true ] || [ -z "${SSH_HOST:-}" ]; then printf '%s\n' '중단: current identity/flap/time/host input gate가 필요하다.' >&2; else HOST_READY=false; if ssh -- "$SSH_HOST" bash -s -- "$REMOTE_HOSTNAME" "$NODE_MACHINE_ID" "$KUBELET_UNIT" "$RUNTIME_UNIT" "$START_RFC3339" "$END_RFC3339" <<'REMOTE'
set -o nounset
expected=$1
expected_machine=$2
kubelet=$3
runtime=$4
start=$5
end=$6
actual=$(hostname -f 2>/dev/null || hostname)
[ "$actual" = "$expected" ] || { printf '중단: host identity 불일치 expected=%s actual=%s\n' "$expected" "$actual" >&2; exit 50; }
[ -f /etc/machine-id ] && [ -r /etc/machine-id ] || { printf '%s\n' '중단: regular readable /etc/machine-id가 없음' >&2; exit 51; }
IFS= read -r actual_machine </etc/machine-id
[ "$actual_machine" = "$expected_machine" ] || { printf '중단: machine-id 불일치 expected=%s actual=%s\n' "$expected_machine" "$actual_machine" >&2; exit 52; }
[ "$(ps -p 1 -o comm=)" = systemd ] || exit 53
systemctl show "$kubelet" "$runtime" -p Id -p LoadState -p ActiveState -p SubState -p ExecMainStartTimestamp -p NRestarts --no-pager || exit 54
tmp=$(mktemp) || exit 55
trap 'rm -f "$tmp"' EXIT
if ! journalctl -u "$kubelet" -u "$runtime" --since "$start" --until "$end" --no-pager >"$tmp"; then printf '%s\n' 'producer 오류: journalctl 실패' >&2; exit 56; fi
if grep -E -i -C 3 'stopped|started|restart|node status|lease|PLEG|runtime|shim|rpc|pressure|oom|read-only|I/O error|timeout|refused|x509|clock' "$tmp"; then exit 0; else rc=$?; [ "$rc" -eq 1 ] && { printf '%s\n' 'filter no-match: 재시작/오류 부재 증거가 아님' >&2; exit 57; }; printf 'filter 오류: grep rc=%s\n' "$rc" >&2; exit 58; fi
REMOTE
then HOST_READY=true; printf 'HOST_READY=true host=%s\n' "$REMOTE_HOSTNAME"; else printf '%s\n' '중단: remote identity/service/pipeline 실패' >&2; fi; fi
```

SSH 인수는 positional로 전달하며 `eval`과 host key 검증 우회를 금지한다. unit은 active service 목록과 배포 계약에서 확인하며 runtime 식별자만 보고 추측하지 않는다. CRI 사용이 필요하면 active kubelet 설정의 단일 endpoint, runtime socket provider exact match, 현재 host/container ID를 재검증한 전용 런북에서만 실행한다.

## 변경과 정상화

이 런북은 `cordon`, `uncordon`, restart, reboot, drain을 직접 실행하지 않는다. 별도 mutation은 fresh Node/Lease UID·RV, workload owner/selector, PDB·quorum·대체 capacity, kubelet/runtime spec·revision·digest, 관측 evidence, 승인·window, 멱등성, stop/rollback을 실행 직전에 직접 gate한다. 공통 failure domain이 보이면 단일 Node 변경을 중단한다.

변경 또는 replacement 뒤 새 UID/RV·owner/containerID/node/host 관계를 재등록한다. 정상화는 `host/runtime/kubelet 원인 해소 → service restart 증가 중단 → Lease 연속 갱신 → Ready=True 무전환 → pressure/taint 정상 → workload Ready → 실제 요청·queue·Job 결과`를 최소 원래 관측 창만큼 검증한다. rollback도 동일하다.

`FUNCTION_EVIDENCE_REF='PASTE_APPROVED_FUNCTION_EVIDENCE_HERE'; NORMAL_READY=false; if ! current_api_guard || [ "$TIME_READY" != true ] || [ -z "$FUNCTION_EVIDENCE_REF" ]; then printf '%s\n' '중단: current API/time과 실제 기능 evidence가 필요하다.' >&2; else REMOTE_HOST_NOW="$(ssh -- "$SSH_HOST" hostname -f 2>/dev/null || ssh -- "$SSH_HOST" hostname)" && REMOTE_MACHINE_NOW="$(ssh -- "$SSH_HOST" cat /etc/machine-id)" && REMOTE_UNITS_NOW="$(ssh -- "$SSH_HOST" systemctl show "$KUBELET_UNIT" "$RUNTIME_UNIT" -p LoadState -p ActiveState)" || { printf '%s\n' '중단: fresh remote identity/service 조회 실패' >&2; exit 1; }; if [ "$REMOTE_HOST_NOW" != "$REMOTE_HOSTNAME" ] || [ "$REMOTE_MACHINE_NOW" != "$NODE_MACHINE_ID" ] || ! grep -q '^LoadState=loaded$' <<<"$REMOTE_UNITS_NOW"; then printf '%s\n' '중단: fresh host machineID/unit identity 불일치' >&2; exit 1; fi; L1="$LEASE_RENEW"; sleep 60; if current_api_guard && N="$(kubectl get node "$NODE" -o json)" && P="$(kubectl get pods -A --field-selector "spec.nodeName=$NODE" -o json)" && [ "$L1" != "$LEASE_RENEW" ] && jq -e 'any(.status.conditions[]; .type=="Ready" and .status=="True") and all(.status.conditions[]; select(.type=="MemoryPressure" or .type=="DiskPressure" or .type=="PIDPressure") | .status=="False")' <<<"$N" >/dev/null && jq -e 'all(.items[]; ([.status.containerStatuses[]?.ready] | all))' <<<"$P" >/dev/null; then NORMAL_READY=true; printf 'NORMAL_READY=true node=%s uid=%s lease=%s function=%s\n' "$NODE" "$NODE_UID" "$LEASE_RENEW" "$FUNCTION_EVIDENCE_REF"; else printf '%s\n' '중단: Lease/Ready/pressure/workload 정상화 실패' >&2; fi; fi`

RBAC은 필요한 `nodes/leases/pods get,list`에 제한한다. audit는 별도 권한의 owner가 Node UID와 창으로 제한해 조회하며 Secret·token·키·개인정보·전체 argv/journal을 저장하지 않는다.
