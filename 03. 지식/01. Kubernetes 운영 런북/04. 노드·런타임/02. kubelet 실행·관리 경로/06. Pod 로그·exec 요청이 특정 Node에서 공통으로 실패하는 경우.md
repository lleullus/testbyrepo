# Pod 로그·exec 요청이 특정 Node에서 공통으로 실패하는 경우

## 입력값과 빠른 수집

실행 hop은 `승인된 작업 PC 또는 bastion → API server → 현재 Pod UID → 현재 Node UID/machineID → kubelet streaming endpoint → 현재 containerID → CRI provider`다. `readyz`, Node `Ready`, log 한 건 또는 exec 한 건의 성공은 다음 hop이나 application 정상을 보장하지 않는다.

| 종류 | 값 | 확정 방법 |
|---|---|---|
| 최초 입력 | `NS`, `POD` | 공통 실패 중 대표 Pod의 exact name |
| API 자동값 | Pod UID/RV, owner, node UID/RV/machineID, regular container/current containerID | 모든 API와 관계 검증 성공 후 등록 |
| 계산 후보 | containerID suffix, escaped identity regex, CRI socket | provider와 endpoint 대조 전 미확정 |
| 운영자 선택값 | container, 승인 SSH host, kubelet/runtime unit, CRI endpoint, UTC 창, 비파괴 exec argv[0] | 후보 출력·CMDB·active kubelet 설정·사건 시각을 확인한 뒤 선택 |

아래 다중 행 Bash는 승인된 `kubectl` context를 가진 같은 shell에서 실행한다. node 전용 도구는 `host_gate` 이후 원격 분기에서만 요구한다.

```bash
TOOLS_READY=false
MISSING=''
for tool in kubectl jq ssh date grep sed mktemp; do command -v "$tool" >/dev/null 2>&1 || MISSING="$MISSING $tool"; done
if [ -n "${BASH_VERSION:-}" ] && [ -z "$MISSING" ]; then TOOLS_READY=true; else printf '중단: Bash 또는 도구 없음:%s\n' "$MISSING" >&2; fi

invalidate(){
  unset API_READY SELECTION_READY TIME_READY HOST_READY ENDPOINT_READY EVIDENCE_READY CRI_EVIDENCE_READY NORMALIZATION_READY POD_JSON POD_UID POD_RV POD_CREATED OWNER_KIND OWNER_NAME OWNER_UID NODE NODE_UID NODE_RV NODE_MACHINE_ID CONTAINER RAW_CONTAINER_ID CONTAINER_ID ID_RE SSH_HOST REMOTE_MACHINE_ID KUBELET_UNIT RUNTIME_UNIT CRI_ENDPOINT CRI_SOCKET RUNTIME_MAIN_PID START_RFC3339 END_RFC3339 START_EPOCH END_EPOCH EXEC_ARG0 LOG_BODY CRI_JSON
  API_READY=false; SELECTION_READY=false; TIME_READY=false; HOST_READY=false; ENDPOINT_READY=false; EVIDENCE_READY=false; CRI_EVIDENCE_READY=false; NORMALIZATION_READY=false
}
invalidate
NS='PASTE_NAMESPACE_HERE'
POD='PASTE_POD_HERE'
INPUT_READY=false
if [ "$TOOLS_READY" = true ] && [ -n "$NS" ] && [ -n "$POD" ] && kubectl get pod "$POD" -n "$NS" >/dev/null 2>&1; then INPUT_READY=true; printf 'INPUT_READY=true ns=%s pod=%s\n' "$NS" "$POD"; else printf '%s\n' '중단: exact Pod 입력/API 실패' >&2; fi

refresh_api(){
  invalidate
  [ "$INPUT_READY" = true ] || return 1
  local p n owner_count
  p="$(kubectl get pod "$POD" -n "$NS" -o json)" || return 1
  NODE="$(jq -er '.spec.nodeName|select(length>0)' <<<"$p")" || return 1
  n="$(kubectl get node "$NODE" -o json)" || return 1
  owner_count="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)]|length' <<<"$p")" || return 1
  [ "$owner_count" -le 1 ] || return 1
  POD_JSON="$p"
  POD_UID="$(jq -er '.metadata.uid' <<<"$p")" || return 1
  POD_RV="$(jq -er '.metadata.resourceVersion' <<<"$p")" || return 1
  POD_CREATED="$(jq -er '.metadata.creationTimestamp' <<<"$p")" || return 1
  OWNER_KIND="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].kind // "None"' <<<"$p")"
  OWNER_NAME="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].name // "-"' <<<"$p")"
  OWNER_UID="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid // "-"' <<<"$p")"
  NODE_UID="$(jq -er '.metadata.uid' <<<"$n")" || return 1
  NODE_RV="$(jq -er '.metadata.resourceVersion' <<<"$n")" || return 1
  NODE_MACHINE_ID="$(jq -er '.status.nodeInfo.machineID|select(length>0)' <<<"$n")" || return 1
  API_READY=true
  printf 'API_READY=true\nPod UID/RV: %s/%s created=%s\nOwner: %s/%s UID=%s\nNode: %s UID/RV=%s/%s machineID=%s\n' "$POD_UID" "$POD_RV" "$POD_CREATED" "$OWNER_KIND" "$OWNER_NAME" "$OWNER_UID" "$NODE" "$NODE_UID" "$NODE_RV" "$NODE_MACHINE_ID"
  jq -r '(.spec.initContainers[]? | "init/\(.name)"),(.spec.containers[]? as $c | (.status.containerStatuses[]?|select(.name==$c.name)) as $s | "regular/\($c.name) id=\($s.containerID // "-") state=\(($s.state|keys[0]) // "-")")' <<<"$p"
}
refresh_api || { invalidate; printf '%s\n' '중단: API identity 등록 실패' >&2; }

fresh_target(){
  local p n owner
  [ "$API_READY" = true ] && [ "$SELECTION_READY" = true ] || return 1
  p="$(kubectl get pod "$POD" -n "$NS" -o json)" || return 1
  n="$(kubectl get node "$NODE" -o json)" || return 1
  [ "$(jq -r '.metadata.uid' <<<"$p")" = "$POD_UID" ] || return 1
  [ "$(jq -r '.spec.nodeName' <<<"$p")" = "$NODE" ] || return 1
  [ "$(jq -r '.metadata.uid' <<<"$n")" = "$NODE_UID" ] || return 1
  [ "$(jq -r '.status.nodeInfo.machineID' <<<"$n")" = "$NODE_MACHINE_ID" ] || return 1
  [ "$(jq -r --arg c "$CONTAINER" '.status.containerStatuses[]?|select(.name==$c)|.containerID // empty' <<<"$p")" = "$RAW_CONTAINER_ID" ] || return 1
  if [ "$OWNER_UID" != '-' ]; then owner="$(kubectl get "$OWNER_KIND" "$OWNER_NAME" -n "$NS" -o json)" || return 1; [ "$(jq -r '.metadata.uid' <<<"$owner")" = "$OWNER_UID" ] || return 1; fi
}

CONTAINER='PASTE_REGULAR_CONTAINER_HERE'
SELECTION_READY=false
if [ "$API_READY" = true ] && RAW_CONTAINER_ID="$(jq -er --arg c "$CONTAINER" '.status.containerStatuses[]|select(.name==$c)|.containerID|select(length>0)' <<<"$POD_JSON")"; then
  case "$RAW_CONTAINER_ID" in containerd://*|cri-o://*) CONTAINER_ID="${RAW_CONTAINER_ID#*://}";; *) CONTAINER_ID='';; esac
  if [[ "$CONTAINER_ID" =~ ^[a-fA-F0-9]{12,128}$ ]]; then ID_RE="$(printf '%s' "$CONTAINER_ID" | sed 's/[][\\.^$*+?(){}|]/\\&/g')"; SELECTION_READY=true; printf 'SELECTION_READY=true container=%s id=%s\n' "$CONTAINER" "$RAW_CONTAINER_ID"; fi
fi
[ "$SELECTION_READY" = true ] || printf '%s\n' '중단: regular container/current containerID 선택 실패' >&2

SSH_HOST='PASTE_APPROVED_SSH_HOST_HERE'
KUBELET_UNIT='PASTE_KUBELET_UNIT_HERE'
RUNTIME_UNIT='PASTE_RUNTIME_UNIT_HERE'
CRI_ENDPOINT='PASTE_EXACT_CRI_ENDPOINT_HERE'
START_RFC3339='PASTE_START_RFC3339_HERE'
END_RFC3339='PASTE_END_RFC3339_HERE'
EXEC_ARG0='PASTE_NON_MUTATING_EXEC_ARG0_HERE'

TIME_READY=false
if [[ "$START_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && [[ "$END_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && [ "$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" = "$START_RFC3339" ] && [ "$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" = "$END_RFC3339" ]; then START_EPOCH="$(date -u -d "$START_RFC3339" +%s)"; END_EPOCH="$(date -u -d "$END_RFC3339" +%s)"; [ "$START_EPOCH" -lt "$END_EPOCH" ] && TIME_READY=true; fi
[ "$TIME_READY" = true ] || printf '%s\n' '중단: canonical UTC round-trip/start<end 실패' >&2

host_gate(){
  HOST_READY=false
  fresh_target || return 1
  [[ "$SSH_HOST" =~ ^[A-Za-z0-9_.@:-]+$ ]] && [[ "$SSH_HOST" != -* ]] || return 1
  REMOTE_MACHINE_ID="$(ssh -o BatchMode=yes -o StrictHostKeyChecking=yes -- "$SSH_HOST" bash -s <<'REMOTE'
set -eu
test -r /etc/machine-id
tr -d '\n' </etc/machine-id
REMOTE
)" || return 1
  [ "$REMOTE_MACHINE_ID" = "$NODE_MACHINE_ID" ] || return 1
  HOST_READY=true
}

endpoint_gate(){
  ENDPOINT_READY=false
  host_gate || return 1
  ssh -o BatchMode=yes -o StrictHostKeyChecking=yes -- "$SSH_HOST" bash -s -- "$KUBELET_UNIT" "$RUNTIME_UNIT" "$CRI_ENDPOINT" <<'REMOTE'
set -eu
ku=$1; ru=$2; expected=$3
[ "$(systemctl is-active "$ku")" = active ]
kp="$(systemctl show "$ku" -p MainPID --value)"; [ "$kp" -gt 1 ]
mapfile -d '' -t argv <"/proc/$kp/cmdline"
vals=(); cfg=''
for ((i=0;i<${#argv[@]};i++)); do case "${argv[i]}" in --container-runtime-endpoint=*) vals+=("${argv[i]#*=}");; --container-runtime-endpoint) ((i++)); vals+=("${argv[i]:-}");; --config=*) cfg="${argv[i]#*=}";; --config) ((i++)); cfg="${argv[i]:-}";; esac; done
if [ -n "$cfg" ]; then
  [[ "$cfg" = /* ]] && [ -f "$cfg" ] && [ -r "$cfg" ]
  mapfile -t from_cfg < <(sed -n 's/^[[:space:]]*containerRuntimeEndpoint:[[:space:]]*["'"']\{0,1\}\([^"'"'[:space:]]*\)["'"']\{0,1\}[[:space:]]*$/\1/p' "$cfg")
  [ "${#from_cfg[@]}" -eq 1 ] && vals+=("${from_cfg[0]}")
fi
mapfile -t unique < <(printf '%s\n' "${vals[@]}" | sed '/^$/d' | sort -u)
[ "${#unique[@]}" -eq 1 ] && [ "${unique[0]}" = "$expected" ] && [[ "$expected" == unix:///* ]]
sock=${expected#unix://}; [ -S "$sock" ]
[ "$(systemctl is-active "$ru")" = active ]
rp="$(systemctl show "$ru" -p MainPID --value)"; [ "$rp" -gt 1 ]
out="$(sudo ss -xlpn)"; line="$(grep -F -- "$sock" <<<"$out")"; [ -n "$line" ]; grep -Fq -- "pid=$rp," <<<"$line"
printf 'endpoint=%s socket=%s providerPID=%s\n' "$expected" "$sock" "$rp"
REMOTE
  local rc=$?
  if [ "$rc" -eq 0 ]; then CRI_SOCKET="${CRI_ENDPOINT#unix://}"; RUNTIME_MAIN_PID="$(ssh -o BatchMode=yes -o StrictHostKeyChecking=yes -- "$SSH_HOST" systemctl show "$RUNTIME_UNIT" -p MainPID --value)"; ENDPOINT_READY=true; fi
}
endpoint_gate || printf '%s\n' '중단: remote identity/active kubelet endpoint/runtime provider 실패' >&2
```

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | 같은 Node의 서로 다른 현재 Pod UID에서 logs 또는 exec가 공통 실패하고 개별 container 선택·RBAC 오류가 배제됨 |
| 적용 | API server→kubelet streaming→runtime/container 경계를 같은 UTC 창과 identity로 연결 가능 |
| 제외 | 단일 Pod logs는 [[27. Pod 로그를 가져올 수 없는 경우]], 단일 exec는 [[33. Pod exec 연결이 실패하는 경우]], `Forbidden`은 각 RBAC 런북 |

검색 문자열: `unable to retrieve container logs`, `unable to upgrade connection`, `error dialing backend`, `proxy error`, `connection refused`, `i/o timeout`, `x509`, `stream copy error`, `container not found`.

## 관측별 진단 경로

| 지금 보이는 것 | 먼저 볼 것 | 다음 경계 | 상세 절 |
|---|---|---|---|
| API 일반 요청도 실패 | readyz HTTP와 원 요청 status | client/API server | [[#API·RBAC와 대표 요청]] |
| logs와 exec가 해당 Node에서 함께 실패 | Pod UID/containerID·backend 오류 | kubelet streaming | [[#kubelet·runtime 증거]] |

| exec만 실패 | Running state·argv[0]·stream 오류 | exec/container | [[33. Pod exec 연결이 실패하는 경우]] |

## API·RBAC와 대표 요청

아래 명령은 현재 target과 exact subresource 권한을 다시 확인한다. `can-i=yes`는 실제 요청 성공을 보장하지 않는다.

```bash
if fresh_target; then kubectl get --raw='/readyz?verbose' >/dev/null && printf '%s\n' 'API_READYZ=true' || printf '%s\n' 'API readyz producer 실패' >&2; else printf '%s\n' '중단: fresh target 실패' >&2; fi
if fresh_target; then kubectl auth can-i get "pods/$POD" --subresource=log -n "$NS"; kubectl auth can-i create "pods/$POD" --subresource=exec -n "$NS"; else printf '%s\n' '중단: fresh target 실패' >&2; fi
```

audit가 활성화된 환경에서는 동일 `auditID`의 timestamp, verb, `pods/log` 또는 `pods/exec`, namespace/name, requestURI, response status와 현재 Pod lifecycle을 승인된 audit reader가 대조한다. actor·token·request body는 저장하지 않는다. audit record가 0개·복수이거나 사건 창 밖이면 증거 gate는 실패다.

## kubelet·runtime 증거

producer 실패, filter no-match, filter 오류를 분리한다. 각 명령은 current target, canonical 시간, remote machine identity와 endpoint/provider를 직접 재검사한다.

```bash
EVIDENCE_READY=false
if [ "$TIME_READY" != true ] || ! endpoint_gate || ! fresh_target; then printf '%s\n' '중단: current time/host/endpoint/target gate 실패' >&2; else
  TMP="$(mktemp)" || exit 1
  ssh -o BatchMode=yes -o StrictHostKeyChecking=yes -- "$SSH_HOST" sudo journalctl -u "$KUBELET_UNIT" --since "$START_RFC3339" --until "$END_RFC3339" -o short-iso --no-pager >"$TMP"
  PRODUCER_RC=$?
  if [ "$PRODUCER_RC" -eq 0 ]; then grep -E -i -C 3 -- "$POD_UID|$ID_RE|log|stream|exec|upgrade|refused|timeout|x509|runtime|not found" "$TMP"; FILTER_RC=$?; else FILTER_RC=99; fi
  rm -f "$TMP"
  printf 'producer_rc=%s filter_rc=%s\n' "$PRODUCER_RC" "$FILTER_RC"
  if [ "$PRODUCER_RC" -eq 0 ] && { [ "$FILTER_RC" -eq 0 ] || [ "$FILTER_RC" -eq 1 ]; }; then EVIDENCE_READY=true; fi
fi

CRI_EVIDENCE_READY=false
if [ "$EVIDENCE_READY" = true ] && endpoint_gate && fresh_target; then
  CRI_JSON="$(ssh -o BatchMode=yes -o StrictHostKeyChecking=yes -- "$SSH_HOST" sudo crictl --runtime-endpoint="$CRI_ENDPOINT" inspect "$CONTAINER_ID")"
  PRODUCER_RC=$?
  if [ "$PRODUCER_RC" -eq 0 ] && jq -e --arg id "$CONTAINER_ID" --arg uid "$POD_UID" --arg ns "$NS" --arg pod "$POD" --arg c "$CONTAINER" '.status.id==$id and .status.labels["io.kubernetes.pod.uid"]==$uid and .status.labels["io.kubernetes.pod.namespace"]==$ns and .status.labels["io.kubernetes.pod.name"]==$pod and .status.labels["io.kubernetes.container.name"]==$c' <<<"$CRI_JSON" >/dev/null; then CRI_EVIDENCE_READY=true; jq -c '{id:.status.id,state:.status.state,createdAt:.status.createdAt,startedAt:.status.startedAt,finishedAt:.status.finishedAt,exitCode:.status.exitCode,podSandboxId:.status.podSandboxId}' <<<"$CRI_JSON"; else printf '중단: CRI producer=%s 또는 exact identity 불일치\n' "$PRODUCER_RC" >&2; fi
fi
```

`filter_rc=1`은 일치 없음이지 오류 부재가 아니다. `connection refused`·timeout은 API server→kubelet listener 경계, `x509`는 kubelet serving certificate·clock·trust 경계, exact CRI `NotFound`는 lifecycle/GC 경계다. TLS 검증 비활성화, broad RBAC, kubelet/runtime restart를 진단에 사용하지 않는다.

## 변경 경계와 정상화

이 문서는 상태 변경 명령을 제공하지 않는다. kubelet/runtime/certificate/network 변경은 fresh Pod·owner·Node UID, remote machineID, active endpoint/provider, evidence, approval ID, 변경 창, PDB/quorum·가용성, idempotence와 rollback 원본을 별도 변경 절차가 명령 직전에 다시 검증할 때만 수행한다.

변경 뒤 기존 Pod가 교체됐으면 기존 이름·UID·containerID와 모든 READY를 폐기한다. controller selector와 current controller UID, owner chain, 변경 이후 creation time, 단일 후보 관계로 새 Pod를 선택한 뒤 `refresh_api`부터 반복한다.

```bash
NORMALIZATION_READY=false
if [ "$TIME_READY" = true ] && endpoint_gate && fresh_target && [ "$CRI_EVIDENCE_READY" = true ] && [[ "$EXEC_ARG0" =~ ^[A-Za-z0-9._/-]+$ ]]; then
  kubectl logs "$POD" -n "$NS" -c "$CONTAINER" --timestamps --since-time="$START_RFC3339" --tail=20 >/dev/null && LOG_RC=0 || LOG_RC=$?
  kubectl exec "$POD" -n "$NS" -c "$CONTAINER" -- "$EXEC_ARG0" >/dev/null && EXEC_RC=0 || EXEC_RC=$?
  if [ "$LOG_RC" -eq 0 ] && [ "$EXEC_RC" -eq 0 ]; then NORMALIZATION_READY=true; printf 'NORMALIZATION_READY=true podUID=%s containerID=%s\n' "$POD_UID" "$RAW_CONTAINER_ID"; else printf '정상화 실패 log_rc=%s exec_rc=%s\n' "$LOG_RC" "$EXEC_RC" >&2; fi
fi
```

최종 정상화에는 같은 Node의 서로 다른 현재 Pod UID에서 logs·exec 반복 성공, kubelet/runtime 오류 재발 없음, Node heartbeat 안정, owner desired/Ready, EndpointSlice와 실제 서비스 기능이 필요하다. `NORMALIZATION_READY=true` 한 번만으로 application 정상은 아니다.

## Prometheus 빠른 관측

```bash
if fresh_target; then NODE_RE="$(printf '%s' "$NODE" | sed 's/[][\\.^$*+?(){}|]/\\&/g')"; printf 'kube_node_status_condition{node=~"^%s$",condition="Ready",status=~"false|unknown"} == 1\n' "$NODE_RE"; printf 'sum by(code,subresource)(rate(apiserver_request_total{resource="pods",subresource=~"log|exec",code=~"4..|5.."}[5m]))\n'; else printf '%s\n' '중단: fresh target 실패' >&2; fi
```

첫 query는 current node identity와 연결된다. 두 번째 API metric에는 Node label이 없을 수 있으므로 특정 Node 귀속 근거가 아니다. 빈 vector는 정상이나 0이 아니라 metric 미수집·label/range/권한 문제 후보며 API·journal 증거로 이관한다.

## 증거·미검증

Pod/owner/Node UID·RV, machineID, regular container/current containerID, canonical 창, auditID와 redacted request identity, producer/filter rc, endpoint/provider PID, CRI 요약, 정상화 결과만 보존한다. Secret·token·private key·전체 argv/config/inspect/journal은 저장하지 않는다.

Kubernetes 1.26 배포판별 kubelet unit, runtime, SSH/CMDB, audit, metric label은 환경별 미검증이다.
