# kubelet이 Node 상태·Lease를 갱신하지 못하는 경우

## 입력값과 빠른 수집

hop은 `kubelet API client → node route/TLS → API server → Node status와 kube-node-lease Lease`다. 작업 PC API 관측과 대상 host의 kubelet journal을 같은 canonical UTC 창으로 묶는다.

| 입력 종류 | 값 | gate |
|---|---|---|
| 최초 입력 | `NODE`, `SSH_TARGET` | exact Node, SSH allowlist |
| API 자동값 | Node UID/RV, Lease UID/RV·holder·renewTime, runtime, Pod owner | 매 조회 fresh exact match |
| 계산 후보값 | kubelet unit, API endpoint host/port | active argv/config에서 단일 후보 추출 |
| 운영자 선택값 | UTC 창, host 관계, 변경 승인·quorum | round-trip과 `start < end` |

**입력 전 모든 identity/resourceVersion/owner/containerID/node/host/relation/time/argv·regex/READY 폐기**

`unset INPUT_READY WINDOW_READY API_READY HOST_READY UNIT_READY EVIDENCE_READY MUTATION_READY NORMAL_READY NODE_UID NODE_RV NODE_MACHINE_ID LEASE_UID LEASE_RV LEASE_RENEW HOST_IDENTITY KUBELET_UNIT API_ENDPOINT START_RFC3339 END_RFC3339 FUNCTION_EVIDENCE_REF; NODE='PASTE_NODE_NAME_HERE'; SSH_TARGET='PASTE_APPROVED_SSH_TARGET_HERE'; MISSING=''; for T in kubectl jq ssh date grep awk sed mktemp curl; do command -v "$T" >/dev/null 2>&1 || MISSING="$MISSING $T"; done; INPUT_READY=false; if [ -n "${BASH_VERSION:-}" ] && [ -z "$MISSING" ] && [[ "$NODE" =~ ^[A-Za-z0-9._-]+$ ]] && [[ "$SSH_TARGET" =~ ^[A-Za-z0-9_.@:-]+$ ]] && [[ "$SSH_TARGET" != -* ]] && N="$(kubectl get node "$NODE" -o json)" && L="$(kubectl get lease "$NODE" -n kube-node-lease -o json)" && [ "$(jq -r '.metadata.name' <<<"$N")" = "$NODE" ] && [ "$(jq -r '.spec.holderIdentity // empty' <<<"$L")" = "$NODE" ]; then NODE_UID="$(jq -er '.metadata.uid' <<<"$N")" && NODE_MACHINE_ID="$(jq -er '.status.nodeInfo.machineID' <<<"$N")" && LEASE_UID="$(jq -er '.metadata.uid' <<<"$L")" && INPUT_READY=true; SSH_OPTS=(-o BatchMode=yes -o StrictHostKeyChecking=yes -o ConnectTimeout=10); printf 'INPUT_READY=%s node=%s uid=%s machineID=%s\n' "$INPUT_READY" "$NODE" "$NODE_UID" "$NODE_MACHINE_ID"; else printf '중단: 입력/Node·Lease 조회 실패 missing=%s\n' "$MISSING" >&2; fi`

`WINDOW_READY=false; API_READY=false; HOST_READY=false; UNIT_READY=false; EVIDENCE_READY=false; MUTATION_READY=false; NORMAL_READY=false; START_RFC3339='PASTE_START_UTC_RFC3339_HERE'; END_RFC3339='PASTE_END_UTC_RFC3339_HERE'; S="$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)"; SRC=$?; E="$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)"; ERC=$?; if [ "$INPUT_READY" = true ] && [[ "$START_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && [[ "$END_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && [ "$SRC" -eq 0 ] && [ "$ERC" -eq 0 ] && [ "$S" = "$START_RFC3339" ] && [ "$E" = "$END_RFC3339" ] && [[ "$S" < "$E" ]]; then WINDOW_READY=true; printf 'WINDOW_READY=true\n'; else printf '%s\n' '중단: UTC RFC3339 창 실패' >&2; fi`

`invalidate_current(){ API_READY=false; HOST_READY=false; UNIT_READY=false; EVIDENCE_READY=false; MUTATION_READY=false; NORMAL_READY=false; }; api_guard(){ API_READY=false; local n l; [ "$WINDOW_READY" = true ] || { invalidate_current; return 1; }; n="$(kubectl get node "$NODE" -o json)" && l="$(kubectl get lease "$NODE" -n kube-node-lease -o json)" || { invalidate_current; return 1; }; [ "$(jq -r '.metadata.uid' <<<"$n")" = "$NODE_UID" ] && [ "$(jq -r '.status.nodeInfo.machineID' <<<"$n")" = "$NODE_MACHINE_ID" ] && [ "$(jq -r '.metadata.uid' <<<"$l")" = "$LEASE_UID" ] && [ "$(jq -r '.spec.holderIdentity // empty' <<<"$l")" = "$NODE" ] || { invalidate_current; return 1; }; NODE_RV="$(jq -er '.metadata.resourceVersion' <<<"$n")" && LEASE_RV="$(jq -er '.metadata.resourceVersion' <<<"$l")" && LEASE_RENEW="$(jq -er '.spec.renewTime' <<<"$l")" && RUNTIME_PROVIDER="$(jq -er '.status.nodeInfo.containerRuntimeVersion|split("://")[0]' <<<"$n")" || { invalidate_current; return 1; }; API_READY=true; }`

**Node·Lease·Pod 관계 수집**

`api_guard || exit; N="$(kubectl get node "$NODE" -o json)" && L="$(kubectl get lease "$NODE" -n kube-node-lease -o json)" && P="$(kubectl get pods -A --field-selector "spec.nodeName=$NODE" -o json)"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ]; then jq -n -r --argjson n "$N" --argjson l "$L" --argjson p "$P" '"=== Node/Lease ===\nNode UID/RV: \($n.metadata.uid)/\($n.metadata.resourceVersion)\nLease UID/RV: \($l.metadata.uid)/\($l.metadata.resourceVersion)\nholder/renew: \($l.spec.holderIdentity)/\($l.spec.renewTime)\nconditions: \([$n.status.conditions[]|"\(.type)=\(.status)@\(.lastHeartbeatTime)"]|join(";"))\n=== Pods ===\n" + ([$p.items[]|"\(.metadata.namespace)/\(.metadata.name) uid=\(.metadata.uid) owner=\([.metadata.ownerReferences[]?|"\(.kind)/\(.name)/\(.uid)"]|join(",")) containerID=\([.status.containerStatuses[]?|.containerID // "-"]|join(","))"]|join("\n"))'; else printf '중단: producer_rc=%s\n' "$PRODUCER_RC" >&2; fi`

Lease UID는 Node UID가 아니다. holder·renewTime·RV를 Node identity와 별도 보존한다.

**host 승인과 guard**

`api_guard || exit; RID="$(ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" 'printf "hostname=%s machine_id=%s boot_id=%s\n" "$(hostname -f 2>/dev/null || hostname)" "$(cat /etc/machine-id 2>/dev/null)" "$(cat /proc/sys/kernel/random/boot_id 2>/dev/null)"')"; RC=$?; printf '%s\n' "$RID"; read -r -p '대상 Node가 맞으면 Node 이름 입력: ' HC; if [ "$RC" -eq 0 ] && [ "$HC" = "$NODE" ] && [[ "$RID" =~ ^hostname=[^[:space:]]+[[:space:]]machine_id=[0-9a-fA-F-]+[[:space:]]boot_id=[0-9a-fA-F-]+$ ]] && [[ "$RID" == *"machine_id=$NODE_MACHINE_ID "* ]]; then HOST_IDENTITY="$RID"; HOST_READY=true; else printf '%s\n' '중단: remote /etc/machine-id와 Node identity 불일치' >&2; exit 1; fi`

`host_guard(){ HOST_READY=false; api_guard || return 1; local r; r="$(ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" 'printf "hostname=%s machine_id=%s boot_id=%s\n" "$(hostname -f 2>/dev/null || hostname)" "$(cat /etc/machine-id 2>/dev/null)" "$(cat /proc/sys/kernel/random/boot_id 2>/dev/null)"')" || { invalidate_current; return 1; }; [ "$r" = "$HOST_IDENTITY" ] && [[ "$r" == *"machine_id=$NODE_MACHINE_ID "* ]] || { invalidate_current; return 1; }; HOST_READY=true; }`

`host_guard || exit; OUT="$(ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" systemctl list-unit-files --type=service --no-legend)"; PRODUCER_RC=$?; U=(); if [ "$PRODUCER_RC" -eq 0 ]; then MATCHES="$(grep -E '^kubelet[A-Za-z0-9_.@-]*\.service[[:space:]]' <<<"$OUT")"; FILTER_RC=$?; [ "$FILTER_RC" -eq 0 ] && mapfile -t U < <(awk '{print $1}' <<<"$MATCHES"); else FILTER_RC=99; fi; if [ "$PRODUCER_RC" -eq 0 ] && [ "$FILTER_RC" -eq 0 ] && [ "${#U[@]}" -eq 1 ] && [[ "${U[0]}" =~ ^kubelet[A-Za-z0-9_.@-]*\.service$ ]]; then KUBELET_UNIT="${U[0]}"; UNIT_READY=true; else printf '중단: unit producer=%s filter=%s count=%s\n' "$PRODUCER_RC" "$FILTER_RC" "${#U[@]}" >&2; fi; unit_guard(){ UNIT_READY=false; host_guard || return 1; local s; s="$(ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" systemctl show "$KUBELET_UNIT" -p Id -p LoadState)" || { invalidate_current; return 1; }; [ "$(sed -n 's/^Id=//p' <<<"$s")" = "$KUBELET_UNIT" ] && [ "$(sed -n 's/^LoadState=//p' <<<"$s")" = loaded ] || { invalidate_current; return 1; }; UNIT_READY=true; }`

## 진입 판단과 분기

검색 문자열: `Failed to update node status`, `Failed to update lease`, `Unauthorized`, `forbidden`, `x509`, `context deadline exceeded`, `i/o timeout`, `connection refused`, `429`.

| 관측 | 의미 범위 | 다음 경로 |
|---|---|---|
| status 실패, Lease 진행 | Node status API 경로만 실패 후보 | status request/audit |
| status·Lease 모두 중단, timeout | node→API network/APF 후보 | route/API owner |
| `401/403/x509` | credential·RBAC·trust/clock 후보 | 보안 owner |
| 여러 Node 동시 정체 | node-local 결론 금지 | control-plane |
| PLEG/relist가 주 오류 | API 보고 장애로 확대 금지 | [[04. kubelet 로그에 PLEG is not healthy가 기록된 경우]] |

**bounded journal**

`unit_guard || exit; TMP="$(mktemp)" || exit 1; ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" sudo journalctl -u "$KUBELET_UNIT" --since "$START_RFC3339" --until "$END_RFC3339" --no-pager >"$TMP"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ]; then grep -E -i -C 3 'update node status|update lease|update status for pod|unauthorized|forbidden|x509|certificate|context deadline exceeded|i/o timeout|connection refused|429|PLEG|relist' "$TMP"; FILTER_RC=$?; else FILTER_RC=99; fi; rm -f "$TMP"; printf 'producer_rc=%s filter_rc=%s\n' "$PRODUCER_RC" "$FILTER_RC"; if [ "$PRODUCER_RC" -eq 0 ] && [ "$FILTER_RC" -eq 0 ]; then EVIDENCE_READY=true; else printf '%s\n' '중단: 실패 또는 무결과를 정상으로 해석하지 않음' >&2; fi`

**active argv에서 API endpoint 엄격 추출과 host 보조 확인**

`unit_guard || exit; [ "$EVIDENCE_READY" = true ] || exit 1; ARGV="$(ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" systemctl show "$KUBELET_UNIT" -p ExecStart --value)"; PRODUCER_RC=$?; EP_MATCHES="$(grep -Eo 'https://[-A-Za-z0-9._]+:[0-9]+' <<<"$ARGV")"; FILTER_RC=$?; API_EPS=(); [ "$FILTER_RC" -eq 0 ] && mapfile -t API_EPS < <(sort -u <<<"$EP_MATCHES"); if [ "$PRODUCER_RC" -eq 0 ] && [ "$FILTER_RC" -eq 0 ] && [ "${#API_EPS[@]}" -eq 1 ]; then API_ENDPOINT="${API_EPS[0]}"; API_HOSTPORT="${API_ENDPOINT#https://}"; printf 'endpoint_candidate=%s\n' "$API_ENDPOINT"; else printf '%s\n' 'active argv에 단일 endpoint가 없음: kubeconfig server를 비밀값 없이 config owner가 확인' >&2; exit 1; fi; unit_guard || exit; ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" getent ahosts "${API_HOSTPORT%:*}" >/dev/null; DNS_RC=$?; ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" curl -sS --max-time 10 -o /dev/null -w 'http=%{http_code}\n' "$API_ENDPOINT/readyz"; CURL_RC=$?; printf 'dns_rc=%s curl_rc=%s\n' "$DNS_RC" "$CURL_RC"`

HTTP `401/403`은 transport 도달 증거이지 kubelet credential 성공이 아니다. `-k`, token 삽입, RBAC 확대를 하지 않는다. API audit은 승인된 담당자가 이 UTC 창의 Node/Lease requestURI, kubelet subject, responseStatus와 stageTimestamp만 조회한다.

## Prometheus query

`api_guard || exit; printf 'time()-kube_lease_renew_time{namespace="kube-node-lease",lease="%s"}\n' "$NODE"; printf 'kube_node_status_condition{node="%s",condition="Ready",status=~"false|unknown"} == 1\n' "$NODE"`

빈 vector는 정상 성공이 아니다. metric 미수집, label 불일치, 조회 범위·권한 실패로 분리하고 현재 Lease API와 journal을 기준으로 한다.

## 변경·rollback·정상화

네트워크, certificate, RBAC 문제가 직접 확인되면 kubelet restart를 하지 않는다. process 자체 wedged 근거가 있을 때만 fresh Node/Lease/host/evidence, 승인 window, PDB/quorum, Job 멱등성, 대체 capacity를 확인한다.

`MUTATION_READY=false; unit_guard || exit; [ "$EVIDENCE_READY" = true ] || exit 1; PODS_NOW="$(kubectl get pods -A --field-selector "spec.nodeName=$NODE" -o json)" || exit 1; jq -r '.items[]|"\(.metadata.namespace)/\(.metadata.name) uid=\(.metadata.uid) owner=\([.metadata.ownerReferences[]?|"\(.kind)/\(.name)/\(.uid)"]|join(","))"' <<<"$PODS_NOW"; read -r -p '승인 ID: ' APPROVAL_ID; read -r -p '승인 변경 창: ' CHANGE_WINDOW; read -r -p '위 current owner의 quorum/PDB/멱등성 확인 후 Node 이름 입력: ' QC; [ -n "$APPROVAL_ID" ] && [ -n "$CHANGE_WINDOW" ] && [ "$QC" = "$NODE" ] && MUTATION_READY=true; [ "$MUTATION_READY" = true ] || exit 1; ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" sudo systemctl restart "$KUBELET_UNIT"; RC=$?; printf 'mutation_rc=%s node_uid=%s node_rv=%s\n' "$RC" "$NODE_UID" "$NODE_RV"; [ "$RC" -eq 0 ] || exit 1`

restart 반복은 rollback이 아니다. config/certificate/RBAC 변경은 승인된 이전 revision과 최소 권한으로만 되돌린다. 실패 시 추가 mutation을 중단한다.

`FUNCTION_EVIDENCE_REF='PASTE_APPROVED_FUNCTION_EVIDENCE_HERE'; NORMAL_READY=false; unit_guard || exit; [ "$(kubectl get node "$NODE" -o jsonpath='{.metadata.uid}')" = "$NODE_UID" ] || { printf '%s\n' 'replacement Node: 전체 재등록 필요' >&2; exit 1; }; L1="$(kubectl get lease "$NODE" -n kube-node-lease -o jsonpath='{.spec.renewTime}')"; sleep 60; api_guard || exit; L2="$(kubectl get lease "$NODE" -n kube-node-lease -o jsonpath='{.spec.renewTime}')"; N2="$(kubectl get node "$NODE" -o json)"; if [[ "$L1" < "$L2" ]] && [ -n "$FUNCTION_EVIDENCE_REF" ] && jq -e 'any(.status.conditions[]; .type=="Ready" and .status=="True")' <<<"$N2" >/dev/null; then NORMAL_READY=true; printf 'NORMAL_READY=true lease=%s->%s function=%s\n' "$L1" "$L2" "$FUNCTION_EVIDENCE_REF"; else printf '%s\n' '중단: Lease/Ready/실제 기능이 진행하지 않음' >&2; fi`

이후 Node heartbeat, Pod UID/status, owner desired/Ready, EndpointSlice와 실제 요청·queue·Job 산출물을 관찰 창 동안 검증한다. replacement는 새 UID·host·runtime·owner/containerID 관계를 처음부터 등록한다.

## 보안·환경 경계

Node/Lease/Pod read 최소 RBAC만 사용한다. Secret/token/client key/kubeconfig 원문/전체 audit·journal은 저장하지 않는다. SSH inventory, kubelet unit/argv, kubeconfig endpoint, audit 위치, APF/network, Prometheus label, workload quorum과 기능 검증은 환경 미검증이다.
