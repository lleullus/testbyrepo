# 같은 Node의 여러 Pod 생성·종료·상태 동기화가 정체된 경우

## 입력값과 빠른 수집

실행 경로는 `API Pod lifecycle → kubelet pod worker/status manager → active CRI endpoint → sandbox/container → Linux host`다. 서로 다른 Pod UID 두 개 이상이 같은 Node·UTC 창에서 정체되어야 이 문서로 진입한다.

| 입력 종류 | 값 | gate |
|---|---|---|
| 최초 입력 | `NODE`, `SSH_TARGET` | exact Node·안전한 SSH 목적지 |
| API 자동값 | Node/Lease RV, Pod UID·owner UID·containerID·node·시간 | fresh API에서 관계 보존 |
| 계산 후보값 | 대표 Pod 쌍, kubelet/runtime unit, CRI endpoint | 임의 첫 후보 선택 금지, strict extraction |
| 운영자 선택값 | canonical UTC 창, host relation, 대표 Pod 쌍, 변경 승인 | `start < end`, UID exact match, quorum/멱등성 |

**입력 전 모든 stale identity/resourceVersion/owner/containerID/node/host/relation/time/argv·regex/downstream READY 폐기**

`unset INPUT_READY WINDOW_READY API_READY HOST_READY UNIT_READY TARGET_READY CRI_READY EVIDENCE_READY MUTATION_READY NORMAL_READY NODE_UID NODE_RV LEASE_RV HOST_IDENTITY KUBELET_UNIT RUNTIME_PROVIDER RUNTIME_UNIT CRI_ENDPOINT CRI_SOCKET NS1 POD1 UID1 NS2 POD2 UID2 START_RFC3339 END_RFC3339; NODE='PASTE_NODE_NAME_HERE'; SSH_TARGET='PASTE_APPROVED_SSH_TARGET_HERE'; MISSING=''; for T in kubectl jq ssh date grep awk sed mktemp; do command -v "$T" >/dev/null 2>&1 || MISSING="$MISSING $T"; done; INPUT_READY=false; if [ -n "${BASH_VERSION:-}" ] && [ -z "$MISSING" ] && [[ "$NODE" =~ ^[A-Za-z0-9._-]+$ ]] && [[ "$SSH_TARGET" =~ ^[A-Za-z0-9_.@:-]+$ ]] && [[ "$SSH_TARGET" != -* ]] && N="$(kubectl get node "$NODE" -o json)"; then NODE_UID="$(jq -r '.metadata.uid' <<<"$N")"; [ -n "$NODE_UID" ] && INPUT_READY=true; SSH_ARGS=(-o BatchMode=yes -o StrictHostKeyChecking=yes -o ConnectTimeout=10 "$SSH_TARGET"); printf 'INPUT_READY=%s node=%s uid=%s\n' "$INPUT_READY" "$NODE" "$NODE_UID"; else printf '중단: 입력 실패 missing=%s\n' "$MISSING" >&2; fi`

`WINDOW_READY=false; START_RFC3339='PASTE_START_UTC_RFC3339_HERE'; END_RFC3339='PASTE_END_UTC_RFC3339_HERE'; S="$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)"; SRC=$?; E="$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)"; ERC=$?; if [ "$INPUT_READY" = true ] && [ "$SRC" -eq 0 ] && [ "$ERC" -eq 0 ] && [ "$S" = "$START_RFC3339" ] && [ "$E" = "$END_RFC3339" ] && [[ "$S" < "$E" ]]; then WINDOW_READY=true; else printf '%s\n' '중단: canonical UTC 창 실패' >&2; fi`

`api_guard(){ API_READY=false; HOST_READY=false; UNIT_READY=false; TARGET_READY=false; CRI_READY=false; EVIDENCE_READY=false; NORMAL_READY=false; unset NODE_RV LEASE_RV RUNTIME_PROVIDER; local n l; [ "$WINDOW_READY" = true ] || return 1; n="$(kubectl get node "$NODE" -o json)" || return 1; l="$(kubectl get lease "$NODE" -n kube-node-lease -o json)" || return 1; [ "$(jq -r '.metadata.uid' <<<"$n")" = "$NODE_UID" ] && [ "$(jq -r '.spec.holderIdentity // empty' <<<"$l")" = "$NODE" ] || return 1; NODE_RV="$(jq -r '.metadata.resourceVersion' <<<"$n")"; LEASE_RV="$(jq -r '.metadata.resourceVersion' <<<"$l")"; RUNTIME_PROVIDER="$(jq -r '.status.nodeInfo.containerRuntimeVersion|split("://")[0]' <<<"$n")"; API_READY=true; }`

**사람이 대표 UID를 선택할 수 있는 세로 출력**

`api_guard || exit; P="$(kubectl get pods -A --field-selector "spec.nodeName=$NODE" -o json)"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ]; then jq -r '.items|sort_by(.metadata.namespace,.metadata.name)[]|"=== Pod ===\n이름: \(.metadata.namespace)/\(.metadata.name)\nUID/RV: \(.metadata.uid)/\(.metadata.resourceVersion)\nNode: \(.spec.nodeName)\nowner: \([.metadata.ownerReferences[]?|"\(.kind)/\(.name)/\(.uid)"]|join(","))\n생성/삭제: \(.metadata.creationTimestamp)/\(.metadata.deletionTimestamp // "-")\n상태: \(.status.phase)\ncontainer lifecycle: \([.status.containerStatuses[]?|"\(.name) id=\(.containerID // "-") restart=\(.restartCount) state=\((.state|keys)[0] // "-")"]|join(";"))"' <<<"$P"; else printf '중단: producer=%s\n' "$PRODUCER_RC" >&2; fi`

**서로 다른 대표 Pod 두 개 선택과 identity 검증**

`api_guard || exit; read -r -p '대표1 namespace: ' NS1; read -r -p '대표1 Pod: ' POD1; read -r -p '대표2 namespace: ' NS2; read -r -p '대표2 Pod: ' POD2; TARGET_READY=false; J1="$(kubectl get pod "$POD1" -n "$NS1" -o json)" && J2="$(kubectl get pod "$POD2" -n "$NS2" -o json)"; RC=$?; if [ "$RC" -eq 0 ] && [ "$(jq -r '.spec.nodeName' <<<"$J1")" = "$NODE" ] && [ "$(jq -r '.spec.nodeName' <<<"$J2")" = "$NODE" ]; then UID1="$(jq -r '.metadata.uid' <<<"$J1")"; UID2="$(jq -r '.metadata.uid' <<<"$J2")"; [ "$UID1" != "$UID2" ] && TARGET_READY=true; fi; [ "$TARGET_READY" = true ] && printf 'TARGET_READY=true %s/%s=%s %s/%s=%s\n' "$NS1" "$POD1" "$UID1" "$NS2" "$POD2" "$UID2" || { printf '%s\n' '중단: 서로 다른 same-node UID 두 개가 아님' >&2; exit 1; }`

`target_guard(){ TARGET_READY=false; api_guard || return 1; local a b; a="$(kubectl get pod "$POD1" -n "$NS1" -o json)" || return 1; b="$(kubectl get pod "$POD2" -n "$NS2" -o json)" || return 1; [ "$(jq -r '.metadata.uid' <<<"$a")" = "$UID1" ] && [ "$(jq -r '.metadata.uid' <<<"$b")" = "$UID2" ] && [ "$(jq -r '.spec.nodeName' <<<"$a")" = "$NODE" ] && [ "$(jq -r '.spec.nodeName' <<<"$b")" = "$NODE" ] || return 1; TARGET_READY=true; }`

**host identity와 unit**

`target_guard || exit; RID="$(ssh "${SSH_ARGS[@]}" -- sh -c 'printf "hostname=%s machine_id=%s boot_id=%s\\n" "$(hostname -f 2>/dev/null || hostname)" "$(cat /etc/machine-id 2>/dev/null)" "$(cat /proc/sys/kernel/random/boot_id 2>/dev/null)"')"; RC=$?; printf '%s\n' "$RID"; read -r -p '대상 Node이면 Node 이름 입력: ' HC; [ "$RC" -eq 0 ] && [ "$HC" = "$NODE" ] && HOST_IDENTITY="$RID" && HOST_READY=true`

`host_guard(){ HOST_READY=false; target_guard || return 1; local r; r="$(ssh "${SSH_ARGS[@]}" -- sh -c 'printf "hostname=%s machine_id=%s boot_id=%s\\n" "$(hostname -f 2>/dev/null || hostname)" "$(cat /etc/machine-id 2>/dev/null)" "$(cat /proc/sys/kernel/random/boot_id 2>/dev/null)"')" || return 1; [ "$r" = "$HOST_IDENTITY" ] || return 1; HOST_READY=true; }`

`host_guard || exit; O="$(ssh "${SSH_ARGS[@]}" -- systemctl list-unit-files --type=service --no-legend)"; PRODUCER_RC=$?; mapfile -t U < <(grep -E '^kubelet[A-Za-z0-9_.@-]*\.service[[:space:]]' <<<"$O" | awk '{print $1}'); FILTER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ] && [ "$FILTER_RC" -eq 0 ] && [ "${#U[@]}" -eq 1 ]; then KUBELET_UNIT="${U[0]}"; UNIT_READY=true; else exit 1; fi`

## 관측별 진단 경로

검색 문자열: `pod_workers`, `syncPod`, `status manager`, `failed to update status`, `KillPod`, `PLEG`, `runtime service unavailable`, `context deadline exceeded`.

| 관측 | 다음 직접 증거 | 경로 |
|---|---|---|
| 두 UID의 create가 함께 정체 | syncPod·sandbox·mount/CNI 최초 오류 | kubelet→개별 dependency |
| 두 UID의 terminate가 함께 정체 | KillPod·stop timeout·shim | kubelet→runtime |
| runtime state는 변하고 API status만 stale | Pod RV·status manager·audit response | kubelet→API |
| image/volume/CNI 오류가 UID마다 다름 | 공통 장애 결론 중단 | 개별 Pod 런북 |

**두 UID bounded kubelet journal**

`host_guard || exit; [ "$UNIT_READY" = true ] || exit 1; T="$(mktemp)" || exit 1; ssh "${SSH_ARGS[@]}" -- sudo journalctl -u "$KUBELET_UNIT" --since "$START_RFC3339" --until "$END_RFC3339" --no-pager >"$T"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ]; then grep -E -i -C 3 "$UID1|$UID2|pod_workers|syncPod|status manager|failed to update status|PLEG|runtime service unavailable|pod sandbox|KillPod|deadline|timeout" "$T"; FILTER_RC=$?; else FILTER_RC=99; fi; rm -f "$T"; printf 'producer=%s filter=%s\n' "$PRODUCER_RC" "$FILTER_RC"; if [ "$PRODUCER_RC" -eq 0 ] && [ "$FILTER_RC" -eq 0 ]; then EVIDENCE_READY=true; else printf '%s\n' '중단: 무결과/수집 실패는 정상 아님' >&2; fi`

UID는 API에서 얻은 allowlist 형식이므로 regex에 사용하기 전 `^[0-9a-fA-F-]+$`인지 확인한다. audit은 승인된 담당자가 같은 창의 UID, requestURI, response code, stageTimestamp만 확인하고 전체 레코드는 저장하지 않는다.

### CRI relation

이 문서 안의 과거 `CRI_READY`나 socket 이름은 재사용하지 않는다. CRI 상태가 실제 분기를 바꿀 때만 [[07. runtime 상태와 Pod container ID가 일치하지 않는 경우]]에서 current 두 Pod UID·containerID, Node UID·원격 machine-id, active kubelet argv 또는 엄격히 검증한 config의 exact endpoint, runtime MainPID와 socket provider를 새로 등록한다. 그 절의 direct current guard를 통과한 `crictl info/inspect` 요약만 이 문서의 CRI 증거로 연결한다.

container inspect가 필요하면 먼저 fresh Pod status의 containerID를 다시 얻고 runtime scheme을 제거한 exact ID와 inspect의 `io.kubernetes.pod.uid`, sandbox relation, created/finished time을 대조한다. 전체 inspect 원문은 저장하지 않는다.

## Prometheus query

`api_guard || exit; printf 'histogram_quantile(0.99,sum by(le)(rate(kubelet_pod_worker_duration_seconds_bucket{job="kubelet"}[5m])))\n'; printf 'sum by(operation_type)(rate(kubelet_runtime_operations_errors_total{job="kubelet"}[5m]))\n'`

job/instance→Node mapping 확인 전에는 원인 근거로 쓰지 않는다. 빈 vector는 성공이 아니며 histogram은 영구 정체를 과소평가한다.

## 변경·정상화

이 문서는 기본적으로 읽기 전용이다. 별도 변경은 fresh 두 UID·Node/host/CRI/evidence, 승인 window, 모든 owner/PDB/quorum, Job 멱등성·외부 side effect, 대체 capacity와 rollback revision이 있을 때만 해당 owner 절차에서 수행한다. Pod delete, runtime/kubelet restart를 자동 제시하지 않는다.

`NORMAL_READY=false; target_guard || exit; host_guard || exit; [ "$(kubectl get node "$NODE" -o jsonpath='{.metadata.uid}')" = "$NODE_UID" ] || { printf '%s\n' 'replacement Node: 전체 재등록' >&2; exit 1; }; P="$(kubectl get pods -A --field-selector "spec.nodeName=$NODE" -o json)" || exit 1; if jq -e 'all(.items[]; ([.status.containerStatuses[]?.ready]|all))' <<<"$P" >/dev/null; then NORMAL_READY=true; printf 'NORMAL_READY=true node=%s; CRI 정상화는 07 런북의 fresh endpoint/provider gate 결과를 별도 첨부\n' "$NODE"; else printf '%s\n' '중단: lifecycle 검증 실패' >&2; fi`

교체 Pod는 새 UID/containerID/owner를 재등록한다. create·terminate·status가 시간순으로 수렴하고 Lease/Ready, owner desired/Ready, EndpointSlice와 실제 요청·queue·Job 산출물이 정상이어야 한다. 변경 검증 실패 시 추가 mutation을 중단하고 승인된 revision만 rollback한다.

## 보안·환경 경계

최소 Node/Lease/Pod read와 승인된 audit 접근만 쓴다. Secret/token/private key/전체 inspect·journal·audit은 저장하지 않는다. SSH inventory, units, CRI socket, audit, Prometheus labels, PDB/quorum/서비스 기능은 환경 미검증이다.
