# kubelet 로그에 PLEG is not healthy가 기록된 경우

## 입력값과 빠른 수집

hop은 `kubelet PLEG relist → kubelet이 실제 사용하는 CRI unix endpoint → active runtime provider/socket → sandbox·container lifecycle`이다. 별도 hop으로 `kubectl → API server → Node·Lease·여러 Pod`와 `SSH → 검증된 Linux host → systemd/journal/kernel`을 대조한다.

| 입력 종류 | 값 | gate |
|---|---|---|
| 최초 입력 | `NODE`, `SSH_TARGET` | exact Node와 안전한 SSH 목적지 |
| API 자동값 | Node UID/RV, Lease RV·holder, runtime provider, Pod UID·owner·containerID | 매 명령 직전 재조회 |
| 계산 후보값 | kubelet/runtime unit, active CRI endpoint/socket | 단일 추출·provider exact match |
| 운영자 선택값 | canonical UTC 창, host relation, kubelet metric instance mapping, 변경 승인·quorum | round-trip, current Node UID mapping, 승인 기록 |

**stale identity·resourceVersion·owner·containerID·node·host·relation·time·argv/regex·READY 폐기와 입력**

`unset INPUT_READY WINDOW_READY API_READY HOST_READY UNIT_READY PLEG_READY CRI_READY EVIDENCE_READY MUTATION_READY NORMAL_READY NODE_UID NODE_RV NODE_MACHINE_ID LEASE_UID LEASE_RV RUNTIME_PROVIDER HOST_IDENTITY KUBELET_UNIT RUNTIME_UNIT CRI_ENDPOINT CRI_SOCKET START_RFC3339 END_RFC3339 FUNCTION_EVIDENCE_REF; NODE='PASTE_NODE_NAME_HERE'; SSH_TARGET='PASTE_APPROVED_SSH_TARGET_HERE'; MISSING=''; for T in kubectl jq ssh date grep awk sed mktemp; do command -v "$T" >/dev/null 2>&1 || MISSING="$MISSING $T"; done; INPUT_READY=false; if [ -n "${BASH_VERSION:-}" ] && [ -z "$MISSING" ] && [[ "$NODE" =~ ^[A-Za-z0-9._-]+$ ]] && [[ "$SSH_TARGET" =~ ^[A-Za-z0-9_.@:-]+$ ]] && [[ "$SSH_TARGET" != -* ]] && N="$(kubectl get node "$NODE" -o json)" && L="$(kubectl get lease "$NODE" -n kube-node-lease -o json)" && [ "$(jq -r '.metadata.name' <<<"$N")" = "$NODE" ] && [ "$(jq -r '.spec.holderIdentity // empty' <<<"$L")" = "$NODE" ]; then NODE_UID="$(jq -er '.metadata.uid' <<<"$N")" && NODE_MACHINE_ID="$(jq -er '.status.nodeInfo.machineID' <<<"$N")" && LEASE_UID="$(jq -er '.metadata.uid' <<<"$L")" && INPUT_READY=true; SSH_OPTS=(-o BatchMode=yes -o StrictHostKeyChecking=yes -o ConnectTimeout=10); printf 'INPUT_READY=%s node=%s uid=%s machineID=%s\n' "$INPUT_READY" "$NODE" "$NODE_UID" "$NODE_MACHINE_ID"; else printf '중단: 입력/Node·Lease 조회 실패 missing=%s\n' "$MISSING" >&2; fi`

`WINDOW_READY=false; API_READY=false; HOST_READY=false; UNIT_READY=false; PLEG_READY=false; CRI_READY=false; EVIDENCE_READY=false; MUTATION_READY=false; NORMAL_READY=false; START_RFC3339='PASTE_START_UTC_RFC3339_HERE'; END_RFC3339='PASTE_END_UTC_RFC3339_HERE'; S="$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)"; SRC=$?; E="$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)"; ERC=$?; if [ "$INPUT_READY" = true ] && [[ "$START_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && [[ "$END_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && [ "$SRC" -eq 0 ] && [ "$ERC" -eq 0 ] && [ "$S" = "$START_RFC3339" ] && [ "$E" = "$END_RFC3339" ] && [[ "$S" < "$E" ]]; then WINDOW_READY=true; else printf '%s\n' '중단: canonical UTC 창 실패' >&2; fi`

`invalidate_current(){ API_READY=false; HOST_READY=false; UNIT_READY=false; PLEG_READY=false; CRI_READY=false; EVIDENCE_READY=false; MUTATION_READY=false; NORMAL_READY=false; }; api_guard(){ API_READY=false; local n l; [ "$WINDOW_READY" = true ] || { invalidate_current; return 1; }; n="$(kubectl get node "$NODE" -o json)" && l="$(kubectl get lease "$NODE" -n kube-node-lease -o json)" || { invalidate_current; return 1; }; [ "$(jq -r '.metadata.uid' <<<"$n")" = "$NODE_UID" ] && [ "$(jq -r '.status.nodeInfo.machineID' <<<"$n")" = "$NODE_MACHINE_ID" ] && [ "$(jq -r '.metadata.uid' <<<"$l")" = "$LEASE_UID" ] && [ "$(jq -r '.spec.holderIdentity // empty' <<<"$l")" = "$NODE" ] || { invalidate_current; return 1; }; NODE_RV="$(jq -er '.metadata.resourceVersion' <<<"$n")" && LEASE_RV="$(jq -er '.metadata.resourceVersion' <<<"$l")" && RUNTIME_PROVIDER="$(jq -er '.status.nodeInfo.containerRuntimeVersion|split("://")[0]' <<<"$n")" || { invalidate_current; return 1; }; API_READY=true; }`

**blast radius와 lifecycle 관계**

`api_guard || exit; PODS_JSON="$(kubectl get pods -A --field-selector "spec.nodeName=$NODE" -o json)"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ]; then jq -r '.items|sort_by(.metadata.namespace,.metadata.name)[]|"=== Pod ===\n이름: \(.metadata.namespace)/\(.metadata.name)\nUID: \(.metadata.uid)\nNode: \(.spec.nodeName)\nowner: \([.metadata.ownerReferences[]?|"\(.kind)/\(.name)/\(.uid)"]|join(","))\n생성/삭제: \(.metadata.creationTimestamp)/\(.metadata.deletionTimestamp // "-")\ncontainer lifecycle: \([.status.containerStatuses[]?|"\(.name) id=\(.containerID // "-") restart=\(.restartCount) state=\((.state|keys)[0] // "-")"]|join(";"))"' <<<"$PODS_JSON"; else printf '중단: pod producer_rc=%s\n' "$PRODUCER_RC" >&2; fi`

단일 Pod만 멈추면 node-wide PLEG로 확대하지 않는다. Pod 이름이 같아도 UID/containerID가 바뀌면 이전 증거를 폐기한다.

**host identity와 fresh guard**

`api_guard || exit; RID="$(ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" 'printf "hostname=%s machine_id=%s boot_id=%s\n" "$(hostname -f 2>/dev/null || hostname)" "$(cat /etc/machine-id 2>/dev/null)" "$(cat /proc/sys/kernel/random/boot_id 2>/dev/null)"')"; RC=$?; printf '%s\n' "$RID"; read -r -p '대상 Node가 맞으면 Node 이름 입력: ' HC; if [ "$RC" -eq 0 ] && [ "$HC" = "$NODE" ] && [[ "$RID" =~ ^hostname=[^[:space:]]+[[:space:]]machine_id=[0-9a-fA-F-]+[[:space:]]boot_id=[0-9a-fA-F-]+$ ]] && [[ "$RID" == *"machine_id=$NODE_MACHINE_ID "* ]]; then HOST_IDENTITY="$RID"; HOST_READY=true; else printf '%s\n' '중단: remote /etc/machine-id와 Node identity 불일치' >&2; fi`

`host_guard(){ HOST_READY=false; api_guard || return 1; local r; r="$(ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" 'printf "hostname=%s machine_id=%s boot_id=%s\n" "$(hostname -f 2>/dev/null || hostname)" "$(cat /etc/machine-id 2>/dev/null)" "$(cat /proc/sys/kernel/random/boot_id 2>/dev/null)"')" || { invalidate_current; return 1; }; [ "$r" = "$HOST_IDENTITY" ] && [[ "$r" == *"machine_id=$NODE_MACHINE_ID "* ]] || { invalidate_current; return 1; }; HOST_READY=true; }`

`host_guard || exit; OUT="$(ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" systemctl list-unit-files --type=service --no-legend)"; PRODUCER_RC=$?; U=(); if [ "$PRODUCER_RC" -eq 0 ]; then MATCHES="$(grep -E '^kubelet[A-Za-z0-9_.@-]*\.service[[:space:]]' <<<"$OUT")"; FILTER_RC=$?; [ "$FILTER_RC" -eq 0 ] && mapfile -t U < <(awk '{print $1}' <<<"$MATCHES"); else FILTER_RC=99; fi; if [ "$PRODUCER_RC" -eq 0 ] && [ "$FILTER_RC" -eq 0 ] && [ "${#U[@]}" -eq 1 ] && [[ "${U[0]}" =~ ^kubelet[A-Za-z0-9_.@-]*\.service$ ]]; then KUBELET_UNIT="${U[0]}"; UNIT_READY=true; else printf '중단: kubelet unit producer=%s filter=%s count=%s\n' "$PRODUCER_RC" "$FILTER_RC" "${#U[@]}" >&2; fi; unit_guard(){ UNIT_READY=false; host_guard || return 1; local s; s="$(ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" systemctl show "$KUBELET_UNIT" -p Id -p LoadState)" || { invalidate_current; return 1; }; [ "$(sed -n 's/^Id=//p' <<<"$s")" = "$KUBELET_UNIT" ] && [ "$(sed -n 's/^LoadState=//p' <<<"$s")" = loaded ] || { invalidate_current; return 1; }; UNIT_READY=true; }`

## 진입 판단과 관측 경로

검색 문자열: `PLEG is not healthy`, `GenericPLEG`, `relist`, `rpc error`, `context deadline exceeded`, `runtime service unavailable`, `I/O error`, `read-only`, `Out of memory`.

| 관측 | 직접 판별 | 다음 경로 |
|---|---|---|
| PLEG와 CRI timeout 동시 | active endpoint의 `crictl info`·runtime journal | runtime |
| PLEG와 여러 Pod status 지연 | UID별 lifecycle과 Lease | kubelet/runtime |
| runtime 응답 정상, kernel I/O/OOM | 같은 UTC kernel 창 | Linux/storage |
| status·Lease API 오류가 주 증거 | PLEG 원인 확정 금지 | [[03. kubelet이 Node 상태·Lease를 갱신하지 못하는 경우]] |
| 한 Pod만 Unknown | node-wide 결론 금지 | Pod lifecycle 런북 |

**PLEG 직접 증거, producer/filter rc 분리**

`unit_guard || exit; TMP="$(mktemp)" || exit 1; ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" sudo journalctl -u "$KUBELET_UNIT" --since "$START_RFC3339" --until "$END_RFC3339" --no-pager >"$TMP"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ]; then grep -E -i -C 3 'PLEG is not healthy|GenericPLEG|relist|container runtime|rpc error|context deadline exceeded|timeout|pod worker' "$TMP"; FILTER_RC=$?; else FILTER_RC=99; fi; rm -f "$TMP"; printf 'producer_rc=%s filter_rc=%s\n' "$PRODUCER_RC" "$FILTER_RC"; if [ "$PRODUCER_RC" -eq 0 ] && [ "$FILTER_RC" -eq 0 ]; then PLEG_READY=true; EVIDENCE_READY=true; else printf '%s\n' '중단: 수집 실패/무결과는 정상 아님' >&2; fi`

**active kubelet endpoint 엄격 추출과 runtime socket provider 관계**

`unit_guard || exit; [ "$PLEG_READY" = true ] || exit 1; ARGV="$(ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" systemctl show "$KUBELET_UNIT" -p ExecStart --value)"; PRODUCER_RC=$?; EP_MATCHES="$(grep -Eo -- 'unix://[^ ;}]+' <<<"$ARGV")"; FILTER_RC=$?; EPS=(); [ "$FILTER_RC" -eq 0 ] && mapfile -t EPS < <(sort -u <<<"$EP_MATCHES"); if [ "$PRODUCER_RC" -eq 0 ] && [ "$FILTER_RC" -eq 0 ] && [ "${#EPS[@]}" -eq 1 ] && [[ "${EPS[0]}" =~ ^unix:///[-A-Za-z0-9._/]+\.sock$ ]]; then CRI_ENDPOINT="${EPS[0]}"; CRI_SOCKET="${CRI_ENDPOINT#unix://}"; else printf '중단: active ExecStart endpoint producer=%s filter=%s count=%s; 추측/fallback 금지\n' "$PRODUCER_RC" "$FILTER_RC" "${#EPS[@]}" >&2; exit 1; fi; case "$RUNTIME_PROVIDER" in containerd) RUNTIME_UNIT=containerd.service;; cri-o) RUNTIME_UNIT=crio.service;; *) printf '중단: provider=%s\n' "$RUNTIME_PROVIDER" >&2; exit 1;; esac; cri_guard(){ CRI_READY=false; unit_guard || return 1; local rs ss stat_out pid line; rs="$(ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" systemctl show "$RUNTIME_UNIT" -p ActiveState -p MainPID)" && ss="$(ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" sudo ss -lxnp)" && stat_out="$(ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" sudo stat -c '%F %n' "$CRI_SOCKET")" || return 1; pid="$(sed -n 's/^MainPID=//p' <<<"$rs")"; line="$(grep -F "$CRI_SOCKET" <<<"$ss")" || return 1; [ "$(sed -n 's/^ActiveState=//p' <<<"$rs")" = active ] && [[ "$pid" =~ ^[1-9][0-9]*$ ]] && grep -Fq "pid=$pid," <<<"$line" && grep -Fq "socket $CRI_SOCKET" <<<"$stat_out" || return 1; CRI_READY=true; }; if cri_guard; then printf 'CRI_READY=true provider=%s unit=%s endpoint=%s\n' "$RUNTIME_PROVIDER" "$RUNTIME_UNIT" "$CRI_ENDPOINT"; else printf '%s\n' '중단: provider MainPID/socket owner 관계 실패' >&2; fi`

`cri_current_guard(){ CRI_READY=false; unit_guard || return 1; local argv_now matches rc expected_unit; argv_now="$(ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" systemctl show "$KUBELET_UNIT" -p ExecStart --value)" || return 1; matches="$(grep -Eo -- 'unix://[^ ;}]+' <<<"$argv_now")"; rc=$?; [ "$rc" -eq 0 ] || return 1; CURRENT_EPS=(); mapfile -t CURRENT_EPS < <(sort -u <<<"$matches"); [ "${#CURRENT_EPS[@]}" -eq 1 ] && [ "${CURRENT_EPS[0]}" = "$CRI_ENDPOINT" ] || return 1; case "$RUNTIME_PROVIDER" in containerd) expected_unit=containerd.service;; cri-o) expected_unit=crio.service;; *) return 1;; esac; [ "$RUNTIME_UNIT" = "$expected_unit" ] || return 1; cri_guard; }`

`cri_current_guard || exit 1; ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" sudo crictl --runtime-endpoint="$CRI_ENDPOINT" info >/dev/null; CRI_RC=$?; printf 'cri_info_rc=%s endpoint=%s\n' "$CRI_RC" "$CRI_ENDPOINT"; [ "$CRI_RC" -eq 0 ] || exit 1`

## Prometheus query

`KUBELET_INSTANCE='PASTE_CONFIRMED_KUBELET_METRIC_INSTANCE_HERE'; METRIC_MAPPING_EVIDENCE='PASTE_APPROVED_METRIC_MAPPING_EVIDENCE_HERE'; QUERY_READY=false; if ! api_guard || [ -z "$KUBELET_INSTANCE" ] || [ -z "$METRIC_MAPPING_EVIDENCE" ]; then printf '%s\n' '중단: current Node UID와 kubelet metric instance의 승인 mapping이 필요하다.' >&2; else WINDOW_SECONDS="$(( $(date -u -d "$END_RFC3339" +%s) - $(date -u -d "$START_RFC3339" +%s) ))"; INSTANCE_LABEL="$(printf '%s' "$KUBELET_INSTANCE" | sed 's/\\/\\\\/g;s/"/\\"/g')"; QUERY_READY=true; printf 'histogram_quantile(0.99,sum by(le)(rate(kubelet_pleg_relist_duration_seconds_bucket{job="kubelet",instance="%s"}[%ss])))\n' "$INSTANCE_LABEL" "$WINDOW_SECONDS"; printf 'sum by(operation_type)(rate(kubelet_runtime_operations_errors_total{job="kubelet",instance="%s"}[%ss]))\n' "$INSTANCE_LABEL" "$WINDOW_SECONDS"; printf 'QUERY_READY=true node_uid=%s mapping=%s\n' "$NODE_UID" "$METRIC_MAPPING_EVIDENCE"; fi`

`METRIC_MAPPING_EVIDENCE`는 current Node UID와 exact `instance` label을 연결한 승인된 target/discovery 증거다. 출력한 완성 PromQL을 `END_RFC3339` 시각에 평가한다. 빈 metric은 성공이 아니라 scrape/label/권한/범위 실패다. completed histogram은 영구 정체를 과소평가하므로 journal이 직접 근거다.

## 변경·rollback·정상화

runtime/host 원인이 없고 kubelet PLEG 자체가 진행하지 않는 직접 근거가 있을 때만 fresh Node/host/CRI/evidence, 승인 window, PDB/quorum, Job 멱등성, 대체 capacity를 확인한다.

`MUTATION_READY=false; cri_current_guard || exit; [ "$PLEG_READY" = true ] || exit 1; PODS_NOW="$(kubectl get pods -A --field-selector "spec.nodeName=$NODE" -o json)" || exit 1; jq -r '.items[]|"\(.metadata.namespace)/\(.metadata.name) uid=\(.metadata.uid) owner=\([.metadata.ownerReferences[]?|"\(.kind)/\(.name)/\(.uid)"]|join(","))"' <<<"$PODS_NOW"; read -r -p '승인 ID: ' APPROVAL_ID; read -r -p '변경 창: ' CHANGE_WINDOW; read -r -p '위 current owner의 quorum/PDB/멱등성 확인 후 Node 이름 입력: ' QC; [ -n "$APPROVAL_ID" ] && [ -n "$CHANGE_WINDOW" ] && [ "$QC" = "$NODE" ] && MUTATION_READY=true; [ "$MUTATION_READY" = true ] || exit 1; ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" sudo systemctl restart "$KUBELET_UNIT"; RC=$?; printf 'mutation_rc=%s node_uid=%s node_rv=%s\n' "$RC" "$NODE_UID" "$NODE_RV"; [ "$RC" -eq 0 ] || exit 1`

반복 restart는 rollback이 아니다. 승인된 config/unit 변경만 이전 revision으로 되돌리며 PLEG 재발, CRI 실패, Ready 감소, quorum·Job 이상이면 추가 변경을 중단한다.

`FUNCTION_EVIDENCE_REF='PASTE_APPROVED_FUNCTION_EVIDENCE_HERE'; NORMAL_READY=false; cri_current_guard || exit; [ "$(kubectl get node "$NODE" -o jsonpath='{.metadata.uid}')" = "$NODE_UID" ] || { printf '%s\n' 'replacement Node: 전체 identity 재등록' >&2; exit 1; }; LEASE_1="$(kubectl get lease "$NODE" -n kube-node-lease -o jsonpath='{.spec.renewTime}')" || exit 1; CHECK_START="$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; sleep 60; cri_current_guard || exit; LEASE_2="$(kubectl get lease "$NODE" -n kube-node-lease -o jsonpath='{.spec.renewTime}')" || exit 1; NODE_NOW="$(kubectl get node "$NODE" -o json)" || exit 1; ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" sudo crictl --runtime-endpoint="$CRI_ENDPOINT" info >/dev/null; RC=$?; P="$(kubectl get pods -A --field-selector "spec.nodeName=$NODE" -o json)" || exit 1; TMP="$(mktemp)" || exit 1; ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" sudo journalctl -u "$KUBELET_UNIT" --since "$CHECK_START" --no-pager >"$TMP"; JOURNAL_RC=$?; if [ "$JOURNAL_RC" -eq 0 ]; then grep -E -i 'PLEG is not healthy|GenericPLEG.*relist.*(failed|timeout)|runtime service unavailable' "$TMP" >/dev/null; PLEG_FILTER_RC=$?; else PLEG_FILTER_RC=99; fi; rm -f "$TMP"; if [ "$RC" -eq 0 ] && [ "$JOURNAL_RC" -eq 0 ] && [ "$PLEG_FILTER_RC" -eq 1 ] && [[ "$LEASE_1" < "$LEASE_2" ]] && [ -n "$FUNCTION_EVIDENCE_REF" ] && jq -e 'any(.status.conditions[]; .type=="Ready" and .status=="True")' <<<"$NODE_NOW" >/dev/null && jq -e 'all(.items[]; ([.status.containerStatuses[]?.ready]|all))' <<<"$P" >/dev/null; then NORMAL_READY=true; printf 'NORMAL_READY=true node=%s uid=%s function=%s\n' "$NODE" "$NODE_UID" "$FUNCTION_EVIDENCE_REF"; else printf '중단: CRI/PLEG/Lease/Ready/Pod/실제 기능 검증 실패 journal_rc=%s filter_rc=%s\n' "$JOURNAL_RC" "$PLEG_FILTER_RC" >&2; fi`

새 PLEG 부재, Lease/Ready 진행, owner desired/Ready, EndpointSlice와 실제 요청·queue·Job 결과까지 확인한다. replacement Node는 새 UID/host/runtime/Pod owner/containerID를 재등록한다.

## 보안·환경 경계

`api_guard || exit; kubectl auth can-i get "node/$NODE"; kubectl auth can-i get "lease/$NODE" -n kube-node-lease; kubectl auth can-i list pods --all-namespaces`

작업 PC의 현재 인증 subject에 `nodes get`, `leases get`(namespace `kube-node-lease`, resource name `$NODE`), `pods list`만 필요한 조회 범위로 확인한다. `crictl`, `journalctl`, `systemctl`은 Kubernetes RBAC이 아니라 승인된 대상 host의 SSH/sudo 정책 범위다. `can-i=yes`는 실제 요청 성공이나 동일 subject를 보장하지 않으므로 승인된 auth inventory 또는 bounded audit에서 subject를 대조한다. Secret/token/private key/kubeconfig/전체 journal·audit은 저장하지 않는다. SSH inventory, unit/endpoint, socket process 표기, journal 보존, Prometheus label, quorum/서비스 기능은 환경 미검증이다.
