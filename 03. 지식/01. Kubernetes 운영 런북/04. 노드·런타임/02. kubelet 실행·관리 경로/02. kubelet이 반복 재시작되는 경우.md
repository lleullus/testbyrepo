# kubelet이 반복 재시작되는 경우

## 입력값과 빠른 수집

hop은 `작업 PC Bash → SSH host key → Linux host → systemd kubelet`과 `kubectl → API server → Node·Lease·Pod`다. 안전한 SSH 배열과 엄격히 검증한 인수만 사용하며 `eval`은 금지한다.

| 입력 종류 | 값 | gate |
|---|---|---|
| 최초 입력 | `NODE`, `SSH_TARGET` | exact Node와 SSH allowlist |
| API 확인 자동값 | Node UID/resourceVersion/machineID, Lease UID/resourceVersion·holder, runtime, Pod UID·owner·containerID | 매 명령 직전 fresh API exact match |
| 계산 후보값 | 실제 kubelet unit, escaped `NODE_RE` | 후보 하나, argv/unit allowlist |
| 운영자 선택값 | UTC 증거 창, host 관계, kubelet metric instance mapping, 변경 승인·quorum 판단 | RFC3339 round-trip, current Node UID mapping, 승인 기록 |

**도구와 입력, 모든 stale 상태 폐기**

`unset INPUT_READY API_READY HOST_READY UNIT_READY WINDOW_READY CYCLE_READY CRI_READY EVIDENCE_READY MUTATION_READY NORMAL_READY NODE_UID NODE_RV NODE_MACHINE_ID LEASE_UID LEASE_RV HOST_IDENTITY KUBELET_UNIT RUNTIME_PROVIDER CRI_ENDPOINT CRI_SOCKET RUNTIME_UNIT START_RFC3339 END_RFC3339 FUNCTION_EVIDENCE_REF; NODE='PASTE_NODE_NAME_HERE'; SSH_TARGET='PASTE_APPROVED_SSH_TARGET_HERE'; INPUT_READY=false; MISSING=''; for T in kubectl jq ssh date grep awk sed mktemp; do command -v "$T" >/dev/null 2>&1 || MISSING="$MISSING $T"; done; if [ -n "${BASH_VERSION:-}" ] && [ -z "$MISSING" ] && [[ "$NODE" =~ ^[A-Za-z0-9._-]+$ ]] && [[ "$SSH_TARGET" =~ ^[A-Za-z0-9_.@:-]+$ ]] && [[ "$SSH_TARGET" != -* ]] && N="$(kubectl get node "$NODE" -o json)" && L="$(kubectl get lease "$NODE" -n kube-node-lease -o json)" && [ "$(jq -r '.metadata.name' <<<"$N")" = "$NODE" ] && [ "$(jq -r '.spec.holderIdentity // empty' <<<"$L")" = "$NODE" ]; then NODE_UID="$(jq -er '.metadata.uid' <<<"$N")" && NODE_MACHINE_ID="$(jq -er '.status.nodeInfo.machineID' <<<"$N")" && LEASE_UID="$(jq -er '.metadata.uid' <<<"$L")" && INPUT_READY=true; SSH_OPTS=(-o BatchMode=yes -o StrictHostKeyChecking=yes -o ConnectTimeout=10); printf 'INPUT_READY=true node=%s uid=%s machineID=%s\n' "$NODE" "$NODE_UID" "$NODE_MACHINE_ID"; else printf '중단: 입력/도구/Node·Lease API 실패 missing=%s\n' "$MISSING" >&2; fi`

**canonical UTC 창**

`WINDOW_READY=false; API_READY=false; HOST_READY=false; UNIT_READY=false; CYCLE_READY=false; CRI_READY=false; EVIDENCE_READY=false; MUTATION_READY=false; NORMAL_READY=false; START_RFC3339='PASTE_START_UTC_RFC3339_HERE'; END_RFC3339='PASTE_END_UTC_RFC3339_HERE'; S="$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)"; SRC=$?; E="$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)"; ERC=$?; if [ "$INPUT_READY" = true ] && [[ "$START_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && [[ "$END_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && [ "$SRC" -eq 0 ] && [ "$ERC" -eq 0 ] && [ "$S" = "$START_RFC3339" ] && [ "$E" = "$END_RFC3339" ] && [[ "$S" < "$E" ]]; then WINDOW_READY=true; printf 'WINDOW_READY=true %s..%s\n' "$S" "$E"; else printf '%s\n' '중단: UTC 창 검증 실패' >&2; fi`

`invalidate_current(){ API_READY=false; HOST_READY=false; UNIT_READY=false; CYCLE_READY=false; CRI_READY=false; EVIDENCE_READY=false; MUTATION_READY=false; NORMAL_READY=false; }; api_guard(){ API_READY=false; local n l; [ "$WINDOW_READY" = true ] || { invalidate_current; return 1; }; n="$(kubectl get node "$NODE" -o json)" && l="$(kubectl get lease "$NODE" -n kube-node-lease -o json)" || { invalidate_current; return 1; }; [ "$(jq -r '.metadata.uid' <<<"$n")" = "$NODE_UID" ] && [ "$(jq -r '.status.nodeInfo.machineID' <<<"$n")" = "$NODE_MACHINE_ID" ] && [ "$(jq -r '.metadata.uid' <<<"$l")" = "$LEASE_UID" ] && [ "$(jq -r '.spec.holderIdentity // empty' <<<"$l")" = "$NODE" ] || { invalidate_current; return 1; }; NODE_RV="$(jq -er '.metadata.resourceVersion' <<<"$n")" && LEASE_RV="$(jq -er '.metadata.resourceVersion' <<<"$l")" && RUNTIME_PROVIDER="$(jq -er '.status.nodeInfo.containerRuntimeVersion|split("://")[0]' <<<"$n")" || { invalidate_current; return 1; }; API_READY=true; }`

**Node·Pod lifecycle와 owner 관계 수집**

`api_guard || exit; PODS_JSON="$(kubectl get pods -A --field-selector "spec.nodeName=$NODE" -o json)"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ]; then jq -r '.items|sort_by(.metadata.namespace,.metadata.name)[]|"=== Pod ===\n이름: \(.metadata.namespace)/\(.metadata.name)\nUID: \(.metadata.uid)\nNode: \(.spec.nodeName)\nowner: \([.metadata.ownerReferences[]?|"\(.kind)/\(.name)/\(.uid)"]|join(","))\ncontainerID: \([.status.containerStatuses[]?|"\(.name)=\(.containerID // "-")"]|join(";"))\nphase/ready/restart: \(.status.phase)/\([.status.containerStatuses[]?.ready]|join(","))/\([.status.containerStatuses[]?.restartCount]|join(","))"' <<<"$PODS_JSON"; else printf '중단: Pod producer_rc=%s\n' "$PRODUCER_RC" >&2; fi`

재수집 전 UID/resourceVersion/owner/containerID/node/host/time/argv·regex와 downstream READY를 모두 폐기한다. 이름이 같은 replacement Pod의 이전 containerID를 재사용하지 않는다.

**host identity 관계 승인과 fresh guard**

`api_guard || exit; RID="$(ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" 'printf "hostname=%s machine_id=%s boot_id=%s\n" "$(hostname -f 2>/dev/null || hostname)" "$(cat /etc/machine-id 2>/dev/null)" "$(cat /proc/sys/kernel/random/boot_id 2>/dev/null)"')"; RRC=$?; printf '%s\n' "$RID"; read -r -p '대상 Node가 맞으면 Node 이름 입력: ' HC; if [ "$RRC" -eq 0 ] && [ "$HC" = "$NODE" ] && [[ "$RID" =~ ^hostname=[^[:space:]]+[[:space:]]machine_id=[0-9a-fA-F-]+[[:space:]]boot_id=[0-9a-fA-F-]+$ ]] && [[ "$RID" == *"machine_id=$NODE_MACHINE_ID "* ]]; then HOST_IDENTITY="$RID"; HOST_READY=true; else printf '%s\n' '중단: remote /etc/machine-id와 Node 관계 미승인' >&2; fi`

`host_guard(){ HOST_READY=false; api_guard || return 1; local r; r="$(ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" 'printf "hostname=%s machine_id=%s boot_id=%s\n" "$(hostname -f 2>/dev/null || hostname)" "$(cat /etc/machine-id 2>/dev/null)" "$(cat /proc/sys/kernel/random/boot_id 2>/dev/null)"')" || { invalidate_current; return 1; }; [ "$r" = "$HOST_IDENTITY" ] && [[ "$r" == *"machine_id=$NODE_MACHINE_ID "* ]] || { invalidate_current; return 1; }; HOST_READY=true; }`

**unit 엄격 확정**

`host_guard || exit; OUT="$(ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" systemctl list-unit-files --type=service --no-legend)"; PRODUCER_RC=$?; UNITS=(); if [ "$PRODUCER_RC" -eq 0 ]; then MATCHES="$(grep -E '^kubelet[A-Za-z0-9_.@-]*\.service[[:space:]]' <<<"$OUT")"; FILTER_RC=$?; [ "$FILTER_RC" -eq 0 ] && mapfile -t UNITS < <(awk '{print $1}' <<<"$MATCHES"); else FILTER_RC=99; fi; if [ "$PRODUCER_RC" -eq 0 ] && [ "$FILTER_RC" -eq 0 ] && [ "${#UNITS[@]}" -eq 1 ] && [[ "${UNITS[0]}" =~ ^kubelet[A-Za-z0-9_.@-]*\.service$ ]]; then KUBELET_UNIT="${UNITS[0]}"; UNIT_READY=true; printf 'UNIT_READY=true unit=%s\n' "$KUBELET_UNIT"; else printf '중단: producer=%s filter=%s candidates=%s\n' "$PRODUCER_RC" "$FILTER_RC" "${#UNITS[@]}" >&2; fi; unit_guard(){ UNIT_READY=false; host_guard || return 1; local s; s="$(ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" systemctl show "$KUBELET_UNIT" -p Id -p LoadState)" || { invalidate_current; return 1; }; [ "$(sed -n 's/^Id=//p' <<<"$s")" = "$KUBELET_UNIT" ] && [ "$(sed -n 's/^LoadState=//p' <<<"$s")" = loaded ] || { invalidate_current; return 1; }; UNIT_READY=true; }`

## 진입 판단과 진단 경로

검색 문자열: `NRestarts`, `Main process exited`, `Start request repeated too quickly`, `watchdog`, `OOM`, `failed to load`, `x509`, `cgroup`, `runtime`, `failed to register node`.

| 관측 | 직접 판별 | 다음 owner |
|---|---|---|
| 각 cycle 같은 non-zero exit와 config 오류 | cycle start/exit와 parser message | config 배포 |
| certificate/clock 오류 반복 | notBefore/notAfter와 host UTC | 보안·시간 |
| CRI timeout이 exit 직전에 반복 | active endpoint와 provider socket | runtime |
| watchdog/OOM/dependency | kernel/systemd 동일 창 | Linux |
| API timeout/register 실패 | endpoint·route·TLS와 Lease | network/control-plane |

**restart cycle과 argv 증거**

`unit_guard || exit; STATE="$(ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" systemctl show "$KUBELET_UNIT" -p ActiveState -p SubState -p NRestarts -p Result -p ExecMainStatus -p ExecMainStartTimestamp -p ExecMainExitTimestamp -p ExecStart)"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ] && grep -Eq '^NRestarts=[1-9][0-9]*$' <<<"$STATE"; then FILTER_RC=0; CYCLE_READY=true; EVIDENCE_READY=true; printf '%s\n' "$STATE" | sed -E 's/(token|password|secret)=[^ ;]+/\1=REDACTED/Ig'; else FILTER_RC=$?; printf '중단: producer=%s restart_filter=%s\n' "$PRODUCER_RC" "$FILTER_RC" >&2; fi`

**cycle bounded journal, producer/filter rc 분리**

`unit_guard || exit; STATE_NOW="$(ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" systemctl show "$KUBELET_UNIT" -p NRestarts)" || exit 1; grep -Eq '^NRestarts=[1-9][0-9]*$' <<<"$STATE_NOW" || { printf '%s\n' '중단: fresh restart cycle evidence가 없음' >&2; exit 1; }; CYCLE_READY=true; TMP="$(mktemp)" || exit 1; ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" sudo journalctl -u "$KUBELET_UNIT" --since "$START_RFC3339" --until "$END_RFC3339" --no-pager >"$TMP"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ]; then grep -E -i -C 3 'starting|started|main process exited|failed|watchdog|oom|config|unknown field|certificate|x509|cgroup|runtime|cri|register node|timeout|refused|no route' "$TMP"; FILTER_RC=$?; else FILTER_RC=99; fi; rm -f "$TMP"; printf 'producer_rc=%s filter_rc=%s\n' "$PRODUCER_RC" "$FILTER_RC"; [ "$PRODUCER_RC" -eq 0 ] || exit 1; [ "$FILTER_RC" -eq 0 ] || printf '%s\n' '무결과는 안정이 아니며 bounded 원문 owner 확인 필요' >&2`

두 cycle 이상의 시작·종료·직접 오류가 같은 순서여야 원인을 연결한다. `Start request repeated too quickly`는 근본 원인이 아니다.

### CRI direct branch

CRI timeout이 cycle 종료 직전에 반복되는 직접 증거가 있을 때만 [[04. 노드·런타임/03. 컨테이너 런타임·CRI 이상/02. CRI endpoint 연결이 거부되거나 timeout인 경우]]로 이동한다. 그 절에서 매 `crictl` 직전에 active kubelet `ExecStart` 또는 strict fallback config의 단일 endpoint를 다시 추출하고, runtime MainPID가 socket owner인지 exact match한다. 이 문서에 저장한 endpoint·provider·READY를 재사용하거나 기본 socket을 추측하지 않는다.

## Prometheus 빠른 관측

`KUBELET_INSTANCE='PASTE_CONFIRMED_KUBELET_METRIC_INSTANCE_HERE'; METRIC_MAPPING_EVIDENCE='PASTE_APPROVED_METRIC_MAPPING_EVIDENCE_HERE'; QUERY_READY=false; if ! api_guard || [ -z "$KUBELET_INSTANCE" ] || [ -z "$METRIC_MAPPING_EVIDENCE" ]; then printf '%s\n' '중단: current Node UID와 kubelet metric instance의 승인 mapping이 필요하다.' >&2; else WINDOW_SECONDS="$(( $(date -u -d "$END_RFC3339" +%s) - $(date -u -d "$START_RFC3339" +%s) ))"; INSTANCE_LABEL="$(printf '%s' "$KUBELET_INSTANCE" | sed 's/\\/\\\\/g;s/"/\\"/g')"; QUERY_READY=true; printf 'changes(process_start_time_seconds{job="kubelet",instance="%s"}[%ss])\n' "$INSTANCE_LABEL" "$WINDOW_SECONDS"; printf 'kube_node_status_condition{node="%s",condition="Ready",status=~"false|unknown"} == 1\n' "$NODE"; printf 'QUERY_READY=true node_uid=%s mapping=%s\n' "$NODE_UID" "$METRIC_MAPPING_EVIDENCE"; fi`

`METRIC_MAPPING_EVIDENCE`는 current Node UID와 exact `instance` label을 연결한 승인된 target/discovery 증거다. 출력한 완성 PromQL을 `END_RFC3339` 시각에 평가한다. 빈 vector는 성공이 아니며 metric/label/권한/범위 실패로 분기한다. journal의 `NRestarts`가 재시작 횟수의 기준이다.

## 제한적 변경·rollback·정상화

fresh target/evidence, 승인 ID·UTC window, PDB/quorum, Job/CronJob 외부 side effect와 멱등성, 대체 capacity를 모두 확인해야 한 번의 controlled restart를 허용한다.

`MUTATION_READY=false; unit_guard || exit; CURRENT_CYCLE="$(ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" systemctl show "$KUBELET_UNIT" -p Id -p ActiveState -p NRestarts -p Result -p ExecMainStatus)" || exit 1; grep -Eq '^NRestarts=[1-9][0-9]*$' <<<"$CURRENT_CYCLE" || { printf '%s\n' '중단: fresh restart cycle evidence가 없음' >&2; exit 1; }; CYCLE_READY=true; EVIDENCE_READY=true; PODS_NOW="$(kubectl get pods -A --field-selector "spec.nodeName=$NODE" -o json)" || exit 1; jq -r '.items[]|"\(.metadata.namespace)/\(.metadata.name) uid=\(.metadata.uid) owner=\([.metadata.ownerReferences[]?|"\(.kind)/\(.name)/\(.uid)"]|join(","))"' <<<"$PODS_NOW"; read -r -p '승인 ID: ' APPROVAL_ID; read -r -p '승인된 변경 창: ' CHANGE_WINDOW; read -r -p '위 current owner의 quorum/PDB/멱등성 확인 후 Node 이름 입력: ' QC; [ -n "$APPROVAL_ID" ] && [ -n "$CHANGE_WINDOW" ] && [ "$QC" = "$NODE" ] && MUTATION_READY=true; [ "$MUTATION_READY" = true ] || exit 1; ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" sudo systemctl restart "$KUBELET_UNIT"; MUTATION_RC=$?; printf 'mutation_rc=%s approval=%s node_uid=%s node_rv=%s\n' "$MUTATION_RC" "$APPROVAL_ID" "$NODE_UID" "$NODE_RV"; [ "$MUTATION_RC" -eq 0 ] || exit 1`

재시작 반복은 rollback이 아니다. config/unit/certificate 변경은 승인된 이전 revision으로만 rollback하고 수동 덮어쓰기나 `reset-failed`는 금지한다. cycle 재발, Ready 감소, quorum/Job 중복 징후가 하나라도 나오면 즉시 중단한다.

`FUNCTION_EVIDENCE_REF='PASTE_APPROVED_FUNCTION_EVIDENCE_HERE'; NORMAL_READY=false; unit_guard || exit; POST_1="$(ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" systemctl show "$KUBELET_UNIT" -p ActiveState -p NRestarts -p ExecMainStatus)" || exit 1; LEASE_1="$(kubectl get lease "$NODE" -n kube-node-lease -o jsonpath='{.spec.renewTime}')" || exit 1; sleep 60; unit_guard || exit; POST_2="$(ssh "${SSH_OPTS[@]}" -- "$SSH_TARGET" systemctl show "$KUBELET_UNIT" -p ActiveState -p NRestarts -p ExecMainStatus)" || exit 1; LEASE_2="$(kubectl get lease "$NODE" -n kube-node-lease -o jsonpath='{.spec.renewTime}')" || exit 1; NODE_NOW="$(kubectl get node "$NODE" -o json)" || exit 1; PODS="$(kubectl get pods -A --field-selector "spec.nodeName=$NODE" -o json)" || exit 1; if [ "$POST_1" = "$POST_2" ] && grep -q '^ActiveState=active$' <<<"$POST_2" && [[ "$LEASE_1" < "$LEASE_2" ]] && [ -n "$FUNCTION_EVIDENCE_REF" ] && jq -e 'any(.status.conditions[]; .type=="Ready" and .status=="True")' <<<"$NODE_NOW" >/dev/null && jq -e 'all(.items[]; ([.status.containerStatuses[]?.ready]|all))' <<<"$PODS" >/dev/null; then NORMAL_READY=true; printf 'NORMAL_READY=true node=%s uid=%s lease=%s function=%s\n' "$NODE" "$NODE_UID" "$LEASE_2" "$FUNCTION_EVIDENCE_REF"; else printf '%s\n' '중단: restart/process/Lease/Ready/Pod/실제 기능 검증 실패' >&2; fi`

관찰 창 동안 `NRestarts`가 더 늘지 않고 Lease·Ready, owner desired/Ready, EndpointSlice, 실제 요청·queue·Job 산출물이 정상이어야 종료한다. replacement는 새 Node UID·host·runtime·Pod owner/containerID 관계를 처음부터 등록한다.

## 보안·감사·환경 경계

최소 RBAC으로 Node/Lease/Pod만 읽는다. audit은 승인된 담당자가 bounded UTC 창의 subject·requestURI·response code만 본다. Secret/token/private key/kubeconfig 원문/전체 journal은 저장하지 않는다. SSH inventory, unit/restart policy, runtime socket provider, Prometheus label, PDB/quorum/서비스 기능은 환경 미검증이다.
