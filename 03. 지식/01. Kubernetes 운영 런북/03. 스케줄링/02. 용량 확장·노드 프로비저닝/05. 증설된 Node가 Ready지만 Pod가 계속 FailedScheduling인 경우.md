# 증설된 Node가 Ready지만 Pod가 계속 FailedScheduling인 경우

## 입력값과 빠른 수집

실행 경로는 `승인된 작업 PC/Bash → Kubernetes API → original Pod·owner / 새 Node / Event / 조건부 PVC·PV·StorageClass·VolumeAttachment`다. `Ready=True`는 kubelet readiness일 뿐 해당 Pod의 scheduler 후보, volume topology 또는 실제 서비스 capacity를 보장하지 않는다.

| 구분 | 값 | 출처와 gate |
|---|---|---|
| 최초 입력 | `NS`, `POD`, `NODE`, `PROVISIONING_ID` | incident 기록의 original Pod와 승인된 Node 공급 record |
| API 확인 자동값 | Pod UID/RV·owner와 owner 현재 UID/RV, Node UID/RV/providerID·Ready 전환, Event | 모든 exact API 호출과 identity 대조 성공 뒤 등록 |
| 계산 후보값 | `POD_RE`, `NODE_RE`, Prometheus label 값 | 현재 이름 escape 후 bounded 출력 |
| 운영자 선택값 | `START_RFC3339`, `END_RFC3339`, 조건부 `PVC` | 승인된 관측 창과 Pod의 PVC 후보 출력에서 선택 |

재등록 전에는 Pod/owner/Node/PVC/PV/StorageClass identity·resourceVersion, containerID, 실행 node·host 관계, 시간, selector·argv/regex와 모든 downstream `READY`를 무효화한다. API 실패 뒤 partial/stale 값은 사용하지 않는다.

### 2. 현재 Pod·owner·Node 자동 등록

`unset OWNER_READY OWNER_RV`

`unset AUTO_READY TIME_READY EVENT_READY STORAGE_READY QUERY_READY POD_UID POD_RV OWNER_KIND OWNER_NAME OWNER_UID NODE_UID NODE_RV PROVIDER_ID READY_STATUS READY_TRANSITION POD_RE NODE_RE POD_LABEL NODE_LABEL START_RFC3339 END_RFC3339 PVC PVC_UID PVC_RV PV PV_UID PV_RV SC SC_UID SC_RV; AUTO_READY=false; if [ "$INPUT_READY" != true ]; then printf '%s\n' '중단: INPUT_READY가 아니다.' >&2; elif POD_JSON="$(kubectl get pod "$POD" -n "$NS" -o json)" && NODE_JSON="$(kubectl get node "$NODE" -o json)"; then POD_UID="$(jq -er '.metadata.uid' <<<"$POD_JSON")" && POD_RV="$(jq -er '.metadata.resourceVersion' <<<"$POD_JSON")" && OWNER_COUNT="$(jq '[.metadata.ownerReferences[]? | select(.controller==true)]|length' <<<"$POD_JSON")" && [ "$OWNER_COUNT" -eq 1 ] && OWNER_KIND="$(jq -er '[.metadata.ownerReferences[]|select(.controller==true)][0].kind' <<<"$POD_JSON")" && OWNER_NAME="$(jq -er '[.metadata.ownerReferences[]|select(.controller==true)][0].name' <<<"$POD_JSON")" && OWNER_UID="$(jq -er '[.metadata.ownerReferences[]|select(.controller==true)][0].uid' <<<"$POD_JSON")" && NODE_UID="$(jq -er '.metadata.uid' <<<"$NODE_JSON")" && NODE_RV="$(jq -er '.metadata.resourceVersion' <<<"$NODE_JSON")" && PROVIDER_ID="$(jq -r '.spec.providerID // empty' <<<"$NODE_JSON")" && READY_STATUS="$(jq -er '[.status.conditions[]? | select(.type=="Ready")][0].status' <<<"$NODE_JSON")" && READY_TRANSITION="$(jq -er '[.status.conditions[]? | select(.type=="Ready")][0].lastTransitionTime' <<<"$NODE_JSON")" && [ "$READY_STATUS" = True ] && POD_RE="$(printf '%s' "$POD" | sed 's/[][\\.^$*+?(){}|]/\\&/g')" && NODE_RE="$(printf '%s' "$NODE" | sed 's/[][\\.^$*+?(){}|]/\\&/g')" && POD_LABEL="$(printf '%s' "$POD" | sed 's/\\/\\\\/g;s/"/\\"/g')" && NODE_LABEL="$(printf '%s' "$NODE" | sed 's/\\/\\\\/g;s/"/\\"/g')" && AUTO_READY=true && export POD_UID POD_RV OWNER_KIND OWNER_NAME OWNER_UID NODE_UID NODE_RV PROVIDER_ID READY_STATUS READY_TRANSITION POD_RE NODE_RE POD_LABEL NODE_LABEL; if [ "$AUTO_READY" = true ]; then printf 'AUTO_READY=true\nPod UID/RV: %s / %s\nOwner: %s/%s uid=%s\nNode UID/RV: %s / %s\nProviderID: %s\nReady/transition: %s / %s\nPOD_RE=%s NODE_RE=%s\n' "$POD_UID" "$POD_RV" "$OWNER_KIND" "$OWNER_NAME" "$OWNER_UID" "$NODE_UID" "$NODE_RV" "${PROVIDER_ID:--}" "$READY_STATUS" "$READY_TRANSITION" "$POD_RE" "$NODE_RE"; else printf '%s\n' '중단: identity/owner/Ready gate 실패' >&2; fi; else printf '%s\n' '중단: Pod 또는 Node API 실패; stale 값 사용 금지' >&2; fi`

Node providerID와 provisioning record의 비민감 instance identity가 exact match하지 않으면 이 문서를 중단한다. Pod가 replacement되면 이름이 같아도 UID가 달라질 수 있으므로 old UID의 Event를 새 Pod에 연결하지 않는다.

**현재 owner 등록과 Pod·Node·owner identity 재검증 함수**

`unset OWNER_READY OWNER_RV; OWNER_READY=false; if [ "$AUTO_READY" != true ]; then printf '%s\n' '중단: AUTO_READY가 아니다.' >&2; elif OWNER_JSON="$(kubectl get "$OWNER_KIND" "$OWNER_NAME" -n "$NS" -o json)" && [ "$(jq -r '.metadata.uid' <<<"$OWNER_JSON")" = "$OWNER_UID" ]; then OWNER_RV="$(jq -er '.metadata.resourceVersion' <<<"$OWNER_JSON")" && OWNER_READY=true && export OWNER_RV; printf 'OWNER_READY=true kind=%s name=%s uid=%s rv=%s\n' "$OWNER_KIND" "$OWNER_NAME" "$OWNER_UID" "$OWNER_RV"; else printf '%s\n' '중단: direct owner API 실패 또는 UID 변경' >&2; fi; require_current_target(){ local p n o; [ "$AUTO_READY" = true ] && [ "$OWNER_READY" = true ] || { printf '%s\n' '중단: AUTO_READY/OWNER_READY가 아니다.' >&2; return 1; }; p="$(kubectl get pod "$POD" -n "$NS" -o json)" && n="$(kubectl get node "$NODE" -o json)" && o="$(kubectl get "$OWNER_KIND" "$OWNER_NAME" -n "$NS" -o json)" || { printf '%s\n' '중단: current target API 재조회 실패' >&2; return 1; }; [ "$(jq -r '.metadata.uid' <<<"$p")" = "$POD_UID" ] && [ "$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid // empty' <<<"$p")" = "$OWNER_UID" ] && [ "$(jq -r '.metadata.uid' <<<"$n")" = "$NODE_UID" ] && [ "$(jq -r '[.status.conditions[]?|select(.type=="Ready")][0].status // empty' <<<"$n")" = True ] && [ "$(jq -r '.metadata.uid' <<<"$o")" = "$OWNER_UID" ] || { printf '%s\n' '중단: Pod/Node/owner identity 또는 Ready 상태 변경; 재등록 필요' >&2; return 1; }; }`

### 3. 관측 창과 Ready 이후 Event

`START_RFC3339='PASTE_START_RFC3339_HERE'; END_RFC3339='PASTE_END_RFC3339_HERE'; TIME_READY=false; EVENT_READY=false; if ! require_current_target; then printf '%s\n' '중단: current target identity가 필요하다.' >&2; elif [[ ! "$START_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] || [[ ! "$END_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]]; then printf '%s\n' '중단: UTC RFC3339 형식이 아님' >&2; elif S="$(date -u -d "$START_RFC3339" +%s)" && E="$(date -u -d "$END_RFC3339" +%s)" && R="$(date -u -d "$READY_TRANSITION" +%s)" && [ "$(date -u -d "@$S" +%Y-%m-%dT%H:%M:%SZ)" = "$START_RFC3339" ] && [ "$(date -u -d "@$E" +%Y-%m-%dT%H:%M:%SZ)" = "$END_RFC3339" ] && [ "$S" -lt "$E" ] && [ "$R" -le "$E" ]; then TIME_READY=true; export START_RFC3339 END_RFC3339; printf 'TIME_READY=true start=%s ready=%s end=%s\n' "$START_RFC3339" "$READY_TRANSITION" "$END_RFC3339"; else printf '%s\n' '중단: date round-trip, start<end 또는 Ready/evidence 시간관계 실패' >&2; fi`

`if [ "$TIME_READY" != true ] || ! require_current_target; then printf '%s\n' '중단: current target과 TIME_READY가 필요하다.' >&2; else EVENT_READY=false; if EVENTS_JSON="$(kubectl get events -n "$NS" --field-selector "involvedObject.kind=Pod,involvedObject.name=$POD,involvedObject.uid=$POD_UID,reason=FailedScheduling" -o json)"; then EVENT_LINES="$(jq -r '.items[] | [(.eventTime // .series.lastObservedTime // .lastTimestamp // .firstTimestamp // .metadata.creationTimestamp), .involvedObject.uid, (.count // 1), .message] | @tsv' <<<"$EVENTS_JSON")"; FILTER_RC=$?; if [ "$FILTER_RC" -ne 0 ]; then printf 'filter 오류: jq rc=%s\n' "$FILTER_RC" >&2; elif [ -z "$EVENT_LINES" ]; then printf '%s\n' 'filter no-match: Event 후보 없음' >&2; else EVENT_COUNT=0; TIME_FILTER_ERROR=false; while IFS=$'\t' read -r T UID COUNT MESSAGE; do if ! TE="$(date -u -d "$T" +%s)"; then TIME_FILTER_ERROR=true; break; elif [ "$TE" -ge "$S" ] && [ "$TE" -ge "$R" ] && [ "$TE" -le "$E" ] && [ "$UID" = "$POD_UID" ]; then EVENT_COUNT="$((EVENT_COUNT+1))"; printf '=== FailedScheduling ===\n시각: %s\nUID: %s\nCount: %s\nMessage: %s\n' "$T" "$UID" "$COUNT" "$MESSAGE"; fi; done <<<"$EVENT_LINES"; if [ "$TIME_FILTER_ERROR" = true ]; then printf '%s\n' 'filter 오류: Event timestamp parse 실패' >&2; elif [ "$EVENT_COUNT" -gt 0 ]; then EVENT_READY=true; else printf '%s\n' 'filter no-match: Ready 이후 현재 UID Event 없음' >&2; fi; fi; else printf '%s\n' 'producer 오류: Event API 실패' >&2; fi; fi`

`EVENT_READY=true`가 아니면 이 런북의 핵심 진입 조건이 아니다. API 오류, 성공한 조회의 no-match, jq/filter 오류를 서로 구분한다.

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | 현재 Node `Ready=True` 이후 current original Pod UID의 `FailedScheduling`이 관측 창에 존재 |
| 적용 | Pod request·placement·owner와 Node label·taint·allocatable 및 PVC topology를 현재 identity로 대조 |
| 제외 | Node가 Ready가 아니면 [[04. 증설된 Node가 클러스터에 등록되거나 Ready가 되지 않는 경우]], 일반 배치 원인은 [[18. Pod 이벤트에 FailedScheduling이 기록된 경우]] |

검색 문자열: `Insufficient`, `Too many pods`, `untolerated taint`, `node affinity`, `topology`, `unbound immediate PersistentVolumeClaims`, `volume node affinity conflict`.

## 관측별 진단 경로

| Event message | 직접 대조 | owner |
|---|---|---|
| taint·selector·affinity 불일치 | current Pod spec ↔ current Node label/taint | workload/template |
| `Insufficient`·`Too many pods` | init 포함 request·overhead ↔ allocatable·기존 Pod | capacity/DaemonSet |
| volume/PVC topology | Pod→PVC→PV claimRef/nodeAffinity→Node zone | storage |
| 모두 일치하나 계속 실패 | current Event와 scheduler 후보 관측 | scheduler/control plane |

## placement·capacity 직접 대조

`if [ "$EVENT_READY" != true ] || ! require_current_target; then printf '%s\n' '중단: 최신 current target과 EVENT_READY가 필요하다.' >&2; elif POD_JSON="$(kubectl get pod "$POD" -n "$NS" -o json)" && NODE_JSON="$(kubectl get node "$NODE" -o json)"; then jq -n --argjson pod "$POD_JSON" --argjson node "$NODE_JSON" '{pod:{uid:$pod.metadata.uid,nodeName:$pod.spec.nodeName,nodeSelector:$pod.spec.nodeSelector,affinity:$pod.spec.affinity,tolerations:$pod.spec.tolerations,topologySpread:$pod.spec.topologySpreadConstraints,overhead:$pod.spec.overhead,containers:[$pod.spec.containers[]|{name,requests:.resources.requests}],initContainers:[$pod.spec.initContainers[]?|{name,requests:.resources.requests}],pvcs:[$pod.spec.volumes[]?|select(.persistentVolumeClaim)|.persistentVolumeClaim.claimName]},node:{uid:$node.metadata.uid,labels:$node.metadata.labels,taints:$node.spec.taints,capacity:$node.status.capacity,allocatable:$node.status.allocatable}}'; else printf '%s\n' '중단: API 실패; 재등록 필요' >&2; fi`

출력은 관계를 보존한다. 일반 container request 합, init container별 최대 request와 Pod overhead를 scheduler 계산 규칙에 맞춰 비교한다. allocatable 전체가 아니라 기존 Pod·DaemonSet request 뒤의 여유를 본다. spec·label map에 Secret 값은 없지만 annotation·env·Secret 원문은 출력하지 않는다.

## 조건부 PVC 관계

`if [ "$EVENT_READY" != true ] || ! require_current_target; then printf '%s\n' '중단: 최신 gate가 필요하다.' >&2; else kubectl get pod "$POD" -n "$NS" -o json | jq -er --arg uid "$POD_UID" 'select(.metadata.uid==$uid) | .spec.volumes[]? | select(.persistentVolumeClaim) | "PVC 후보: \(.persistentVolumeClaim.claimName)"'; fi`

복수 후보를 임의로 고르지 않는다. Event가 volume 경로를 가리킬 때만 선택한다.

`PVC='PASTE_PVC_FROM_CANDIDATES_HERE'; STORAGE_READY=false; unset PVC_UID PVC_RV PV PV_UID PV_RV SC SC_UID SC_RV; if [ "$AUTO_READY" != true ] || [ "$EVENT_READY" != true ] || [ -z "$PVC" ]; then printf '%s\n' '중단: gate와 PVC 선택이 필요하다.' >&2; elif POD_JSON="$(kubectl get pod "$POD" -n "$NS" -o json)" && [ "$(jq -r '.metadata.uid' <<<"$POD_JSON")" = "$POD_UID" ] && jq -e --arg pvc "$PVC" 'any(.spec.volumes[]?; .persistentVolumeClaim.claimName==$pvc)' <<<"$POD_JSON" >/dev/null && PVC_JSON="$(kubectl get pvc "$PVC" -n "$NS" -o json)"; then PVC_UID="$(jq -er '.metadata.uid' <<<"$PVC_JSON")" && PVC_RV="$(jq -er '.metadata.resourceVersion' <<<"$PVC_JSON")" && PV="$(jq -er '.spec.volumeName' <<<"$PVC_JSON")" && SC="$(jq -er '.spec.storageClassName' <<<"$PVC_JSON")" && PV_JSON="$(kubectl get pv "$PV" -o json)" && SC_JSON="$(kubectl get storageclass "$SC" -o json)" && PV_UID="$(jq -er '.metadata.uid' <<<"$PV_JSON")" && PV_RV="$(jq -er '.metadata.resourceVersion' <<<"$PV_JSON")" && SC_UID="$(jq -er '.metadata.uid' <<<"$SC_JSON")" && SC_RV="$(jq -er '.metadata.resourceVersion' <<<"$SC_JSON")" && [ "$(jq -r '.spec.claimRef.uid' <<<"$PV_JSON")" = "$PVC_UID" ] && STORAGE_READY=true && export PVC PVC_UID PVC_RV PV PV_UID PV_RV SC SC_UID SC_RV; if [ "$STORAGE_READY" = true ]; then jq -n --argjson pvc "$PVC_JSON" --argjson pv "$PV_JSON" --argjson sc "$SC_JSON" '{pvc:{name:$pvc.metadata.name,uid:$pvc.metadata.uid,phase:$pvc.status.phase,pv:$pvc.spec.volumeName},pv:{name:$pv.metadata.name,uid:$pv.metadata.uid,claimRef:$pv.spec.claimRef,nodeAffinity:$pv.spec.nodeAffinity},storageClass:{name:$sc.metadata.name,uid:$sc.metadata.uid,binding:$sc.volumeBindingMode,provisioner:$sc.provisioner}}'; else printf '%s\n' '중단: PVC→PV claimRef identity 불일치' >&2; fi; else printf '%s\n' '중단: 후보 관계/API 실패; stale storage 값 금지' >&2; fi`

`if [ "$STORAGE_READY" != true ] || ! require_current_target; then printf '%s\n' '중단: current target과 STORAGE_READY가 필요하다.' >&2; elif CURRENT_PVC_JSON="$(kubectl get pvc "$PVC" -n "$NS" -o json)" && CURRENT_PV_JSON="$(kubectl get pv "$PV" -o json)" && [ "$(jq -r '.metadata.uid' <<<"$CURRENT_PVC_JSON")" = "$PVC_UID" ] && [ "$(jq -r '.metadata.uid' <<<"$CURRENT_PV_JSON")" = "$PV_UID" ] && [ "$(jq -r '.spec.claimRef.uid' <<<"$CURRENT_PV_JSON")" = "$PVC_UID" ] && VA_JSON="$(kubectl get volumeattachments.storage.k8s.io -o json)"; then jq -n --argjson pv "$CURRENT_PV_JSON" '{pv:{name:$pv.metadata.name,uid:$pv.metadata.uid,csiDriver:($pv.spec.csi.driver // "-"),nodeAffinity:$pv.spec.nodeAffinity}}'; VA_COUNT="$(jq --arg pv "$PV" '[.items[] | select(.spec.source.persistentVolumeName==$pv)] | length' <<<"$VA_JSON")"; if [ "$VA_COUNT" -eq 0 ]; then printf '%s\n' 'VolumeAttachment: 없음 (Pending Pod에서는 attach 성공 증거가 아님)'; else jq -r --arg pv "$PV" '.items[] | select(.spec.source.persistentVolumeName==$pv) | "VolumeAttachment: \(.metadata.name) uid=\(.metadata.uid) attacher=\(.spec.attacher) node=\(.spec.nodeName) attached=\(.status.attached // false)"' <<<"$VA_JSON"; fi; else printf '%s\n' '중단: current PVC/PV/claimRef 또는 VolumeAttachment producer 실패' >&2; fi`

## Prometheus 쿼리 생성

`QUERY_READY=false; if [ "$EVENT_READY" != true ] || ! require_current_target; then printf '%s\n' '중단: current identity gate가 필요하다.' >&2; else QUERY_READY=true; printf 'kube_node_status_condition{node="%s",condition="Ready",status="true"} == 1\n' "$NODE_LABEL"; printf 'kube_pod_status_scheduled{namespace="%s",pod="%s",condition="false"} == 1\n' "$(printf '%s' "$NS" | sed 's/\\/\\\\/g;s/"/\\"/g')" "$POD_LABEL"; fi`

현재 UID는 metric label로 보장되지 않으므로 API/Event identity를 대체하지 않는다. 외부 UI의 빈 결과는 metric 미수집·label 불일치·시간 범위 오류와 API/query 실패를 분리하고 0이나 정상으로 해석하지 않는다.

## 복구·replacement·정상화

이 문서는 `label`, `taint`, `patch`, Pod 삭제, scale 또는 추가 증설 명령을 제공하지 않는다. 변경 절차는 실행 직전에 fresh Pod UID/RV·owner UID, Node UID/RV/providerID, selector/affinity/spec, provisioning template revision·image digest, PVC/PV identity, evidence 창, 승인 ID·변경 창, PDB·quorum, 멱등성·중복 side effect, rollback을 직접 gate해야 한다. 하나라도 API 실패·identity 변경이면 중단한다.

변경 후 original Pod가 사라지면 replacement의 새 UID/RV·owner UID·Node UID를 재등록하고 old/new Event를 섞지 않는다. 정상화는 `template/record → Node 사실 → scheduler 후보 조건 → Pod bind → container Ready → Endpoint/Job 산출물 → 실제 요청·중복 side effect 없음` 순으로 확인한다. rollback 후에도 같은 관계와 관찰 창을 다시 검증한다.

RBAC은 `pods/nodes/events/persistentvolumeclaims/persistentvolumes/storageclasses/volumeattachments.storage.k8s.io get,list` 중 실제 분기에 필요한 범위만 사용한다. scheduler/audit 로그는 별도 권한·보존정책이 확인된 owner가 시간·Pod UID로 제한해 조회하며 credential·Secret·개인정보를 저장하지 않는다.
