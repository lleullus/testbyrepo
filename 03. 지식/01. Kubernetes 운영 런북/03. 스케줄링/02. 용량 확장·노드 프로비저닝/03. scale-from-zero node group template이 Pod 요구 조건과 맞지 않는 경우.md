# scale-from-zero node group template이 Pod 요구 조건과 맞지 않는 경우

## 입력값과 빠른 수집

0대 node group에는 live Node가 없으므로 Node 이름이나 allocatable을 추측하지 않는다. 같은 Pending Pod UID의 요구 조건, autoscaler의 group 선택·template 평가와 승인된 template revision의 값을 같은 시간 창에서 연결한다.

| 구분 | 값 | gate |
|---|---|---|
| 최초 입력 | `NS`, `POD` | exact-name Pending·unbound Pod |
| API 자동값 | Pod/current owner UID·resourceVersion, selector, request·placement·PVC, autoscaler Pod/container/current owner/Node identity | 관계와 current containerID 일치 |
| 계산 후보 | `POD_RE`, log regex, PVC→PV→CSI/topology 관계 | parser와 API producer 성공 전 미확정 |
| 운영자 선택 | UTC 창, node group/template source·revision·scope, 불일치 종류와 양쪽 값 | 승인된 inventory·template evidence 대조 |

실행 위치는 승인된 kubeconfig가 있는 작업 PC 또는 bastion의 같은 Bash다. hop은 `작업 host → API server → Pending Pod/Event → autoscaler Pod/container log → 승인된 node group/template 관리면`이다. 마지막 hop은 별도 승인 경로이며 provider endpoint·CLI를 발명하지 않는다. SSH 절차는 positional argument와 원격 host/account/scope 재검증을 사용하고 `eval`은 금지한다.

request는 일반 container 합, init container의 자원별 최대와 Pod overhead로 계산한다. extended resource 이름·수량도 보존한다. `matchExpressions` owner selector는 문자열로 축약하지 않아 replacement 자동화를 중단한다.

`fresh_target(){ local p o host uid ouid node cids; [ "$AUTO_READY" = true ] || return 1; host="$(hostname -f)" || return 1; [ "$host" = "$HOST_ID" ] || { printf '%s\n' '중단: 실행 host 변경' >&2; return 1; }; p="$(kubectl get pod "$POD" -n "$NS" -o json)" || return 1; o="$(kubectl get "$OWNER_RESOURCE" "$OWNER_NAME" -n "$NS" -o json)" || return 1; uid="$(jq -r '.metadata.uid' <<<"$p")"; ouid="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid//empty' <<<"$p")"; node="$(jq -r '.spec.nodeName//"UNBOUND"' <<<"$p")"; cids="$(jq -r '[.status.initContainerStatuses[]?.containerID//empty,.status.containerStatuses[]?.containerID//empty]|sort|join(",")|if .=="" then "NONE" else . end' <<<"$p")"; if [ "$uid" != "$POD_UID" ] || [ "$ouid" != "$OWNER_UID" ] || [ "$(jq -r '.metadata.uid' <<<"$o")" != "$OWNER_UID" ] || [ "$node" != "$POD_NODE" ] || [ "$cids" != "$CONTAINER_IDS" ]; then printf '%s\n' '중단: Pod/current owner/containerID/node identity 변경; 전체 재수집' >&2; return 1; fi; POD_RV="$(jq -r '.metadata.resourceVersion' <<<"$p")"; export POD_RV; }`

### autoscaler Pod·container 선택과 자동 identity

external mode이면 Kubernetes log 명령을 사용하지 않고 승인된 감사 경로에서 같은 Pod UID·group·template revision·시간을 확인한다. 아래 gate는 in-cluster autoscaler 전용이다.

`AS_NS='PASTE_TEMPLATE_AUTOSCALER_NAMESPACE_HERE'; AS_POD='PASTE_TEMPLATE_AUTOSCALER_POD_HERE'; AS_CONTAINER='PASTE_TEMPLATE_AUTOSCALER_CONTAINER_HERE'; AS_SELECTION_READY=false; if [ "$AUTO_READY" != true ] || ! fresh_target; then printf '%s\n' '중단: fresh target 실패' >&2; elif [ -z "$AS_NS" ] || [ -z "$AS_POD" ] || [ -z "$AS_CONTAINER" ]; then printf '%s\n' '중단: in-cluster autoscaler 선택값 필요' >&2; else AS_SELECTION_READY=true; export AS_NS AS_POD AS_CONTAINER; printf 'AS_SELECTION_READY=true ns=%s pod=%s container=%s\n' "$AS_NS" "$AS_POD" "$AS_CONTAINER"; fi`

`unset AS_READY WINDOW_READY EVENT_READY LOG_READY STORAGE_READY TEMPLATE_READY CHANGE_READY AS_UID AS_RV AS_OWNER_KIND AS_OWNER_RESOURCE AS_OWNER_NAME AS_OWNER_UID AS_NODE AS_NODE_UID AS_CONTAINER_ID; AS_READY=false; if [ "$AS_SELECTION_READY" != true ] || ! fresh_target; then printf '%s\n' '중단: autoscaler selection/fresh target 실패' >&2; elif AJSON="$(kubectl get pod "$AS_POD" -n "$AS_NS" -o json)" && jq -e --arg c "$AS_CONTAINER" 'any(.spec.containers[]?;.name==$c)' <<<"$AJSON" >/dev/null; then AS_UID="$(jq -r '.metadata.uid' <<<"$AJSON")"; AS_RV="$(jq -r '.metadata.resourceVersion' <<<"$AJSON")"; AS_OWNER_KIND="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].kind//empty' <<<"$AJSON")"; AS_OWNER_NAME="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].name//empty' <<<"$AJSON")"; AS_OWNER_UID="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid//empty' <<<"$AJSON")"; AS_OWNER_RESOURCE="$(printf '%s' "$AS_OWNER_KIND" | tr '[:upper:]' '[:lower:]')"; AS_NODE="$(jq -r '.spec.nodeName//empty' <<<"$AJSON")"; AS_CONTAINER_ID="$(jq -r --arg c "$AS_CONTAINER" '[.status.containerStatuses[]?|select(.name==$c)][0].containerID//empty' <<<"$AJSON")"; if AOJSON="$(kubectl get "$AS_OWNER_RESOURCE" "$AS_OWNER_NAME" -n "$AS_NS" -o json)" && ANJSON="$(kubectl get node "$AS_NODE" -o json)" && [ "$(jq -r '.metadata.uid' <<<"$AOJSON")" = "$AS_OWNER_UID" ]; then AS_NODE_UID="$(jq -r '.metadata.uid' <<<"$ANJSON")"; if [ -n "$AS_UID" ] && [ -n "$AS_CONTAINER_ID" ] && [ -n "$AS_NODE_UID" ]; then AS_READY=true; export AS_UID AS_RV AS_OWNER_KIND AS_OWNER_RESOURCE AS_OWNER_NAME AS_OWNER_UID AS_NODE AS_NODE_UID AS_CONTAINER_ID; printf 'AS_READY=true pod=%s[%s] owner=%s/%s[%s] container=%s[%s] node=%s[%s]\n' "$AS_POD" "$AS_UID" "$AS_OWNER_KIND" "$AS_OWNER_NAME" "$AS_OWNER_UID" "$AS_CONTAINER" "$AS_CONTAINER_ID" "$AS_NODE" "$AS_NODE_UID"; else printf '%s\n' '중단: autoscaler containerID/node identity 불완전' >&2; fi; else printf '%s\n' '중단: autoscaler current owner 또는 Node 조회 실패' >&2; fi; else printf '%s\n' '중단: autoscaler Pod/container 조회 실패' >&2; fi`

`fresh_all(){ local a ao an; fresh_target || return 1; [ "$AS_READY" = true ] || return 1; a="$(kubectl get pod "$AS_POD" -n "$AS_NS" -o json)" || return 1; ao="$(kubectl get "$AS_OWNER_RESOURCE" "$AS_OWNER_NAME" -n "$AS_NS" -o json)" || return 1; an="$(kubectl get node "$AS_NODE" -o json)" || return 1; if [ "$(jq -r '.metadata.uid' <<<"$a")" != "$AS_UID" ] || [ "$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid//empty' <<<"$a")" != "$AS_OWNER_UID" ] || [ "$(jq -r '.metadata.uid' <<<"$ao")" != "$AS_OWNER_UID" ] || [ "$(jq -r '.metadata.uid' <<<"$an")" != "$AS_NODE_UID" ] || [ "$(jq -r '.spec.nodeName//empty' <<<"$a")" != "$AS_NODE" ] || [ "$(jq -r --arg c "$AS_CONTAINER" '[.status.containerStatuses[]?|select(.name==$c)][0].containerID//empty' <<<"$a")" != "$AS_CONTAINER_ID" ]; then printf '%s\n' '중단: autoscaler current owner/containerID/node identity 변경' >&2; return 1; fi; AS_RV="$(jq -r '.metadata.resourceVersion' <<<"$a")"; export AS_RV; }`

### UTC evidence window와 log regex

`START_RFC3339='PASTE_TEMPLATE_START_UTC_HERE'; END_RFC3339='PASTE_TEMPLATE_END_UTC_HERE'; LOG_RE='PASTE_TEMPLATE_VERIFIED_LOG_REGEX_HERE'; WINDOW_READY=false; RFC3339_RE='^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$'; grep -E "$LOG_RE" </dev/null >/dev/null 2>&1; REGEX_RC=$?; if [ "$AS_READY" != true ] || ! fresh_all; then printf '%s\n' '중단: fresh identity 실패' >&2; elif [[ ! "$START_RFC3339" =~ $RFC3339_RE ]] || [[ ! "$END_RFC3339" =~ $RFC3339_RE ]] || [ -z "$LOG_RE" ] || [ "$REGEX_RC" -eq 2 ]; then printf '%s\n' '중단: canonical UTC 또는 regex parser 검증 실패' >&2; elif START_EPOCH="$(date -u -d "$START_RFC3339" +%s 2>/dev/null)" && END_EPOCH="$(date -u -d "$END_RFC3339" +%s 2>/dev/null)" && [ "$(date -u -d "@$START_EPOCH" '+%Y-%m-%dT%H:%M:%SZ')" = "$START_RFC3339" ] && [ "$(date -u -d "@$END_EPOCH" '+%Y-%m-%dT%H:%M:%SZ')" = "$END_RFC3339" ] && [ "$START_EPOCH" -lt "$END_EPOCH" ] && [ "$END_EPOCH" -le "$(date -u +%s)" ]; then WINDOW_READY=true; export START_RFC3339 END_RFC3339 START_EPOCH END_EPOCH LOG_RE; printf 'WINDOW_READY=true %s..%s regex=%s\n' "$START_RFC3339" "$END_RFC3339" "$LOG_RE"; else printf '%s\n' '중단: time round-trip/start<end/미래 상한 실패' >&2; fi`

## 읽기 전용 진단

**UID·시간 제한 FailedScheduling**

`EVENT_READY=false; if [ "$WINDOW_READY" != true ] || ! fresh_all; then printf '%s\n' '중단: 최신 window/identity 실패' >&2; elif EVENTS_JSON="$(kubectl get events -n "$NS" --field-selector "involvedObject.kind=Pod,involvedObject.name=$POD,involvedObject.uid=$POD_UID,reason=FailedScheduling" -o json)"; then EVENT_ROWS="$(jq -r --argjson s "$START_EPOCH" --argjson e "$END_EPOCH" '.items[]|(.eventTime//.series.lastObservedTime//.lastTimestamp//.firstTimestamp) as $raw|($raw|strings|sub("\\.[0-9]+Z$";"Z")|fromdateiso8601?) as $t|select($t!=null and $t>=$s and $t<=$e)|[.metadata.uid,(.count//1),$raw,.message]|@tsv' <<<"$EVENTS_JSON")"; FILTER_RC=$?; if [ "$FILTER_RC" -eq 0 ] && [ -n "$EVENT_ROWS" ]; then EVENT_READY=true; printf '%s\n' "$EVENT_ROWS"; printf 'EVENT_READY=true pod_uid=%s\n' "$POD_UID"; elif [ "$FILTER_RC" -eq 0 ]; then printf '%s\n' '중단: window 안 Event 없음; 정상으로 해석 금지' >&2; else printf '중단: jq filter rc=%s\n' "$FILTER_RC" >&2; fi; else PRODUCER_RC=$?; printf '중단: Event producer rc=%s; filter 미실행\n' "$PRODUCER_RC" >&2; fi`

**autoscaler log producer·시간 filter·regex filter 분리**

`LOG_READY=false; if [ "$EVENT_READY" != true ] || ! fresh_all; then printf '%s\n' '중단: 최신 Event/identity 실패' >&2; elif RAW_LOG="$(kubectl logs "$AS_POD" -n "$AS_NS" -c "$AS_CONTAINER" --timestamps --since-time="$START_RFC3339" --tail=500)"; then WINDOW_LOG="$(awk -v s="$START_RFC3339" -v e="$END_RFC3339" '$1>=s && $1<=e' <<<"$RAW_LOG")"; AWK_RC=$?; if [ "$AWK_RC" -ne 0 ]; then printf '중단: time filter rc=%s\n' "$AWK_RC" >&2; else MATCHED="$(grep -E -i -C 3 -- "$LOG_RE" <<<"$WINDOW_LOG")"; GREP_RC=$?; if [ "$GREP_RC" -eq 0 ]; then LOG_READY=true; printf '%s\n' "$MATCHED"; printf 'LOG_READY=true pod_uid=%s autoscaler_uid=%s\n' "$POD_UID" "$AS_UID"; elif [ "$GREP_RC" -eq 1 ]; then printf '%s\n' '필터 no-match: group 선택·template 평가·schema·시간 재확인' >&2; else printf '중단: grep filter rc=%s\n' "$GREP_RC" >&2; fi; fi; else PRODUCER_RC=$?; printf '중단: log producer rc=%s\n' "$PRODUCER_RC" >&2; fi`

로그에는 대상 Pod UID, 선택된 0대 node group, template source/revision과 거부 조건이 함께 있어야 한다. provider별 문자열을 추측하지 않는다. `pods/log get`, `events list`, 객체 `get`은 서로 다른 RBAC 확인이다. audit는 승인된 경로에서 단일 auditID와 current lifecycle UID만 연결하며 actor 상세·token을 보존하지 않는다.

### PVC·PV·CSI topology 관계

`STORAGE_READY=false; if [ "$LOG_READY" != true ] || ! fresh_all; then printf '%s\n' '중단: fresh log/identity 실패' >&2; elif [ "${#PVC_NAMES[@]}" -eq 0 ]; then STORAGE_READY=true; printf '%s\n' 'STORAGE_READY=true PVC 없음'; else STORAGE_FAILED=false; for CLAIM in "${PVC_NAMES[@]}"; do if ! CJSON="$(kubectl get pvc "$CLAIM" -n "$NS" -o json)"; then STORAGE_FAILED=true; break; fi; SC="$(jq -r '.spec.storageClassName//empty' <<<"$CJSON")"; PV="$(jq -r '.spec.volumeName//empty' <<<"$CJSON")"; if [ -z "$SC" ] || ! SJSON="$(kubectl get storageclass "$SC" -o json)"; then STORAGE_FAILED=true; break; fi; printf '=== PVC: %s ===\nUID: %s\nPhase: %s\nPV: %s\nStorageClass: %s\nBinding: %s\nProvisioner: %s\n' "$CLAIM" "$(jq -r '.metadata.uid' <<<"$CJSON")" "$(jq -r '.status.phase//"-"' <<<"$CJSON")" "${PV:--}" "$SC" "$(jq -r '.volumeBindingMode//"-"' <<<"$SJSON")" "$(jq -r '.provisioner//"-"' <<<"$SJSON")"; if [ -n "$PV" ]; then if ! VJSON="$(kubectl get pv "$PV" -o json)"; then STORAGE_FAILED=true; break; fi; printf 'PV UID: %s\nCSI driver: %s\nClaim UID: %s\nNode affinity: %s\n' "$(jq -r '.metadata.uid' <<<"$VJSON")" "$(jq -r '.spec.csi.driver//"-"' <<<"$VJSON")" "$(jq -r '.spec.claimRef.uid//"-"' <<<"$VJSON")" "$(jq -c '.spec.nodeAffinity//{}' <<<"$VJSON")"; if [ "$(jq -r '.spec.claimRef.uid//empty' <<<"$VJSON")" != "$(jq -r '.metadata.uid' <<<"$CJSON")" ]; then STORAGE_FAILED=true; break; fi; fi; done; if [ "$STORAGE_FAILED" = false ]; then STORAGE_READY=true; printf 'STORAGE_READY=true pvc_count=%s\n' "${#PVC_NAMES[@]}"; else printf '%s\n' '중단: PVC→PV→CSI/StorageClass producer 또는 claim UID 관계 실패' >&2; fi; fi`

`WaitForFirstConsumer`의 Pending PVC와 빈 PV는 정상 중간 상태일 수 있다. Bound PV가 있으면 claim UID, CSI driver와 `nodeAffinity`를 template zone과 대조한다. `VolumeAttachment`는 아직 Node가 없는 scale-from-zero 평가 단계에는 기대하지 않는다.

## template evidence와 불일치 확정

승인된 template 관리면의 읽기 전용 결과에서 identifier·revision·scope와 필요한 값만 입력한다. 전체 provider 응답이나 credential은 저장하지 않는다.

`NODE_GROUP_ID='PASTE_TEMPLATE_NODE_GROUP_ID_HERE'; TEMPLATE_SOURCE_ID='PASTE_TEMPLATE_SOURCE_ID_HERE'; TEMPLATE_REVISION='PASTE_TEMPLATE_REVISION_HERE'; TEMPLATE_SCOPE='PASTE_TEMPLATE_SCOPE_HERE'; TEMPLATE_EVIDENCE_ID='PASTE_TEMPLATE_EVIDENCE_ID_HERE'; TEMPLATE_PROCEDURE_ID='PASTE_TEMPLATE_PROCEDURE_ID_HERE'; MISMATCH_KIND='PASTE_TEMPLATE_MISMATCH_KIND_HERE'; POD_REQUIREMENT='PASTE_TEMPLATE_POD_REQUIREMENT_HERE'; TEMPLATE_VALUE='PASTE_TEMPLATE_OBSERVED_VALUE_HERE'; TEMPLATE_READY=false; if [ "$STORAGE_READY" != true ] || ! fresh_all; then printf '%s\n' '중단: fresh storage/log evidence가 아니다.' >&2; elif [[ ! "$MISMATCH_KIND" =~ ^(resources|placement|architecture|topology)$ ]] || [ -z "$NODE_GROUP_ID" ] || [ -z "$TEMPLATE_SOURCE_ID" ] || [ -z "$TEMPLATE_REVISION" ] || [ -z "$TEMPLATE_SCOPE" ] || [ -z "$TEMPLATE_EVIDENCE_ID" ] || [ -z "$TEMPLATE_PROCEDURE_ID" ] || [ -z "$POD_REQUIREMENT" ] || [ -z "$TEMPLATE_VALUE" ]; then printf '%s\n' '중단: template identity/evidence/비교값 불완전' >&2; else TEMPLATE_READY=true; export NODE_GROUP_ID TEMPLATE_SOURCE_ID TEMPLATE_REVISION TEMPLATE_SCOPE TEMPLATE_EVIDENCE_ID TEMPLATE_PROCEDURE_ID MISMATCH_KIND POD_REQUIREMENT TEMPLATE_VALUE; printf 'TEMPLATE_READY=true group=%s source=%s revision=%s scope=%s mismatch=%s evidence=%s\nPod requirement=%s\nTemplate value=%s\n' "$NODE_GROUP_ID" "$TEMPLATE_SOURCE_ID" "$TEMPLATE_REVISION" "$TEMPLATE_SCOPE" "$MISMATCH_KIND" "$TEMPLATE_EVIDENCE_ID" "$POD_REQUIREMENT" "$TEMPLATE_VALUE"; fi`

- `resources`: 일반 합·init 최대·overhead·extended resource와 template allocatable을 자원별로 비교한다.
- `placement`: required selector/affinity와 key/operator/value, taint/effect/toleration을 term별로 비교한다. preferred 불일치는 hard 거부가 아니다.
- `architecture`: workload 정책과 image manifest가 지원하는 architecture를 template과 비교한다.
- `topology`: Pod 조건과 PVC→PV nodeAffinity/CSI topology를 template zone과 비교한다.
- template revision 또는 scope가 autoscaler log와 다르면 원인 확정을 중단한다.

## 상태 변경 경계

template, label·taint, allocatable, architecture·zone, Pod selector/affinity/toleration 또는 PVC를 변경·삭제하는 명령을 제공하지 않는다. 변경은 workload·template·storage owner가 승인한 관리면에서만 수행한다.

`APPROVAL_ID='PASTE_TEMPLATE_APPROVAL_ID_HERE'; SAFETY_EVIDENCE_ID='PASTE_TEMPLATE_SAFETY_EVIDENCE_ID_HERE'; ROLLBACK_REF='PASTE_TEMPLATE_ROLLBACK_REF_HERE'; IDEMPOTENCE_RECORD='PASTE_TEMPLATE_IDEMPOTENCE_RECORD_HERE'; CHANGE_RFC3339='PASTE_TEMPLATE_CHANGE_UTC_HERE'; CHANGE_READY=false; if [ "$TEMPLATE_READY" != true ] || ! fresh_all; then printf '%s\n' '중단: fresh template evidence가 아니다.' >&2; elif [ -z "$APPROVAL_ID" ] || [ -z "$SAFETY_EVIDENCE_ID" ] || [ -z "$ROLLBACK_REF" ] || [ -z "$IDEMPOTENCE_RECORD" ]; then printf '%s\n' '중단: approval/safety/rollback/idempotence 기록 필요' >&2; elif [[ ! "$CHANGE_RFC3339" =~ $RFC3339_RE ]] || ! CHANGE_EPOCH="$(date -u -d "$CHANGE_RFC3339" +%s 2>/dev/null)" || [ "$(date -u -d "@$CHANGE_EPOCH" '+%Y-%m-%dT%H:%M:%SZ')" != "$CHANGE_RFC3339" ] || [ "$CHANGE_EPOCH" -lt "$END_EPOCH" ] || [ "$CHANGE_EPOCH" -gt "$(date -u +%s)" ]; then printf '%s\n' '중단: 변경 시각 canonical/round-trip/evidence 이후 조건 실패' >&2; else CHANGE_READY=true; export APPROVAL_ID SAFETY_EVIDENCE_ID ROLLBACK_REF IDEMPOTENCE_RECORD CHANGE_RFC3339 CHANGE_EPOCH; printf 'CHANGE_READY=true group=%s revision=%s approval=%s change=%s rollback=%s\n' "$NODE_GROUP_ID" "$TEMPLATE_REVISION" "$APPROVAL_ID" "$CHANGE_RFC3339" "$ROLLBACK_REF"; fi`

이 gate는 mutation이 아니다. 외부 변경 직전 current target/autoscaler identity, template revision·selector, PDB·quorum·Ready·Endpoint baseline, 비용·tenant 영향과 이미 적용됐는지 여부를 다시 확인한다. scale-from-zero 미시도, 잘못된 architecture/zone, Ready 감소, 동일 거부가 보이면 중단하고 검증된 `ROLLBACK_REF`만 사용한다.

## replacement 재등록과 정상화

template owner가 확인한 새 Node와 controller selector의 사후 Pod를 입력한다. 기존 Pending Pod가 그대로 bind됐다면 같은 이름·UID를 허용하고, 실제 교체된 경우에만 새 UID와 변경 시각 이후 creation을 요구한다.

`NEW_POD='PASTE_TEMPLATE_REPLACEMENT_POD_HERE'; NEW_NODE='PASTE_TEMPLATE_REGISTERED_NODE_HERE'; unset REPLACEMENT_READY NORMAL_READY NEW_POD_UID NEW_POD_RV NEW_NODE_UID NEW_CREATED NEW_CONTAINER_IDS POST_LIFECYCLE; REPLACEMENT_READY=false; if [ "$CHANGE_READY" != true ]; then printf '%s\n' '중단: CHANGE_READY가 아니다.' >&2; elif CANDIDATES="$(kubectl get pods -n "$NS" -l "$OWNER_SELECTOR" -o json)" && [ "$(jq --arg p "$NEW_POD" '[.items[]|select(.metadata.name==$p)]|length' <<<"$CANDIDATES")" -eq 1 ] && NJSON="$(kubectl get node "$NEW_NODE" -o json)" && OJSON="$(kubectl get "$OWNER_RESOURCE" "$OWNER_NAME" -n "$NS" -o json)"; then NP="$(jq -c --arg p "$NEW_POD" '.items[]|select(.metadata.name==$p)' <<<"$CANDIDATES")"; NEW_POD_UID="$(jq -r '.metadata.uid' <<<"$NP")"; NEW_POD_RV="$(jq -r '.metadata.resourceVersion' <<<"$NP")"; NEW_NODE_UID="$(jq -r '.metadata.uid' <<<"$NJSON")"; NEW_CREATED="$(jq -r '.metadata.creationTimestamp' <<<"$NP")"; NEW_CONTAINER_IDS="$(jq -r '[.status.initContainerStatuses[]?.containerID//empty,.status.containerStatuses[]?.containerID//empty]|sort|join(",")' <<<"$NP")"; POST_LIFECYCLE=''; if [ "$NEW_POD" = "$POD" ] && [ "$NEW_POD_UID" = "$POD_UID" ]; then POST_LIFECYCLE='same-pod'; elif [ "$NEW_POD_UID" != "$POD_UID" ] && [ "$(date -u -d "$NEW_CREATED" +%s)" -ge "$CHANGE_EPOCH" ]; then POST_LIFECYCLE='replacement'; fi; if [ "$(jq -r '.metadata.uid' <<<"$OJSON")" = "$OWNER_UID" ] && [ "$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid//empty' <<<"$NP")" = "$OWNER_UID" ] && [ "$(jq -r '.spec.nodeName//empty' <<<"$NP")" = "$NEW_NODE" ] && [ -n "$NEW_NODE_UID" ] && [ -n "$POST_LIFECYCLE" ]; then REPLACEMENT_READY=true; export NEW_POD NEW_POD_UID NEW_POD_RV NEW_NODE NEW_NODE_UID NEW_CREATED NEW_CONTAINER_IDS POST_LIFECYCLE; printf 'REPLACEMENT_READY=true lifecycle=%s pod=%s[%s] rv=%s node=%s[%s] created=%s containerIDs=%s\n' "$POST_LIFECYCLE" "$NEW_POD" "$NEW_POD_UID" "$NEW_POD_RV" "$NEW_NODE" "$NEW_NODE_UID" "$NEW_CREATED" "${NEW_CONTAINER_IDS:--}"; else printf '%s\n' '중단: selector/current owner/node/lifecycle/change-time relation 불일치' >&2; fi; else printf '%s\n' '중단: post-change Pod 단일 후보 또는 current Node/owner 조회 실패' >&2; fi`

`NORMAL_READY=false; if [ "$REPLACEMENT_READY" != true ] || ! NP="$(kubectl get pod "$NEW_POD" -n "$NS" -o json)" || ! NN="$(kubectl get node "$NEW_NODE" -o json)" || ! OO="$(kubectl get "$OWNER_RESOURCE" "$OWNER_NAME" -n "$NS" -o json)"; then printf '%s\n' '중단: 정상화 producer 실패' >&2; elif [ "$(jq -r '.metadata.uid' <<<"$NP")" != "$NEW_POD_UID" ] || [ "$(jq -r '.metadata.uid' <<<"$NN")" != "$NEW_NODE_UID" ] || [ "$(jq -r '.metadata.uid' <<<"$OO")" != "$OWNER_UID" ] || [ "$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid//empty' <<<"$NP")" != "$OWNER_UID" ] || [ "$(jq -r '.spec.nodeName//empty' <<<"$NP")" != "$NEW_NODE" ]; then printf '%s\n' '중단: replacement current identity 불일치' >&2; elif [ "$(jq -r '[.status.conditions[]?|select(.type=="Ready")][0].status//""' <<<"$NN")" = True ] && [ "$(jq -r '.status.phase' <<<"$NP")" = Running ] && jq -e '(.status.containerStatuses|length)>0 and all(.status.containerStatuses[];.ready==true and (.containerID//"")!="")' <<<"$NP" >/dev/null; then NEW_CONTAINER_IDS="$(jq -r '[.status.containerStatuses[].containerID]|sort|join(",")' <<<"$NP")"; NORMAL_READY=true; printf 'NORMAL_READY=true pod=%s[%s] node=%s[%s] arch=%s containerIDs=%s\n' "$NEW_POD" "$NEW_POD_UID" "$NEW_NODE" "$NEW_NODE_UID" "$(jq -r '.status.nodeInfo.architecture' <<<"$NN")" "$NEW_CONTAINER_IDS"; else printf '%s\n' '중단: Node Ready 또는 Pod/container Running/Ready 미수렴' >&2; fi`

정상화는 새 Node의 template revision에 해당하는 label·taint·architecture·zone·allocatable, Pod requirement 충족, PVC binding/topology, owner desired/Ready, 같은 `FailedScheduling`과 template 거부 재발 없음, 조건부 Service→EndpointSlice→새 Pod UID와 실제 요청·Job 결과를 관찰 창 동안 확인한다. 새 Node나 Pod `Running`만으로 닫지 않는다.

## Prometheus 빠른 관측

`if [ "$EVENT_READY" != true ] || ! fresh_all; then printf '%s\n' '중단: latest Event identity gate 실패' >&2; else printf 'kube_pod_info{namespace="%s",pod="%s",uid="%s"}\n' "$NS" "$POD" "$POD_UID"; printf 'kube_pod_status_unschedulable{namespace="%s",pod="%s"} == 1\n' "$NS" "$POD"; fi`

빈 결과는 0이 아니다. metric 부재·label 불일치·cluster 범위·scrape 실패를 구분한다. live Node가 없는 group의 template revision·label·taint·가상 allocatable은 이 질의로 증명할 수 없다.

## 증거 보존과 환경별 미검증

실행 host, UTC window, Pod/current owner/autoscaler/current autoscaler owner/container/Node UID·resourceVersion·관계, request·placement, PVC→PV→CSI/topology, Event와 producer/filter rc, 비식별 로그, group/template source·revision·scope·양쪽 비교값, 승인·safety·idempotence·rollback·변경 시각, replacement와 실제 기능을 보존한다. Secret·token·credential·개인키·전체 provider 응답·전체 로그는 저장하지 않는다.

scale-from-zero 구현체, template metadata·allocatable 산정, provider API, image manifest, PDB/quorum, Service endpoint와 실제 Kubernetes 1.26 결과는 환경별 미검증이다. 이 문서 작성 중 실제 인프라 명령은 실행하지 않았다.
