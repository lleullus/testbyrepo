# node group 최대 크기·인프라 quota 때문에 scale-up이 중단된 경우

## 입력값과 빠른 수집

`Pending`이나 `Insufficient`만으로 공급 한계를 확정하지 않는다. 같은 Pod UID의 `FailedScheduling`, autoscaler가 선택한 node group과 실제 시도, 승인된 provider evidence의 scope·시각이 연결돼야 한다.

| 구분 | 값 | gate |
|---|---|---|
| 최초 입력 | `NS`, `POD` | exact-name Pending·unbound Pod |
| API 자동값 | Pod·owner UID/resourceVersion, selector, request·placement, autoscaler Pod/container·Node identity | current owner와 containerID 관계 일치 |
| 계산 후보 | `POD_RE`, 로그 regex | escape·regex parser 검증 전 미확정 |
| 운영자 선택 | autoscaler 위치, UTC 창, node group/scope, 한계 종류, provider evidence | 인벤토리·비식별 로그·승인된 외부 기록 대조 |

실행 위치는 승인된 kubeconfig가 있는 작업 PC 또는 bastion의 같은 Bash다. Kubernetes hop은 `작업 host → API server → Pending Pod/Event → autoscaler Pod/container log`다. provider API·node group 관리면은 별도 승인 경로이며 이 문서에서 endpoint나 CLI를 발명하지 않는다. SSH가 필요한 외부 절차는 작업 PC 변수를 자동 전달한다고 가정하지 않고 positional argument와 원격 host/account/scope 재검증을 사용하며 `eval`은 금지한다.

**대상 Pod·current owner 자동 등록**

일반 container request 합, init container의 자원별 최대와 Pod overhead가 실질 요구량이다. limit은 scale-up 필요량이 아니다. `matchExpressions` owner selector는 문자열로 축약하지 않으며 이 문서의 자동 replacement 경로를 사용하지 않는다.

**모든 후속 Kubernetes 명령이 호출하는 fresh target 함수**

`fresh_target(){ local p o host uid ouid node cids; [ "$AUTO_READY" = true ] || return 1; host="$(hostname -f)" || return 1; [ "$host" = "$HOST_ID" ] || { printf '%s\n' '중단: 실행 host 변경' >&2; return 1; }; p="$(kubectl get pod "$POD" -n "$NS" -o json)" || return 1; o="$(kubectl get "$OWNER_RESOURCE" "$OWNER_NAME" -n "$NS" -o json)" || return 1; uid="$(jq -r '.metadata.uid' <<<"$p")"; ouid="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid//empty' <<<"$p")"; node="$(jq -r '.spec.nodeName//"UNBOUND"' <<<"$p")"; cids="$(jq -r '[.status.initContainerStatuses[]?.containerID//empty,.status.containerStatuses[]?.containerID//empty]|sort|join(",")|if .=="" then "NONE" else . end' <<<"$p")"; if [ "$uid" != "$POD_UID" ] || [ "$ouid" != "$OWNER_UID" ] || [ "$(jq -r '.metadata.uid' <<<"$o")" != "$OWNER_UID" ] || [ "$node" != "$POD_NODE" ] || [ "$cids" != "$CONTAINER_IDS" ]; then printf '%s\n' '중단: Pod/current owner/containerID/node identity 변경; 전체 재수집' >&2; return 1; fi; POD_RV="$(jq -r '.metadata.resourceVersion' <<<"$p")"; export POD_RV; }`

### autoscaler Pod·container 선택과 자동 identity

외부 autoscaler면 아래 Kubernetes log 경로를 사용하지 않고 승인된 감사 시스템에서 동일 필드를 확인한다. 이 문서의 자동 gate는 `in-cluster`만 지원한다.

`AS_NS='PASTE_LIMIT_AUTOSCALER_NAMESPACE_HERE'; AS_POD='PASTE_LIMIT_AUTOSCALER_POD_HERE'; AS_CONTAINER='PASTE_LIMIT_AUTOSCALER_CONTAINER_HERE'; AS_SELECTION_READY=false; if [ "$AUTO_READY" != true ] || ! fresh_target; then printf '%s\n' '중단: fresh target 실패' >&2; elif [ -z "$AS_NS" ] || [ -z "$AS_POD" ] || [ -z "$AS_CONTAINER" ]; then printf '%s\n' '중단: in-cluster autoscaler 선택값 필요' >&2; else AS_SELECTION_READY=true; export AS_NS AS_POD AS_CONTAINER; printf 'AS_SELECTION_READY=true ns=%s pod=%s container=%s\n' "$AS_NS" "$AS_POD" "$AS_CONTAINER"; fi`

`unset AS_READY WINDOW_READY EVENT_READY LOG_READY LIMIT_READY CHANGE_READY AS_UID AS_RV AS_OWNER_KIND AS_OWNER_RESOURCE AS_OWNER_NAME AS_OWNER_UID AS_NODE AS_NODE_UID AS_CONTAINER_ID; AS_READY=false; if [ "$AS_SELECTION_READY" != true ] || ! fresh_target; then printf '%s\n' '중단: autoscaler selection/fresh target 실패' >&2; elif AJSON="$(kubectl get pod "$AS_POD" -n "$AS_NS" -o json)" && jq -e --arg c "$AS_CONTAINER" 'any(.spec.containers[]?;.name==$c)' <<<"$AJSON" >/dev/null; then AS_UID="$(jq -r '.metadata.uid' <<<"$AJSON")"; AS_RV="$(jq -r '.metadata.resourceVersion' <<<"$AJSON")"; AS_OWNER_KIND="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].kind//empty' <<<"$AJSON")"; AS_OWNER_NAME="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].name//empty' <<<"$AJSON")"; AS_OWNER_UID="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid//empty' <<<"$AJSON")"; AS_OWNER_RESOURCE="$(printf '%s' "$AS_OWNER_KIND" | tr '[:upper:]' '[:lower:]')"; AS_NODE="$(jq -r '.spec.nodeName//empty' <<<"$AJSON")"; AS_CONTAINER_ID="$(jq -r --arg c "$AS_CONTAINER" '[.status.containerStatuses[]?|select(.name==$c)][0].containerID//empty' <<<"$AJSON")"; if AOJSON="$(kubectl get "$AS_OWNER_RESOURCE" "$AS_OWNER_NAME" -n "$AS_NS" -o json)" && ANJSON="$(kubectl get node "$AS_NODE" -o json)" && [ "$(jq -r '.metadata.uid' <<<"$AOJSON")" = "$AS_OWNER_UID" ]; then AS_NODE_UID="$(jq -r '.metadata.uid' <<<"$ANJSON")"; if [ -n "$AS_UID" ] && [ -n "$AS_CONTAINER_ID" ] && [ -n "$AS_NODE_UID" ]; then AS_READY=true; export AS_UID AS_RV AS_OWNER_KIND AS_OWNER_RESOURCE AS_OWNER_NAME AS_OWNER_UID AS_NODE AS_NODE_UID AS_CONTAINER_ID; printf 'AS_READY=true pod=%s uid=%s owner=%s/%s[%s] container=%s[%s] node=%s[%s]\n' "$AS_POD" "$AS_UID" "$AS_OWNER_KIND" "$AS_OWNER_NAME" "$AS_OWNER_UID" "$AS_CONTAINER" "$AS_CONTAINER_ID" "$AS_NODE" "$AS_NODE_UID"; else printf '%s\n' '중단: autoscaler containerID/node identity 불완전' >&2; fi; else printf '%s\n' '중단: autoscaler current owner 또는 Node 조회 실패' >&2; fi; else printf '%s\n' '중단: autoscaler Pod/container 조회 실패' >&2; fi`

`fresh_all(){ local a ao an; fresh_target || return 1; [ "$AS_READY" = true ] || return 1; a="$(kubectl get pod "$AS_POD" -n "$AS_NS" -o json)" || return 1; ao="$(kubectl get "$AS_OWNER_RESOURCE" "$AS_OWNER_NAME" -n "$AS_NS" -o json)" || return 1; an="$(kubectl get node "$AS_NODE" -o json)" || return 1; if [ "$(jq -r '.metadata.uid' <<<"$a")" != "$AS_UID" ] || [ "$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid//empty' <<<"$a")" != "$AS_OWNER_UID" ] || [ "$(jq -r '.metadata.uid' <<<"$ao")" != "$AS_OWNER_UID" ] || [ "$(jq -r '.metadata.uid' <<<"$an")" != "$AS_NODE_UID" ] || [ "$(jq -r '.spec.nodeName//empty' <<<"$a")" != "$AS_NODE" ] || [ "$(jq -r --arg c "$AS_CONTAINER" '[.status.containerStatuses[]?|select(.name==$c)][0].containerID//empty' <<<"$a")" != "$AS_CONTAINER_ID" ]; then printf '%s\n' '중단: autoscaler current owner/containerID/node identity 변경' >&2; return 1; fi; AS_RV="$(jq -r '.metadata.resourceVersion' <<<"$a")"; export AS_RV; }`

### UTC evidence window와 로그 정규식

`START_RFC3339='PASTE_LIMIT_START_UTC_HERE'; END_RFC3339='PASTE_LIMIT_END_UTC_HERE'; LOG_RE='PASTE_LIMIT_VERIFIED_LOG_REGEX_HERE'; WINDOW_READY=false; RFC3339_RE='^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$'; grep -E "$LOG_RE" </dev/null >/dev/null 2>&1; REGEX_RC=$?; if [ "$AS_READY" != true ] || ! fresh_all; then printf '%s\n' '중단: fresh identity 실패' >&2; elif [[ ! "$START_RFC3339" =~ $RFC3339_RE ]] || [[ ! "$END_RFC3339" =~ $RFC3339_RE ]] || [ -z "$LOG_RE" ] || [ "$REGEX_RC" -eq 2 ]; then printf '%s\n' '중단: canonical UTC 또는 regex parser 검증 실패' >&2; elif START_EPOCH="$(date -u -d "$START_RFC3339" +%s 2>/dev/null)" && END_EPOCH="$(date -u -d "$END_RFC3339" +%s 2>/dev/null)" && [ "$(date -u -d "@$START_EPOCH" '+%Y-%m-%dT%H:%M:%SZ')" = "$START_RFC3339" ] && [ "$(date -u -d "@$END_EPOCH" '+%Y-%m-%dT%H:%M:%SZ')" = "$END_RFC3339" ] && [ "$START_EPOCH" -lt "$END_EPOCH" ] && [ "$END_EPOCH" -le "$(date -u +%s)" ]; then WINDOW_READY=true; export START_RFC3339 END_RFC3339 START_EPOCH END_EPOCH LOG_RE; printf 'WINDOW_READY=true %s..%s regex=%s\n' "$START_RFC3339" "$END_RFC3339" "$LOG_RE"; else printf '%s\n' '중단: time round-trip/start<end/미래 상한 실패' >&2; fi`

## 읽기 전용 진단

**UID·시간 제한 FailedScheduling**

`EVENT_READY=false; if [ "$WINDOW_READY" != true ] || ! fresh_all; then printf '%s\n' '중단: 최신 window/identity 실패' >&2; elif EVENTS_JSON="$(kubectl get events -n "$NS" --field-selector "involvedObject.kind=Pod,involvedObject.name=$POD,involvedObject.uid=$POD_UID,reason=FailedScheduling" -o json)"; then EVENT_ROWS="$(jq -r --argjson s "$START_EPOCH" --argjson e "$END_EPOCH" '.items[]|(.eventTime//.series.lastObservedTime//.lastTimestamp//.firstTimestamp) as $raw|($raw|strings|sub("\\.[0-9]+Z$";"Z")|fromdateiso8601?) as $t|select($t!=null and $t>=$s and $t<=$e)|[.metadata.uid,(.count//1),$raw,.message]|@tsv' <<<"$EVENTS_JSON")"; FILTER_RC=$?; if [ "$FILTER_RC" -eq 0 ] && [ -n "$EVENT_ROWS" ]; then EVENT_READY=true; printf '%s\n' "$EVENT_ROWS"; printf 'EVENT_READY=true pod_uid=%s\n' "$POD_UID"; elif [ "$FILTER_RC" -eq 0 ]; then printf '%s\n' '중단: window 안 Event 없음; 정상으로 해석 금지' >&2; else printf '중단: jq filter rc=%s\n' "$FILTER_RC" >&2; fi; else PRODUCER_RC=$?; printf '중단: Event producer rc=%s; filter 미실행\n' "$PRODUCER_RC" >&2; fi`

Event message에 taint·affinity·topology·PVC 원인이 섞이면 공급 한계로 확정하지 않는다.

**autoscaler log producer·시간 filter·regex filter 분리**

`LOG_READY=false; if [ "$EVENT_READY" != true ] || ! fresh_all; then printf '%s\n' '중단: 최신 Event/identity 실패' >&2; elif RAW_LOG="$(kubectl logs "$AS_POD" -n "$AS_NS" -c "$AS_CONTAINER" --timestamps --since-time="$START_RFC3339" --tail=500)"; then WINDOW_LOG="$(awk -v s="$START_RFC3339" -v e="$END_RFC3339" '$1>=s && $1<=e' <<<"$RAW_LOG")"; AWK_RC=$?; if [ "$AWK_RC" -ne 0 ]; then printf '중단: time filter rc=%s\n' "$AWK_RC" >&2; else MATCHED="$(grep -E -i -C 3 -- "$LOG_RE" <<<"$WINDOW_LOG")"; GREP_RC=$?; if [ "$GREP_RC" -eq 0 ]; then LOG_READY=true; printf '%s\n' "$MATCHED"; printf 'LOG_READY=true pod_uid=%s autoscaler_uid=%s\n' "$POD_UID" "$AS_UID"; elif [ "$GREP_RC" -eq 1 ]; then printf '%s\n' '필터 no-match: watch 범위·schema·시간 재확인' >&2; else printf '중단: grep filter rc=%s\n' "$GREP_RC" >&2; fi; fi; else PRODUCER_RC=$?; printf '중단: log producer rc=%s\n' "$PRODUCER_RC" >&2; fi`

`pods/log`에는 대상 namespace의 `get`, Event에는 `events list`, Pod·owner·Node에는 해당 resource의 `get` 권한이 필요하다. `auth can-i=yes`는 실제 요청 성공을 보장하지 않는다. audit가 필요하면 승인된 경로에서 단일 auditID의 timestamp·verb·resource/subresource·namespace/name·requestURI·status와 current UID 관계만 확인하고 actor 상세·token은 저장하지 않는다.

## 공급 한계 evidence 연결

승인된 provider 결과에서 credential·전체 응답이 아닌 식별자·scope·수치·시각만 가져온다. `LIMIT_KIND`는 `max`, `quota`, `capacity` 중 하나다.

`NODE_GROUP_ID='PASTE_LIMIT_NODE_GROUP_ID_HERE'; NODE_GROUP_SCOPE='PASTE_LIMIT_NODE_GROUP_SCOPE_HERE'; LIMIT_KIND='PASTE_LIMIT_KIND_HERE'; LIMIT_EVIDENCE_ID='PASTE_LIMIT_EVIDENCE_ID_HERE'; PROVIDER_PROCEDURE_ID='PASTE_LIMIT_PROVIDER_PROCEDURE_ID_HERE'; LIMIT_READY=false; if [ "$LOG_READY" != true ] || ! fresh_all; then printf '%s\n' '중단: fresh log evidence가 아니다.' >&2; elif [[ ! "$LIMIT_KIND" =~ ^(max|quota|capacity)$ ]] || [ -z "$NODE_GROUP_ID" ] || [ -z "$NODE_GROUP_SCOPE" ] || [ -z "$LIMIT_EVIDENCE_ID" ] || [ -z "$PROVIDER_PROCEDURE_ID" ]; then printf '%s\n' '중단: group/scope/한계/evidence/procedure 불완전' >&2; else LIMIT_READY=true; export NODE_GROUP_ID NODE_GROUP_SCOPE LIMIT_KIND LIMIT_EVIDENCE_ID PROVIDER_PROCEDURE_ID; printf 'LIMIT_READY=true group=%s scope=%s kind=%s evidence=%s procedure=%s\n' "$NODE_GROUP_ID" "$NODE_GROUP_SCOPE" "$LIMIT_KIND" "$LIMIT_EVIDENCE_ID" "$PROVIDER_PROCEDURE_ID"; fi`

- `max`: 선택된 group의 현재 크기·max와 scale 요청량이 같은 scope·시각에서 일치해야 한다.
- `quota`: account/project/subscription·region, quota 항목, 현재 사용량·limit·요청량이 일치해야 한다. Kubernetes `ResourceQuota`와 혼동하지 않는다.
- `capacity`: 실패 zone/type/pool과 요청량, PVC/PV topology 및 대체 architecture·비용·tenant 영향을 함께 확인한다.
- provider 응답 부재, scope 불일치 또는 log no-match는 한계 증거가 아니다.

## 상태 변경 경계

이 문서는 max·quota·zone/type·Pod request를 변경하거나 Pod를 삭제하는 명령을 제공하지 않는다. 변경은 current identity, provider evidence, 비용·tenant·architecture·volume topology, availability/quorum, 멱등성, 승인된 rollback이 갖춰진 기존 관리면에서만 수행한다.

`APPROVAL_ID='PASTE_LIMIT_APPROVAL_ID_HERE'; SAFETY_EVIDENCE_ID='PASTE_LIMIT_SAFETY_EVIDENCE_ID_HERE'; ROLLBACK_REF='PASTE_LIMIT_ROLLBACK_REF_HERE'; IDEMPOTENCE_RECORD='PASTE_LIMIT_IDEMPOTENCE_RECORD_HERE'; CHANGE_RFC3339='PASTE_LIMIT_CHANGE_UTC_HERE'; CHANGE_READY=false; if [ "$LIMIT_READY" != true ] || ! fresh_all; then printf '%s\n' '중단: fresh limit evidence가 아니다.' >&2; elif [ -z "$APPROVAL_ID" ] || [ -z "$SAFETY_EVIDENCE_ID" ] || [ -z "$ROLLBACK_REF" ] || [ -z "$IDEMPOTENCE_RECORD" ]; then printf '%s\n' '중단: approval/safety/rollback/idempotence 기록 필요' >&2; elif [[ ! "$CHANGE_RFC3339" =~ $RFC3339_RE ]] || ! CHANGE_EPOCH="$(date -u -d "$CHANGE_RFC3339" +%s 2>/dev/null)" || [ "$(date -u -d "@$CHANGE_EPOCH" '+%Y-%m-%dT%H:%M:%SZ')" != "$CHANGE_RFC3339" ] || [ "$CHANGE_EPOCH" -lt "$END_EPOCH" ] || [ "$CHANGE_EPOCH" -gt "$(date -u +%s)" ]; then printf '%s\n' '중단: 변경 시각 canonical/round-trip/evidence 이후 조건 실패' >&2; else CHANGE_READY=true; export APPROVAL_ID SAFETY_EVIDENCE_ID ROLLBACK_REF IDEMPOTENCE_RECORD CHANGE_RFC3339 CHANGE_EPOCH; printf 'CHANGE_READY=true group=%s kind=%s approval=%s change=%s rollback=%s\n' "$NODE_GROUP_ID" "$LIMIT_KIND" "$APPROVAL_ID" "$CHANGE_RFC3339" "$ROLLBACK_REF"; fi`

이 gate는 mutation이 아니다. 외부 변경 직전에도 Pod·owner·autoscaler identity, 현재 provider 값, 승인 범위와 이미 적용됐는지 여부를 다시 확인한다. 새 비용/tenant 한도 초과, 잘못된 zone/type, Ready 감소, 동일 provider 오류면 중단하고 검증된 `ROLLBACK_REF` 절차로만 복귀한다.

## replacement 재등록과 정상화

capacity owner가 확인한 새 Node와 controller selector에서 확인한 사후 Pod를 입력한다. 기존 Pending Pod가 그대로 bind됐다면 같은 이름·UID를 허용하고, 실제 교체된 경우에만 새 UID와 변경 시각 이후 creation을 요구한다.

`NEW_POD='PASTE_LIMIT_REPLACEMENT_POD_HERE'; NEW_NODE='PASTE_LIMIT_REGISTERED_NODE_HERE'; unset REPLACEMENT_READY NORMAL_READY NEW_POD_UID NEW_POD_RV NEW_NODE_UID NEW_CREATED NEW_CONTAINER_IDS POST_LIFECYCLE; REPLACEMENT_READY=false; if [ "$CHANGE_READY" != true ]; then printf '%s\n' '중단: CHANGE_READY가 아니다.' >&2; elif CANDIDATES="$(kubectl get pods -n "$NS" -l "$OWNER_SELECTOR" -o json)" && [ "$(jq --arg p "$NEW_POD" '[.items[]|select(.metadata.name==$p)]|length' <<<"$CANDIDATES")" -eq 1 ] && NJSON="$(kubectl get node "$NEW_NODE" -o json)" && OJSON="$(kubectl get "$OWNER_RESOURCE" "$OWNER_NAME" -n "$NS" -o json)"; then NP="$(jq -c --arg p "$NEW_POD" '.items[]|select(.metadata.name==$p)' <<<"$CANDIDATES")"; NEW_POD_UID="$(jq -r '.metadata.uid' <<<"$NP")"; NEW_POD_RV="$(jq -r '.metadata.resourceVersion' <<<"$NP")"; NEW_NODE_UID="$(jq -r '.metadata.uid' <<<"$NJSON")"; NEW_CREATED="$(jq -r '.metadata.creationTimestamp' <<<"$NP")"; NEW_CONTAINER_IDS="$(jq -r '[.status.initContainerStatuses[]?.containerID//empty,.status.containerStatuses[]?.containerID//empty]|sort|join(",")' <<<"$NP")"; POST_LIFECYCLE=''; if [ "$NEW_POD" = "$POD" ] && [ "$NEW_POD_UID" = "$POD_UID" ]; then POST_LIFECYCLE='same-pod'; elif [ "$NEW_POD_UID" != "$POD_UID" ] && [ "$(date -u -d "$NEW_CREATED" +%s)" -ge "$CHANGE_EPOCH" ]; then POST_LIFECYCLE='replacement'; fi; if [ "$(jq -r '.metadata.uid' <<<"$OJSON")" = "$OWNER_UID" ] && [ "$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid//empty' <<<"$NP")" = "$OWNER_UID" ] && [ "$(jq -r '.spec.nodeName//empty' <<<"$NP")" = "$NEW_NODE" ] && [ -n "$NEW_NODE_UID" ] && [ -n "$POST_LIFECYCLE" ]; then REPLACEMENT_READY=true; export NEW_POD NEW_POD_UID NEW_POD_RV NEW_NODE NEW_NODE_UID NEW_CREATED NEW_CONTAINER_IDS POST_LIFECYCLE; printf 'REPLACEMENT_READY=true lifecycle=%s pod=%s[%s] rv=%s node=%s[%s] created=%s containerIDs=%s\n' "$POST_LIFECYCLE" "$NEW_POD" "$NEW_POD_UID" "$NEW_POD_RV" "$NEW_NODE" "$NEW_NODE_UID" "$NEW_CREATED" "${NEW_CONTAINER_IDS:--}"; else printf '%s\n' '중단: selector/current owner/node/lifecycle/change-time relation 불일치' >&2; fi; else printf '%s\n' '중단: post-change Pod 단일 후보 또는 current Node/owner 조회 실패' >&2; fi`

`NORMAL_READY=false; if [ "$REPLACEMENT_READY" != true ] || ! NP="$(kubectl get pod "$NEW_POD" -n "$NS" -o json)" || ! NN="$(kubectl get node "$NEW_NODE" -o json)" || ! OO="$(kubectl get "$OWNER_RESOURCE" "$OWNER_NAME" -n "$NS" -o json)"; then printf '%s\n' '중단: 정상화 producer 실패' >&2; elif [ "$(jq -r '.metadata.uid' <<<"$NP")" != "$NEW_POD_UID" ] || [ "$(jq -r '.metadata.uid' <<<"$NN")" != "$NEW_NODE_UID" ] || [ "$(jq -r '.metadata.uid' <<<"$OO")" != "$OWNER_UID" ] || [ "$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid//empty' <<<"$NP")" != "$OWNER_UID" ] || [ "$(jq -r '.spec.nodeName//empty' <<<"$NP")" != "$NEW_NODE" ]; then printf '%s\n' '중단: replacement current identity 불일치' >&2; elif [ "$(jq -r '[.status.conditions[]?|select(.type=="Ready")][0].status//""' <<<"$NN")" = True ] && [ "$(jq -r '.status.phase' <<<"$NP")" = Running ] && jq -e '(.status.containerStatuses|length)>0 and all(.status.containerStatuses[];.ready==true and (.containerID//"")!="")' <<<"$NP" >/dev/null; then NEW_CONTAINER_IDS="$(jq -r '[.status.containerStatuses[].containerID]|sort|join(",")' <<<"$NP")"; NORMAL_READY=true; printf 'NORMAL_READY=true pod=%s uid=%s node=%s[%s] containerIDs=%s\n' "$NEW_POD" "$NEW_POD_UID" "$NEW_NODE" "$NEW_NODE_UID" "$NEW_CONTAINER_IDS"; else printf '%s\n' '중단: Node Ready 또는 Pod/container Running/Ready 미수렴' >&2; fi`

`NORMAL_READY=true`는 Kubernetes 객체 경로만 통과했다. owner desired/Ready, 같은 오류 Event와 provider 실패 재발 없음, 조건부 Service→EndpointSlice port→새 Pod UID 관계, 실제 요청·핵심 기능 또는 Job 산출물을 승인된 관찰 창 동안 확인해야 정상화다. Node 수 증가나 Pod `Running`만으로 닫지 않는다.

## Prometheus 빠른 관측

metric과 label이 실제 존재할 때 생성 명령의 출력만 Prometheus/Grafana에 붙인다.

`if [ "$EVENT_READY" != true ] || ! fresh_all; then printf '%s\n' '중단: latest Event identity gate 실패' >&2; else printf 'kube_pod_info{namespace="%s",pod="%s",uid="%s"}\n' "$NS" "$POD" "$POD_UID"; printf 'kube_pod_status_unschedulable{namespace="%s",pod="%s"} == 1\n' "$NS" "$POD"; fi`

빈 결과는 0이나 정상 상태가 아니다. metric 부재, label allowlist, cluster/namespace 범위와 scrape 실패를 구분한다. 이 질의는 node group max, provider quota·capacity 또는 autoscaler decision을 증명하지 않는다.

## 증거 보존과 환경별 미검증

실행 host, UTC window, Pod/current owner/autoscaler/current autoscaler owner/container/Node UID·resourceVersion·관계, request·placement·PVC 참조, Event와 producer/filter rc, 비식별 로그, node group·scope·한계 evidence, 승인·safety·idempotence·rollback·변경 시각, replacement와 실제 기능을 보존한다. Secret·token·credential·개인키·전체 provider 응답·전체 로그는 저장하지 않는다.

autoscaler 제품·log schema, provider inventory/API, quota·capacity scope, PDB/quorum, Service endpoint와 실제 Kubernetes 1.26 결과는 환경별 미검증이다. 이 문서 작성 중 실제 인프라 명령은 실행하지 않았다.
