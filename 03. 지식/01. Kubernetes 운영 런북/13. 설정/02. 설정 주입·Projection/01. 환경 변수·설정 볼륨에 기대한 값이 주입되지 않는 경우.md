# 환경 변수·설정 볼륨에 기대한 값이 주입되지 않는 경우

## 입력값과 빠른 수집

명령은 승인된 작업 PC 또는 bastion의 같은 Bash에서 실행한다. 경로는 `kubectl → API server → 현재 Pod UID → 선택 container instance → env 또는 volume projection → application`이다. API spec 성공은 process 환경·파일 내용·application 사용 성공을 보장하지 않는다. 값, 파일 원문, Secret, token과 내부 주소는 출력하지 않는다.

| 구분 | 값 | 출처와 gate |
|---|---|---|
| 최초 입력 | `NS`, `POD` | exact-name Pod 조회가 성공해야 한다. |
| API 확인 자동값 | `POD_UID`, `POD_RV`, `NODE`, `OWNER_*`, 선택 container의 `CONTAINER_ID`, Pod/volume relation | 현재 API 응답에서만 등록한다. |
| 계산 후보값 | `POD_RE`, `CONTAINER_RE`, file metadata query | escape와 현재 identity 대조 후에만 사용한다. |
| 운영자 선택값 | `CONTAINER`, `INJECTION_KIND`, `FILE_PATH`, 증거 시간 | 관계 출력에서 고른다. `INJECTION_KIND`는 `env` 또는 `volume`이다. |

standalone Pod에 owner가 새로 붙는 경우와 선택한 container kind가 바뀌는 경우까지 막도록 helper를 다음 정의로 갱신한다.

`refresh_current(){ CURRENT_READY=false; CURRENT_POD_JSON="$(kubectl get pod "$POD" -n "$NS" -o json 2>/dev/null)" || { unset AUTO_READY SELECTION_READY EVIDENCE_TIME_READY FILE_READY LOG_READY QUERY_READY CHANGE_READY APPLY_READY ROLLBACK_READY NORMAL_READY; printf '%s\n' '중단: Pod 재조회 실패, downstream 무효' >&2; return 1; }; CURRENT_UID="$(jq -r '.metadata.uid // empty' <<<"$CURRENT_POD_JSON")"; CURRENT_RV="$(jq -r '.metadata.resourceVersion // empty' <<<"$CURRENT_POD_JSON")"; CURRENT_OWNER_UID="$(jq -r '[.metadata.ownerReferences[]? | select(.controller==true)] | if length==1 then .[0].uid else empty end' <<<"$CURRENT_POD_JSON")"; CURRENT_CONTAINER_ID=''; if [ "$SELECTION_READY" = true ]; then CURRENT_CONTAINER_ID="$(jq -r --arg c "$CONTAINER" --arg k "$CONTAINER_KIND" 'if $k=="regular" then [.status.containerStatuses[]? | select(.name==$c)] else [.status.initContainerStatuses[]? | select(.name==$c)] end | if length==1 then .[0].containerID // empty else empty end' <<<"$CURRENT_POD_JSON")"; fi; if [ "$AUTO_READY" = true ] && [ "$CURRENT_UID" = "$POD_UID" ] && [ -n "$CURRENT_RV" ] && { { [ -z "$OWNER_UID" ] && [ -z "$CURRENT_OWNER_UID" ]; } || { [ -n "$OWNER_UID" ] && [ "$CURRENT_OWNER_UID" = "$OWNER_UID" ]; }; } && { [ "$SELECTION_READY" != true ] || [ "$CURRENT_CONTAINER_ID" = "$CONTAINER_ID" ]; }; then CURRENT_READY=true; printf 'CURRENT_READY=true uid=%s rv=%s owner_uid=%s container_kind=%s container_id=%s\n' "$CURRENT_UID" "$CURRENT_RV" "${CURRENT_OWNER_UID:--}" "${CONTAINER_KIND:--}" "${CURRENT_CONTAINER_ID:--}"; return 0; fi; unset AUTO_READY SELECTION_READY EVIDENCE_TIME_READY FILE_READY LOG_READY QUERY_READY CHANGE_READY APPLY_READY ROLLBACK_READY NORMAL_READY; printf '%s\n' '중단: Pod/owner/container lifecycle 변경, 첫 수집부터 반복' >&2; return 1; }`

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | 새 Pod의 env·volume·projected·downwardAPI 주입 결과가 승인 선언과 다른 경우 |
| 적용 | current Pod UID와 exact container instance, source→mount relation, 비민감 기능/metadata를 대조할 수 있는 경우 |
| 제외 | object 이름·key가 다르면 [[01. ConfigMap·Secret의 참조 이름·key·값이 기대와 다른 경우]], object 부재는 [[26. Pod가 ConfigMap·Secret을 찾지 못하는 경우]], 실행 중 갱신은 [[01. ConfigMap·Secret 변경 후 워크로드가 이전 설정을 계속 사용하는 경우]] |

검색 문자열: `envFrom`, `valueFrom`, `fieldRef`, `resourceFieldRef`, `projected`, `volumeMounts`, `subPath`, `CreateContainerConfigError`, `FailedMount`.

## 관측별 진단 경로

| 지금 보이는 것 | 먼저 볼 것 | 다음 경계 | 상세 절 |
|---|---|---|---|
| env 이름/source 불일치 | container와 source relation | owner template·object key | [[#1. container와 env 주입]] |
| 파일 없음·mode/path 불일치 | mount→volume→source relation | projection·kubelet·image | [[#2. volume projection과 파일 metadata]] |

### 1. container와 env 주입

**선택값 등록**

`CONTAINER='PASTE_CONTAINER_HERE'; CONTAINER_KIND='PASTE_REGULAR_OR_INIT_HERE'; INJECTION_KIND='PASTE_ENV_OR_VOLUME_HERE'; FILE_PATH='PASTE_FILE_PATH_OR_DASH_HERE'`

`unset SELECTION_READY CURRENT_READY EVIDENCE_TIME_READY FILE_READY LOG_READY QUERY_READY CHANGE_READY APPLY_READY ROLLBACK_READY NORMAL_READY CONTAINER_ID CONTAINER_STARTED CONTAINER_RE POD_RE; SELECTION_READY=false; if [ "$AUTO_READY" != true ] || { [ "$INJECTION_KIND" != env ] && [ "$INJECTION_KIND" != volume ]; }; then printf '%s\n' '중단: AUTO_READY와 injection kind 확인' >&2; elif STATUS_COUNT="$(jq -r --arg c "$CONTAINER" '[.status.initContainerStatuses[]?,.status.containerStatuses[]? | select(.name==$c)] | length' <<<"$POD_JSON")" && [ "$STATUS_COUNT" -eq 1 ]; then CONTAINER_ID="$(jq -r --arg c "$CONTAINER" '[.status.initContainerStatuses[]?,.status.containerStatuses[]? | select(.name==$c)][0].containerID // empty' <<<"$POD_JSON")"; CONTAINER_STARTED="$(jq -r --arg c "$CONTAINER" '[.status.initContainerStatuses[]?,.status.containerStatuses[]? | select(.name==$c)][0].state.running.startedAt // "-"' <<<"$POD_JSON")"; if [ -n "$CONTAINER_ID" ] && { [ "$INJECTION_KIND" = env ] || [ "$FILE_PATH" != - ]; }; then CONTAINER_RE="$(printf '%s' "$CONTAINER" | sed 's/[][\\.^$*+?(){}|]/\\&/g')"; POD_RE="$(printf '%s' "$POD" | sed 's/[][\\.^$*+?(){}|]/\\&/g')"; SELECTION_READY=true; printf 'SELECTION_READY=true container=%s id=%s started=%s kind=%s path=%s\n' "$CONTAINER" "$CONTAINER_ID" "$CONTAINER_STARTED" "$INJECTION_KIND" "$FILE_PATH"; else unset CONTAINER_ID CONTAINER_STARTED CONTAINER_RE POD_RE; printf '%s\n' '중단: 현재 container ID 또는 file path 없음' >&2; fi; else printf '%s\n' '중단: container status가 정확히 하나가 아님' >&2; fi`

`CONTAINER_KIND_READY=false; if [ "$SELECTION_READY" != true ] || { [ "$CONTAINER_KIND" != regular ] && [ "$CONTAINER_KIND" != init ]; }; then SELECTION_READY=false; printf '%s\n' '중단: regular|init container kind를 확인한다.' >&2; else KIND_ID="$(jq -r --arg c "$CONTAINER" --arg k "$CONTAINER_KIND" 'if $k=="regular" then [.status.containerStatuses[]? | select(.name==$c)] else [.status.initContainerStatuses[]? | select(.name==$c)] end | if length==1 then .[0].containerID // empty else empty end' <<<"$POD_JSON")"; if [ "$KIND_ID" = "$CONTAINER_ID" ]; then CONTAINER_KIND_READY=true; export CONTAINER_KIND; printf 'CONTAINER_KIND_READY=true kind=%s name=%s id=%s\n' "$CONTAINER_KIND" "$CONTAINER" "$CONTAINER_ID"; else SELECTION_READY=false; unset CONTAINER_ID CONTAINER_STARTED CONTAINER_RE POD_RE; printf '%s\n' '중단: container kind/name/current containerID 관계 불일치' >&2; fi; fi`

env는 Pod 생성 시 process snapshot이다. 실행 중 object 변경으로 자동 갱신되지 않는다. 실제 값 전체를 `printenv`로 출력하지 않는다. 승인된 image에 shell이 있을 때 이름 존재만 검사한다.

`if [ "$SELECTION_READY" != true ] || [ "$INJECTION_KIND" != env ] || ! refresh_current; then printf '%s\n' '중단: 최신 env container identity가 필요하다.' >&2; else kubectl exec "$POD" -n "$NS" -c "$CONTAINER" -- sh -c 'command -v printenv >/dev/null 2>&1 || exit 3; printenv | cut -d= -f1 | sort -u'; fi`

exit 3은 image 도구 부재이며 env 부재가 아니다. 값 확인은 application owner의 비민감 schema/version diagnostic으로 대체한다.

### 2. volume projection과 파일 metadata

`if [ "$SELECTION_READY" != true ] || [ "$INJECTION_KIND" != volume ] || ! refresh_current; then printf '%s\n' '중단: 최신 volume container identity가 필요하다.' >&2; else jq -r --arg c "$CONTAINER" '.spec.initContainers[]?,.spec.containers[]? | select(.name==$c) | .volumeMounts[]? | "container=\($c) mount=\(.name) path=\(.mountPath) subPath=\(.subPath // "") readOnly=\(.readOnly // false)"' <<<"$CURRENT_POD_JSON"; jq -r '.spec.volumes[]? | "volume=\(.name) source="+(if .configMap then "ConfigMap/\(.configMap.name)" elif .secret then "Secret/\(.secret.secretName)" elif .projected then "projected" elif .downwardAPI then "downwardAPI" else "other" end)+" items="+([(.configMap.items[]?,.secret.items[]?) | "\(.key):\(.path)"]|join(","))' <<<"$CURRENT_POD_JSON"; fi`

mount name→volume name→source와 projected의 **모든 source**를 대조한다. `subPath`는 이후 projection 갱신을 받지 않는다. 파일 metadata 조회는 path를 shell 코드에 삽입하지 않고 positional argument로 전달한다.

`FILE_READY=false; if [ "$SELECTION_READY" != true ] || [ "$INJECTION_KIND" != volume ] || ! refresh_current; then printf '%s\n' '중단: 최신 volume identity가 필요하다.' >&2; elif kubectl exec "$POD" -n "$NS" -c "$CONTAINER" -- sh -c 'command -v stat >/dev/null 2>&1 || exit 3; test -e "$1" || exit 4; stat -c "%F %a %s %Y %n" -- "$1"' sh "$FILE_PATH"; then FILE_READY=true; printf 'FILE_READY=true pod_uid=%s container_id=%s\n' "$POD_UID" "$CONTAINER_ID"; else RC=$?; printf 'FILE_READY=false rc=%s (3=tool 없음, 4=file 없음, 그 외 exec/API 실패)\n' "$RC" >&2; fi`

파일 존재는 내용·형식·process reload를 보장하지 않는다. Secret 파일을 hash·cat하지 않는다.

### 3. downwardAPI

`if [ "$SELECTION_READY" != true ] || ! refresh_current; then printf '%s\n' '중단: 최신 container identity가 필요하다.' >&2; else jq -r --arg c "$CONTAINER" '.spec.initContainers[]?,.spec.containers[]? | select(.name==$c) | .env[]? | if .valueFrom.fieldRef then "env=\(.name) fieldPath=\(.valueFrom.fieldRef.fieldPath)" elif .valueFrom.resourceFieldRef then "env=\(.name) resource=\(.valueFrom.resourceFieldRef.resource) sourceContainer=\(.valueFrom.resourceFieldRef.containerName // $c) divisor=\(.valueFrom.resourceFieldRef.divisor // "1")" else empty end' <<<"$CURRENT_POD_JSON"; jq -r '"pod_uid=\(.metadata.uid) namespace=\(.metadata.namespace) node=\(.spec.nodeName // "-") podIP_present=\(.status.podIP != null)"' <<<"$CURRENT_POD_JSON"; fi`

내부 주소 값은 출력하지 않는다. fieldPath와 현재 metadata/status, resourceFieldRef의 정확한 container·request/limit을 대조한다.

### 4. 기능 증거와 producer-filter 경계

**엄격한 증거 창 입력**

`START_RFC3339='PASTE_EVIDENCE_START_UTC_HERE'; END_RFC3339='PASTE_EVIDENCE_END_UTC_HERE'`

`unset EVIDENCE_TIME_READY LOG_READY START_EPOCH END_EPOCH; EVIDENCE_TIME_READY=false; if [[ "$START_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ && "$END_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && START_EPOCH="$(date -u -d "$START_RFC3339" +%s 2>/dev/null)" && END_EPOCH="$(date -u -d "$END_RFC3339" +%s 2>/dev/null)" && [ "$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ')" = "$START_RFC3339" ] && [ "$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ')" = "$END_RFC3339" ] && [ "$START_EPOCH" -lt "$END_EPOCH" ]; then EVIDENCE_TIME_READY=true; printf 'EVIDENCE_TIME_READY=true start=%s end=%s\n' "$START_RFC3339" "$END_RFC3339"; else printf '%s\n' '중단: canonical UTC RFC3339 round-trip/order 실패' >&2; fi`

`LOG_READY=false; PRODUCER_RC=0; FILTER_RC=0; TMP=''; if [ "$EVIDENCE_TIME_READY" != true ] || [ "$SELECTION_READY" != true ] || ! refresh_current; then printf '%s\n' '중단: 최신 identity와 evidence window 필요' >&2; elif TMP="$(mktemp)"; then if kubectl logs "$POD" -n "$NS" -c "$CONTAINER" --timestamps --since-time="$START_RFC3339" --tail=300 >"$TMP"; then PRODUCER_RC=0; else PRODUCER_RC=$?; fi; if [ "$PRODUCER_RC" -ne 0 ]; then printf 'producer_rc=%s filter_rc=not-run\n' "$PRODUCER_RC" >&2; elif grep -E -i -C 3 'config|reload|parse|invalid|permission|denied|failed|error' "$TMP"; then FILTER_RC=0; LOG_READY=true; printf 'producer_rc=0 filter_rc=0 match=true\n'; else FILTER_RC=$?; if [ "$FILTER_RC" -eq 1 ]; then LOG_READY=true; printf 'producer_rc=0 filter_rc=1 match=false (오류 없음 아님)\n'; else printf 'producer_rc=0 filter_rc=%s filter_error=true\n' "$FILTER_RC" >&2; fi; fi; rm -f "$TMP"; else printf '%s\n' '중단: 임시 파일 생성 실패' >&2; fi`

filter 무일치는 오류 부재가 아니다. 같은 창의 승인된 app diagnostic에서 비민감 config version/schema와 실제 요청 결과를 확인한다.

## 변경·rollback gate

live Pod를 patch/delete하거나 값 확인을 위해 재시작하지 않는다. 변경은 승인된 owner manifest로 relation-select replacement 또는 full Pod refresh가 필요한 항목만 수행한다. StatefulSet/quorum, Job side effect와 외부 호출의 멱등성을 먼저 확인한다.

`APPROVAL_ID='PASTE_APPROVAL_ID_HERE'; CHANGE_START='PASTE_CHANGE_START_UTC_HERE'; CHANGE_END='PASTE_CHANGE_END_UTC_HERE'; APPROVED_MANIFEST='PASTE_APPROVED_OWNER_MANIFEST_HERE'; ROLLBACK_MANIFEST='PASTE_ROLLBACK_OWNER_MANIFEST_HERE'; IDEMPOTENCE_EVIDENCE='PASTE_IDEMPOTENCE_EVIDENCE_HERE'; QUORUM_EVIDENCE='PASTE_QUORUM_EVIDENCE_OR_NA_HERE'; CHANGE_INPUT_READY=true`

`unset CHANGE_READY APPLY_READY ROLLBACK_READY NORMAL_READY; CHANGE_READY=false; if [ "$CHANGE_INPUT_READY" != true ] || [ "$SELECTION_READY" != true ] || ! refresh_current; then printf '%s\n' '중단: fresh Pod/container evidence 필요' >&2; elif [[ "$CHANGE_START" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ && "$CHANGE_END" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && CS="$(date -u -d "$CHANGE_START" +%s 2>/dev/null)" && CE="$(date -u -d "$CHANGE_END" +%s 2>/dev/null)" && [ "$(date -u -d "$CHANGE_START" '+%Y-%m-%dT%H:%M:%SZ')" = "$CHANGE_START" ] && [ "$(date -u -d "$CHANGE_END" '+%Y-%m-%dT%H:%M:%SZ')" = "$CHANGE_END" ] && [ "$CS" -lt "$CE" ] && NOW="$(date -u +%s)" && [ "$NOW" -ge "$CS" ] && [ "$NOW" -le "$CE" ] && [ -n "$APPROVAL_ID" ] && [ -r "$APPROVED_MANIFEST" ] && [ -r "$ROLLBACK_MANIFEST" ] && [ -n "$IDEMPOTENCE_EVIDENCE" ] && [ -n "$QUORUM_EVIDENCE" ]; then CHANGE_READY=true; printf 'CHANGE_READY=true approval=%s pod_uid=%s container_id=%s\n' "$APPROVAL_ID" "$POD_UID" "$CONTAINER_ID"; else printf '%s\n' '중단: 승인 창·rollback·멱등성·quorum gate 실패' >&2; fi`

이 런북은 범용 owner manifest를 실행하지 않는다. 별도 승인 mutation 절차가 current owner kind/name/UID·generation·revision, Pod template의 현재 relation, image digest, selector, 가용성·quorum, 승인 창, 멱등성과 rollback revision을 **적용 명령 자체에서** 다시 확인해야 한다. 새 Pod를 만드는 변경이면 기존 Pod 이름·UID·containerID를 정상화 gate로 재사용하지 않는다.

rollback 명령도 제공하지 않는다. 별도 절차는 실제 적용 revision·현재 owner UID·새 Pod UID와 image digest·rollback 원본·승인 창을 직접 검사하고, rollback 뒤 모든 Pod/container/relation과 실제 기능을 첫 단계부터 재등록한다.

## 정상화와 쿼리

정상화는 `실제 기능 → app config version/schema → env 또는 file metadata → current container ID → 새 Pod UID → owner availability` 순서다. 같은 오류·restart·Warning이 승인 관찰 창에 재발하지 않아야 한다. `Running`, file 존재, log filter 무일치, apply 성공은 단독 성공이 아니다.

KSM metric이 실제로 존재할 때만 escaped 완성 PromQL을 생성한다.

`QUERY_READY=false; if [ "$SELECTION_READY" != true ] || ! refresh_current; then printf '%s\n' '중단: 최신 identity 필요' >&2; elif [ -n "$POD_RE" ] && [ -n "$CONTAINER_RE" ]; then QUERY_READY=true; printf 'kube_pod_info{namespace="%s",pod=~"^%s$",uid="%s"}\n' "$NS" "$POD_RE" "$POD_UID"; printf 'kube_pod_container_status_ready{namespace="%s",pod=~"^%s$",uid="%s",container=~"^%s$"}==1\n' "$NS" "$POD_RE" "$POD_UID" "$CONTAINER_RE"; printf 'sum by(namespace,pod,uid,container)(increase(kube_pod_container_status_restarts_total{namespace="%s",pod=~"^%s$",uid="%s",container=~"^%s$"}[15m]))\n' "$NS" "$POD_RE" "$POD_UID" "$CONTAINER_RE"; fi`

빈 query 결과를 Ready, restart 0 또는 주입 성공으로 해석하지 않는다. metric/label 부재, 수집 실패, 범위 오류를 API·기능 증거와 분리한다.
