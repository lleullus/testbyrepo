# 프로세스가 D 상태에서 오래 멈춘 경우

## 입력값과 빠른 수집

연결 경로는 `작업 PC/Bastion Bash → kubectl → API server → Node 객체`와 `같은 host의 ssh → 승인된 endpoint → systemUUID가 일치하는 Linux host → PID namespace → process → syscall/wchan/stack → 실제 mount·device 또는 remote filesystem`이다. SSH 이름과 Node 이름, container PID와 host PID, `wchan` 문자열과 실제 storage 관계를 같은 것으로 보지 않는다.

| 종류 | 변수 | 등록 계약 |
|---|---|---|
| 최초 입력 | `NODE`, `HOST` | exact Node 이름과 옵션 형태가 아닌 승인된 SSH endpoint |
| API 확인 자동값 | `NODE_UID`, `NODE_RV`, `NODE_SYSTEM_UUID` | Node API 성공 후 원격 DMI UUID와 대조 |
| 계산 후보값 | `INSTANCE_RE`, `PID_START_TICKS` | regex escape 또는 `/proc/PID/stat` 계산값이며 재검증 전 stale |
| 운영자 선택값 | `PID`, `START_RFC3339`, `END_RFC3339`, `EXPORTER_INSTANCE` | bounded 목록·장애 증거를 확인한 뒤 각각 한 번 입력 |

### 2. 매 호출 fresh gate와 제한 원격 함수

```bash
fresh_node() {
  local j u r
  [ "$INPUT_READY" = true ] || return 1
  j="$(kubectl get node "$NODE" -o json)" || return 1
  u="$(jq -r '.metadata.uid // empty' <<<"$j")"
  r="$(jq -r '.metadata.resourceVersion // empty' <<<"$j")"
  [ "$u" = "$NODE_UID" ] && [ "$r" = "$NODE_RV" ] || { printf '중단: Node lifecycle 변경 uid=%s rv=%s\n' "$u" "$r" >&2; return 1; }
}
node_read() {
  local op="$1"
  shift
  fresh_node || { printf '%s\n' '중단: fresh Node gate 실패' >&2; return 1; }
  ssh -- "$HOST" bash -s -- "$NODE_SYSTEM_UUID" "$op" "$@" <<'REMOTE'
set -o pipefail
expected=$1; op=$2; shift 2
actual=$(tr '[:upper:]' '[:lower:]' </sys/class/dmi/id/product_uuid 2>/dev/null | tr -d '-') || exit 20
expected=$(printf '%s' "$expected" | tr '[:upper:]' '[:lower:]' | tr -d '-')
[ -n "$actual" ] && [ "$actual" = "$expected" ] || exit 21
pid_identity() {
  pid=$1
  [[ "$pid" =~ ^[1-9][0-9]*$ ]] && [ -r "/proc/$pid/stat" ] || exit 30
  stat=$(cat "/proc/$pid/stat") || exit 31
  rest=${stat##*) }
  set -- $rest
  start=${20}
  ns=$(readlink "/proc/$pid/ns/pid") || exit 32
  ppid=$(sed -n 's/^PPid:[[:space:]]*//p' "/proc/$pid/status") || exit 33
  cgroup=$(tr '\n' ';' <"/proc/$pid/cgroup") || exit 34
  printf '%s\t%s\t%s\t%s\n' "$start" "$ns" "$ppid" "$cgroup"
}
case "$op" in
  identity) printf 'hostname=%s\nsystem_uuid=%s\n' "$(hostname --fqdn 2>/dev/null || hostname)" "$actual" ;;
  list_d) ps -e -o pid=,ppid=,state=,wchan:32,etimes=,comm= | grep -E '^[[:space:]]*[0-9]+[[:space:]]+[0-9]+[[:space:]]+D' | sed -n '1,20p'; status=("${PIPESTATUS[@]}"); [ "${status[0]}" -eq 0 ] || exit "${status[0]}"; { [ "${status[1]}" -eq 0 ] || [ "${status[1]}" -eq 1 ]; } || exit "${status[1]}"; [ "${status[2]}" -eq 0 ] || exit "${status[2]}" ;;
  pid_identity) pid_identity "$1" ;;
  sample) pid=$1; start=$2; ns=$3; for i in 1 2 3; do now=$(pid_identity "$pid") || exit; now_start=${now%%$'\t'*}; [ "$now_start" = "$start" ] || exit 35; now_ns=$(printf '%s' "$now" | cut -f2); [ "$now_ns" = "$ns" ] || exit 36; date -u '+sample-rfc3339=%Y-%m-%dT%H:%M:%SZ'; ps -p "$pid" -o pid=,ppid=,stat=,wchan=,etimes=,lstart=,comm=; sleep 10; done ;;
  detail) pid=$1; start=$2; ns=$3; now=$(pid_identity "$pid") || exit; [ "${now%%$'\t'*}" = "$start" ] || exit 35; [ "$(printf '%s' "$now" | cut -f2)" = "$ns" ] || exit 36; sed -n '/^\(Name\|State\|Pid\|PPid\|NSpid\):/p' "/proc/$pid/status"; cat "/proc/$pid/wchan"; cat "/proc/$pid/syscall"; cat "/proc/$pid/stack" ;;
  kernel) start=$1; end=$2; journalctl -k --since "$start" --until "$end" --no-pager | grep -E -i -C 3 'I/O error|timeout|timed out|hung task|blocked|reset|read-only|filesystem|mount|nfs|cifs|blk|scsi|nvme'; status=("${PIPESTATUS[@]}"); [ "${status[0]}" -eq 0 ] || exit "${status[0]}"; [ "${status[1]}" -eq 0 ] || [ "${status[1]}" -eq 1 ] || exit "${status[1]}" ;;
  *) exit 22 ;;
esac
REMOTE
}
```

`unset REMOTE_HOSTNAME REMOTE_SYSTEM_UUID; REMOTE_READY=false; TARGET_READY=false; TIME_READY=false; QUERY_READY=false; DIAG_READY=false; NORMALIZE_READY=false; if O="$(node_read identity)"; then REMOTE_HOSTNAME="$(sed -n 's/^hostname=//p' <<<"$O")"; REMOTE_SYSTEM_UUID="$(sed -n 's/^system_uuid=//p' <<<"$O")"; if [ -n "$REMOTE_HOSTNAME" ] && [ -n "$REMOTE_SYSTEM_UUID" ]; then REMOTE_READY=true; export REMOTE_HOSTNAME REMOTE_SYSTEM_UUID; printf 'REMOTE_READY=true hostname=%s systemUUID=%s\n' "$REMOTE_HOSTNAME" "$REMOTE_SYSTEM_UUID"; else printf '%s\n' '중단: 원격 identity 불완전' >&2; fi; else printf '%s\n' '중단: SSH/원격 identity 실패' >&2; fi; unset O`

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | 동일 host PID identity가 bounded 반복에서 계속 `D`인 경우 |
| 적용 | PID start ticks·namespace·PPID와 wait channel·stack·kernel 시간을 연결할 수 있는 경우 |
| 제외 | PIDPressure는 전용 Node 런북, 단발 `D`는 추가 관측, container PID만 알면 host PID를 추정하지 않고 runtime/workload owner로 이관한다. |

검색 문자열: `D`, `uninterruptible`, `io_schedule`, `wait_on_page`, `hung task`, `I/O error`, `nfs`, `cifs`.

## 대상 PID와 시간 등록

먼저 bounded 목록 producer를 실행한다. 빈 출력은 no-match이며 `ps`/filter 오류와 구분한다.

`if [ "$REMOTE_READY" != true ]; then printf '%s\n' '중단: REMOTE_READY가 아니다.' >&2; elif L="$(node_read list_d)"; then if [ -n "$L" ]; then printf '%s\n' "$L"; else printf '%s\n' 'no-match: 현재 D-state PID 없음'; fi; else printf '%s\n' '중단: process producer/filter 오류' >&2; fi; unset L`

`unset PID PID_START_TICKS PID_NS PID_PPID PID_CGROUP; TARGET_READY=false; DIAG_READY=false; NORMALIZE_READY=false; PID='PASTE_PID_HERE'; if [ "$REMOTE_READY" != true ] || [[ ! "$PID" =~ ^[1-9][0-9]*$ ]]; then printf '%s\n' '중단: 원격 gate 또는 PID 형식 오류' >&2; elif I="$(node_read pid_identity "$PID")"; then IFS=$'\t' read -r PID_START_TICKS PID_NS PID_PPID PID_CGROUP <<<"$I"; if [ -n "$PID_START_TICKS" ] && [ -n "$PID_NS" ] && [ -n "$PID_PPID" ]; then TARGET_READY=true; export PID PID_START_TICKS PID_NS PID_PPID PID_CGROUP; printf 'TARGET_READY=true pid=%s start_ticks=%s pid_ns=%s ppid=%s\n' "$PID" "$PID_START_TICKS" "$PID_NS" "$PID_PPID"; else printf '%s\n' '중단: PID identity 불완전' >&2; fi; else printf '%s\n' '중단: PID lookup 실패 또는 lifecycle 변경' >&2; fi; unset I`

`unset START_RFC3339 END_RFC3339 START_EPOCH END_EPOCH; TIME_READY=false; DIAG_READY=false; NORMALIZE_READY=false; START_RFC3339='PASTE_START_RFC3339_HERE'; END_RFC3339='PASTE_END_RFC3339_HERE'; RE='^[0-9]{4}-(0[1-9]|1[0-2])-([0-2][0-9]|3[01])T([01][0-9]|2[0-3]):[0-5][0-9]:[0-5][0-9]Z$'; if [ "$TARGET_READY" != true ] || [[ ! "$START_RFC3339" =~ $RE ]] || [[ ! "$END_RFC3339" =~ $RE ]]; then printf '%s\n' '중단: target 또는 canonical UTC RFC3339 gate 실패' >&2; elif [ "$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" != "$START_RFC3339" ] || [ "$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" != "$END_RFC3339" ]; then printf '%s\n' '중단: date round-trip 실패' >&2; else START_EPOCH="$(date -u -d "$START_RFC3339" +%s)"; END_EPOCH="$(date -u -d "$END_RFC3339" +%s)"; if [ "$START_EPOCH" -lt "$END_EPOCH" ]; then TIME_READY=true; export START_RFC3339 END_RFC3339; printf 'TIME_READY=true start=%s end=%s\n' "$START_RFC3339" "$END_RFC3339"; else printf '%s\n' '중단: 시간 순서 오류' >&2; fi; fi; unset RE`

## 진단 파이프라인

`DIAG_READY=false; if [ "$TIME_READY" != true ]; then printf '%s\n' '중단: TIME_READY가 아니다.' >&2; elif S="$(node_read sample "$PID" "$PID_START_TICKS" "$PID_NS")"; then DIAG_READY=true; printf '%s\n' "$S"; else printf '%s\n' '중단: PID가 사라졌거나 start/namespace/state producer가 변경됨' >&2; fi; unset S`

`if [ "$DIAG_READY" = true ] && D="$(node_read detail "$PID" "$PID_START_TICKS" "$PID_NS")"; then printf '%s\n' "$D"; else printf '%s\n' '중단: PID detail fresh gate/권한/producer 실패' >&2; fi; unset D`

`if [ "$DIAG_READY" = true ] && K="$(node_read kernel "$START_RFC3339" "$END_RFC3339")"; then if [ -n "$K" ]; then printf '%s\n' "$K"; else printf '%s\n' 'no-match: bounded kernel filter 결과 없음; 보존·권한·시간 범위를 별도 확인'; fi; else printf '%s\n' '중단: journal producer 또는 filter 오류' >&2; fi; unset K`

세 sample 모두 같은 PID/start/namespace에서 `D`이고 wait·stack과 같은 시각의 실제 mount/device/remote filesystem 오류가 연결돼야 원인을 좁힌다. `wchan` 이름, 현재 `df`, 빈 journal만으로 storage 원인을 확정하지 않는다. 경로를 확인해야 할 경우 전체 fd 목록 대신 승인된 단일 fd의 유형·mount identity만 최소 수집하며 원문 path·peer를 사건 기록에 저장하지 않는다.

## Prometheus 쿼리와 빈 metric 분기

`if [ "$QUERY_READY" = true ] && fresh_node; then printf 'node_processes_state{instance=~"^%s$",state="D"}\n' "$INSTANCE_RE"; printf 'rate(node_pressure_io_waiting_seconds_total{instance=~"^%s$"}[5m])\n' "$INSTANCE_RE"; printf 'absent(node_processes_state{instance=~"^%s$",state="D"})\n' "$INSTANCE_RE"; else printf '%s\n' '중단: QUERY_READY/fresh Node gate 실패' >&2; fi`

`absent(...) == 1`이면 정상 0이 아니라 collector 미수집, target down, label·datasource·시간 범위 오류를 분리한다. query parser/API 오류는 no-match가 아니다. 이 집계는 PID identity를 제공하지 않는다.

## 변경 금지와 정상화

이 문서는 `kill`, attach, unmount/remount, `fsck`, reboot, sysctl/cgroup, kubelet/runtime restart, Pod 삭제를 실행하지 않으며 CRI/socket을 사용하지 않는다. 별도 변경은 fresh identity·직접 증거·승인 ID와 정확한 RBAC/SSH 주체·audit 위치·window·blast radius·PDB/quorum·멱등성·중단 조건·검증 가능한 rollback이 모두 있을 때만 별도 절차로 수행한다. 변경 후 replacement PID/Pod/Node를 이전 identity로 간주하지 않고 전체 등록을 새로 한다.

`NORMALIZE_READY=false; if [ "$DIAG_READY" != true ]; then printf '%s\n' '중단: DIAG_READY가 아니다.' >&2; elif V="$(node_read sample "$PID" "$PID_START_TICKS" "$PID_NS")"; then NORMALIZE_READY=true; printf '%s\n' "$V"; else printf '%s\n' '중단: 정상화 fresh identity/producer 실패; 종료·PID reuse는 정상화가 아님' >&2; fi; unset V`

같은 identity에서 `D` 반복과 새 I/O 오류가 멈추고, 공유 mount/device 영향이 해소되며, workload의 새 UID·owner·Ready·EndpointSlice와 실제 요청/Job 결과가 정상이어야 한다. PID 종료나 한 번의 `S/R`은 단독 정상화가 아니다.

## 증거·환경 한계

Node UID/RV/systemUUID, 원격 host identity, PID/start ticks/namespace/PPID/cgroup, canonical UTC window, bounded state·wchan·syscall·stack 결정 행, kernel producer/no-match/filter 결과, 실제 storage/Linux owner와 기능 검증을 보존한다. argv·환경변수·전체 stack/fd/path·Secret·token·개인정보는 저장하지 않는다. 실제 인프라 명령은 문서 작성 중 실행하지 않았으며 proc 권한, symbol, journald, PSI/exporter 보존은 환경별 미검증이다.
