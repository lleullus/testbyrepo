# file descriptor가 고갈되어 too many open files가 발생한 경우

## 입력값과 빠른 수집

hop은 `작업 Bash → API server → Node identity`와 `ssh → 검증된 Linux host → PID/start time/PID namespace → /proc/PID/limits·fd count` 또는 `검증된 host → system file table`이다. `EMFILE` process limit과 `ENFILE` system table을 분리하며 FD target path·socket peer·내용은 출력하지 않는다.

| 종류 | 변수 | gate |
|---|---|---|
| 최초 입력 | `NODE`, `HOST` | 승인된 exact Node와 SSH endpoint |
| API 확인 자동값 | `NODE_UID`, `NODE_RV`, `NODE_SYSTEM_UUID` | API 성공과 원격 UUID 일치 |
| 계산 후보값 | `PID_START_TICKS`, `PID_NS`, `INSTANCE_RE` | 현재 process identity와 escaped query 후보 |
| 운영자 선택값 | `PID`, `JOURNAL_UNIT`, `START_RFC3339`, `END_RFC3339`, `EXPORTER_INSTANCE` | 직접 오류·service·monitoring 근거 뒤 각각 한 번 입력 |

### 1. 공통 identity lifecycle

```bash
fresh_node() {
  local j u r
  [ "$INPUT_READY" = true ] || return 1
  j="$(kubectl get node "$NODE" -o json)" || return 1
  u="$(jq -r '.metadata.uid // empty' <<<"$j")"; r="$(jq -r '.metadata.resourceVersion // empty' <<<"$j")"
  [ "$u" = "$NODE_UID" ] && [ "$r" = "$NODE_RV" ] || { printf '중단: Node lifecycle 변경 uid=%s rv=%s\n' "$u" "$r" >&2; return 1; }
}
node_read() {
  local op="$1"; shift
  fresh_node || return 1
  ssh -- "$HOST" bash -s -- "$NODE_SYSTEM_UUID" "$op" "$@" <<'REMOTE'
set -o pipefail
expected=$1; op=$2; shift 2
actual=$(tr '[:upper:]' '[:lower:]' </sys/class/dmi/id/product_uuid 2>/dev/null | tr -d '-') || exit 20
expected=$(printf '%s' "$expected" | tr '[:upper:]' '[:lower:]' | tr -d '-')
[ -n "$actual" ] && [ "$actual" = "$expected" ] || exit 21
proc_identity() {
  pid=$1; [[ "$pid" =~ ^[1-9][0-9]*$ ]] && [ -r "/proc/$pid/stat" ] || exit 30
  stat=$(cat "/proc/$pid/stat") || exit 31; rest=${stat##*) }; set -- $rest
  printf '%s\t%s\t%s\t%s\n' "${20}" "$(readlink "/proc/$pid/ns/pid")" "$(sed -n 's/^PPid:[[:space:]]*//p' "/proc/$pid/status")" "$(tr '\n' ';' <"/proc/$pid/cgroup")"
}
check_proc() {
  pid=$1; start=$2; ns=$3; now=$(proc_identity "$pid") || exit
  [ "${now%%$'\t'*}" = "$start" ] && [ "$(printf '%s' "$now" | cut -f2)" = "$ns" ] || exit 35
}
case "$op" in
  identity) printf 'hostname=%s\nsystem_uuid=%s\n' "$(hostname --fqdn 2>/dev/null || hostname)" "$actual" ;;
  proc_identity) proc_identity "$1" ;;
  fd) pid=$1; start=$2; ns=$3; check_proc "$pid" "$start" "$ns"; date -u '+observed-rfc3339=%Y-%m-%dT%H:%M:%SZ'; grep -E '^Max open files' "/proc/$pid/limits"; printf 'fd_count='; if [ -r "/proc/$pid/fd" ]; then find "/proc/$pid/fd" -maxdepth 1 -type l -printf . 2>/dev/null | wc -c; else printf '%s\n' unavailable; fi; ps -p "$pid" -o pid=,ppid=,stat=,etimes=,lstart=,comm= ;;
  system) date -u '+observed-rfc3339=%Y-%m-%dT%H:%M:%SZ'; printf 'file_nr='; tr '\t' ' ' </proc/sys/fs/file-nr; printf 'file_max='; cat /proc/sys/fs/file-max ;;
  service_identity) unit=$1; [[ "$unit" =~ ^[A-Za-z0-9_.@-]+\.service$ ]] || exit 40; systemctl show "$unit" -p Id -p LoadState -p FragmentPath -p MainPID -p InvocationID --value | sed -n '1,5p' ;;
  journal) unit=$1; fragment=$2; mainpid=$3; invocation=$4; start=$5; end=$6; [[ "$unit" =~ ^[A-Za-z0-9_.@-]+\.service$ ]] || exit 40; out=$(systemctl show "$unit" -p FragmentPath -p MainPID -p InvocationID --value) || exit 41; [ "$(printf '%s\n' "$out" | sed -n '1p')" = "$fragment" ] && [ "$(printf '%s\n' "$out" | sed -n '2p')" = "$mainpid" ] && [ "$(printf '%s\n' "$out" | sed -n '3p')" = "$invocation" ] || exit 42; journalctl -u "$unit" --since "$start" --until "$end" --no-pager | grep -E -i -C 3 'too many open files|EMFILE|ENFILE|accept4?|socket|file descriptor|failed'; status=("${PIPESTATUS[@]}"); [ "${status[0]}" -eq 0 ] || exit "${status[0]}"; [ "${status[1]}" -eq 0 ] || [ "${status[1]}" -eq 1 ] || exit "${status[1]}" ;;
  *) exit 22 ;;
esac
REMOTE
}
```

`unset REMOTE_HOSTNAME REMOTE_SYSTEM_UUID; REMOTE_READY=false; TARGET_READY=false; SERVICE_READY=false; TIME_READY=false; QUERY_READY=false; DIAG_READY=false; NORMALIZE_READY=false; if O="$(node_read identity)"; then REMOTE_HOSTNAME="$(sed -n 's/^hostname=//p' <<<"$O")"; REMOTE_SYSTEM_UUID="$(sed -n 's/^system_uuid=//p' <<<"$O")"; if [ -n "$REMOTE_HOSTNAME" ] && [ -n "$REMOTE_SYSTEM_UUID" ]; then REMOTE_READY=true; export REMOTE_HOSTNAME REMOTE_SYSTEM_UUID; printf 'REMOTE_READY=true hostname=%s systemUUID=%s\n' "$REMOTE_HOSTNAME" "$REMOTE_SYSTEM_UUID"; else printf '%s\n' '중단: 원격 identity 불완전' >&2; fi; else printf '%s\n' '중단: SSH/원격 identity 실패' >&2; fi; unset O`

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | `too many open files`, `EMFILE`, `ENFILE`, `accept4`/`socket` FD 오류 |
| 적용 | 동일 PID lifecycle의 limits/count와 host file table을 같은 UTC window에서 비교 가능 |
| 제외 | fork/cgroup PID는 PID 런북, inode/space는 filesystem 런북, 단순 연결 실패는 network 런북 |

검색 문자열: `too many open files`, `EMFILE`, `ENFILE`, `Max open files`, `file-nr`, `file-max`.

`unset START_RFC3339 END_RFC3339 START_EPOCH END_EPOCH; TIME_READY=false; DIAG_READY=false; NORMALIZE_READY=false; START_RFC3339='PASTE_START_RFC3339_HERE'; END_RFC3339='PASTE_END_RFC3339_HERE'; RE='^[0-9]{4}-(0[1-9]|1[0-2])-([0-2][0-9]|3[01])T([01][0-9]|2[0-3]):[0-5][0-9]:[0-5][0-9]Z$'; if [ "$TARGET_READY" != true ] || [[ ! ## 직접 진단

`DIAG_READY=false; if [ "$TIME_READY" != true ]; then printf '%s\n' '중단: TIME_READY가 아니다.' >&2; elif F="$(node_read fd "$PID" "$PID_START_TICKS" "$PID_NS")" && S="$(node_read system)"; then DIAG_READY=true; printf '%s\n%s\n' "$F" "$S"; else printf '%s\n' '중단: process replacement/권한/FD producer 실패' >&2; fi; unset F S`

`if [ "$DIAG_READY" = true ] && [ "$SERVICE_READY" = true ] && J="$(node_read journal "$JOURNAL_UNIT" "$UNIT_FRAGMENT" "$UNIT_MAINPID" "$UNIT_INVOCATION_ID" "$START_RFC3339" "$END_RFC3339")"; then if [ -n "$J" ]; then printf '%s\n' "$J"; else printf '%s\n' 'no-match: 동일 service invocation의 bounded 필터 결과 없음'; fi; else printf '%s\n' '중단: service replacement 또는 journal producer/filter 오류' >&2; fi; unset J`

open count가 soft limit에 도달하고 동일 PID에서 `EMFILE`이 맞으면 process-local 경로를 지지한다. 여러 process의 `ENFILE`과 `file-nr` max 접근이 같은 시각이면 host table 경로다. `ulimit -n`은 작업 shell 값이므로 대상 process 근거가 아니다. 성공한 producer의 no-match, producer/권한 실패, filter 오류를 구분한다. 현재 count 감소는 과거 peak 부재를 보장하지 않는다. container/runtime 연결은 cgroup 문자열만으로 확정하지 않으며 CRI/socket은 사용하지 않는다.

## Prometheus 완성 쿼리

`unset EXPORTER_INSTANCE INSTANCE_RE; QUERY_READY=false; EXPORTER_INSTANCE='PASTE_NODE_EXPORTER_INSTANCE_HERE'; if [ "$INPUT_READY" = true ] && fresh_node && [ -n "$EXPORTER_INSTANCE" ]; then INSTANCE_RE="$(printf '%s' "$EXPORTER_INSTANCE" | sed 's/[][\\.^$*+?(){}|]/\\&/g')"; QUERY_READY=true; export INSTANCE_RE; printf 'QUERY_READY=true node=%s uid=%s instance_re=^%s$\n' "$NODE" "$NODE_UID" "$INSTANCE_RE"; else printf '%s\n' '중단: query identity gate 실패' >&2; fi`

`if [ "$QUERY_READY" = true ] && fresh_node; then printf '100*node_filefd_allocated{instance=~"^%s$"}/node_filefd_maximum{instance=~"^%s$"}\n' "$INSTANCE_RE" "$INSTANCE_RE"; printf 'predict_linear(node_filefd_allocated{instance=~"^%s$"}[1h],4*3600)>node_filefd_maximum{instance=~"^%s$"}\n' "$INSTANCE_RE" "$INSTANCE_RE"; printf 'absent(node_filefd_allocated{instance=~"^%s$"})\n' "$INSTANCE_RE"; else printf '%s\n' '중단: query fresh gate 실패' >&2; fi`

`absent==1`은 정상 0이 아니며 producer/collector, target, exact label mapping, datasource/time 범위를 확인한다. query/API 오류와 no series를 분리한다. host metric은 process `RLIMIT_NOFILE`을 대신하지 않는다.

## 변경 격리와 정상화

limit/sysctl/cgroup 변경, kill/restart, Pod 삭제, kubelet/runtime restart, reboot를 실행하지 않는다. 별도 변경에는 fresh Node/process/service identity, 직접 증거, owner 승인 ID·정확한 RBAC/SSH/audit 주체, window, blast radius, quorum/PDB·Job side effect, 멱등성, 중단 조건과 검증된 rollback이 필요하다. replacement PID/Pod/service invocation은 이전 evidence를 재사용하지 않고 전체 등록한다.

`NORMALIZE_READY=false; if [ "$DIAG_READY" != true ]; then printf '%s\n' '중단: DIAG_READY가 아니다.' >&2; elif F="$(node_read fd "$PID" "$PID_START_TICKS" "$PID_NS")" && S="$(node_read system)"; then NORMALIZE_READY=true; printf '%s\n%s\n' "$F" "$S"; else printf '%s\n' '중단: 정상화 fresh producer 실패; PID 교체는 정상화가 아님' >&2; fi; unset F S`

bounded window에서 새 `EMFILE/ENFILE`이 멈추고 process count와 system table이 안정되며, 영향 workload의 새 UID·owner·Ready·EndpointSlice와 실제 요청/Job 결과가 정상이어야 한다. count 한 번 감소나 Pod `Running`은 충분하지 않다.

## 증거·환경 한계

Node UID/RV/systemUUID, 원격 identity, PID/start/namespace/PPID/cgroup, service FragmentPath/MainPID/InvocationID, canonical UTC window, soft/hard limit·FD count·file-nr/max, producer/no-match/filter 분기와 기능 결과를 보존한다. FD path·symlink·peer·payload·argv·Secret·token·개인정보는 저장하지 않는다. 실제 인프라 명령은 실행하지 않았고 procfs/systemd/journal/exporter 보존은 환경별 미검증이다.
