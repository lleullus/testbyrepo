# sysctl 값이 승인 기준과 다르거나 적용되지 않는 경우

## 입력값과 빠른 수집

실행 hop은 `승인된 작업 PC/Bash → Kubernetes API`와 `작업 PC/Bash → SSH → Node host namespace → sysctl/procfs·loader journal`이다. container·Pod network namespace의 값은 host 값과 별도 identity로 취급한다.

| 구분 | 값 | 계약 |
|---|---|---|
| 최초 입력 | `NODE`, `SSH_DEST`, `EXPECTED_MACHINE_ID` | 승인 inventory와 exact match |
| API 확인 자동값 | `NODE_UID`, `SYSTEM_UUID`, `HOST_BOOT_ID`, `KERNEL_RELEASE` | API와 SSH producer가 모두 성공해야 유효 |
| 계산 후보값 | `SYSCTL_RE` | exact key를 ERE escape한 로그 필터 후보 |
| 운영자 선택값 | `SYSCTL_KEY`, `APPROVED_VALUE`, `APPROVAL_ID`, `START_UTC`, `END_UTC` | 승인 원본과 canonical UTC 창 |

### 1. 대상과 승인 기준 고정

```bash
TARGET_READY=false; SELECTION_READY=false
unset NODE_UID SYSTEM_UUID API_BOOT_ID HOST_BOOT_ID KERNEL_RELEASE SYSCTL_KEY SYSCTL_RE APPROVED_VALUE APPROVAL_ID START_UTC END_UTC PID PID_START_TICKS MOUNT_ID DEVICE_ID CGROUP_ID NETNS_ID NETWORK_ID ARGV_RE READ_READY JOURNAL_READY FILTER_READY READ_OUT JOURNAL_OUT FILTERED_OUT
SSH_OPTS=(-o BatchMode=yes -o ConnectTimeout=10 -o StrictHostKeyChecking=yes -o LogLevel=ERROR)
canonical_utc() { local value=${1:-}; [[ "$value" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && [ "$(date -u -d "$value" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" = "$value" ]; }
window_valid() { canonical_utc "${START_UTC:-}" && canonical_utc "${END_UTC:-}" && [[ "$START_UTC" < "$END_UTC" ]]; }
NODE='PASTE_SYSCTL_NODE_HERE'; SSH_DEST='PASTE_SYSCTL_SSH_DEST_HERE'; EXPECTED_MACHINE_ID='PASTE_SYSCTL_MACHINE_ID_HERE'
MISSING=''; for tool in kubectl jq ssh grep sed date; do command -v "$tool" >/dev/null 2>&1 || MISSING="$MISSING $tool"; done
if [ -z "${BASH_VERSION:-}" ] || [ -n "$MISSING" ] || [[ ! "$SSH_DEST" =~ ^[A-Za-z0-9][A-Za-z0-9._@:-]*$ ]] || [[ ! "$EXPECTED_MACHINE_ID" =~ ^[0-9a-f]{32}$ ]]; then printf '중단: Bash/도구/입력 gate 실패:%s\n' "$MISSING" >&2; else
  J="$(kubectl get node "$NODE" -o json)"; API_RC=$?; [ "$API_RC" -eq 0 ] && NODE_UID="$(jq -r '.metadata.uid // empty' <<<"$J")" && SYSTEM_UUID="$(jq -r '.status.nodeInfo.systemUUID // empty' <<<"$J")" && API_BOOT_ID="$(jq -r '.status.nodeInfo.bootID // empty' <<<"$J")"
  R="$(ssh "${SSH_OPTS[@]}" -- "$SSH_DEST" sh -s -- "$EXPECTED_MACHINE_ID" <<'REMOTE'
set -eu
[ "$(cat /etc/machine-id)" = "$1" ] || exit 42
printf '%s\t%s\n' "$(cat /proc/sys/kernel/random/boot_id)" "$(uname -r)"
REMOTE
)"; SSH_RC=$?; [ "$SSH_RC" -eq 0 ] && IFS=$'\t' read -r HOST_BOOT_ID KERNEL_RELEASE <<<"$R"
  if [ "$API_RC" -eq 0 ] && [ -n "$NODE_UID" ] && [ -n "$SYSTEM_UUID" ] && [ "$API_BOOT_ID" = "$HOST_BOOT_ID" ] && [ -n "$KERNEL_RELEASE" ]; then TARGET_READY=true; printf 'TARGET_READY=true node=%s uid=%s systemUUID=%s machine-id=%s boot-id=%s kernel=%s\n' "$NODE" "$NODE_UID" "$SYSTEM_UUID" "$EXPECTED_MACHINE_ID" "$HOST_BOOT_ID" "$KERNEL_RELEASE"; else unset NODE_UID SYSTEM_UUID API_BOOT_ID HOST_BOOT_ID KERNEL_RELEASE; printf '중단: identity 생산 실패 api_rc=%s ssh_rc=%s\n' "$API_RC" "$SSH_RC" >&2; fi
fi
unset J R
```

```bash
SELECTION_READY=false; READ_READY=false; JOURNAL_READY=false; FILTER_READY=false
unset SYSCTL_KEY SYSCTL_RE APPROVED_VALUE APPROVAL_ID START_UTC END_UTC READ_OUT JOURNAL_OUT FILTERED_OUT
SYSCTL_KEY='PASTE_SYSCTL_KEY_HERE'; APPROVED_VALUE='PASTE_SYSCTL_APPROVED_VALUE_HERE'; APPROVAL_ID='PASTE_SYSCTL_APPROVAL_ID_HERE'; START_UTC='PASTE_SYSCTL_START_UTC_HERE'; END_UTC='PASTE_SYSCTL_END_UTC_HERE'
if [[ "$SYSCTL_KEY" =~ ^[A-Za-z0-9_][A-Za-z0-9_.-]{0,255}$ ]] && [ -n "$APPROVED_VALUE" ] && [ -n "$APPROVAL_ID" ] && window_valid; then SYSCTL_RE="$(printf '%s' "$SYSCTL_KEY" | sed 's/[][\\.^$*+?(){}|]/\\&/g')"; SELECTION_READY=true; printf 'SELECTION_READY=true key=%s approved=%s approval=%s regex=%s\n' "$SYSCTL_KEY" "$APPROVED_VALUE" "$APPROVAL_ID" "$SYSCTL_RE"; else unset SYSCTL_KEY SYSCTL_RE APPROVED_VALUE APPROVAL_ID START_UTC END_UTC; printf '%s\n' '중단: 승인 기준/UTC 입력 오류' >&2; fi
```

### 2. fresh identity를 포함한 host 읽기

```bash
sysctl_read() {
  local mode=${1:-} j uid system boot rc
  READ_READY=false; unset READ_OUT
  [ "$TARGET_READY" = true ] && [ "$SELECTION_READY" = true ] && window_valid && [ "$(printf '%s' "$SYSCTL_KEY" | sed 's/[][\\.^$*+?(){}|]/\\&/g')" = "$SYSCTL_RE" ] || { printf '%s\n' '중단: gate 미충족 또는 선택값 변경' >&2; return 1; }
  j="$(kubectl get node "$NODE" -o json)"; rc=$?; [ "$rc" -eq 0 ] || { TARGET_READY=false; printf 'fresh API 실패 rc=%s\n' "$rc" >&2; return "$rc"; }
  uid="$(jq -r '.metadata.uid // empty' <<<"$j")"; system="$(jq -r '.status.nodeInfo.systemUUID // empty' <<<"$j")"; boot="$(jq -r '.status.nodeInfo.bootID // empty' <<<"$j")"
  [ "$uid" = "$NODE_UID" ] && [ "$system" = "$SYSTEM_UUID" ] && [ "$boot" = "$HOST_BOOT_ID" ] || { TARGET_READY=false; printf '%s\n' '중단: identity/lifecycle 변경' >&2; return 1; }
  READ_OUT="$(ssh "${SSH_OPTS[@]}" -- "$SSH_DEST" sh -s -- "$EXPECTED_MACHINE_ID" "$HOST_BOOT_ID" "$SYSCTL_KEY" "$APPROVED_VALUE" "$mode" <<'REMOTE'
set -eu
[ "$(cat /etc/machine-id)" = "$1" ] || exit 42
[ "$(cat /proc/sys/kernel/random/boot_id)" = "$2" ] || exit 43
key=$3; approved=$4; mode=$5
case "$mode" in
  runtime) command -v sysctl >/dev/null 2>&1 || exit 69; value=$(sysctl -n "$key"); [ "$value" = "$approved" ] && match=true || match=false; printf 'key=%s\nvalue=%s\napproved=%s\nmatch=%s\nobserved=%s\n' "$key" "$value" "$approved" "$match" "$(date -u '+%Y-%m-%dT%H:%M:%SZ')" ;;
  loader) command -v systemctl >/dev/null 2>&1 || exit 69; [ "$(cat /proc/1/comm)" = systemd ] || exit 70; systemctl show systemd-sysctl.service -p Id -p LoadState -p ActiveState -p Result -p FragmentPath -p DropInPaths ;;
  sources) command -v awk >/dev/null 2>&1 || exit 69; for d in /etc/sysctl.d /run/sysctl.d /usr/local/lib/sysctl.d /usr/lib/sysctl.d /lib/sysctl.d; do [ -d "$d" ] || continue; for f in "$d"/*.conf; do [ -f "$f" ] && [ -r "$f" ] || continue; awk -v key="$key" 'BEGIN{FS="="} /^[[:space:]]*[#;]/ {next} NF>=2 {lhs=$1; gsub(/^[[:space:]]+|[[:space:]]+$/, "", lhs); if(lhs==key) printf "%s:%d:%s\n", FILENAME, FNR, $0}' "$f" || exit $?; done; done; if [ -r /etc/sysctl.conf ]; then awk -v key="$key" 'BEGIN{FS="="} /^[[:space:]]*[#;]/ {next} NF>=2 {lhs=$1; gsub(/^[[:space:]]+|[[:space:]]+$/, "", lhs); if(lhs==key) printf "%s:%d:%s\n", FILENAME, FNR, $0}' /etc/sysctl.conf || exit $?; fi ;;
  *) exit 64 ;;
esac
REMOTE
)"; rc=$?
  if [ "$rc" -ne 0 ]; then unset READ_OUT; printf 'remote producer 실패 mode=%s rc=%s\n' "$mode" "$rc" >&2; return "$rc"; fi
  READ_READY=true; printf '%s\n' "$READ_OUT"
}
```

`sysctl_read runtime`

`sysctl_read loader`

`sysctl_read sources`

source 출력은 주석을 제외한 exact assignment 후보만 보여 주지만 loader precedence를 확정하지 않는다. 같은 key의 precedence는 실제 loader 문서와 basename ordering을 대조한다. 빈 출력은 source 없음이 아니라 권한·loader 차이 후보다. config 전체와 민감값은 저장하지 않는다.

### 3. loader journal

```bash
JOURNAL_READY=false; FILTER_READY=false; unset JOURNAL_OUT FILTERED_OUT
if [ "$TARGET_READY" = true ] && [ "$SELECTION_READY" = true ] && window_valid && [ "$(printf '%s' "$SYSCTL_KEY" | sed 's/[][\\.^$*+?(){}|]/\\&/g')" = "$SYSCTL_RE" ]; then
  J="$(kubectl get node "$NODE" -o json)"; API_RC=$?
  if [ "$API_RC" -eq 0 ] && [ "$(jq -r '.metadata.uid' <<<"$J")" = "$NODE_UID" ] && [ "$(jq -r '.status.nodeInfo.systemUUID' <<<"$J")" = "$SYSTEM_UUID" ] && [ "$(jq -r '.status.nodeInfo.bootID' <<<"$J")" = "$HOST_BOOT_ID" ]; then
    JOURNAL_OUT="$(ssh "${SSH_OPTS[@]}" -- "$SSH_DEST" sh -s -- "$EXPECTED_MACHINE_ID" "$HOST_BOOT_ID" "$KERNEL_RELEASE" "$START_UTC" "$END_UTC" <<'REMOTE'
set -eu
[ "$(cat /etc/machine-id)" = "$1" ] || exit 42
[ "$(cat /proc/sys/kernel/random/boot_id)" = "$2" ] || exit 43
[ "$(uname -r)" = "$3" ] || exit 44
[ "$(cat /proc/1/comm)" = systemd ] || exit 70
exec journalctl -u systemd-sysctl.service --utc --since "$4" --until "$5" -o short-iso-precise --no-pager --lines=500
REMOTE
)"; PRODUCER_RC=$?
    if [ "$PRODUCER_RC" -eq 0 ]; then JOURNAL_READY=true; FILTERED_OUT="$(grep -E -i -C 3 -- "$SYSCTL_RE|cannot stat|permission denied|invalid argument|read-only|not permitted|failed" <<<"$JOURNAL_OUT")"; FILTER_RC=$?; case "$FILTER_RC" in 0) FILTER_READY=true; printf '%s\n' "$FILTERED_OUT";; 1) unset FILTERED_OUT; printf '%s\n' '필터 일치 없음: 정상 아님' >&2;; *) unset JOURNAL_OUT FILTERED_OUT; JOURNAL_READY=false; printf '필터 오류 rc=%s\n' "$FILTER_RC" >&2;; esac; else unset JOURNAL_OUT; printf 'journal producer 실패 rc=%s\n' "$PRODUCER_RC" >&2; fi
  else TARGET_READY=false; printf 'fresh API 실패/identity 변경 rc=%s\n' "$API_RC" >&2; fi
fi
unset J
```

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | exact key·승인값·scope와 runtime 불일치 또는 적용 오류가 있음 |
| 적용 | 같은 host namespace·boot·kernel에서 runtime, source, loader를 대조 가능 |
| 제외 | 승인 원본 불명, Pod namespace 전용 값, admission/unsafe sysctl 거부 |

검색 문자열: `sysctl`, `cannot stat`, `permission denied`, `invalid argument`, `read-only`, `allowedUnsafeSysctls`.

## 분기와 판정

- runtime이 다르고 source도 다르면 승인 revision·배포 경로 문제다.
- source는 맞고 loader journal이 실패하면 loader/kernel/policy 경계다.
- runtime은 맞지만 source가 다르면 현재 값만으로 지속성을 보장하지 않는다.
- key 조회 실패는 key 부재·namespace·kernel feature·권한을 구분하기 전까지 정상이나 불일치로 기록하지 않는다.
- Pod 원인이면 Pod UID·container ID·network namespace inode를 fresh하게 고정하고 host 값을 재사용하지 않는다. RBAC·admission·unsafe sysctl 정책을 우회하지 않는다.

## 변경 경계와 정상화

이 문서는 `sysctl -w`, `sysctl --system`, config edit, loader restart, Pod patch/delete와 reboot를 실행하지 않는다. 별도 변경은 fresh identity·runtime/source/journal, 승인 ID·window·scope, workload availability·PDB/quorum, exact desired value, idempotence, 중단 조건과 원 파일/revision으로의 검증 가능한 rollback을 갖춰야 한다. 변경 직전 identity나 runtime이 달라지면 중단한다.

정상화는 매번 `sysctl_read runtime`, `sysctl_read sources`, fresh loader journal을 다시 실행하고 승인 scope의 값·source·loader가 일치하는지 본다. Pod scope라면 교체 Pod의 새 UID/container ID/netns를 재등록하고 실제 요청·network 기능을 확인한다. 값 한 번 일치, loader active, Node Ready는 실제 기능 정상화를 단독 보장하지 않는다.

보존: Node UID·systemUUID, machine-id·boot ID·kernel, key/value·승인 revision/scope, source의 해당 행, producer/filter rc, UTC journal, namespace identity, 기능 검증과 rollback 결과. config 전체·Secret·token·private key는 저장하지 않는다.

## 환경별 미검증

배포판 loader precedence, namespace scope, kernel feature, unsafe sysctl/admission, 실제 Kubernetes 1.26 환경은 미검증이며 명령을 실행하지 않았다.
