# conntrack table이 가득 찬 경우

## 입력값과 빠른 수집

실행 경로는 `승인된 작업 PC의 Bash → 검증된 SSH host key → machine-id·boot-id → 선택한 PID lifecycle → 그 PID의 Linux network namespace → nf_conntrack count/max·per-CPU 통계 → node-wide kernel journal → 실제 영향 flow`다. Kubernetes namespace, Linux network namespace와 conntrack zone은 서로 다른 식별자다.

| 구분 | 값 | 출처와 gate |
|---|---|---|
| 최초 입력 | `INCIDENT_ID`, `SSH_TARGET`, `EXPECTED_MACHINE_ID` | 사건 기록과 승인된 inventory |
| 시스템 자동값 | `BOOT_ID`, `HOST_NETNS_INODE`, `NETNS_PID`, `PID_START_TICKS`, `NETNS_INODE`, `CGROUP_HASH` | 원격 호출마다 PID 재사용·netns 교체 여부를 다시 검증한다. |
| 계산 후보값 | canonical UTC epoch, t0/t1 count·max·counter delta | producer가 모두 성공한 동일 netns 표본만 비교한다. |
| 운영자 선택값 | `NETNS_PID_INPUT`, `ARGV_REGEX` | 승인된 host/process inventory에서 고른다. host netns는 보통 PID 1이지만 자동 가정하지 않는다. |

각 고유 `PASTE_..._HERE` 입력칸은 한 번만 나온다. PID 종료·재사용, netns 재생성, host reboot 또는 counter reset이 있으면 모든 표본과 하위 `READY`를 폐기한다.

### 1. 실행 계약과 host 등록

```bash
TOOLS_READY=false
MISSING_TOOLS=''
for TOOL in ssh date base64 mktemp; do
  command -v "$TOOL" >/dev/null 2>&1 || MISSING_TOOLS="$MISSING_TOOLS $TOOL"
done
if [ -n "${BASH_VERSION:-}" ] && [ -z "$MISSING_TOOLS" ]; then
  TOOLS_READY=true
else
  printf '중단: Bash 또는 로컬 도구 누락:%s\n' "$MISSING_TOOLS" >&2
fi

clear_target_state() {
  unset ACTUAL_MACHINE_ID BOOT_ID REMOTE_HOSTNAME HOST_NETNS_INODE NETNS_PID_INPUT NETNS_PID PID_START_TICKS NETNS_INODE CGROUP_HASH ARGV_REGEX ARGV_REGEX_B64 START_RFC3339 END_RFC3339 START_EPOCH END_EPOCH T0_EPOCH T0_COUNT T0_MAX T0_DROP T0_INSERT_FAILED T0_EARLY_DROP T1_EPOCH T1_COUNT T1_MAX T1_DROP T1_INSERT_FAILED T1_EARLY_DROP
  HOST_READY=false
  TIME_READY=false
  NETNS_READY=false
  EVIDENCE_READY=false
  SAMPLE_READY=false
  AUTHORIZATION_READY=false
  MUTATION_READY=false
  ROLLBACK_READY=false
  NORMALIZATION_READY=false
}

register_host() {
  clear_target_state
  if [ "$TOOLS_READY" != true ]; then printf '%s\n' '중단: TOOLS_READY가 아니다.' >&2; return 1; fi
  if ! [[ "$SSH_TARGET" =~ ^([A-Za-z0-9._-]+@)?[A-Za-z0-9._-]+$ ]] || ! [[ "$EXPECTED_MACHINE_ID" =~ ^[0-9a-f]{32}$ ]]; then
    printf '%s\n' '중단: SSH target 또는 machine-id 형식 오류' >&2
    return 1
  fi
  local RESULT RC
  RESULT="$(ssh -o BatchMode=yes -o StrictHostKeyChecking=yes -o ConnectTimeout=10 -- "$SSH_TARGET" bash -s -- "$EXPECTED_MACHINE_ID" <<'REMOTE'
set -uo pipefail
expected=$1
actual=$(< /etc/machine-id) || exit 20
boot=$(< /proc/sys/kernel/random/boot_id) || exit 20
netns=$(readlink /proc/1/ns/net) || exit 20
name=$(hostname) || exit 20
[ "$actual" = "$expected" ] || exit 21
printf '%s\t%s\t%s\t%s\n' "$actual" "$boot" "$name" "$netns"
REMOTE
)"
  RC=$?
  if [ "$RC" -ne 0 ]; then clear_target_state; printf '중단: host 등록 실패 rc=%s\n' "$RC" >&2; return "$RC"; fi
  IFS=$'\t' read -r ACTUAL_MACHINE_ID BOOT_ID REMOTE_HOSTNAME HOST_NETNS_INODE <<<"$RESULT"
  if [ -z "$ACTUAL_MACHINE_ID" ] || [ -z "$BOOT_ID" ] || [ -z "$HOST_NETNS_INODE" ]; then clear_target_state; printf '%s\n' '중단: host identity 불완전' >&2; return 1; fi
  HOST_READY=true
  printf 'HOST_READY=true host=%s machine-id=%s boot-id=%s host-netns=%s incident=%s\n' "$REMOTE_HOSTNAME" "$ACTUAL_MACHINE_ID" "$BOOT_ID" "$HOST_NETNS_INODE" "$INCIDENT_ID"
}

host_call() {
  [ "${HOST_READY:-false}" = true ] || { printf '%s\n' '중단: HOST_READY가 아니다.' >&2; return 1; }
  local OP=$1
  shift
  ssh -o BatchMode=yes -o StrictHostKeyChecking=yes -o ConnectTimeout=10 -- "$SSH_TARGET" bash -s -- "$EXPECTED_MACHINE_ID" "$BOOT_ID" "$OP" "$@" <<'REMOTE'
set -uo pipefail
expected_machine=$1; expected_boot=$2; op=$3; shift 3
[ "$(< /etc/machine-id)" = "$expected_machine" ] || exit 21
[ "$(< /proc/sys/kernel/random/boot_id)" = "$expected_boot" ] || exit 22

process_start_ticks() {
  pid=$1
  stat=$(< "/proc/$pid/stat") || exit 30
  rest=${stat#*) }
  set -- $rest
  printf '%s\n' "${20}"
}

guard_netns() {
  pid=$1; expected_start=$2; expected_netns=$3; expected_cgroup=$4; regex_b64=$5
  [[ "$pid" =~ ^[1-9][0-9]*$ ]] || exit 30
  [ "$(process_start_ticks "$pid")" = "$expected_start" ] || exit 31
  [ "$(readlink "/proc/$pid/ns/net")" = "$expected_netns" ] || exit 32
  current_cgroup=$(sha256sum "/proc/$pid/cgroup") || exit 33
  [ "${current_cgroup%% *}" = "$expected_cgroup" ] || exit 34
  regex_file=$(mktemp) || exit 35
  argv_file=$(mktemp) || { rm -f "$regex_file"; exit 35; }
  set +e
  printf '%s' "$regex_b64" | base64 -d >"$regex_file"
  decode_status=("${PIPESTATUS[@]}")
  set -e
  if [ "${decode_status[0]}" -ne 0 ] || [ "${decode_status[1]}" -ne 0 ]; then rm -f "$regex_file" "$argv_file"; exit 36; fi
  tr '\0' ' ' <"/proc/$pid/cmdline" >"$argv_file" || { rm -f "$regex_file" "$argv_file"; exit 37; }
  set +e
  grep -E -q -f "$regex_file" "$argv_file"
  filter_rc=$?
  set -e
  rm -f "$regex_file" "$argv_file"
  [ "$filter_rc" -eq 0 ] || exit 38
}

sample_conntrack() {
  pid=$1
  command -v nsenter >/dev/null 2>&1 || exit 40
  command -v conntrack >/dev/null 2>&1 || exit 40
  count=$(nsenter -t "$pid" -n -- cat /proc/sys/net/netfilter/nf_conntrack_count) || exit 41
  maximum=$(nsenter -t "$pid" -n -- cat /proc/sys/net/netfilter/nf_conntrack_max) || exit 42
  [[ "$count" =~ ^[0-9]+$ ]] && [[ "$maximum" =~ ^[1-9][0-9]*$ ]] || exit 43
  stats=$(mktemp) || exit 44
  summary=$(mktemp) || { rm -f "$stats"; exit 44; }
  nsenter -t "$pid" -n -- conntrack -S >"$stats" || { rm -f "$stats" "$summary"; exit 45; }
  awk 'BEGIN{d=0;i=0;e=0} {for(n=1;n<=NF;n++){split($n,a,"="); if(a[1]=="drop")d+=a[2]; else if(a[1]=="insert_failed")i+=a[2]; else if(a[1]=="early_drop")e+=a[2]}} END{print d,i,e}' "$stats" >"$summary" || { rm -f "$stats" "$summary"; exit 46; }
  read -r drop insert_failed early_drop <"$summary" || { rm -f "$stats" "$summary"; exit 46; }
  rm -f "$stats" "$summary"
  [[ "$drop" =~ ^[0-9]+$ ]] && [[ "$insert_failed" =~ ^[0-9]+$ ]] && [[ "$early_drop" =~ ^[0-9]+$ ]] || exit 46
  printf '%s\t%s\t%s\t%s\t%s\n' "$count" "$maximum" "$drop" "$insert_failed" "$early_drop"
}

case "$op" in
  netns-register)
    pid=$1; regex_b64=$2
    [[ "$pid" =~ ^[1-9][0-9]*$ ]] || exit 30
    start=$(process_start_ticks "$pid") || exit 30
    netns=$(readlink "/proc/$pid/ns/net") || exit 30
    cgroup=$(sha256sum "/proc/$pid/cgroup") || exit 30
    guard_netns "$pid" "$start" "$netns" "${cgroup%% *}" "$regex_b64"
    printf '%s\t%s\t%s\n' "$start" "$netns" "${cgroup%% *}"
    ;;
  sample)
    guard_netns "$1" "$2" "$3" "$4" "$5"
    sample_conntrack "$1"
    ;;
  journal)
    guard_netns "$1" "$2" "$3" "$4" "$5"
    start=$6; end=$7
    set +e
    journalctl -k --since "$start" --until "$end" --no-pager -n 300 | grep -E -i -C 3 -- 'nf_conntrack|table full|dropping packet|insert_failed|early_drop|conntrack.*drop'
    status=("${PIPESTATUS[@]}")
    set -e
    [ "${status[0]}" -eq 0 ] || exit 60
    [ "${status[1]}" -eq 0 ] && exit 0
    [ "${status[1]}" -eq 1 ] && exit 61
    exit 62
    ;;
  *) exit 64 ;;
esac
REMOTE
}

INCIDENT_ID='PASTE_INCIDENT_ID_HERE'
SSH_TARGET='PASTE_SSH_TARGET_HERE'
EXPECTED_MACHINE_ID='PASTE_EXPECTED_MACHINE_ID_HERE'
clear_target_state
```

```bash
register_host
```

### 2. canonical UTC 증거 창

```bash
START_RFC3339='PASTE_START_RFC3339_HERE'
END_RFC3339='PASTE_END_RFC3339_HERE'
TIME_READY=false
unset START_EPOCH END_EPOCH
RFC3339_RE='^[0-9]{4}-(0[1-9]|1[0-2])-([0-2][0-9]|3[01])T([01][0-9]|2[0-3]):[0-5][0-9]:[0-5][0-9]Z$'
if [ "${HOST_READY:-false}" != true ]; then
  printf '%s\n' '중단: HOST_READY가 아니다.' >&2
elif ! [[ "$START_RFC3339" =~ $RFC3339_RE ]] || ! [[ "$END_RFC3339" =~ $RFC3339_RE ]]; then
  printf '%s\n' '중단: UTC RFC3339 형식 오류' >&2
elif [ "$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" != "$START_RFC3339" ] || [ "$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" != "$END_RFC3339" ]; then
  printf '%s\n' '중단: date round-trip 실패' >&2
else
  START_EPOCH=$(date -u -d "$START_RFC3339" +%s)
  END_EPOCH=$(date -u -d "$END_RFC3339" +%s)
  if (( START_EPOCH >= END_EPOCH || END_EPOCH - START_EPOCH > 86400 )); then
    unset START_EPOCH END_EPOCH
    printf '%s\n' '중단: 시작은 종료보다 앞서야 하고 창은 24시간 이하여야 한다.' >&2
  else
    TIME_READY=true
    printf 'TIME_READY=true start=%s end=%s seconds=%s\n' "$START_RFC3339" "$END_RFC3339" "$((END_EPOCH-START_EPOCH))"
  fi
fi
```

### 3. network namespace identity 등록

`NETNS_PID_INPUT`은 조사할 netns에 현재 속한 승인된 process PID다. `ARGV_REGEX`는 executable과 비민감 고정 인수만 최소 매칭한다. raw argv·environment·cgroup 원문은 출력하지 않는다.

```bash
NETNS_PID_INPUT='PASTE_CONFIRMED_NETNS_PID_HERE'
ARGV_REGEX='PASTE_NON_SECRET_ARGV_REGEX_HERE'
NETNS_READY=false
EVIDENCE_READY=false
SAMPLE_READY=false
unset NETNS_PID PID_START_TICKS NETNS_INODE CGROUP_HASH ARGV_REGEX_B64 T0_EPOCH T0_COUNT T0_MAX T0_DROP T0_INSERT_FAILED T0_EARLY_DROP T1_EPOCH T1_COUNT T1_MAX T1_DROP T1_INSERT_FAILED T1_EARLY_DROP
if [ "${HOST_READY:-false}" != true ] || ! [[ "$NETNS_PID_INPUT" =~ ^[1-9][0-9]*$ ]] || [ -z "$ARGV_REGEX" ]; then
  printf '%s\n' '중단: host, PID 또는 argv regex 입력 오류' >&2
else
  TMP=$(mktemp)
  printf '%s' "$ARGV_REGEX" >"$TMP"
  ARGV_REGEX_B64=$(base64 -w0 "$TMP")
  ENCODE_RC=$?
  rm -f "$TMP"
  if [ "$ENCODE_RC" -ne 0 ] || ! [[ "$ARGV_REGEX_B64" =~ ^[A-Za-z0-9+/=]+$ ]]; then
    unset ARGV_REGEX_B64
    printf '%s\n' '중단: regex encoding 실패' >&2
  elif RESULT=$(host_call netns-register "$NETNS_PID_INPUT" "$ARGV_REGEX_B64"); then
    IFS=$'\t' read -r PID_START_TICKS NETNS_INODE CGROUP_HASH <<<"$RESULT"
    if [ -n "$PID_START_TICKS" ] && [ -n "$NETNS_INODE" ] && [[ "$CGROUP_HASH" =~ ^[0-9a-f]{64}$ ]]; then
      NETNS_PID=$NETNS_PID_INPUT
      NETNS_READY=true
      printf 'NETNS_READY=true pid=%s start-ticks=%s netns=%s cgroup-hash=%s host-netns=%s\n' "$NETNS_PID" "$PID_START_TICKS" "$NETNS_INODE" "$CGROUP_HASH" "$HOST_NETNS_INODE"
    else
      unset NETNS_PID PID_START_TICKS NETNS_INODE CGROUP_HASH
      printf '%s\n' '중단: netns identity 결과 불완전' >&2
    fi
  else
    unset NETNS_PID PID_START_TICKS NETNS_INODE CGROUP_HASH
    printf '%s\n' '중단: PID 종료·재사용, argv 불일치 또는 netns 등록 실패' >&2
  fi
fi
```

`NETNS_INODE`가 `HOST_NETNS_INODE`와 같을 때만 host netns라고 기록한다. 다르면 별도 Linux netns이며 node-wide journal을 해당 netns 전용 로그로 해석하지 않는다.

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | `nf_conntrack: table full, dropping packet` 또는 같은 창의 count/max 포화와 `insert_failed`·`early_drop` 증가가 직접 관측된 경우 |
| 적용 | current machine·boot·PID start·netns·counter 원점을 유지할 수 있는 경우 |
| 제외 | host rule `DROP`/`REJECT`가 최초 증거면 [[05. iptables·nftables rule에서 packet drop이 확인된 경우]], route·MTU·CNI·외부 장비가 먼저면 해당 owner 경로로 이동한다. |

검색 문자열: `nf_conntrack: table full`, `dropping packet`, `nf_conntrack_count`, `nf_conntrack_max`, `insert_failed`, `early_drop`, `conntrack zone`.

## 관측별 진단 경로

| 지금 보이는 것 | 직접 증거 | 다음 행동 |
|---|---|---|
| count가 max에 접근하고 insert 실패 증가 | 같은 netns t0/t1·kernel 창 | 현재 고갈과 실제 flow 영향 대조 |
| 현재 count는 낮지만 사건 창에 table full | boot·journal 시각·counter reset 여부 | 과거 고갈로 보존하고 현재값과 분리 |
| drop만 증가하고 count 여유 있음 | insert_failed·early_drop·rule counter | firewall 또는 다른 conntrack 오류 분기 |
| PID/netns가 교체됨 | start ticks·netns inode mismatch | 모든 표본 폐기 후 replacement 등록 |
| 특정 zone만 의심됨 | 단일 backend의 CT zone rule·별도 계측 | zone owner 이관; namespace 전체 count를 zone 값으로 표현하지 않음 |

## 읽기 전용 count·통계

아래 표본은 원격에서 현재 PID lifecycle과 netns를 직접 검증하고, 그 netns의 count/max와 `conntrack -S` producer가 모두 성공한 경우에만 유효하다.

```bash
EVIDENCE_READY=false
unset T0_EPOCH T0_COUNT T0_MAX T0_DROP T0_INSERT_FAILED T0_EARLY_DROP
if [ "${NETNS_READY:-false}" != true ]; then
  printf '%s\n' '중단: NETNS_READY가 아니다.' >&2
elif RESULT=$(host_call sample "$NETNS_PID" "$PID_START_TICKS" "$NETNS_INODE" "$CGROUP_HASH" "$ARGV_REGEX_B64"); then
  IFS=$'\t' read -r T0_COUNT T0_MAX T0_DROP T0_INSERT_FAILED T0_EARLY_DROP <<<"$RESULT"
  T0_EPOCH=$(date -u +%s)
  if [[ "$T0_COUNT" =~ ^[0-9]+$ ]] && [[ "$T0_MAX" =~ ^[1-9][0-9]*$ ]] && (( T0_COUNT <= T0_MAX )); then
    EVIDENCE_READY=true
    printf 'EVIDENCE_READY=true count=%s max=%s usage-percent=%s drop=%s insert-failed=%s early-drop=%s observed-at=%s\n' "$T0_COUNT" "$T0_MAX" "$((100*T0_COUNT/T0_MAX))" "$T0_DROP" "$T0_INSERT_FAILED" "$T0_EARLY_DROP" "$(date -u '+%Y-%m-%dT%H:%M:%SZ')"
  else
    unset T0_EPOCH T0_COUNT T0_MAX T0_DROP T0_INSERT_FAILED T0_EARLY_DROP
    printf '%s\n' '중단: count/max 결과 불완전 또는 모순' >&2
  fi
else
  printf '%s\n' '중단: identity, procfs 또는 conntrack 통계 producer 실패' >&2
fi
```

빈 통계, `conntrack` 부재, 권한 실패와 netns 소멸은 0이 아니다. `drop` 하나만으로 table full을 확정하지 않는다.

승인된 관찰 간격 뒤 같은 원점의 t1을 등록한다.

```bash
SAMPLE_READY=false
unset T1_EPOCH T1_COUNT T1_MAX T1_DROP T1_INSERT_FAILED T1_EARLY_DROP
if [ "${EVIDENCE_READY:-false}" != true ]; then
  printf '%s\n' '중단: EVIDENCE_READY가 아니다.' >&2
elif RESULT=$(host_call sample "$NETNS_PID" "$PID_START_TICKS" "$NETNS_INODE" "$CGROUP_HASH" "$ARGV_REGEX_B64"); then
  IFS=$'\t' read -r T1_COUNT T1_MAX T1_DROP T1_INSERT_FAILED T1_EARLY_DROP <<<"$RESULT"
  T1_EPOCH=$(date -u +%s)
  if (( T1_EPOCH > T0_EPOCH )) && [ "$T1_MAX" = "$T0_MAX" ] && (( T1_DROP >= T0_DROP && T1_INSERT_FAILED >= T0_INSERT_FAILED && T1_EARLY_DROP >= T0_EARLY_DROP )); then
    SAMPLE_READY=true
    printf 'SAMPLE_READY=true seconds=%s count=%s/%s drop-delta=%s insert-failed-delta=%s early-drop-delta=%s\n' "$((T1_EPOCH-T0_EPOCH))" "$T1_COUNT" "$T1_MAX" "$((T1_DROP-T0_DROP))" "$((T1_INSERT_FAILED-T0_INSERT_FAILED))" "$((T1_EARLY_DROP-T0_EARLY_DROP))"
  else
    unset T1_EPOCH T1_COUNT T1_MAX T1_DROP T1_INSERT_FAILED T1_EARLY_DROP
    printf '%s\n' '중단: max 변경, counter reset 또는 시간 원점 불일치' >&2
  fi
else
  printf '%s\n' '중단: t1 producer 또는 current identity 검증 실패' >&2
fi
```

통계는 kernel·conntrack-tools 버전에 따라 per-CPU 누적값이다. 동일 도구·netns·boot 원점만 비교한다. count가 max에 접근하면서 `insert_failed`·`early_drop`가 증가하고 실제 요청 실패와 겹칠 때만 현재 포화 경로를 지지한다.

```bash
if [ "${NETNS_READY:-false}" != true ] || [ "${TIME_READY:-false}" != true ]; then
  printf '%s\n' '중단: NETNS_READY와 TIME_READY가 필요하다.' >&2
else
  host_call journal "$NETNS_PID" "$PID_START_TICKS" "$NETNS_INODE" "$CGROUP_HASH" "$ARGV_REGEX_B64" "$START_RFC3339" "$END_RFC3339"
  RC=$?
  case "$RC" in
    0) printf '%s\n' 'journal producer와 filter 성공' ;;
    60) printf '%s\n' '중단: journal producer 실패' >&2 ;;
    61) printf '%s\n' 'filter 일치 없음: 오류 부재가 아니라 이 창에 일치 행 없음' >&2 ;;
    62) printf '%s\n' '중단: journal filter 실패' >&2 ;;
    *) printf '중단: host/PID/netns identity guard 실패 rc=%s\n' "$RC" >&2 ;;
  esac
fi
```

kernel journal은 node-wide다. 같은 시각의 선택 netns 표본과 actual flow 영향이 없으면 특정 netns·zone의 사건으로 귀속하지 않는다. 전체 conntrack entry와 journal 원문은 덤프하지 않는다.

## zone·Kubernetes 경계

zone별 capacity가 직접 제공되지 않는 환경에서 namespace 전체 count/max를 zone 값으로 표현하지 않는다. CT zone을 조사해야 하면 [[05. iptables·nftables rule에서 packet drop이 확인된 경우]]의 단일 backend·exact rule identity 방식으로 zone assignment rule만 확인하고, 전체 ruleset을 읽지 않는다.

이 문서에는 `kubectl`·CRI 명령이 없다. Kubernetes 영향으로 연결하려면 current Node name·UID·provider ID·InternalIP와 승인 machine-id, current Pod namespace/name/UID·owner kind/name/UID·nodeName·container kind/name/current containerID와 active runtime endpoint를 별도 절차에서 먼저 연결한다. Kubernetes namespace 이름을 Linux netns identity로 사용하지 않는다.

## 변경 전 안전 gate

이 문서는 `conntrack -F`, entry delete, `nf_conntrack_max`·timeout sysctl 변경, rule 변경, service restart와 node reboot 명령을 제공하지 않는다. 별도 변경 절차는 fresh host·PID·netns와 같은 원점의 count/max·counter·journal·실제 영향, approval ID·실행자·변경 창, zone/workload blast radius, availability, desired 상태 no-op, persistent source, 검증된 rollback 또는 보상과 즉시 중단 조건을 모두 가져야 한다.

host reboot, netns/PID 교체, module reload 또는 counter reset 뒤에는 기존 identity·표본을 재사용하지 않는다. replacement를 단일 후보로 다시 등록한다. 이미 손실된 요청과 CNI·firewall·application 복구는 conntrack 조치가 보장하지 않는다.

## 정상화 검증

```bash
clear_target_state
register_host
printf '%s\n' '다음: 새 UTC 창과 current netns PID lifecycle을 다시 등록하고 count/max·counter t0/t1·journal·실제 기능을 재검증한다.'
```

정상화는 새 lifecycle의 bounded 관찰 창에서 count가 max에 지속 접근하지 않고, `insert_failed`·`early_drop`·`drop`가 증가하지 않으며 새 `table full` journal이 없고, 동일 source namespace·protocol·endpoint의 실제 요청·error rate·latency가 회복된 상태다. 현재 count 한 번, Node Ready 또는 Pod Running만으로 과거 고갈과 근본 원인 해결을 선언하지 않는다.

## 증거·완료

보존: 사건 ID, SSH host key 경로, machine-id·boot-id, PID/start ticks/netns inode/cgroup hash/argv match 결과, host netns와의 관계, canonical UTC 창, t0/t1 count/max·필요 counter와 원점, journal producer/filter 상태, zone 확인 방식, 비식별 workload 영향, owner·승인·정상화 결과.

저장 금지: conntrack entry 전체, raw argv·environment·cgroup, IP·port·MAC·token·개인키·개인정보와 전체 journal. 결론은 `current exhaustion`, `historical exhaustion`, `counter reset`, `zone/netns 미확정`, `firewall/다른 계층`, `증거 부족` 중 하나로 쓴다.
