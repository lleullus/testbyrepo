# iptables·nftables rule에서 packet drop이 확인된 경우

## 입력값과 빠른 수집

실행 경로는 `승인된 작업 PC의 Bash → 검증된 SSH host key → machine-id·boot-id → host network namespace → active netfilter backend → family·table·chain·rule → packet/byte counter → 실제 실패 flow`다. Pod netns, 외부 방화벽, CNI 정책과 host netfilter를 같은 경로로 취급하지 않는다.

| 구분 | 값 | 출처와 gate |
|---|---|---|
| 최초 입력 | `INCIDENT_ID`, `SSH_TARGET`, `EXPECTED_MACHINE_ID` | 사건 기록과 승인된 inventory. SSH로 발견한 machine-id를 승인값으로 사용하지 않는다. |
| 시스템 자동값 | `BOOT_ID`, `HOST_NETNS_INODE`, `RULE_FINGERPRINT` | 원격 호출마다 다시 검증한다. fingerprint는 counter를 제외한 exact rule 구조다. |
| 계산 후보값 | `START_EPOCH`, `END_EPOCH`, `T0_*`, `T1_*` | canonical UTC 창과 같은 counter 원점의 두 표본이다. |
| 운영자 선택값 | `BACKEND`, `FAMILY`, `TABLE_NAME`, `CHAIN_NAME`, `RULE_ID` | 현재 host의 단일 active backend와 승인된 drop 증거에서 선택한다. |

각 고유 `PASTE_..._HERE` 입력칸은 한 번만 나온다. backend, rule, host 또는 boot가 바뀌면 이전 counter·시간·evidence와 모든 하위 `READY`를 폐기한다.

### 1. 실행 계약과 host 등록

```bash
TOOLS_READY=false
MISSING_TOOLS=''
for TOOL in ssh date; do
  command -v "$TOOL" >/dev/null 2>&1 || MISSING_TOOLS="$MISSING_TOOLS $TOOL"
done
if [ -n "${BASH_VERSION:-}" ] && [ -z "$MISSING_TOOLS" ]; then
  TOOLS_READY=true
else
  printf '중단: Bash 또는 로컬 도구 누락:%s\n' "$MISSING_TOOLS" >&2
fi

clear_target_state() {
  unset ACTUAL_MACHINE_ID BOOT_ID REMOTE_HOSTNAME HOST_NETNS_INODE BACKEND FAMILY TABLE_NAME CHAIN_NAME RULE_ID RULE_FINGERPRINT START_RFC3339 END_RFC3339 START_EPOCH END_EPOCH T0_PACKETS T0_BYTES T0_EPOCH T1_PACKETS T1_BYTES T1_EPOCH VERDICT
  HOST_READY=false
  TIME_READY=false
  RULE_READY=false
  EVIDENCE_READY=false
  SAMPLE_READY=false
  AUTHORIZATION_READY=false
  MUTATION_READY=false
  ROLLBACK_READY=false
  NORMALIZATION_READY=false
}

register_host() {
  clear_target_state
  if [ "$TOOLS_READY" != true ]; then
    printf '%s\n' '중단: TOOLS_READY가 아니다.' >&2
    return 1
  fi
  if ! [[ "$SSH_TARGET" =~ ^([A-Za-z0-9._-]+@)?[A-Za-z0-9._-]+$ ]] || ! [[ "$EXPECTED_MACHINE_ID" =~ ^[0-9a-f]{32}$ ]]; then
    printf '%s\n' '중단: 승인된 SSH target 또는 machine-id 형식 오류' >&2
    return 1
  fi
  local RESULT RC
  RESULT="$(ssh -o BatchMode=yes -o StrictHostKeyChecking=yes -o ConnectTimeout=10 -- "$SSH_TARGET" bash -s -- "$EXPECTED_MACHINE_ID" <<'REMOTE'
set -uo pipefail
expected=$1
actual=$(< /etc/machine-id) || exit 20
boot=$(< /proc/sys/kernel/random/boot_id) || exit 20
netns=$(readlink /proc/1/ns/net) || exit 20
name=$(hostname) || exit 20
[ "$actual" = "$expected" ] || exit 21
printf '%s\t%s\t%s\t%s\n' "$actual" "$boot" "$name" "$netns"
REMOTE
)"
  RC=$?
  if [ "$RC" -ne 0 ]; then
    clear_target_state
    printf '중단: host 등록 실패 rc=%s\n' "$RC" >&2
    return "$RC"
  fi
  IFS=$'\t' read -r ACTUAL_MACHINE_ID BOOT_ID REMOTE_HOSTNAME HOST_NETNS_INODE <<<"$RESULT"
  if [ -z "$ACTUAL_MACHINE_ID" ] || [ -z "$BOOT_ID" ] || [ -z "$HOST_NETNS_INODE" ]; then
    clear_target_state
    printf '%s\n' '중단: host identity 결과 불완전' >&2
    return 1
  fi
  HOST_READY=true
  printf 'HOST_READY=true host=%s machine-id=%s boot-id=%s netns=%s incident=%s\n' "$REMOTE_HOSTNAME" "$ACTUAL_MACHINE_ID" "$BOOT_ID" "$HOST_NETNS_INODE" "$INCIDENT_ID"
}

host_call() {
  [ "${HOST_READY:-false}" = true ] || { printf '%s\n' '중단: HOST_READY가 아니다.' >&2; return 1; }
  local OP=$1
  shift
  ssh -o BatchMode=yes -o StrictHostKeyChecking=yes -o ConnectTimeout=10 -- "$SSH_TARGET" bash -s -- "$EXPECTED_MACHINE_ID" "$BOOT_ID" "$HOST_NETNS_INODE" "$OP" "$@" <<'REMOTE'
set -uo pipefail
expected_machine=$1; expected_boot=$2; expected_netns=$3; op=$4; shift 4
[ "$(< /etc/machine-id)" = "$expected_machine" ] || exit 21
[ "$(< /proc/sys/kernel/random/boot_id)" = "$expected_boot" ] || exit 22
[ "$(readlink /proc/1/ns/net)" = "$expected_netns" ] || exit 23

rule_data() {
  backend=$1; family=$2; table=$3; chain=$4; rule_id=$5
  [[ "$table" =~ ^[A-Za-z0-9_.-]{1,64}$ ]] || exit 30
  [[ "$chain" =~ ^[A-Za-z0-9_.-]{1,128}$ ]] || exit 30
  [[ "$rule_id" =~ ^[1-9][0-9]*$ ]] || exit 30
  for tool in awk mktemp sha256sum sed; do command -v "$tool" >/dev/null 2>&1 || exit 31; done
  raw=$(mktemp) || exit 32
  selected=$(mktemp) || { rm -f "$raw"; exit 32; }
  canonical=$(mktemp) || { rm -f "$raw" "$selected"; exit 32; }
  case "$backend:$family" in
    iptables:ipv4|ip6tables:ipv6)
      command -v "$backend" >/dev/null 2>&1 || { rm -f "$raw" "$selected" "$canonical"; exit 33; }
      "$backend" -t "$table" -L "$chain" -v -n -x --line-numbers >"$raw" || { rm -f "$raw" "$selected" "$canonical"; exit 34; }
      awk -v id="$rule_id" 'NR>2 && $1==id {print}' "$raw" >"$selected" || { rm -f "$raw" "$selected" "$canonical"; exit 35; }
      count=$(awk 'END{print NR+0}' "$selected")
      [ "$count" -eq 1 ] || { rm -f "$raw" "$selected" "$canonical"; exit 36; }
      awk '{for(i=4;i<=NF;i++) printf "%s%s",$i,(i<NF?OFS:ORS)}' "$selected" >"$canonical" || { rm -f "$raw" "$selected" "$canonical"; exit 35; }
      read -r id packets bytes verdict _ <"$selected"
      ;;
    nft:ip|nft:ip6|nft:inet|nft:bridge)
      command -v nft >/dev/null 2>&1 || { rm -f "$raw" "$selected" "$canonical"; exit 33; }
      nft -a list chain "$family" "$table" "$chain" >"$raw" || { rm -f "$raw" "$selected" "$canonical"; exit 34; }
      awk -v id="$rule_id" '$0 ~ ("handle " id "([^0-9]|$)") {print}' "$raw" >"$selected" || { rm -f "$raw" "$selected" "$canonical"; exit 35; }
      count=$(awk 'END{print NR+0}' "$selected")
      [ "$count" -eq 1 ] || { rm -f "$raw" "$selected" "$canonical"; exit 36; }
      sed -E 's/counter packets [0-9]+ bytes [0-9]+/counter packets # bytes #/' "$selected" >"$canonical" || { rm -f "$raw" "$selected" "$canonical"; exit 35; }
      read -r packets bytes verdict < <(awk '{p="";b="";v="unknown"; for(i=1;i<=NF;i++){if($i=="packets")p=$(i+1); if($i=="bytes")b=$(i+1); if($i=="drop"||$i=="reject")v=$i} print p,b,v}' "$selected")
      id=$rule_id
      ;;
    *) rm -f "$raw" "$selected" "$canonical"; exit 30 ;;
  esac
  [[ "$packets" =~ ^[0-9]+$ ]] && [[ "$bytes" =~ ^[0-9]+$ ]] || { rm -f "$raw" "$selected" "$canonical"; exit 37; }
  digest=$(sha256sum "$canonical") || { rm -f "$raw" "$selected" "$canonical"; exit 38; }
  rm -f "$raw" "$selected" "$canonical"
  printf '%s\t%s\t%s\t%s\n' "${digest%% *}" "$packets" "$bytes" "$verdict"
}

guard_rule() {
  expected=$6
  current=$(rule_data "$1" "$2" "$3" "$4" "$5") || return
  IFS=$'\t' read -r digest packets bytes verdict <<<"$current"
  [ "$digest" = "$expected" ] || exit 39
}

case "$op" in
  rule-register) rule_data "$@" ;;
  rule-sample)
    guard_rule "$@"
    rule_data "${@:1:5}"
    ;;
  journal)
    guard_rule "${@:1:6}"
    start=$7; end=$8
    set +e
    journalctl -k --since "$start" --until "$end" --no-pager -n 300 | grep -E -i -C 3 -- 'netfilter|iptables|nft|drop|reject|martian|conntrack|packet'
    status=("${PIPESTATUS[@]}")
    set -e
    [ "${status[0]}" -eq 0 ] || exit 60
    [ "${status[1]}" -eq 0 ] && exit 0
    [ "${status[1]}" -eq 1 ] && exit 61
    exit 62
    ;;
  *) exit 64 ;;
esac
REMOTE
}

INCIDENT_ID='PASTE_INCIDENT_ID_HERE'
SSH_TARGET='PASTE_SSH_TARGET_HERE'
EXPECTED_MACHINE_ID='PASTE_EXPECTED_MACHINE_ID_HERE'
clear_target_state
```

`EXPECTED_MACHINE_ID`는 승인 inventory에서 가져온다. host key 오류, machine-id 불일치와 boot 변경은 우회하지 않는다.

```bash
register_host
```

### 2. canonical UTC 증거 창

```bash
START_RFC3339='PASTE_START_RFC3339_HERE'
END_RFC3339='PASTE_END_RFC3339_HERE'
TIME_READY=false
unset START_EPOCH END_EPOCH
RFC3339_RE='^[0-9]{4}-(0[1-9]|1[0-2])-([0-2][0-9]|3[01])T([01][0-9]|2[0-3]):[0-5][0-9]:[0-5][0-9]Z$'
if [ "${HOST_READY:-false}" != true ]; then
  printf '%s\n' '중단: HOST_READY가 아니다.' >&2
elif ! [[ "$START_RFC3339" =~ $RFC3339_RE ]] || ! [[ "$END_RFC3339" =~ $RFC3339_RE ]]; then
  printf '%s\n' '중단: UTC RFC3339 형식 오류' >&2
elif [ "$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" != "$START_RFC3339" ] || [ "$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" != "$END_RFC3339" ]; then
  printf '%s\n' '중단: 존재하지 않거나 canonical하지 않은 시각' >&2
else
  START_EPOCH=$(date -u -d "$START_RFC3339" +%s)
  END_EPOCH=$(date -u -d "$END_RFC3339" +%s)
  if (( START_EPOCH >= END_EPOCH || END_EPOCH - START_EPOCH > 86400 )); then
    unset START_EPOCH END_EPOCH
    printf '%s\n' '중단: 시작은 종료보다 앞서야 하고 창은 24시간 이하여야 한다.' >&2
  else
    TIME_READY=true
    printf 'TIME_READY=true start=%s end=%s seconds=%s\n' "$START_RFC3339" "$END_RFC3339" "$((END_EPOCH-START_EPOCH))"
  fi
fi
```

### 3. active rule identity 등록

`BACKEND`는 `iptables`, `ip6tables`, `nft` 중 하나다. iptables 계열의 `RULE_ID`는 `--line-numbers`의 exact rule 번호이고 nft의 값은 exact handle이다. version만으로 active backend를 고르지 않는다.

```bash
BACKEND='PASTE_CONFIRMED_BACKEND_HERE'
FAMILY='PASTE_CONFIRMED_FAMILY_HERE'
TABLE_NAME='PASTE_CONFIRMED_TABLE_HERE'
CHAIN_NAME='PASTE_CONFIRMED_CHAIN_HERE'
RULE_ID='PASTE_EXACT_RULE_NUMBER_OR_HANDLE_HERE'
RULE_READY=false
EVIDENCE_READY=false
SAMPLE_READY=false
unset RULE_FINGERPRINT T0_PACKETS T0_BYTES T0_EPOCH T1_PACKETS T1_BYTES T1_EPOCH VERDICT
if [ "${HOST_READY:-false}" != true ]; then
  printf '%s\n' '중단: HOST_READY가 아니다.' >&2
elif RESULT=$(host_call rule-register "$BACKEND" "$FAMILY" "$TABLE_NAME" "$CHAIN_NAME" "$RULE_ID"); then
  IFS=$'\t' read -r RULE_FINGERPRINT T0_PACKETS T0_BYTES VERDICT <<<"$RESULT"
  if [[ "$RULE_FINGERPRINT" =~ ^[0-9a-f]{64}$ ]] && [[ "$T0_PACKETS" =~ ^[0-9]+$ ]] && [[ "$T0_BYTES" =~ ^[0-9]+$ ]] && { [ "$VERDICT" = DROP ] || [ "$VERDICT" = REJECT ] || [ "$VERDICT" = drop ] || [ "$VERDICT" = reject ]; }; then
    T0_EPOCH=$(date -u +%s)
    RULE_READY=true
    printf 'RULE_READY=true backend=%s family=%s table=%s chain=%s rule=%s fingerprint=%s verdict=%s t0-packets=%s t0-bytes=%s\n' "$BACKEND" "$FAMILY" "$TABLE_NAME" "$CHAIN_NAME" "$RULE_ID" "$RULE_FINGERPRINT" "$VERDICT" "$T0_PACKETS" "$T0_BYTES"
  else
    unset RULE_FINGERPRINT T0_PACKETS T0_BYTES T0_EPOCH VERDICT
    printf '%s\n' '중단: rule 결과 불완전 또는 verdict가 DROP/REJECT가 아님' >&2
  fi
else
  unset RULE_FINGERPRINT T0_PACKETS T0_BYTES T0_EPOCH VERDICT
  printf '%s\n' '중단: backend producer 실패, rule 없음 또는 복수 match' >&2
fi
```

`RULE_READY=true`는 exact rule 구조와 counter 원점을 고정한 것이지, 그 rule이 현재 실패 flow를 drop했다는 뜻이 아니다. nft rule에 counter 표현식이 없거나 verdict가 `drop`·`reject`가 아니면 이 경로를 중단한다.

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | exact host rule의 `DROP`·`REJECT`와 packet/byte counter 증가가 실제 장애 창에 관측된 경우 |
| 적용 | machine·boot·host netns와 단일 backend·family·table·chain·rule identity를 유지할 수 있는 경우 |
| 제외 | conntrack table full은 [[06. conntrack table이 가득 찬 경우]], route 불일치는 [[03. route·default gateway가 기대 경로와 다른 경우]], Pod/CNI/NetworkPolicy 또는 외부 장비가 최초 증거이면 해당 owner 경로로 이동한다. |

검색 문자열: `DROP`, `REJECT`, `verdict drop`, `policy drop`, `pkts`, `bytes`, `handle`, `iptables-legacy`, `iptables-nft`, `nftables`.

## 관측별 진단 경로

| 지금 보이는 것 | 직접 증거 | 다음 행동 |
|---|---|---|
| rule 구조가 등록 직후 바뀜 | fingerprint mismatch | 모든 rule·counter 상태를 폐기하고 현재 active backend부터 재등록 |
| verdict와 counter가 같은 장애 창에 증가 | t0/t1 delta와 실제 flow 시각·방향 | firewall owner에게 exact rule identity와 비식별 flow 이관 |
| counter가 증가하지 않음 | 같은 원점·producer 성공 | stale rule, 다른 hook/node/backend 또는 다른 계층 확인 |
| `table full`·`insert_failed`가 먼저임 | kernel 창과 conntrack 통계 | [[06. conntrack table이 가득 찬 경우]] |
| rule은 유지되지만 route·application 실패 | current route/listener 기능 | route·listener·MTU 경로 |

## 읽기 전용 증거

아래 명령은 현재 machine·boot·netns와 rule fingerprint를 원격에서 직접 다시 확인한 뒤 counter만 반환한다. 전체 ruleset이나 주소·mark·comment를 출력하지 않는다.

```bash
if [ "${RULE_READY:-false}" != true ]; then
  printf '%s\n' '중단: RULE_READY가 아니다.' >&2
elif RESULT=$(host_call rule-sample "$BACKEND" "$FAMILY" "$TABLE_NAME" "$CHAIN_NAME" "$RULE_ID" "$RULE_FINGERPRINT"); then
  IFS=$'\t' read -r CURRENT_FINGERPRINT CURRENT_PACKETS CURRENT_BYTES CURRENT_VERDICT <<<"$RESULT"
  if [ "$CURRENT_FINGERPRINT" = "$RULE_FINGERPRINT" ]; then
    EVIDENCE_READY=true
    printf 'EVIDENCE_READY=true packets=%s bytes=%s verdict=%s observed-at=%s\n' "$CURRENT_PACKETS" "$CURRENT_BYTES" "$CURRENT_VERDICT" "$(date -u '+%Y-%m-%dT%H:%M:%SZ')"
  else
    EVIDENCE_READY=false
    printf '%s\n' '중단: rule fingerprint 변경' >&2
  fi
else
  EVIDENCE_READY=false
  printf '%s\n' '중단: identity 변경 또는 counter producer 실패' >&2
fi
```

승인된 관찰 간격과 실제 실패 요청이 지난 뒤 같은 원점의 t1을 등록한다. 이 명령은 traffic을 만들지 않는다.

```bash
SAMPLE_READY=false
unset T1_PACKETS T1_BYTES T1_EPOCH
if [ "${EVIDENCE_READY:-false}" != true ]; then
  printf '%s\n' '중단: EVIDENCE_READY가 아니다.' >&2
elif RESULT=$(host_call rule-sample "$BACKEND" "$FAMILY" "$TABLE_NAME" "$CHAIN_NAME" "$RULE_ID" "$RULE_FINGERPRINT"); then
  IFS=$'\t' read -r CURRENT_FINGERPRINT T1_PACKETS T1_BYTES CURRENT_VERDICT <<<"$RESULT"
  T1_EPOCH=$(date -u +%s)
  if [ "$CURRENT_FINGERPRINT" = "$RULE_FINGERPRINT" ] && (( T1_EPOCH > T0_EPOCH )) && (( T1_PACKETS >= T0_PACKETS && T1_BYTES >= T0_BYTES )); then
    SAMPLE_READY=true
    printf 'SAMPLE_READY=true seconds=%s packet-delta=%s byte-delta=%s verdict=%s\n' "$((T1_EPOCH-T0_EPOCH))" "$((T1_PACKETS-T0_PACKETS))" "$((T1_BYTES-T0_BYTES))" "$CURRENT_VERDICT"
  else
    unset T1_PACKETS T1_BYTES T1_EPOCH
    printf '%s\n' '중단: rule replacement, counter reset 또는 시간 원점 불일치' >&2
  fi
else
  printf '%s\n' '중단: t1 producer 실패' >&2
fi
```

delta가 0이면 현재 요청의 drop 원인으로 확정하지 않는다. delta가 양수여도 같은 hook·방향·protocol·port와 실패 시각이 맞아야 한다. 실제 IP, MAC, mark와 전체 rule 원문은 저장하지 않는다.

```bash
if [ "${RULE_READY:-false}" != true ] || [ "${TIME_READY:-false}" != true ]; then
  printf '%s\n' '중단: RULE_READY와 TIME_READY가 필요하다.' >&2
else
  host_call journal "$BACKEND" "$FAMILY" "$TABLE_NAME" "$CHAIN_NAME" "$RULE_ID" "$RULE_FINGERPRINT" "$START_RFC3339" "$END_RFC3339"
  RC=$?
  case "$RC" in
    0) printf '%s\n' 'journal producer와 filter 성공' ;;
    60) printf '%s\n' '중단: journal producer 실패' >&2 ;;
    61) printf '%s\n' 'filter 일치 없음: 오류 부재가 아니라 이 창에 일치 행 없음' >&2 ;;
    62) printf '%s\n' '중단: journal filter 실패' >&2 ;;
    *) printf '중단: host/rule identity guard 실패 rc=%s\n' "$RC" >&2 ;;
  esac
fi
```

kernel journal은 node-wide 증거이며 특정 rule의 counter가 아니다. producer 성공, filter no-match와 filter 오류를 분리하고, exact rule delta와 실제 flow 시간축이 없으면 원인을 확정하지 않는다.

## Kubernetes·namespace 경계

이 Linux host 런북은 `kubectl`·CRI를 실행하지 않는다. Kubernetes 영향으로 연결하려면 별도 절차에서 current Node name·UID·provider ID·InternalIP가 승인 machine-id와 일치하고, current Pod namespace/name/UID·owner kind/name/UID·`spec.nodeName`·container kind/name/current containerID가 그 Node와 active runtime endpoint에 연결됨을 먼저 증명한다. 이 관계 없이 Node 이름이나 Pod IP만으로 host rule 결과를 workload에 귀속하지 않는다.

## 변경 전 안전 gate

이 문서에는 iptables/nft rule add·delete·flush, counter reset, sysctl, service restart와 rollback 명령이 없다. 별도 변경 절차는 fresh host와 exact rule fingerprint, 같은 창의 t0/t1·실제 flow 증거, approval ID·실행자·변경 시각, rule owner, blast radius·가용성, desired 상태의 no-op 조건, persistent source, 검증된 rollback과 즉시 중단 조건을 모두 가져야 한다.

rule 번호·nft handle·chain이 교체되거나 ruleset reload·host reboot가 있으면 기존 이름을 재사용하지 않는다. 모든 자동값과 `READY`를 폐기하고 active backend부터 단일 후보를 다시 등록한다. `flush`나 counter reset은 rollback이 아니다.

## 정상화 검증

```bash
clear_target_state
register_host
printf '%s\n' '다음: 새 UTC 창과 현재 active backend·exact rule을 다시 선택·등록하고 counter t0/t1과 실제 기능을 재검증한다.'
```

정상화는 새 lifecycle에서 원인 rule의 예상하지 않은 counter 증가가 bounded 관찰 창 동안 재발하지 않고, 동일 source namespace·protocol·endpoint의 승인된 실제 요청과 핵심 기능이 회복된 상태다. rule이 삭제됐다면 current ruleset의 의도한 대체 rule·hook 관계와 rollback 가능성을 새 identity로 확인한다. Node Ready, counter delta 0 한 번 또는 단일 요청 성공만으로 닫지 않는다.

## 증거·완료

보존: 사건 ID, SSH host key 확인 경로, machine-id·boot-id·host netns, backend·family·table·chain·exact rule ID·fingerprint·verdict, canonical UTC 창, t0/t1 counter와 원점, producer/filter 상태, 비식별 flow 방향·protocol·port, owner·승인·정상화 결과.

저장 금지: 전체 ruleset·journal, source/destination IP, MAC, mark, comment, credential·token·개인키·개인정보. 결론은 `host rule drop`, `counter 상관 없음`, `rule replacement`, `conntrack`, `다른 namespace/backend`, `증거 부족` 중 하나로 쓴다.
