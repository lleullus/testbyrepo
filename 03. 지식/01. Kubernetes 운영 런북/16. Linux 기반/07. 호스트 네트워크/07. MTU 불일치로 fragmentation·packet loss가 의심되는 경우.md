# MTU 불일치로 fragmentation·packet loss가 의심되는 경우

## 입력값과 빠른 수집

실행 경로는 `승인된 작업 PC의 Bash → 검증된 SSH host key → machine-id·boot-id → host network namespace → ifindex·interface MTU → exact source/destination route → underlay·tunnel·peer → PMTU ICMP → 승인된 bounded probe → 실제 application flow`다. host route를 Pod·VRF·다른 netns 경로로 재사용하지 않는다.

| 구분 | 값 | 출처와 gate |
|---|---|---|
| 최초 입력 | `INCIDENT_ID`, `SSH_TARGET`, `EXPECTED_MACHINE_ID`, `SOURCE_IP`, `DESTINATION_IP` | 실제 실패 flow와 승인된 inventory. DNS·naming convention으로 추측하지 않는다. |
| 시스템 자동값 | `BOOT_ID`, `HOST_NETNS_INODE`, `IFINDEX`, `INTERFACE_MTU`, `PATH_FINGERPRINT` | 모든 read·journal·probe가 원격에서 직접 다시 검증한다. |
| 계산 후보값 | canonical UTC epoch, `BASELINE_PAYLOAD`, `SUSPECT_PAYLOAD` | payload는 승인 후 검증하는 후보이며 path MTU와 동일하지 않다. |
| 운영자 선택값 | `IFACE`, 조건부 payload·`APPROVAL_ID` | 현재 route/interface 출력과 실제 endpoint owner 승인에서 선택한다. |

각 고유 `PASTE_..._HERE` 입력칸은 한 번만 나온다. host reboot, interface/netns 교체 또는 route fingerprint 변경 시 기존 증거·probe 승인과 모든 하위 `READY`를 폐기한다.

### 1. 실행 계약과 host 등록

```bash
TOOLS_READY=false
MISSING_TOOLS=''
for TOOL in ssh date; do
  command -v "$TOOL" >/dev/null 2>&1 || MISSING_TOOLS="$MISSING_TOOLS $TOOL"
done
if [ -n "${BASH_VERSION:-}" ] && [ -z "$MISSING_TOOLS" ]; then
  TOOLS_READY=true
else
  printf '중단: Bash 또는 로컬 도구 누락:%s\n' "$MISSING_TOOLS" >&2
fi

clear_target_state() {
  unset ACTUAL_MACHINE_ID BOOT_ID REMOTE_HOSTNAME HOST_NETNS_INODE IFACE IFINDEX NETNS_INODE INTERFACE_MTU PATH_FINGERPRINT START_RFC3339 END_RFC3339 START_EPOCH END_EPOCH PEER_EVIDENCE_ID BASELINE_PAYLOAD SUSPECT_PAYLOAD APPROVAL_ID
  HOST_READY=false
  TIME_READY=false
  PATH_READY=false
  EVIDENCE_READY=false
  PEER_EVIDENCE_READY=false
  AUTHORIZATION_READY=false
  PROBE_READY=false
  MUTATION_READY=false
  ROLLBACK_READY=false
  NORMALIZATION_READY=false
}

register_host() {
  clear_target_state
  if [ "$TOOLS_READY" != true ]; then printf '%s\n' '중단: TOOLS_READY가 아니다.' >&2; return 1; fi
  if ! [[ "$SSH_TARGET" =~ ^([A-Za-z0-9._-]+@)?[A-Za-z0-9._-]+$ ]] || ! [[ "$EXPECTED_MACHINE_ID" =~ ^[0-9a-f]{32}$ ]]; then
    printf '%s\n' '중단: SSH target 또는 machine-id 형식 오류' >&2
    return 1
  fi
  if ! [[ "$SOURCE_IP" =~ ^[0-9A-Fa-f:.]+$ ]] || ! [[ "$DESTINATION_IP" =~ ^[0-9A-Fa-f:.]+$ ]]; then
    printf '%s\n' '중단: source/destination은 숫자 IP여야 한다.' >&2
    return 1
  fi
  local RESULT RC
  RESULT="$(ssh -o BatchMode=yes -o StrictHostKeyChecking=yes -o ConnectTimeout=10 -- "$SSH_TARGET" bash -s -- "$EXPECTED_MACHINE_ID" <<'REMOTE'
set -uo pipefail
expected=$1
actual=$(< /etc/machine-id) || exit 20
boot=$(< /proc/sys/kernel/random/boot_id) || exit 20
netns=$(readlink /proc/1/ns/net) || exit 20
name=$(hostname) || exit 20
[ "$actual" = "$expected" ] || exit 21
printf '%s\t%s\t%s\t%s\n' "$actual" "$boot" "$name" "$netns"
REMOTE
)"
  RC=$?
  if [ "$RC" -ne 0 ]; then clear_target_state; printf '중단: host 등록 실패 rc=%s\n' "$RC" >&2; return "$RC"; fi
  IFS=$'\t' read -r ACTUAL_MACHINE_ID BOOT_ID REMOTE_HOSTNAME HOST_NETNS_INODE <<<"$RESULT"
  if [ -z "$ACTUAL_MACHINE_ID" ] || [ -z "$BOOT_ID" ] || [ -z "$HOST_NETNS_INODE" ]; then clear_target_state; printf '%s\n' '중단: host identity 불완전' >&2; return 1; fi
  HOST_READY=true
  printf 'HOST_READY=true host=%s machine-id=%s boot-id=%s host-netns=%s incident=%s\n' "$REMOTE_HOSTNAME" "$ACTUAL_MACHINE_ID" "$BOOT_ID" "$HOST_NETNS_INODE" "$INCIDENT_ID"
}

host_call() {
  [ "${HOST_READY:-false}" = true ] || { printf '%s\n' '중단: HOST_READY가 아니다.' >&2; return 1; }
  local OP=$1
  shift
  ssh -o BatchMode=yes -o StrictHostKeyChecking=yes -o ConnectTimeout=10 -- "$SSH_TARGET" bash -s -- "$EXPECTED_MACHINE_ID" "$BOOT_ID" "$OP" "$@" <<'REMOTE'
set -uo pipefail
expected_machine=$1; expected_boot=$2; op=$3; shift 3
[ "$(< /etc/machine-id)" = "$expected_machine" ] || exit 21
[ "$(< /proc/sys/kernel/random/boot_id)" = "$expected_boot" ] || exit 22

family_flag() {
  if [[ "$1" == *.* && "$1" != *:* && "$2" == *.* && "$2" != *:* ]]; then
    printf '%s' '-4'
  elif [[ "$1" == *:* && "$2" == *:* && "$1" != *.* && "$2" != *.* ]]; then
    printf '%s' '-6'
  else
    exit 30
  fi
}

path_data() {
  iface=$1; source=$2; destination=$3
  [[ "$iface" =~ ^[A-Za-z0-9_.:@-]{1,64}$ ]] || exit 31
  command -v ip >/dev/null 2>&1 || exit 32
  command -v sha256sum >/dev/null 2>&1 || exit 32
  ifindex=$(< "/sys/class/net/$iface/ifindex") || exit 33
  netns=$(readlink /proc/1/ns/net) || exit 33
  mtu=$(< "/sys/class/net/$iface/mtu") || exit 33
  [[ "$ifindex" =~ ^[1-9][0-9]*$ ]] && [[ "$mtu" =~ ^[1-9][0-9]*$ ]] || exit 33
  family=$(family_flag "$source" "$destination") || exit 30
  route=$(ip "$family" route get "$destination" from "$source") || exit 34
  [ -n "$route" ] || exit 34
  case " $route " in *" dev $iface "*) ;; *) exit 35 ;; esac
  fingerprint_input=$(printf '%s\n%s\n%s\n%s\n%s\n' "$ifindex" "$netns" "$mtu" "$family" "$route") || exit 36
  digest=$(sha256sum <<<"$fingerprint_input") || exit 36
  printf '%s\t%s\t%s\t%s\n' "$ifindex" "$netns" "$mtu" "${digest%% *}"
}

guard_path() {
  current=$(path_data "$1" "$2" "$3") || return
  IFS=$'\t' read -r ifindex netns mtu fingerprint <<<"$current"
  [ "$ifindex" = "$4" ] && [ "$netns" = "$5" ] && [ "$mtu" = "$6" ] && [ "$fingerprint" = "$7" ] || exit 37
}

case "$op" in
  path-register) path_data "$@" ;;
  path-read)
    guard_path "$@"
    iface=$1
    operstate=$(< "/sys/class/net/$iface/operstate") || exit 40
    rx_drop=$(< "/sys/class/net/$iface/statistics/rx_dropped") || exit 40
    tx_drop=$(< "/sys/class/net/$iface/statistics/tx_dropped") || exit 40
    rx_err=$(< "/sys/class/net/$iface/statistics/rx_errors") || exit 40
    tx_err=$(< "/sys/class/net/$iface/statistics/tx_errors") || exit 40
    printf 'interface=%s ifindex=%s netns=%s mtu=%s state=%s rx-drop=%s tx-drop=%s rx-error=%s tx-error=%s path-fingerprint=%s\n' "$iface" "$4" "$5" "$6" "$operstate" "$rx_drop" "$tx_drop" "$rx_err" "$tx_err" "$7"
    ;;
  journal)
    guard_path "${@:1:7}"
    start=$8; end=$9
    set +e
    journalctl -k --since "$start" --until "$end" --no-pager -n 300 | grep -E -i -C 3 -- 'fragment|frag needed|packet too big|icmp|pmtu|mtu|message too long|drop'
    status=("${PIPESTATUS[@]}")
    set -e
    [ "${status[0]}" -eq 0 ] || exit 60
    [ "${status[1]}" -eq 0 ] && exit 0
    [ "${status[1]}" -eq 1 ] && exit 61
    exit 62
    ;;
  probe)
    guard_path "${@:1:7}"
    payload=$8; approval=$9; peer_evidence=${10}
    [[ "$payload" =~ ^[0-9]+$ ]] && (( payload >= 16 && payload <= 9000 )) || exit 70
    [[ "$approval" =~ ^[A-Za-z0-9._:-]{3,128}$ ]] || exit 71
    [[ "$peer_evidence" =~ ^[A-Za-z0-9._:-]{3,128}$ ]] || exit 71
    family=$(family_flag "$2" "$3") || exit 30
    command -v ping >/dev/null 2>&1 || exit 72
    exec ping "$family" -n -I "$1" -M do -c 3 -W 2 -s "$payload" "$3"
    ;;
  *) exit 64 ;;
esac
REMOTE
}

INCIDENT_ID='PASTE_INCIDENT_ID_HERE'
SSH_TARGET='PASTE_SSH_TARGET_HERE'
EXPECTED_MACHINE_ID='PASTE_EXPECTED_MACHINE_ID_HERE'
SOURCE_IP='PASTE_ACTUAL_SOURCE_IP_HERE'
DESTINATION_IP='PASTE_ACTUAL_DESTINATION_IP_HERE'
clear_target_state
```

```bash
register_host
```

### 2. canonical UTC 증거 창

```bash
START_RFC3339='PASTE_START_RFC3339_HERE'
END_RFC3339='PASTE_END_RFC3339_HERE'
TIME_READY=false
unset START_EPOCH END_EPOCH
RFC3339_RE='^[0-9]{4}-(0[1-9]|1[0-2])-([0-2][0-9]|3[01])T([01][0-9]|2[0-3]):[0-5][0-9]:[0-5][0-9]Z$'
if [ "${HOST_READY:-false}" != true ]; then
  printf '%s\n' '중단: HOST_READY가 아니다.' >&2
elif ! [[ "$START_RFC3339" =~ $RFC3339_RE ]] || ! [[ "$END_RFC3339" =~ $RFC3339_RE ]]; then
  printf '%s\n' '중단: UTC RFC3339 형식 오류' >&2
elif [ "$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" != "$START_RFC3339" ] || [ "$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" != "$END_RFC3339" ]; then
  printf '%s\n' '중단: date round-trip 실패' >&2
else
  START_EPOCH=$(date -u -d "$START_RFC3339" +%s)
  END_EPOCH=$(date -u -d "$END_RFC3339" +%s)
  if (( START_EPOCH >= END_EPOCH || END_EPOCH - START_EPOCH > 86400 )); then
    unset START_EPOCH END_EPOCH
    printf '%s\n' '중단: 시작은 종료보다 앞서야 하고 창은 24시간 이하여야 한다.' >&2
  else
    TIME_READY=true
    printf 'TIME_READY=true start=%s end=%s seconds=%s\n' "$START_RFC3339" "$END_RFC3339" "$((END_EPOCH-START_EPOCH))"
  fi
fi
```

### 3. interface·path identity 등록

`IFACE`는 현재 host의 승인된 route/interface 출력에서 선택한다. 이름만 보존하지 않고 host netns·ifindex·MTU와 exact source/destination route fingerprint를 묶는다.

```bash
IFACE='PASTE_CONFIRMED_INTERFACE_HERE'
PATH_READY=false
EVIDENCE_READY=false
PEER_EVIDENCE_READY=false
AUTHORIZATION_READY=false
PROBE_READY=false
unset IFINDEX NETNS_INODE INTERFACE_MTU PATH_FINGERPRINT BASELINE_PAYLOAD SUSPECT_PAYLOAD APPROVAL_ID
if [ "${HOST_READY:-false}" != true ]; then
  printf '%s\n' '중단: HOST_READY가 아니다.' >&2
elif RESULT=$(host_call path-register "$IFACE" "$SOURCE_IP" "$DESTINATION_IP"); then
  IFS=$'\t' read -r IFINDEX NETNS_INODE INTERFACE_MTU PATH_FINGERPRINT <<<"$RESULT"
  if [ "$NETNS_INODE" = "$HOST_NETNS_INODE" ] && [[ "$IFINDEX" =~ ^[1-9][0-9]*$ ]] && [[ "$INTERFACE_MTU" =~ ^[1-9][0-9]*$ ]] && [[ "$PATH_FINGERPRINT" =~ ^[0-9a-f]{64}$ ]]; then
    PATH_READY=true
    printf 'PATH_READY=true interface=%s ifindex=%s netns=%s mtu=%s fingerprint=%s\n' "$IFACE" "$IFINDEX" "$NETNS_INODE" "$INTERFACE_MTU" "$PATH_FINGERPRINT"
  else
    unset IFINDEX NETNS_INODE INTERFACE_MTU PATH_FINGERPRINT
    printf '%s\n' '중단: host netns/interface/path identity 불일치' >&2
  fi
else
  unset IFINDEX NETNS_INODE INTERFACE_MTU PATH_FINGERPRINT
  printf '%s\n' '중단: address family 불일치, route 없음 또는 선택 interface 불일치' >&2
fi
```

등록 성공은 path가 정상이라는 뜻이 아니다. 현재 source/destination lookup이 선택 interface를 사용한다는 lifecycle만 고정한다.

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | 큰 payload·TLS·장시간 연결에서만 timeout/retransmit이 증가하거나 `fragmentation needed`·`Packet Too Big`·`Message too long`이 실제 창에 있는 경우 |
| 적용 | machine·boot·host netns·ifindex·MTU·same-family source/destination route와 peer 증거를 유지할 수 있는 경우 |
| 제외 | 작은 payload도 모든 경로에서 loss, link down·carrier/driver error, route 불일치, firewall/conntrack drop 또는 Pod/CNI identity가 먼저면 해당 전문 경로로 이동한다. |

검색 문자열: `MTU`, `PMTU`, `fragmentation needed`, `Packet Too Big`, `Message too long`, `ICMP type 3 code 4`, `ICMPv6 type 2`, `retransmit`, `rx_dropped`, `tx_dropped`.

## 관측별 진단 경로

| 지금 보이는 것 | 직접 증거 | 다음 행동 |
|---|---|---|
| source interface MTU와 route가 유지됨 | current path fingerprint·counter | peer/underlay/tunnel MTU와 반대 방향 확인 |
| route 또는 MTU가 바뀜 | fingerprint mismatch | 모든 evidence·probe 승인 폐기 후 replacement 등록 |
| 큰 payload만 실패하고 PMTU ICMP가 있음 | bounded probe·journal·peer 시각 | 실제 병목 hop owner에게 이관 |
| 큰 payload만 실패하지만 ICMP 없음 | 양 끝 route와 firewall exact counter | PMTU black-hole 후보; rule/장비 owner 확인 |
| 모든 크기에서 drop/error 증가 | interface counter·carrier/driver | MTU로 확정하지 않고 link/driver owner 이관 |
| host path는 맞지만 Pod만 실패 | current Pod/node/container/netns relation | CNI·Pod netns 전문 경로 |

## 읽기 전용 path·journal 증거

```bash
EVIDENCE_READY=false
if [ "${PATH_READY:-false}" != true ]; then
  printf '%s\n' '중단: PATH_READY가 아니다.' >&2
elif host_call path-read "$IFACE" "$SOURCE_IP" "$DESTINATION_IP" "$IFINDEX" "$NETNS_INODE" "$INTERFACE_MTU" "$PATH_FINGERPRINT"; then
  EVIDENCE_READY=true
  printf 'EVIDENCE_READY=true observed-at=%s\n' "$(date -u '+%Y-%m-%dT%H:%M:%SZ')"
else
  printf '%s\n' '중단: interface/route replacement 또는 producer 실패' >&2
fi
```

출력의 MTU·operstate·drop/error counter는 local interface 관측이다. tunnel overhead, peer MTU, 중간 장비 path MTU와 application 성공을 보장하지 않는다. t0/t1 counter를 비교할 때는 같은 boot·ifindex 원점을 사용하고 reset·wrap·interface replacement를 분리한다.

```bash
if [ "${PATH_READY:-false}" != true ] || [ "${TIME_READY:-false}" != true ]; then
  printf '%s\n' '중단: PATH_READY와 TIME_READY가 필요하다.' >&2
else
  host_call journal "$IFACE" "$SOURCE_IP" "$DESTINATION_IP" "$IFINDEX" "$NETNS_INODE" "$INTERFACE_MTU" "$PATH_FINGERPRINT" "$START_RFC3339" "$END_RFC3339"
  RC=$?
  case "$RC" in
    0) printf '%s\n' 'journal producer와 filter 성공' ;;
    60) printf '%s\n' '중단: journal producer 실패' >&2 ;;
    61) printf '%s\n' 'filter 일치 없음: PMTU 정상 증거가 아니라 이 창에 일치 행 없음' >&2 ;;
    62) printf '%s\n' '중단: journal filter 실패' >&2 ;;
    *) printf '중단: host/path identity guard 실패 rc=%s\n' "$RC" >&2 ;;
  esac
fi
```

IPv4 `fragmentation needed`와 IPv6 `Packet Too Big`의 source 반환 여부, 실제 장애와 timestamp를 대조한다. 메시지가 없다는 사실은 PMTU 문제 부재를 증명하지 않는다. firewall 확인은 [[05. iptables·nftables rule에서 packet drop이 확인된 경우]]의 단일 backend·exact rule 절차로 수행하며 두 backend 결과를 합치지 않는다.

## peer·overlay 증거 gate

상대 endpoint owner에게 같은 canonical 창의 machine/device identity, source 반대 방향 route, interface/tunnel MTU, encapsulation 방식, PMTU ICMP와 drop/error counter를 요청한다. 실제 peer identity가 확인되기 전에는 `PEER_EVIDENCE_READY=true`로 만들지 않는다. 이 문서에서 임의 peer host에 SSH하거나 private address를 탐색하지 않는다. `PEER_EVIDENCE_ID`는 이 확인 묶음의 비민감 사건 기록 ID다.

다음 조건을 사건 기록에서 모두 확인한 경우에만 현재 Bash에 peer evidence gate를 세운다.

```bash
PEER_EVIDENCE_READY=false
PEER_EVIDENCE_ID='PASTE_PEER_EVIDENCE_RECORD_ID_HERE'
if [ "${EVIDENCE_READY:-false}" != true ] || [ "${TIME_READY:-false}" != true ]; then
  printf '%s\n' '중단: source evidence와 canonical 창이 필요하다.' >&2
elif ! [[ "$PEER_EVIDENCE_ID" =~ ^[A-Za-z0-9._:-]{3,128}$ ]]; then
  printf '%s\n' '중단: peer evidence record ID 형식 오류' >&2
else
  PEER_EVIDENCE_READY=true
  printf 'PEER_EVIDENCE_READY=true record=%s\n' "$PEER_EVIDENCE_ID"
fi
```

record ID는 peer identity·출처·시각·owner와 reverse-path 증거가 사건 기록에 이미 묶인 경우에만 입력한다. 일반적인 VXLAN/Geneve overhead 수치만으로 실제 encapsulation과 병목 MTU를 확정하지 않는다.

## 승인된 bounded probe

ping은 진단 기본값이 아니며 duplicate·application 검증기가 아니다. 실제 destination owner가 ICMP 각 payload 3회를 승인하고, 해당 host netns·source/interface가 원래 flow와 유효하게 연결될 때만 사용한다. internet, 임의 private target과 다른 tenant에는 실행하지 않는다.

```bash
BASELINE_PAYLOAD='PASTE_APPROVED_BASELINE_PAYLOAD_HERE'
SUSPECT_PAYLOAD='PASTE_APPROVED_SUSPECT_PAYLOAD_HERE'
APPROVAL_ID='PASTE_PACKET_TEST_APPROVAL_ID_HERE'
AUTHORIZATION_READY=false
PROBE_READY=false
if [ "${EVIDENCE_READY:-false}" != true ] || [ "${PEER_EVIDENCE_READY:-false}" != true ]; then
  printf '%s\n' '중단: source와 peer evidence가 모두 필요하다.' >&2
elif ! [[ "$BASELINE_PAYLOAD" =~ ^[0-9]+$ ]] || ! [[ "$SUSPECT_PAYLOAD" =~ ^[0-9]+$ ]] || (( BASELINE_PAYLOAD < 16 || SUSPECT_PAYLOAD <= BASELINE_PAYLOAD || SUSPECT_PAYLOAD > 9000 )); then
  printf '%s\n' '중단: payload는 16..9000 범위에서 baseline < suspect여야 한다.' >&2
elif ! [[ "$APPROVAL_ID" =~ ^[A-Za-z0-9._:-]{3,128}$ ]]; then
  printf '%s\n' '중단: packet test 승인 ID 형식 오류' >&2
else
  AUTHORIZATION_READY=true
  PROBE_READY=true
  printf 'PROBE_READY=true baseline=%s suspect=%s approval=%s max-packets=6\n' "$BASELINE_PAYLOAD" "$SUSPECT_PAYLOAD" "$APPROVAL_ID"
fi
```

각 probe 명령은 current host·boot·netns·ifindex·MTU·route fingerprint와 approval ID를 직접 검사한다. 첫 명령 실패 시 두 번째를 자동 실행하지 않는다.

```bash
if [ "${PROBE_READY:-false}" != true ] || [ "${AUTHORIZATION_READY:-false}" != true ]; then
  printf '%s\n' '중단: PROBE_READY와 AUTHORIZATION_READY가 필요하다.' >&2
elif host_call probe "$IFACE" "$SOURCE_IP" "$DESTINATION_IP" "$IFINDEX" "$NETNS_INODE" "$INTERFACE_MTU" "$PATH_FINGERPRINT" "$BASELINE_PAYLOAD" "$APPROVAL_ID" "$PEER_EVIDENCE_ID"; then
  printf '%s\n' 'baseline probe 완료; 성공은 application 정상이나 MTU 정상의 단독 증거가 아니다.'
else
  PROBE_READY=false
  printf '%s\n' 'baseline 실패: suspect probe를 실행하지 않고 route·policy·peer·link를 재확인한다.' >&2
fi
```

```bash
if [ "${PROBE_READY:-false}" != true ] || [ "${AUTHORIZATION_READY:-false}" != true ]; then
  printf '%s\n' '중단: current probe authorization이 없다.' >&2
elif host_call probe "$IFACE" "$SOURCE_IP" "$DESTINATION_IP" "$IFINDEX" "$NETNS_INODE" "$INTERFACE_MTU" "$PATH_FINGERPRINT" "$SUSPECT_PAYLOAD" "$APPROVAL_ID" "$PEER_EVIDENCE_ID"; then
  printf '%s\n' 'suspect probe 완료; 실제 application의 TCP/TLS/대용량 응답을 별도 확인한다.'
else
  printf '%s\n' 'suspect payload 실패: PMTU 후보이며 journal·peer·underlay 직접 증거 없이는 원인 확정 금지' >&2
fi
```

`Message too long`, `Frag needed`, `Packet Too Big`, success/loss 수와 payload를 기록한다. 자동 binary search, flood, loop, 병렬 probe와 반복 실행은 금지한다.

## Kubernetes·CRI 경계

이 Linux host 절차에는 `kubectl`·CRI 명령이 없다. source가 Pod라면 먼저 current Pod namespace/name/UID·owner kind/name/UID·nodeName, regular/init container kind/name/current containerID와 Pod netns inode를 current Node name/UID·provider ID·InternalIP, 승인 machine-id와 active runtime endpoint에 연결한다. 이 관계와 Pod route/interface identity 없이 host path 결과를 Pod MTU 원인으로 승격하지 않는다. replacement Pod는 이름을 재사용하지 않고 selector·owner UID·creation time으로 단일 후보를 다시 등록한다.

## 변경 전 안전 gate

이 문서는 MTU·route·firewall·CNI·sysctl 변경, interface reset, netns 삭제, Pod/Node 재시작과 rollback 명령을 제공하지 않는다. 별도 변경 절차는 fresh path fingerprint와 양 endpoint 직접 증거, approval ID·실행자·변경 시각, exact interface/tunnel/config source, blast radius·availability, desired 상태 no-op, 실제 rollback, 중단 조건과 기능 검증을 모두 가져야 한다.

변경 또는 replacement 뒤에는 기존 ifindex·MTU·route·peer·probe approval을 폐기하고 양 끝 identity를 다시 등록한다. 특정 payload 완화는 Service·TLS·application과 근본 PMTU ICMP 전달의 복구를 보장하지 않는다.

## 정상화 검증

```bash
clear_target_state
register_host
printf '%s\n' '다음: 새 UTC 창과 current interface/path를 등록하고 peer 증거, bounded probe와 실제 application 기능을 재검증한다.'
```

정상화는 새 lifecycle에서 양 endpoint의 interface/tunnel/path MTU·route와 PMTU ICMP 전달이 승인 설계에 맞고, bounded 창 동안 큰 응답·TLS·장시간 연결의 timeout/retransmit, fragmentation/Packet Too Big 오류와 drop/error 증가가 재발하지 않으며 실제 핵심 기능이 회복된 상태다. ping 성공, interface MTU 한 값, Node Ready 또는 Pod Running만으로 닫지 않는다.

## 증거·완료

보존: 사건 ID, SSH host key 경로, machine-id·boot-id·host netns, source/destination 승인 출처, interface·ifindex·MTU·path fingerprint, canonical UTC 창, bounded counter·journal producer/filter 상태, peer identity·reverse path·overlay 근거, probe 승인·payload·최대 횟수·결과, 실제 application·owner·정상화 결과.
