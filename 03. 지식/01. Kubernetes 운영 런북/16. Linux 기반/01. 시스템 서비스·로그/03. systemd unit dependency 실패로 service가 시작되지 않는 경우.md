# systemd unit dependency 실패로 service가 시작되지 않는 경우

## 입력값과 빠른 수집

실행 경로는 `운영자 Bash -> 승인 SSH endpoint -> machine/boot -> PID 1 systemd -> service transaction -> hard dependency job -> mount/device/socket/path/process`다. `After`는 순서만 나타내며 dependency 성공을 요구하지 않는다.

| 종류 | 값 | 등록 기준 |
|---|---|---|
| 최초 입력 | `SSH_DEST`, `SERVICE` | 승인 endpoint와 실패한 exact unit |
| host 자동값 | machine/boot, service Id·InvocationID, hard relation, fragment | fresh 원격 조회 성공 후 등록 |
| 계산 후보값 | relation에 있는 dependency 후보 | journal 직접 실패와 교집합 확인 전 미확정 |
| 운영자 선택값 | expected machine/boot, `DEPENDENCY_INPUT`, UTC 범위 | 자산·graph·동일 transaction journal로 확정 |

로컬 변수는 SSH에 자동 전달하지 않는다. 모든 원격 값은 positional argument이며 `eval`하지 않는다.

```bash
unset INPUT_READY GRAPH_READY SELECTION_READY TIME_READY EVIDENCE_READY RESOURCE_READY MUTATION_READY ROLLBACK_READY NORMALIZATION_READY MACHINE_ID BOOT_ID SERVICE_ID SERVICE_INVOCATION DEPENDENCY_ID DEPENDENCY_INVOCATION DEPENDENCY_PID DEPENDENCY_PID_START START_RFC3339 END_RFC3339
SSH_DEST='PASTE_APPROVED_SSH_DEST_HERE'
SERVICE='PASTE_EXACT_FAILED_SERVICE_UNIT_HERE'
INPUT_READY=false
if [[ "$SSH_DEST" =~ ^[A-Za-z0-9][A-Za-z0-9._-]*(@[A-Za-z0-9][A-Za-z0-9._-]*)?$ && "$SERVICE" =~ ^[A-Za-z0-9_.@:-]+\.(service|mount|device|socket|path|target)$ ]] && command -v ssh >/dev/null && command -v date >/dev/null && command -v sed >/dev/null && command -v grep >/dev/null; then INPUT_READY=true; fi
printf 'INPUT_READY=%s ssh=%s service=%s\n' "$INPUT_READY" "$SSH_DEST" "$SERVICE"
```

### graph·dependency refresh

호출할 때마다 graph, dependency/resource identity, 시간, evidence와 모든 변경 이후 READY를 먼저 폐기한다. service와 dependency 조회 중 하나라도 실패하면 부분값을 등록하지 않는다.

```bash
refresh_dependency_target() {
  local graph dep s e se ee relations relation_match candidate
  unset GRAPH_READY SELECTION_READY TIME_READY EVIDENCE_READY RESOURCE_READY MUTATION_READY ROLLBACK_READY NORMALIZATION_READY MACHINE_ID BOOT_ID SERVICE_ID SERVICE_INVOCATION DEPENDENCY_ID DEPENDENCY_INVOCATION DEPENDENCY_PID DEPENDENCY_PID_START SERVICE_FRAGMENT DEPENDENCY_FRAGMENT DEPENDENCY_WHAT DEPENDENCY_WHERE START_RFC3339 END_RFC3339 START_EPOCH END_EPOCH
  GRAPH_READY=false; SELECTION_READY=false; TIME_READY=false; EVIDENCE_READY=false; RESOURCE_READY=false; MUTATION_READY=false; ROLLBACK_READY=false; NORMALIZATION_READY=false
  [[ "$INPUT_READY" == true ]] || { printf '%s\n' '중단: INPUT_READY가 아니다.' >&2; return 1; }
  [[ "$EXPECTED_MACHINE_ID_INPUT" =~ ^[0-9a-fA-F]{32}$ && "$EXPECTED_BOOT_ID_INPUT" =~ ^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$ && "$DEPENDENCY_INPUT" =~ ^[A-Za-z0-9_.@:-]+\.(service|mount|device|socket|path|target)$ ]] || { printf '%s\n' '중단: machine/boot/dependency 선택 오류' >&2; return 1; }
  [[ "$START_INPUT" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ && "$END_INPUT" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] || return 1
  s=$(date -u -d "$START_INPUT" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null) && e=$(date -u -d "$END_INPUT" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null) && [[ "$s" == "$START_INPUT" && "$e" == "$END_INPUT" ]] || { printf '%s\n' '중단: UTC round-trip 실패' >&2; return 1; }
  se=$(date -u -d "$s" +%s) && ee=$(date -u -d "$e" +%s) && (( se < ee && ee <= $(date -u +%s) && ee - se <= 86400 )) || { printf '%s\n' '중단: start<end/future/24h 범위 실패' >&2; return 1; }
  graph=$(ssh -- "$SSH_DEST" bash -s -- "$EXPECTED_MACHINE_ID_INPUT" "$EXPECTED_BOOT_ID_INPUT" "$SERVICE" <<'REMOTE'
set -eu
expected_machine=$1; expected_boot=$2; service=$3
test "$(tr -d '\n' </etc/machine-id)" = "$expected_machine"; test "$(tr -d '\n' </proc/sys/kernel/random/boot_id)" = "$expected_boot"; pid1=$(ps -p 1 -o comm=); pid1=${pid1//[[:space:]]/}; test "$pid1" = systemd
printf 'machine-id=%s\nboot-id=%s\n' "$expected_machine" "$expected_boot"
systemctl show -p Id -p LoadState -p ActiveState -p SubState -p Result -p InvocationID -p Requires -p Requisite -p BindsTo -p Wants -p After -p Before -p Conflicts -p FragmentPath -p DropInPaths -- "$service"
systemctl list-dependencies --all --plain --no-pager -- "$service"
REMOTE
  ) || { printf '%s\n' 'producer 실패: service graph 조회' >&2; return 1; }
  grep -Fxq "machine-id=$EXPECTED_MACHINE_ID_INPUT" <<<"$graph" && grep -Fxq "boot-id=$EXPECTED_BOOT_ID_INPUT" <<<"$graph" && grep -Fxq "Id=$SERVICE" <<<"$graph" || { printf '%s\n' 'no match: service identity 불일치' >&2; return 1; }
  relations=$(sed -n 's/^Requires=//p; s/^Requisite=//p; s/^BindsTo=//p' <<<"$graph") || { printf '%s\n' 'filter 오류: hard relation 파싱 실패' >&2; return 1; }
  relation_match=false; for candidate in $relations; do if [[ "$candidate" == "$DEPENDENCY_INPUT" ]]; then relation_match=true; break; fi; done
  [[ "$relation_match" == true ]] || { printf '%s\n' 'no match: 선택 dependency가 hard relation에 없음' >&2; return 1; }
  dep=$(ssh -- "$SSH_DEST" bash -s -- "$EXPECTED_MACHINE_ID_INPUT" "$EXPECTED_BOOT_ID_INPUT" "$DEPENDENCY_INPUT" <<'REMOTE'
set -eu
expected_machine=$1; expected_boot=$2; dep=$3
test "$(tr -d '\n' </etc/machine-id)" = "$expected_machine"; test "$(tr -d '\n' </proc/sys/kernel/random/boot_id)" = "$expected_boot"
systemctl show -p Id -p LoadState -p ActiveState -p SubState -p Result -p InvocationID -p ExecMainCode -p ExecMainStatus -p MainPID -p FragmentPath -p DropInPaths -p What -p Where -- "$dep"
pid=$(systemctl show --value -p MainPID -- "$dep")
if [[ "$pid" =~ ^[1-9][0-9]*$ && -r "/proc/$pid/stat" ]]; then stat=$(cat "/proc/$pid/stat"); rest=${stat##*) }; printf 'pid-start=%s\npid-ns=%s\nmnt-ns=%s\nnet-ns=%s\n' "$(cut -d' ' -f20 <<<"$rest")" "$(readlink "/proc/$pid/ns/pid")" "$(readlink "/proc/$pid/ns/mnt")" "$(readlink "/proc/$pid/ns/net")"; fi
REMOTE
  ) || { printf '%s\n' 'producer 실패: dependency 조회' >&2; return 1; }
  grep -Fxq "Id=$DEPENDENCY_INPUT" <<<"$dep" || { printf '%s\n' 'no match: dependency Id 불일치' >&2; return 1; }
  MACHINE_ID=$EXPECTED_MACHINE_ID_INPUT; BOOT_ID=$EXPECTED_BOOT_ID_INPUT; SERVICE_ID=$SERVICE; DEPENDENCY_ID=$DEPENDENCY_INPUT
  SERVICE_INVOCATION=$(sed -n 's/^InvocationID=//p' <<<"$graph"); SERVICE_FRAGMENT=$(sed -n 's/^FragmentPath=//p' <<<"$graph")
  DEPENDENCY_INVOCATION=$(sed -n 's/^InvocationID=//p' <<<"$dep"); DEPENDENCY_PID=$(sed -n 's/^MainPID=//p' <<<"$dep"); DEPENDENCY_PID_START=$(sed -n 's/^pid-start=//p' <<<"$dep"); DEPENDENCY_FRAGMENT=$(sed -n 's/^FragmentPath=//p' <<<"$dep"); DEPENDENCY_WHAT=$(sed -n 's/^What=//p' <<<"$dep"); DEPENDENCY_WHERE=$(sed -n 's/^Where=//p' <<<"$dep")
  START_RFC3339=$s; END_RFC3339=$e; START_EPOCH=$se; END_EPOCH=$ee
  GRAPH_READY=true; SELECTION_READY=true; TIME_READY=true
  printf 'GRAPH_READY=true SELECTION_READY=true TIME_READY=true machine=%s boot=%s service=%s dependency=%s dep_invocation=%s dep_pid=%s dep_start=%s what=%s where=%s\n' "$MACHINE_ID" "$BOOT_ID" "$SERVICE_ID" "$DEPENDENCY_ID" "${DEPENDENCY_INVOCATION:--}" "${DEPENDENCY_PID:-0}" "${DEPENDENCY_PID_START:--}" "${DEPENDENCY_WHAT:--}" "${DEPENDENCY_WHERE:--}"
  printf '%s\n%s\n' "$graph" "$dep"
}
```

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | `Result=dependency`, `Dependency failed` 또는 실패한 dependency job이 직접 보임 |
| 적용 | service hard relation과 실패한 exact dependency를 같은 machine·boot·transaction 시간에 연결 가능 |
| 제외 | 단일 process 실패는 [[01. 호스트에서 systemd service가 inactive·failed인 경우]], 반복은 [[02. systemd service가 반복 재시작되고 start-limit에 도달한 경우]], timeout은 [[04. systemd service 시작·종료가 timeout에서 멈춘 경우]] |

검색 문자열: `Dependency failed`, `A dependency job for`, `Result=dependency`, `Requires`, `Requisite`, `BindsTo`, `not-found`, `bad-setting`.

## 양쪽 bounded evidence

먼저 `refresh_dependency_target`을 실행한다. `Wants`·`After`만 있는 unit이나 복수 후보의 첫 항목을 자동 선택하지 않는다.

```bash
EVIDENCE_READY=false; if [[ "$GRAPH_READY" == true && "$SELECTION_READY" == true && "$TIME_READY" == true ]]; then if ssh -- "$SSH_DEST" bash -s -- "$MACHINE_ID" "$BOOT_ID" "$SERVICE_ID" "$SERVICE_INVOCATION" "$DEPENDENCY_ID" "$DEPENDENCY_INVOCATION" "$DEPENDENCY_PID" "$DEPENDENCY_PID_START" "$START_RFC3339" "$END_RFC3339" <<'REMOTE'
set -u
expected_machine=$1; expected_boot=$2; service=$3; service_invocation=$4; dep=$5; dep_invocation=$6; dep_pid=$7; dep_pid_start=$8; start=$9; end=${10}
[[ "$(tr -d '\n' </etc/machine-id)" == "$expected_machine" && "$(tr -d '\n' </proc/sys/kernel/random/boot_id)" == "$expected_boot" ]] || exit 10
[[ "$(systemctl show --value -p Id -- "$service")" == "$service" && "$(systemctl show --value -p Id -- "$dep")" == "$dep" ]] || exit 11
[[ "$(systemctl show --value -p InvocationID -- "$service")" == "$service_invocation" && "$(systemctl show --value -p InvocationID -- "$dep")" == "$dep_invocation" && "$(systemctl show --value -p MainPID -- "$dep")" == "$dep_pid" ]] || exit 12
if [[ "$dep_pid" =~ ^[1-9][0-9]*$ ]]; then stat=$(cat "/proc/$dep_pid/stat") || exit 13; rest=${stat##*) }; [[ "$(cut -d' ' -f20 <<<"$rest")" == "$dep_pid_start" ]] || exit 14; fi
for pair in "service:$service" "dependency:$dep"; do label=${pair%%:*}; unit=${pair#*:}; tmp=$(mktemp) || exit 12; journalctl --utc --boot="$expected_boot" -u "$unit" --since "$start" --until "$end" --no-pager -o short-iso-precise >"$tmp"; producer_rc=$?; printf '%s_producer_rc=%s\n' "$label" "$producer_rc"; (( producer_rc == 0 )) || { rm -f "$tmp"; exit 20; }; if [[ "$label" == service ]]; then pattern='dependency failed|dependency job|cannot add dependency job|failed to start|requires|requisite|binds to|job .*failed|timeout'; else pattern='failed|failure|not found|bad-setting|permission denied|mount|device|socket|path|timeout|condition|assert|exit-code|i/o error|read-only'; fi; grep -E -i -C 3 "$pattern" "$tmp"; filter_rc=$?; rm -f "$tmp"; printf '%s_filter_rc=%s\n' "$label" "$filter_rc"; case $filter_rc in 0) :;; 1) exit 30;; *) exit 31;; esac; done
REMOTE
then EVIDENCE_READY=true; else printf '중단: 양쪽 evidence 실패 rc=%s\n' "$?" >&2; fi; else printf '%s\n' '중단: fresh graph/selection/time READY가 아니다.' >&2; fi; printf 'EVIDENCE_READY=%s\n' "$EVIDENCE_READY"
```

성공 gate는 service hard relation, dependency exact Id, dependency의 직접 실패가 service의 dependency 실패보다 먼저라는 시간 순서다. 한쪽 filter 무결과는 원인 미확정이며 `Condition`·`Assert` 실패는 process crash가 아니다.

## resource identity 분기

mount/device만 자동 명령을 제공한다. socket/path는 실제 endpoint/path와 owner가 확인된 별도 읽기 절차로 이관한다.

```bash
RESOURCE_READY=false; if [[ "$EVIDENCE_READY" == true ]]; then if ssh -- "$SSH_DEST" bash -s -- "$MACHINE_ID" "$BOOT_ID" "$DEPENDENCY_ID" "$DEPENDENCY_INVOCATION" "$DEPENDENCY_WHAT" "$DEPENDENCY_WHERE" <<'REMOTE'
set -eu
expected_machine=$1; expected_boot=$2; dep=$3; expected_invocation=$4; expected_what=$5; expected_where=$6
test "$(tr -d '\n' </etc/machine-id)" = "$expected_machine"; test "$(tr -d '\n' </proc/sys/kernel/random/boot_id)" = "$expected_boot"; test "$(systemctl show --value -p Id -- "$dep")" = "$dep"
test "$(systemctl show --value -p InvocationID -- "$dep")" = "$expected_invocation"; test "$(systemctl show --value -p What -- "$dep")" = "$expected_what"; test "$(systemctl show --value -p Where -- "$dep")" = "$expected_where"
kind=${dep##*.}
case $kind in mount) where=$(systemctl show --value -p Where -- "$dep"); test -n "$where"; findmnt -T "$where" -o TARGET,SOURCE,FSTYPE,OPTIONS,MAJ:MIN;; device) what=$(systemctl show --value -p What -- "$dep"); test -n "$what"; test -e "$what"; stat -Lc 'path=%n type=%F major=%t minor=%T inode=%i' -- "$what";; *) printf 'unsupported-resource-kind=%s\n' "$kind"; exit 40;; esac
REMOTE
then RESOURCE_READY=true; else printf '중단: resource identity 실패/미지원 rc=%s\n' "$?" >&2; fi; else printf '%s\n' '중단: EVIDENCE_READY가 아니다.' >&2; fi; printf 'RESOURCE_READY=%s\n' "$RESOURCE_READY"
```

mount/device 부재나 identity 교체는 정상화가 아니라 stale evidence 폐기 조건이다. mount는 target·source·filesystem·major:minor, device는 path·major:minor·inode를 함께 기록한다. socket은 address·port·protocol·network namespace와 listener PID/start ticks, path는 mount namespace·inode·owner를 확인한다.

## 변경 경계와 정상화

진단 중 start/restart, daemon-reload, unit 편집, mount/remount, device 조작과 socket 삭제를 실행하지 않는다. 별도 변경은 실행 직전 refresh와 evidence를 다시 통과하고 승인 ID, 변경 창, consumer 가용성·quorum, 멱등성·side effect, 중단 조건과 설정/데이터 rollback 원본을 확인해야 한다. 삭제·device 작업은 원본 없이 rollback 가능하다고 기록하지 않는다.

교체 dependency/process/mount/device/socket은 모든 READY를 폐기하고 새 identity로 등록한다. 정상화는 resource -> dependency 의도 상태와 새 identity -> service transaction -> 실제 consumer 연결·업무 기능 순으로 새 관찰 창에서 검증한다. 같은 job failure가 재발하지 않아야 하며 `active` 한 번이나 빈 journal은 성공이 아니다.

## 증거와 미검증

machine/boot, service/dependency exact Id·InvocationID·hard relation, PID/start ticks·namespace, canonical UTC, 양쪽 producer/filter status, mount/device/socket/path identity, 승인·rollback과 실제 기능 결과만 보존한다. unit 원문, 전체 journal, argv/environment와 credential은 저장하지 않는다.

- [systemd.unit](https://www.freedesktop.org/software/systemd/man/249/systemd.unit.html)

- 실제 배포판, systemd graph, SSH inventory와 resource별 복구는 환경별 미검증이다.
