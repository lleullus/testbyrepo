# 장애 시간대의 journald 로그를 찾을 수 없는 경우

## 입력값과 빠른 수집

실행 경로는 `운영자 Bash -> 승인 SSH endpoint -> machine/boot -> systemd-journald -> exact unit selector -> UTC realtime range -> journal file/cursor/storage -> 선택적 외부 log backend`다. 빈 출력은 오류 부재가 아니며 producer 실패, selector no-match, boot 미보존, 시간 변환 오류와 retention을 분리한다.

| 종류 | 값 | 등록 기준 |
|---|---|---|
| 최초 입력 | `SSH_DEST`, `JOURNAL_UNIT` | 승인 endpoint와 실제 unit |
| host 자동값 | machine/boot, unit Id·fragment, journald state, clock | fresh 원격 조회 성공 후 등록 |
| 계산 후보값 | storage mode·보존 경계 | current config/header만으로 과거 삭제를 확정하지 않음 |
| 운영자 선택값 | expected machine/boot와 UTC evidence 범위 | 자산·boot list·사건 시각에서 선택, refresh마다 검증 |

```bash
unset INPUT_READY TARGET_READY TIME_READY GAP_READY STORAGE_READY MUTATION_READY ROLLBACK_READY NORMALIZATION_READY MACHINE_ID BOOT_ID UNIT_ID UNIT_FRAGMENT JOURNALD_INVOCATION START_RFC3339 END_RFC3339 START_EPOCH END_EPOCH
SSH_DEST='PASTE_APPROVED_SSH_DEST_HERE'
JOURNAL_UNIT='PASTE_CONFIRMED_JOURNAL_UNIT_HERE'
INPUT_READY=false
if [[ "$SSH_DEST" =~ ^[A-Za-z0-9][A-Za-z0-9._-]*(@[A-Za-z0-9][A-Za-z0-9._-]*)?$ && "$JOURNAL_UNIT" =~ ^[A-Za-z0-9_.@:-]+\.(service|scope|slice)$ ]] && command -v ssh >/dev/null && command -v date >/dev/null && command -v sed >/dev/null; then INPUT_READY=true; fi
printf 'INPUT_READY=%s ssh=%s unit=%s\n' "$INPUT_READY" "$SSH_DEST" "$JOURNAL_UNIT"
```

```bash
if [[ "$INPUT_READY" == true ]]; then ssh -- "$SSH_DEST" bash -s <<'REMOTE'
set -eu
test -r /etc/machine-id; test -r /proc/sys/kernel/random/boot_id; pid1=$(ps -p 1 -o comm=); pid1=${pid1//[[:space:]]/}; test "$pid1" = systemd
printf 'host=%s\nmachine-id=%s\nboot-id=%s\n' "$(hostname -f 2>/dev/null || hostname)" "$(tr -d '\n' </etc/machine-id)" "$(tr -d '\n' </proc/sys/kernel/random/boot_id)"
journalctl --list-boots --no-pager
REMOTE
else printf '%s\n' '중단: INPUT_READY가 아니다.' >&2; fi
```

```bash
EXPECTED_MACHINE_ID_INPUT='PASTE_VERIFIED_MACHINE_ID_HERE'
EXPECTED_BOOT_ID_INPUT='PASTE_VERIFIED_BOOT_ID_HERE'
START_INPUT='PASTE_EVIDENCE_START_UTC_RFC3339_HERE'
END_INPUT='PASTE_EVIDENCE_END_UTC_RFC3339_HERE'
```

### journal target refresh

refresh는 host/unit/time, gap/storage evidence와 변경·rollback·정상화 상태를 먼저 폐기한다. 조회 실패 후 과거 boot나 시간 범위가 남지 않는다.

```bash
refresh_journal_target() {
  local out s e se ee
  unset TARGET_READY TIME_READY GAP_READY STORAGE_READY MUTATION_READY ROLLBACK_READY NORMALIZATION_READY MACHINE_ID BOOT_ID UNIT_ID UNIT_FRAGMENT JOURNALD_INVOCATION START_RFC3339 END_RFC3339 START_EPOCH END_EPOCH
  TARGET_READY=false; TIME_READY=false; GAP_READY=false; STORAGE_READY=false; MUTATION_READY=false; ROLLBACK_READY=false; NORMALIZATION_READY=false
  [[ "$INPUT_READY" == true && "$EXPECTED_MACHINE_ID_INPUT" =~ ^[0-9a-fA-F]{32}$ && "$EXPECTED_BOOT_ID_INPUT" =~ ^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$ ]] || { printf '%s\n' '중단: input/machine/boot gate 실패' >&2; return 1; }
  [[ "$START_INPUT" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ && "$END_INPUT" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] || return 1
  s=$(date -u -d "$START_INPUT" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null) && e=$(date -u -d "$END_INPUT" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null) && [[ "$s" == "$START_INPUT" && "$e" == "$END_INPUT" ]] || { printf '%s\n' '중단: UTC round-trip 실패' >&2; return 1; }
  se=$(date -u -d "$s" +%s) && ee=$(date -u -d "$e" +%s) && (( se < ee && ee <= $(date -u +%s) && ee - se <= 86400 )) || { printf '%s\n' '중단: start<end/future/24h gate 실패' >&2; return 1; }
  out=$(ssh -- "$SSH_DEST" bash -s -- "$EXPECTED_MACHINE_ID_INPUT" "$EXPECTED_BOOT_ID_INPUT" "$JOURNAL_UNIT" <<'REMOTE'
set -eu
expected_machine=$1; expected_boot=$2; unit=$3
test "$(tr -d '\n' </etc/machine-id)" = "$expected_machine"; test "$(tr -d '\n' </proc/sys/kernel/random/boot_id)" = "$expected_boot"; pid1=$(ps -p 1 -o comm=); pid1=${pid1//[[:space:]]/}; test "$pid1" = systemd
printf 'machine-id=%s\nboot-id=%s\n' "$expected_machine" "$expected_boot"
systemctl show -p Id -p LoadState -p ActiveState -p FragmentPath -p DropInPaths -- "$unit"
systemctl show -p Id -p ActiveState -p SubState -p InvocationID -- systemd-journald.service
timedatectl show -p Timezone -p NTPSynchronized -p LocalRTC
journalctl --list-boots --no-pager
REMOTE
  ) || { printf '%s\n' 'producer 실패: SSH/host/unit/journald 조회' >&2; return 1; }
  grep -Fxq "machine-id=$EXPECTED_MACHINE_ID_INPUT" <<<"$out" && grep -Fxq "boot-id=$EXPECTED_BOOT_ID_INPUT" <<<"$out" && grep -Fxq "Id=$JOURNAL_UNIT" <<<"$out" || { printf '%s\n' 'no match: machine/boot/unit identity 불일치' >&2; return 1; }
  MACHINE_ID=$EXPECTED_MACHINE_ID_INPUT; BOOT_ID=$EXPECTED_BOOT_ID_INPUT; UNIT_ID=$JOURNAL_UNIT; UNIT_FRAGMENT=$(sed -n '0,/^FragmentPath=/{s/^FragmentPath=//p}' <<<"$out"); JOURNALD_INVOCATION=$(sed -n '/^Id=systemd-journald.service$/,$ { s/^InvocationID=//p; }' <<<"$out"); START_RFC3339=$s; END_RFC3339=$e; START_EPOCH=$se; END_EPOCH=$ee
  TARGET_READY=true; TIME_READY=true
  printf 'TARGET_READY=true TIME_READY=true machine=%s boot=%s unit=%s start=%s end=%s\n' "$MACHINE_ID" "$BOOT_ID" "$UNIT_ID" "$START_RFC3339" "$END_RFC3339"
  printf '%s\n' "$out"
}
```

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | exact unit·boot·UTC 범위의 bounded 조회가 비거나 기대 구간이 없음 |
| 적용 | machine·boot·unit·clock·journal storage와 source를 대조 가능 |
| 제외 | `Suppressed` 또는 rotation/retention 직접 증거가 있으면 [[06. journald rate limit·rotation으로 로그가 누락된 경우]], application이 journald에 쓰지 않으면 해당 stdout/file/remote sink owner로 이동 |

검색 문자열: `No entries`, `-- No entries --`, `Suppressed`, `rate limit`, `journal`, `vacuum`, `rotate`.

## exact range 무결과 확인

먼저 `refresh_journal_target`을 실행한다. 다음 명령은 원격 identity를 다시 확인하고 producer 성공과 행 수를 분리한다. unit journal에 행이 있으면 출력하지 않고 first/last cursor·timestamp만 표시한다.

```bash
GAP_READY=false; if [[ "$TARGET_READY" == true && "$TIME_READY" == true ]]; then if ssh -- "$SSH_DEST" bash -s -- "$MACHINE_ID" "$BOOT_ID" "$UNIT_ID" "$JOURNALD_INVOCATION" "$START_RFC3339" "$END_RFC3339" <<'REMOTE'
set -u
expected_machine=$1; expected_boot=$2; unit=$3; expected_journald_invocation=$4; start=$5; end=$6
[[ "$(tr -d '\n' </etc/machine-id)" == "$expected_machine" && "$(tr -d '\n' </proc/sys/kernel/random/boot_id)" == "$expected_boot" ]] || exit 10
[[ "$(systemctl show --value -p Id -- "$unit")" == "$unit" ]] || exit 11
[[ "$(systemctl show --value -p InvocationID -- systemd-journald.service)" == "$expected_journald_invocation" ]] || exit 12
tmp=$(mktemp) || exit 12; trap 'rm -f "$tmp"' EXIT
journalctl --utc --boot="$expected_boot" -u "$unit" --since "$start" --until "$end" --no-pager -o export >"$tmp"; producer_rc=$?; printf 'producer_rc=%s\n' "$producer_rc"; (( producer_rc == 0 )) || exit 20
entries=$(grep -c '^__CURSOR=' "$tmp"); filter_rc=$?; case $filter_rc in 0) :;; 1) entries=0;; *) printf 'filter_rc=%s\n' "$filter_rc"; exit 31;; esac
printf 'filter_rc=%s entries=%s\n' "$filter_rc" "$entries"
if (( entries == 0 )); then printf '%s\n' 'gap-confirmed: exact selector 범위에 entry 0'; exit 0; fi
grep -m1 -E '^(__CURSOR|__REALTIME_TIMESTAMP)=' "$tmp"; first_rc=$?; grep -E '^(__CURSOR|__REALTIME_TIMESTAMP)=' "$tmp" | tail -n 2; last_pipeline_rc=(${PIPESTATUS[@]}); printf 'first_filter_rc=%s last_producer_rc=%s last_filter_rc=%s\n' "$first_rc" "${last_pipeline_rc[0]}" "${last_pipeline_rc[1]}"; exit 40
REMOTE
then GAP_READY=true; else rc=$?; if [[ "$rc" == 40 ]]; then printf '%s\n' '중단: entry가 있어 완전 무결과가 아니다.' >&2; else printf '중단: gap 조회 실패 rc=%s\n' "$rc" >&2; fi; fi; else printf '%s\n' '중단: fresh target/time READY가 아니다.' >&2; fi; printf 'GAP_READY=%s\n' "$GAP_READY"
```

`GAP_READY=true`는 exact query가 성공했고 entry가 0이라는 뜻뿐이다. 사건이 없었다거나 journald가 삭제했다고 증명하지 않는다. `journalctl --boot` 미지원·권한 오류·boot 부재는 producer 실패다.

| 지금 보이는 것 | 먼저 볼 것 | 그다음 어디를 볼지 | 상세 절 |
|---|---|---|---|
| 다른 unit에는 행이 있으나 exact unit만 0 | 실제 logger destination과 unit Id | application/file/remote sink owner | [[#storage·source 경계]] |
| 선택 boot 자체가 목록에 없음 | boot ID와 persistent/volatile storage | journald storage owner | [[#storage·source 경계]] |
| 범위 앞뒤에는 있으나 중간만 공백 | clock jump·suppression·vacuum/rotation 증거 | rate/rotation 런북 | [[06. journald rate limit·rotation으로 로그가 누락된 경우]] |
| 모든 journal source 조회 실패 | journald service·filesystem·권한 | host/journald owner | [[#storage·source 경계]] |

## storage·source 경계

다음 명령은 gap이 검증된 current target에서만 current storage/header/config source를 읽는다. current disk usage와 config는 이미 삭제된 entry의 직접 증거가 아니다.

```bash
STORAGE_READY=false; if [[ "$GAP_READY" == true ]]; then if ssh -- "$SSH_DEST" bash -s -- "$MACHINE_ID" "$BOOT_ID" "$JOURNALD_INVOCATION" <<'REMOTE'
set -u
expected_machine=$1; expected_boot=$2; expected_journald_invocation=$3
[[ "$(tr -d '\n' </etc/machine-id)" == "$expected_machine" && "$(tr -d '\n' </proc/sys/kernel/random/boot_id)" == "$expected_boot" ]] || exit 10
[[ "$(systemctl show --value -p InvocationID -- systemd-journald.service)" == "$expected_journald_invocation" ]] || exit 11
journalctl --disk-usage --no-pager || exit 20
journalctl --header --no-pager || exit 21
if command -v systemd-analyze >/dev/null; then tmp=$(mktemp) || exit 12; trap 'rm -f "$tmp"' EXIT; systemd-analyze cat-config systemd/journald.conf >"$tmp"; producer_rc=$?; printf 'config_producer_rc=%s\n' "$producer_rc"; (( producer_rc == 0 )) || exit 22; grep -E '^[[:space:]]*(Storage|SystemMaxUse|SystemKeepFree|SystemMaxFileSize|SystemMaxFiles|RuntimeMaxUse|RuntimeKeepFree|RuntimeMaxFileSize|RuntimeMaxFiles|MaxRetentionSec|RateLimitIntervalSec|RateLimitBurst|Compress|Seal)=' "$tmp"; filter_rc=$?; printf 'config_filter_rc=%s\n' "$filter_rc"; case $filter_rc in 0|1) exit 0;; *) exit 31;; esac; else printf '%s\n' 'unsupported: systemd-analyze unavailable'; exit 41; fi
REMOTE
then STORAGE_READY=true; else printf '중단: storage evidence 실패/미지원 rc=%s\n' "$?" >&2; fi; else printf '%s\n' '중단: GAP_READY가 아니다.' >&2; fi; printf 'STORAGE_READY=%s\n' "$STORAGE_READY"
```

config filter `rc=1`은 explicit 설정이 없다는 뜻이며 배포판 default를 확정하지 않는다. application이 file/syslog/remote backend로만 쓰면 journal 무결과는 정상 경로일 수 있으므로 실제 logger 설정과 backend query를 승인된 owner가 확인한다. 외부 query는 machine ID·unit·canonical UTC를 연결하고 성공/빈 결과/API 오류를 분리해야 한다.

## 변경 경계와 정상화

진단 중 vacuum, rotate, journald restart, 설정 변경과 journal 파일 삭제를 실행하지 않는다. 별도 변경은 fresh machine/boot/unit/gap/storage evidence, 승인 ID·권한·변경 창, filesystem headroom, 소비자 영향, 멱등성, 중단 조건과 설정 rollback 원본을 재확인해야 한다. vacuum·삭제는 보존 복제본 없이는 rollback 불가다.

machine·boot·unit이 교체되면 모든 READY를 폐기하고 입력부터 재등록한다. 정상화는 storage/source -> exact unit의 새 bounded entry와 cursor 연속성 -> 실제 service 기능과 외부 log 전달 -> 재발 관찰 순으로 확인한다. 새 로그 한 행이나 빈 filter만으로 정상화하지 않는다.

## 증거와 미검증

machine/boot/unit·fragment, strict UTC 범위, clock, producer/filter RC와 entry count, cursor/timestamp 경계, header·disk usage·config source 및 unsupported 상태, 승인·rollback·실제 기능을 보존한다. 전체 journal, message body, token·private key·개인정보는 저장하지 않는다.
