# journald rate limit·rotation으로 로그가 누락된 경우

## 입력값과 빠른 수집

실행 경로는 `운영자 Bash -> 승인 SSH endpoint -> machine/boot -> exact source unit -> systemd-journald rate limiter -> active/archived journal files·retention -> 선택적 external log sink`다. suppression과 rotation은 독립적이며 로그 공백만으로 어느 쪽도 확정하지 않는다.

| 종류 | 값 | 등록 기준 |
|---|---|---|
| 최초 입력 | `SSH_DEST`, `SOURCE_UNIT` | 승인 endpoint와 로그 source exact unit |
| host 자동값 | machine/boot, source Id·InvocationID, journald InvocationID, config source | fresh 원격 조회 성공 후 등록 |
| 계산 후보값 | `SOURCE_UNIT_RE`, 현재 storage 사용량·보존 경계 | escape·bounded 출력 및 직접 증거 전 미확정 |
| 운영자 선택값 | expected machine/boot와 UTC gap 범위 | 자산·boot list·사건 시각에서 선택, refresh마다 검증 |

```bash
unset INPUT_READY TARGET_READY TIME_READY RATE_READY ROTATION_READY MUTATION_READY ROLLBACK_READY NORMALIZATION_READY MACHINE_ID BOOT_ID SOURCE_ID SOURCE_INVOCATION JOURNALD_INVOCATION SOURCE_UNIT_RE START_RFC3339 END_RFC3339
SSH_DEST='PASTE_APPROVED_SSH_DEST_HERE'
SOURCE_UNIT='PASTE_EXACT_SOURCE_UNIT_HERE'
INPUT_READY=false
if [[ "$SSH_DEST" =~ ^[A-Za-z0-9][A-Za-z0-9._-]*(@[A-Za-z0-9][A-Za-z0-9._-]*)?$ && "$SOURCE_UNIT" =~ ^[A-Za-z0-9_.@:-]+\.(service|scope|slice)$ ]] && command -v ssh >/dev/null && command -v date >/dev/null && command -v sed >/dev/null && command -v grep >/dev/null; then INPUT_READY=true; fi
printf 'INPUT_READY=%s ssh=%s source=%s\n' "$INPUT_READY" "$SSH_DEST" "$SOURCE_UNIT"
```

### rate·rotation target refresh

refresh는 target/time, rate/rotation evidence와 모든 변경 이후 READY를 먼저 폐기한다. source 또는 journald가 재시작되면 InvocationID가 바뀌므로 과거 counter/config 관측과 이어 붙이지 않는다.

```bash
refresh_journald_target() {
  local out s e se ee re
  unset TARGET_READY TIME_READY RATE_READY ROTATION_READY MUTATION_READY ROLLBACK_READY NORMALIZATION_READY MACHINE_ID BOOT_ID SOURCE_ID SOURCE_INVOCATION JOURNALD_INVOCATION SOURCE_FRAGMENT SOURCE_UNIT_RE START_RFC3339 END_RFC3339 START_EPOCH END_EPOCH
  TARGET_READY=false; TIME_READY=false; RATE_READY=false; ROTATION_READY=false; MUTATION_READY=false; ROLLBACK_READY=false; NORMALIZATION_READY=false
  [[ "$INPUT_READY" == true && "$EXPECTED_MACHINE_ID_INPUT" =~ ^[0-9a-fA-F]{32}$ && "$EXPECTED_BOOT_ID_INPUT" =~ ^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$ ]] || { printf '%s\n' '중단: input/machine/boot gate 실패' >&2; return 1; }
  [[ "$START_INPUT" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ && "$END_INPUT" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] || return 1
  s=$(date -u -d "$START_INPUT" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null) && e=$(date -u -d "$END_INPUT" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null) && [[ "$s" == "$START_INPUT" && "$e" == "$END_INPUT" ]] || { printf '%s\n' '중단: UTC round-trip 실패' >&2; return 1; }
  se=$(date -u -d "$s" +%s) && ee=$(date -u -d "$e" +%s) && (( se < ee && ee <= $(date -u +%s) && ee - se <= 86400 )) || { printf '%s\n' '중단: start<end/future/24h gate 실패' >&2; return 1; }
  out=$(ssh -- "$SSH_DEST" bash -s -- "$EXPECTED_MACHINE_ID_INPUT" "$EXPECTED_BOOT_ID_INPUT" "$SOURCE_UNIT" <<'REMOTE'
set -eu
expected_machine=$1; expected_boot=$2; source=$3
test "$(tr -d '\n' </etc/machine-id)" = "$expected_machine"; test "$(tr -d '\n' </proc/sys/kernel/random/boot_id)" = "$expected_boot"; pid1=$(ps -p 1 -o comm=); pid1=${pid1//[[:space:]]/}; test "$pid1" = systemd
printf 'machine-id=%s\nboot-id=%s\n' "$expected_machine" "$expected_boot"
systemctl show -p Id -p LoadState -p ActiveState -p InvocationID -p FragmentPath -- "$source"
systemctl show -p Id -p ActiveState -p SubState -p InvocationID -p FragmentPath -p DropInPaths -- systemd-journald.service
timedatectl show -p Timezone -p NTPSynchronized -p LocalRTC
REMOTE
  ) || { printf '%s\n' 'producer 실패: SSH/host/source/journald 조회' >&2; return 1; }
  grep -Fxq "machine-id=$EXPECTED_MACHINE_ID_INPUT" <<<"$out" && grep -Fxq "boot-id=$EXPECTED_BOOT_ID_INPUT" <<<"$out" && grep -Fxq "Id=$SOURCE_UNIT" <<<"$out" && grep -Fxq 'Id=systemd-journald.service' <<<"$out" || { printf '%s\n' 'no match: current identity 불일치' >&2; return 1; }
  re=$(sed 's/[][\\.^$*+?(){}|]/\\&/g' <<<"$SOURCE_UNIT") || return 1
  MACHINE_ID=$EXPECTED_MACHINE_ID_INPUT; BOOT_ID=$EXPECTED_BOOT_ID_INPUT; SOURCE_ID=$SOURCE_UNIT; SOURCE_UNIT_RE=$re
  SOURCE_INVOCATION=$(sed -n '0,/^InvocationID=/{s/^InvocationID=//p}' <<<"$out"); JOURNALD_INVOCATION=$(sed -n '/^Id=systemd-journald.service$/,$ { s/^InvocationID=//p; }' <<<"$out"); SOURCE_FRAGMENT=$(sed -n '0,/^FragmentPath=/{s/^FragmentPath=//p}' <<<"$out")
  START_RFC3339=$s; END_RFC3339=$e; START_EPOCH=$se; END_EPOCH=$ee; TARGET_READY=true; TIME_READY=true
  printf 'TARGET_READY=true TIME_READY=true machine=%s boot=%s source=%s source_invocation=%s journald_invocation=%s regex=%s\n' "$MACHINE_ID" "$BOOT_ID" "$SOURCE_ID" "${SOURCE_INVOCATION:--}" "${JOURNALD_INVOCATION:--}" "$SOURCE_UNIT_RE"
  printf '%s\n' "$out"
}
```

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | bounded 공백과 `Suppressed`·rate-limit 또는 boot/file/retention 경계가 함께 관측됨 |
| 적용 | source unit·machine·boot·UTC·journald invocation·storage identity를 연결 가능 |
| 제외 | 직접 단서 없이 unit만 비면 [[05. 장애 시간대의 journald 로그를 찾을 수 없는 경우]], source가 journald에 쓰지 않으면 실제 file/syslog/remote sink로 이동 |

검색 문자열: `Suppressed`, `rate limit`, `RateLimitIntervalSec`, `RateLimitBurst`, `rotate`, `vacuum`, `SystemMaxUse`, `RuntimeMaxUse`, `MaxRetentionSec`.

## rate-limit evidence

먼저 `refresh_journald_target`을 실행한다. source journal과 journald 자체 메시지를 별도 producer로 수집하며 각 filter의 no-match와 오류를 분리한다.

```bash
RATE_READY=false; if [[ "$TARGET_READY" == true && "$TIME_READY" == true ]]; then if ssh -- "$SSH_DEST" bash -s -- "$MACHINE_ID" "$BOOT_ID" "$SOURCE_ID" "$SOURCE_INVOCATION" "$JOURNALD_INVOCATION" "$SOURCE_UNIT_RE" "$START_RFC3339" "$END_RFC3339" <<'REMOTE'
set -u
expected_machine=$1; expected_boot=$2; source=$3; expected_source_invocation=$4; expected_journald_invocation=$5; source_re=$6; start=$7; end=$8
[[ "$(tr -d '\n' </etc/machine-id)" == "$expected_machine" && "$(tr -d '\n' </proc/sys/kernel/random/boot_id)" == "$expected_boot" ]] || exit 10
[[ "$(systemctl show --value -p Id -- "$source")" == "$source" ]] || exit 11
[[ "$(systemctl show --value -p InvocationID -- "$source")" == "$expected_source_invocation" && "$(systemctl show --value -p InvocationID -- systemd-journald.service)" == "$expected_journald_invocation" ]] || exit 12
source_tmp=$(mktemp) || exit 12; journal_tmp=$(mktemp) || { rm -f "$source_tmp"; exit 12; }; trap 'rm -f "$source_tmp" "$journal_tmp"' EXIT
journalctl --utc --boot="$expected_boot" -u "$source" --since "$start" --until "$end" --no-pager -o short-iso-precise >"$source_tmp"; source_producer_rc=$?
journalctl --utc --boot="$expected_boot" -u systemd-journald.service --since "$start" --until "$end" --no-pager -o short-iso-precise >"$journal_tmp"; journal_producer_rc=$?
printf 'source_producer_rc=%s journald_producer_rc=%s\n' "$source_producer_rc" "$journal_producer_rc"; (( source_producer_rc == 0 && journal_producer_rc == 0 )) || exit 20
grep -E -i -C 2 'failed|error|timeout|rate limit|suppressed' "$source_tmp"; source_filter_rc=$?; printf 'source_filter_rc=%s\n' "$source_filter_rc"; (( source_filter_rc == 0 || source_filter_rc == 1 )) || exit 31
grep -E -i -C 3 "(Suppressed|rate limit).*$source_re|$source_re.*(Suppressed|rate limit)" "$journal_tmp"; suppression_filter_rc=$?; printf 'suppression_filter_rc=%s\n' "$suppression_filter_rc"
case $suppression_filter_rc in 0) exit 0;; 1) printf '%s\n' 'no match: source-specific suppression 직접 증거 없음' >&2; exit 30;; *) exit 31;; esac
REMOTE
then RATE_READY=true; else printf '중단: rate evidence 미확정 rc=%s\n' "$?" >&2; fi; else printf '%s\n' '중단: fresh target/time READY가 아니다.' >&2; fi; printf 'RATE_READY=%s\n' "$RATE_READY"
```

`Suppressed N messages`는 해당 source와 시간의 suppression을 지지하지만 `N`을 사건 전체 누락 수로 보지 않는다. suppression filter 무결과는 rate limit 배제가 아니다. source가 재시작됐다면 누적량이나 전후 발생률을 같은 원점으로 비교하지 않는다.

## rotation·retention evidence

rotation 조사는 rate evidence 성공 여부와 독립적이지만 fresh target/time을 직접 요구한다. current file/header/config는 과거 vacuum이나 삭제를 단독 증명하지 않는다.

```bash
ROTATION_READY=false; if [[ "$TARGET_READY" == true && "$TIME_READY" == true ]]; then if ssh -- "$SSH_DEST" bash -s -- "$MACHINE_ID" "$BOOT_ID" "$JOURNALD_INVOCATION" <<'REMOTE'
set -u
expected_machine=$1; expected_boot=$2; expected_journald_invocation=$3
[[ "$(tr -d '\n' </etc/machine-id)" == "$expected_machine" && "$(tr -d '\n' </proc/sys/kernel/random/boot_id)" == "$expected_boot" ]] || exit 10
[[ "$(systemctl show --value -p InvocationID -- systemd-journald.service)" == "$expected_journald_invocation" ]] || exit 11
journalctl --list-boots --no-pager || exit 20
journalctl --disk-usage --no-pager || exit 21
journalctl --header --no-pager || exit 22
if command -v systemd-analyze >/dev/null; then tmp=$(mktemp) || exit 12; trap 'rm -f "$tmp"' EXIT; systemd-analyze cat-config systemd/journald.conf >"$tmp"; producer_rc=$?; printf 'config_producer_rc=%s\n' "$producer_rc"; (( producer_rc == 0 )) || exit 23; grep -E '^[[:space:]]*(Storage|SystemMaxUse|SystemKeepFree|SystemMaxFileSize|SystemMaxFiles|RuntimeMaxUse|RuntimeKeepFree|RuntimeMaxFileSize|RuntimeMaxFiles|MaxRetentionSec|MaxFileSec|RateLimitIntervalSec|RateLimitBurst|Compress|Seal)=' "$tmp"; filter_rc=$?; printf 'config_filter_rc=%s\n' "$filter_rc"; case $filter_rc in 0|1) exit 0;; *) exit 31;; esac; else printf '%s\n' 'unsupported: systemd-analyze unavailable'; exit 41; fi
REMOTE
then ROTATION_READY=true; else printf '중단: rotation evidence 실패/미지원 rc=%s\n' "$?" >&2; fi; else printf '%s\n' '중단: fresh target/time READY가 아니다.' >&2; fi; printf 'ROTATION_READY=%s\n' "$ROTATION_READY"
```

boot 부재, disk usage, header 또는 config 값 하나만으로 rotation·volatile storage·삭제를 확정하지 않는다. `config_filter_rc=1`은 explicit override 없음이며 default는 미확정이다. rotation 원인 확정에는 사건 전후 cursor/file time 경계, 보존 설정·disk pressure 또는 vacuum 작업 기록이 같은 boot/time에서 필요하다.

## 변경 경계와 정상화

진단 중 rate-limit·retention 변경, vacuum, rotate, journald restart와 파일 삭제를 실행하지 않는다. 별도 변경은 실행 직전 refresh와 관련 evidence를 다시 통과하고 source/journald current InvocationID, 승인 ID·권한·변경 창, journal filesystem headroom, 로그 폭주 영향, 멱등성, 중단 기준과 config rollback 원본을 확인해야 한다. vacuum·삭제는 보존 복제본 없이는 rollback 불가다.

교체 machine/boot/source/journald는 모든 READY를 폐기하고 입력부터 재등록한다. 정상화는 disk/retention -> suppression·rotation 직접 증거 재발 없음 -> exact source의 cursor가 이어지는 bounded 기록 -> 외부 log 전달과 실제 service 기능 순으로 새 관찰 창에서 검증한다. 로그가 한 행 다시 보인다는 사실만으로 정상화하지 않는다.

## 증거와 미검증
