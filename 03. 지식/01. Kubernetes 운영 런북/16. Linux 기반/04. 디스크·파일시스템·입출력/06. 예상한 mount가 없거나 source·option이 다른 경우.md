# 예상한 mount가 없거나 source·option이 다른 경우

## 입력값과 빠른 수집

실행 경로는 `승인된 작업 PC/bastion → kubectl API의 Node UID·machineID → SSH host의 hostname·machine-id → host mount namespace → process mount namespace → source/device → filesystem → target → application`이다. 모든 명령은 같은 Bash에서 실행한다. host의 `findmnt` 성공은 process namespace의 mount를 보장하지 않는다.

| 구분 | 값 | 출처와 gate |
|---|---|---|
| 최초 입력 | `CTX`, `NODE`, `NODE_HOST`, `MOUNT_TARGET`, evidence window | incident와 승인된 host/target |
| API 확인 자동값 | `NODE_UID`, `API_MACHINE_ID` | Node API, 원격 machine-id와 exact match |
| 계산 후보값 | `MOUNT_UNIT`, target regex, mount ID·major:minor | target과 현재 mount에서 계산하고 검증 전 미확정 |
| 운영자 선택값 | `EXPECTED_SOURCE`, `PID`, `ARGV_RE` | 계약·inventory와 실제 consumer 관측 뒤 선택 |

### 1. 입력·초기화

`TOOLS_READY=false; MISSING=''; for T in kubectl jq ssh date grep sed; do command -v "$T" >/dev/null 2>&1 || MISSING="$MISSING $T"; done; if [ -n "${BASH_VERSION:-}" ] && [ -z "$MISSING" ]; then TOOLS_READY=true; else printf '중단: Bash 또는 도구가 없음:%s\n' "$MISSING" >&2; fi`

identity refresh 또는 실패 전에는 host machine identity, PID start, mount/device, cgroup/netns, time, argv regex와 모든 downstream gate를 버린다.

`invalidate_all(){ unset HOST_READY TARGET_READY PROCESS_READY TIME_READY EVIDENCE_READY UNIT_READY CHANGE_READY RECOVERY_READY NODE_UID API_MACHINE_ID REMOTE_HOSTNAME REMOTE_MACHINE_ID NODE_RE MOUNT_ID MOUNT_SOURCE MOUNT_FSTYPE MOUNT_MAJOR_MINOR MOUNT_OPTIONS MOUNT_UNIT EXPECTED_SOURCE PID PID_START CGROUP_NS NET_NS MNT_NS ARGV_RE START_CANONICAL END_CANONICAL START_EPOCH END_EPOCH PRODUCER_RC FILTER_RC EVIDENCE_AT CHANGE_APPLIED; HOST_READY=false; TARGET_READY=false; PROCESS_READY=false; TIME_READY=false; EVIDENCE_READY=false; UNIT_READY=false; CHANGE_READY=false; RECOVERY_READY=false; }; invalidate_all; CTX='PASTE_CONTEXT_HERE'; NODE='PASTE_NODE_HERE'; NODE_HOST='PASTE_NODE_HOST_HERE'; MOUNT_TARGET='PASTE_MOUNT_TARGET_HERE'; START_RFC3339='PASTE_START_RFC3339_HERE'; END_RFC3339='PASTE_END_RFC3339_HERE'; INPUT_READY=false; if [ "$TOOLS_READY" != true ] || [[ "$CTX" == PASTE_* ]] || [[ "$NODE" == PASTE_* ]] || [[ "$NODE_HOST" == PASTE_* ]] || [[ ! "$NODE_HOST" =~ ^[A-Za-z0-9._:-]+$ ]] || [[ ! "$MOUNT_TARGET" =~ ^/[A-Za-z0-9._/-]+$ ]]; then printf '%s\n' '중단: 승인된 context·Node·host·절대 target이 필요하다.' >&2; elif kubectl --context "$CTX" get node "$NODE" -o name >/dev/null 2>&1; then INPUT_READY=true; export CTX NODE NODE_HOST MOUNT_TARGET; printf 'INPUT_READY=true node=%s target=%s\n' "$NODE" "$MOUNT_TARGET"; else printf '%s\n' '중단: exact-name Node 조회 실패' >&2; fi`

### 2. host identity와 current guard

`refresh_host(){ local j r rh rm; invalidate_all; [ "$INPUT_READY" = true ] || return 1; j="$(kubectl --context "$CTX" get node "$NODE" -o json)" || return 1; NODE_UID="$(jq -er '.metadata.uid' <<<"$j")" || return 1; API_MACHINE_ID="$(jq -er '.status.nodeInfo.machineID' <<<"$j")" || return 1; jq -e --arg h "$NODE_HOST" --arg n "$NODE" '(.metadata.name==$n) and (($h==$n) or any(.status.addresses[]?;.address==$h))' <<<"$j" >/dev/null || { invalidate_all; return 1; }; r="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s <<< 'set -u; hostname -s; cat /etc/machine-id')" || { invalidate_all; return 1; }; rh="${r%%$'\n'*}"; rm="${r#*$'\n'}"; rm="${rm%%$'\n'*}"; jq -e --arg h "$rh" --arg n "$NODE" '($h==$n) or any(.status.addresses[]?;.address==$h)' <<<"$j" >/dev/null && [ "$rm" = "$API_MACHINE_ID" ] || { invalidate_all; return 1; }; REMOTE_HOSTNAME="$rh"; REMOTE_MACHINE_ID="$rm"; NODE_RE="$(printf '%s' "$NODE" | sed 's/[][\\.^$*+?(){}|]/\\&/g')"; HOST_READY=true; export NODE_UID API_MACHINE_ID REMOTE_HOSTNAME REMOTE_MACHINE_ID NODE_RE; printf 'HOST_READY=true node=%s uid=%s machine-id=%s remote=%s\n' "$NODE" "$NODE_UID" "$API_MACHINE_ID" "$REMOTE_HOSTNAME"; }; refresh_host`

`guard_host(){ local j r rh rm; [ "$HOST_READY" = true ] || return 1; j="$(kubectl --context "$CTX" get node "$NODE" -o json)" || { invalidate_all; return 1; }; [ "$(jq -r '.metadata.uid // empty' <<<"$j")" = "$NODE_UID" ] && [ "$(jq -r '.status.nodeInfo.machineID // empty' <<<"$j")" = "$API_MACHINE_ID" ] || { invalidate_all; return 1; }; r="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s <<< 'set -u; hostname -s; cat /etc/machine-id')" || { invalidate_all; return 1; }; rh="${r%%$'\n'*}"; rm="${r#*$'\n'}"; rm="${rm%%$'\n'*}"; [ "$rh" = "$REMOTE_HOSTNAME" ] && [ "$rm" = "$REMOTE_MACHINE_ID" ] || { invalidate_all; return 1; }; }`

`TIME_READY=false; EVIDENCE_READY=false; CHANGE_READY=false; unset START_CANONICAL END_CANONICAL START_EPOCH END_EPOCH; if ! guard_host; then printf '%s\n' '중단: current host guard 실패' >&2; elif [[ "$START_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && [[ "$END_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && START_CANONICAL="$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" && END_CANONICAL="$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" && [ "$START_CANONICAL" = "$START_RFC3339" ] && [ "$END_CANONICAL" = "$END_RFC3339" ] && START_EPOCH="$(date -u -d "$START_RFC3339" +%s)" && END_EPOCH="$(date -u -d "$END_RFC3339" +%s)" && [ "$START_EPOCH" -lt "$END_EPOCH" ]; then TIME_READY=true; export START_RFC3339 END_RFC3339 START_EPOCH END_EPOCH; printf 'TIME_READY=true start=%s end=%s\n' "$START_RFC3339" "$END_RFC3339"; else printf '%s\n' '중단: strict UTC round-trip 또는 start<end 실패' >&2; fi`

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | 기대 target 부재 또는 active source·fstype·options 불일치 |
| 적용 | host와 process mount namespace, mount ID·major:minor, 선언과 active 상태를 분리 가능 |
| 제외 | Pod Event의 `FailedMount`·CSI staging/publish는 [[25. Pod 이벤트에 FailedMount·FailedAttachVolume이 기록된 경우]], corruption은 [[07. filesystem corruption 오류가 관측된 경우]] |

검색 문자열: `mount`, `automount`, `wrong fs type`, `no such device`, `UUID`, `dependency failed`, `read-only`.

## 관측별 진단 경로

| 지금 보이는 것 | 먼저 볼 것 | 그다음 어디를 볼지 | 상세 절 |
|---|---|---|---|
| exact target mount가 없음 | `findmnt -M`의 producer rc와 unit 선언 | dependency·device·network source owner | [[#target·process·namespace 등록\|target 상태]] |
| source·fstype·options가 기대와 다름 | mount ID·major:minor·선언 `What` | 선언 drift 또는 다른 namespace | [[#evidence window와 producer/filter rc\|evidence]] |
| host에는 있고 process namespace에는 없음 | PID start·mount namespace·propagation | container/runtime 또는 service namespace owner | [[#target·process·namespace 등록\|namespace 비교]] |
| mount 복구 후 application 실패 | 실제 read/write·데이터 일관성 | filesystem·storage·application owner | [[#정상화·증거\|정상화]] |

## target·process·namespace 등록

**exact target 존재 여부를 producer rc와 분리**

`if ! guard_host; then printf '%s\n' '중단: current host guard 실패' >&2; else MOUNT_LOOKUP="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s -- "$MOUNT_TARGET" <<< 'set -u; command -v findmnt >/dev/null || exit 127; findmnt -rn -M "$1" -o ID,TARGET,SOURCE,FSTYPE,MAJ:MIN,OPTIONS')"; PRODUCER_RC=$?; case "$PRODUCER_RC" in 0) [ -n "$MOUNT_LOOKUP" ] && printf '%s\n' "$MOUNT_LOOKUP" || printf '%s\n' '중단: 성공 rc지만 빈 mount 출력' >&2 ;; 1) printf 'exact target mount 없음 target=%s; 정상으로 해석하지 않음\n' "$MOUNT_TARGET" ;; *) printf 'findmnt producer 실패 rc=%s\n' "$PRODUCER_RC" >&2 ;; esac; unset MOUNT_LOOKUP; fi`

`EXPECTED_SOURCE='PASTE_EXPECTED_SOURCE_HERE'; TARGET_READY=false; PROCESS_READY=false; EVIDENCE_READY=false; UNIT_READY=false; CHANGE_READY=false; unset MOUNT_ID MOUNT_SOURCE MOUNT_FSTYPE MOUNT_MAJOR_MINOR MOUNT_OPTIONS PID PID_START CGROUP_NS NET_NS MNT_NS; if [ "$TIME_READY" != true ] || ! guard_host || [[ "$EXPECTED_SOURCE" == PASTE_* ]] || [[ ! "$EXPECTED_SOURCE" =~ ^(/[A-Za-z0-9._/+:-]+|[A-Za-z0-9._:-]+:/[A-Za-z0-9._/-]+)$ ]]; then printf '%s\n' '중단: current host/time과 승인된 source가 필요하다.' >&2; else MF="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s -- "$MOUNT_TARGET" <<< 'set -u; command -v findmnt >/dev/null || exit 127; findmnt -rn -M "$1" -o ID,SOURCE,FSTYPE,MAJ:MIN,OPTIONS')"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ] && read -r MOUNT_ID MOUNT_SOURCE MOUNT_FSTYPE MOUNT_MAJOR_MINOR MOUNT_OPTIONS <<<"$MF" && [[ "$MOUNT_ID" =~ ^[0-9]+$ ]]; then TARGET_READY=true; export EXPECTED_SOURCE MOUNT_ID MOUNT_SOURCE MOUNT_FSTYPE MOUNT_MAJOR_MINOR MOUNT_OPTIONS; printf 'TARGET_READY=true target=%s current-source=%s expected-source=%s fstype=%s major-minor=%s mount-id=%s options=%s\n' "$MOUNT_TARGET" "$MOUNT_SOURCE" "$EXPECTED_SOURCE" "$MOUNT_FSTYPE" "$MOUNT_MAJOR_MINOR" "$MOUNT_ID" "$MOUNT_OPTIONS"; else printf '중단: exact target가 없거나 findmnt producer 실패 rc=%s\n' "$PRODUCER_RC" >&2; fi; unset MF; fi`

`guard_target(){ local f id src fs mm opt; guard_host || return 1; [ "$TARGET_READY" = true ] || return 1; f="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s -- "$MOUNT_TARGET" <<< 'set -u; command -v findmnt >/dev/null || exit 127; findmnt -rn -M "$1" -o ID,SOURCE,FSTYPE,MAJ:MIN,OPTIONS')" || { TARGET_READY=false; PROCESS_READY=false; EVIDENCE_READY=false; CHANGE_READY=false; return 1; }; read -r id src fs mm opt <<<"$f"; [ "$id" = "$MOUNT_ID" ] && [ "$src" = "$MOUNT_SOURCE" ] && [ "$fs" = "$MOUNT_FSTYPE" ] && [ "$mm" = "$MOUNT_MAJOR_MINOR" ] && [ "$opt" = "$MOUNT_OPTIONS" ] || { TARGET_READY=false; PROCESS_READY=false; EVIDENCE_READY=false; CHANGE_READY=false; return 1; }; }`

`PID='PASTE_PID_HERE'; ARGV_RE='PASTE_ARGV_REGEX_HERE'; PROCESS_READY=false; if ! guard_target || ! [[ "$PID" =~ ^[1-9][0-9]*$ ]] || [[ "$ARGV_RE" == PASTE_* ]]; then printf '%s\n' '중단: current target과 실제 consumer PID·argv regex가 필요하다.' >&2; elif grep -E -- "$ARGV_RE" </dev/null >/dev/null 2>&1 || [ "$?" -eq 1 ]; then PF="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s -- "$PID" "$ARGV_RE" <<< 'set -u; p="$1"; re="$2"; s="$(cat "/proc/$p/stat")" || exit 3; r="${s##*) }"; set -- $r; tr "\0" "\n" < "/proc/$p/cmdline" | grep -Eq -- "$re" || exit 4; printf "%s %s %s %s\n" "${20}" "$(readlink "/proc/$p/ns/cgroup")" "$(readlink "/proc/$p/ns/net")" "$(readlink "/proc/$p/ns/mnt")"')"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ] && read -r PID_START CGROUP_NS NET_NS MNT_NS <<<"$PF"; then PROCESS_READY=true; export PID ARGV_RE PID_START CGROUP_NS NET_NS MNT_NS; printf 'PROCESS_READY=true pid=%s start=%s cgroup-ns=%s netns=%s mntns=%s\n' "$PID" "$PID_START" "$CGROUP_NS" "$NET_NS" "$MNT_NS"; else printf '중단: process identity rc=%s\n' "$PRODUCER_RC" >&2; fi; unset PF; else printf '%s\n' '중단: invalid argv regex' >&2; fi`

`guard_process(){ local f st cg nn mn; guard_target || return 1; [ "$PROCESS_READY" = true ] || return 1; f="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s -- "$PID" "$ARGV_RE" <<< 'set -u; p="$1"; re="$2"; s="$(cat "/proc/$p/stat")" || exit 3; r="${s##*) }"; set -- $r; tr "\0" "\n" < "/proc/$p/cmdline" | grep -Eq -- "$re" || exit 4; printf "%s %s %s %s\n" "${20}" "$(readlink "/proc/$p/ns/cgroup")" "$(readlink "/proc/$p/ns/net")" "$(readlink "/proc/$p/ns/mnt")"')" || { PROCESS_READY=false; EVIDENCE_READY=false; CHANGE_READY=false; return 1; }; read -r st cg nn mn <<<"$f"; [ "$st" = "$PID_START" ] && [ "$cg" = "$CGROUP_NS" ] && [ "$nn" = "$NET_NS" ] && [ "$mn" = "$MNT_NS" ] || { PROCESS_READY=false; EVIDENCE_READY=false; CHANGE_READY=false; return 1; }; }`

**host와 process namespace를 같은 current identity로 비교**

`if guard_process; then ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s -- "$PID" "$MOUNT_TARGET" <<< 'set -u; p="$1"; t="$2"; printf "%s\n" "=== host mount namespace ==="; sudo -n findmnt -rn -T "$t" -o ID,TARGET,SOURCE,FSTYPE,MAJ:MIN,OPTIONS,PROPAGATION || exit 2; printf "%s\n" "=== process mount namespace ==="; sudo -n nsenter -t "$p" -m -- findmnt -rn -T "$t" -o ID,TARGET,SOURCE,FSTYPE,MAJ:MIN,OPTIONS,PROPAGATION'; else printf '%s\n' '중단: current process/target guard 실패' >&2; fi`

두 결과 중 하나가 비면 성공이 아니다. host와 process의 mount ID·source·major:minor가 다르면 namespace 경계로 기록한다. process가 Kubernetes Pod라면 Pod UID·node·containerID lifecycle을 해당 storage 런북에서 다시 등록한 뒤 host 변경을 중단한다.

## evidence window와 producer/filter rc

`EVIDENCE_READY=false; if [ "$TIME_READY" != true ] || ! guard_process; then printf '%s\n' '중단: current identity/time gate 실패' >&2; else RAW="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s -- "$START_RFC3339" "$END_RFC3339" <<< 'set -u; command -v journalctl >/dev/null || exit 127; sudo -n journalctl -k --since "$1" --until "$2" -o short-iso-precise --no-pager')"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -ne 0 ]; then printf 'journal producer 실패 rc=%s\n' "$PRODUCER_RC" >&2; else MATCH="$(grep -E -i -C 3 -- 'mount|automount|dependency|wrong fs type|no such device|UUID|mapper|read-only|I/O error' <<<"$RAW")"; FILTER_RC=$?; if [ "$FILTER_RC" -eq 0 ]; then EVIDENCE_AT="$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; EVIDENCE_READY=true; printf '%s\n' "$MATCH" | sed -n '1,80p'; printf 'EVIDENCE_READY=true at=%s\n' "$EVIDENCE_AT"; elif [ "$FILTER_RC" -eq 1 ]; then printf '%s\n' 'filter 일치 없음; mount 정상이나 오류 부재로 해석하지 않음' >&2; else printf 'filter 실패 rc=%s\n' "$FILTER_RC" >&2; fi; fi; unset RAW MATCH; fi`

fstab·unit 원문에는 credential이 있을 수 있으므로 전체를 출력하지 않는다. unit이 실제 owner일 때만 계산 이름을 등록하고 bounded property만 읽는다.

`UNIT_READY=false; if ! guard_target; then printf '%s\n' '중단: current target guard 실패' >&2; else MOUNT_UNIT="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s -- "$MOUNT_TARGET" <<< 'set -u; command -v systemd-escape >/dev/null || exit 127; systemd-escape -p --suffix=mount "$1"')"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ] && [[ "$MOUNT_UNIT" =~ ^[A-Za-z0-9_.@-]+\.mount$ ]]; then SHOW="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s -- "$MOUNT_UNIT" <<< 'sudo -n systemctl show "$1" --property=Id,LoadState,ActiveState,SubState,Where,What,Options,CanStart --no-pager')"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ] && grep -Fxq "Id=$MOUNT_UNIT" <<<"$SHOW" && grep -Fxq "Where=$MOUNT_TARGET" <<<"$SHOW" && grep -Fxq "What=$EXPECTED_SOURCE" <<<"$SHOW"; then UNIT_READY=true; export MOUNT_UNIT; printf '%s\n' "$SHOW"; printf 'UNIT_READY=true unit=%s\n' "$MOUNT_UNIT"; else printf '중단: unit property producer/identity 실패 rc=%s\n' "$PRODUCER_RC" >&2; fi; else printf '중단: mount unit 계산 실패 rc=%s\n' "$PRODUCER_RC" >&2; fi; unset SHOW; fi`

## 변경 안전 경계·재등록

이 문서는 `mount`, `umount`, `remount`, `systemctl start/stop/restart`, fstab·unit 수정, device rescan과 filesystem repair 명령을 제공하지 않는다. target가 없을 때는 current mount identity가 없으므로 present target용 `TARGET_READY`·`PROCESS_READY`를 변경 admission으로 재사용할 수 없다. 선언 원본, source availability, filesystem 상태, application quorum·availability, 승인 ID·window, idempotence, 중단 조건과 rollback 원본을 별도 변경 절차에서 새로 등록한다.

외부 변경 뒤 `refresh_host`부터 다시 수행해 exact target의 새 mount ID·source·major:minor·options를 등록한다. replacement process가 생기면 새 PID·start tick·cgroup/netns/mntns·argv regex를 다시 등록한다. 승인된 application read/write와 데이터 일관성, Pod라면 새 Pod UID·node·container lifecycle과 실제 요청을 확인한다.

## 정상화·증거

정상화는 `source/device → filesystem → exact mount ID·options → process namespace → application` 순서다. mount가 보이거나 unit이 active인 것만으로 성공이 아니다. 새 journal 오류가 없고 actual source가 승인 계약과 맞으며 실제 기능이 bounded observation window에서 회복돼야 한다.

보존 항목은 Node UID/machineID, remote machine-id, target, host/process mount ID·source·fstype·major:minor·options·namespace, PID start/cgroup/netns/mntns, UTC 창, producer/filter rc, unit bounded property, 승인·rollback·기능 결과다. argv·fstab 원문, credential, tenant path와 전체 journal은 저장하지 않는다.
