# disk latency와 I-O queue가 지속적으로 높은 경우

## 입력값과 빠른 수집

이 런북은 승인된 작업 PC 또는 bastion의 **같은 Bash**에서 실행한다. 연결 경로는 `작업 host의 kubectl·ssh → API server의 Node UID/machineID → 승인된 SSH host의 hostname/machine-id → block device → partition/mapper → filesystem/mount → process cgroup·namespace → workload`다. `127.0.0.1`이나 CRI endpoint를 사용하는 흐름이 아니며 runtime socket을 추측하지 않는다.

| 구분 | 값 | 출처와 gate |
|---|---|---|
| 최초 입력 | `CTX`, `NODE`, `NODE_HOST`, `START_RFC3339`, `END_RFC3339` | 승인 기록의 context·Node·SSH 주소와 장애 전후 UTC 창 |
| API 확인 자동값 | `NODE_UID`, `API_MACHINE_ID` | 현재 Node API에서 읽고 매 원격 명령 직전에 재검증 |
| 계산 후보값 | `NODE_RE`, device의 KNAME·major:minor·sysfs path | API·`lsblk`·sysfs 출력으로 만들며 일치 전에는 미확정 |
| 운영자 선택값 | `DEVICE_PATH`, 조건부 `PID`, `ARGV_RE` | 실제 latency source와 process I/O 증거를 본 뒤 선택 |

### 1. 실행 위치·도구·최초 입력

**도구 확인**

`TOOLS_READY=false; MISSING=''; for T in kubectl jq ssh date grep sed; do command -v "$T" >/dev/null 2>&1 || MISSING="$MISSING $T"; done; if [ -n "${BASH_VERSION:-}" ] && [ -z "$MISSING" ]; then TOOLS_READY=true; printf 'TOOLS_READY=true bash=%s\n' "$BASH_VERSION"; else printf '중단: Bash 또는 도구가 없음:%s\n' "$MISSING" >&2; fi`

**전체 identity와 downstream gate 무효화·입력**

`invalidate_all(){ unset HOST_READY DEVICE_READY PROCESS_READY TIME_READY EVIDENCE_READY CHANGE_READY RECOVERY_READY NODE_UID API_MACHINE_ID REMOTE_HOSTNAME REMOTE_MACHINE_ID NODE_RE DEVICE_PATH KNAME MAJOR_MINOR DEVICE_TYPE DEVICE_SYSFS DEVICE_SIZE MOUNT_ID MOUNT_SOURCE MOUNT_TARGET PID PID_START CGROUP_NS NET_NS MNT_NS ARGV_RE START_CANONICAL END_CANONICAL START_EPOCH END_EPOCH EVIDENCE_AT EVIDENCE_EPOCH EVIDENCE_NODE_UID EVIDENCE_MAJOR_MINOR PRODUCER_RC FILTER_RC CHANGE_APPLIED; HOST_READY=false; DEVICE_READY=false; PROCESS_READY=false; TIME_READY=false; EVIDENCE_READY=false; CHANGE_READY=false; RECOVERY_READY=false; }; invalidate_all; CTX='PASTE_CONTEXT_HERE'; NODE='PASTE_NODE_HERE'; NODE_HOST='PASTE_NODE_HOST_HERE'; START_RFC3339='PASTE_START_RFC3339_HERE'; END_RFC3339='PASTE_END_RFC3339_HERE'; INPUT_READY=false; if [ "$TOOLS_READY" != true ] || [ -z "$CTX" ] || [ -z "$NODE" ] || [ -z "$NODE_HOST" ] || [[ "$CTX" == PASTE_* ]] || [[ "$NODE" == PASTE_* ]] || [[ "$NODE_HOST" == PASTE_* ]] || ! [[ "$NODE_HOST" =~ ^[A-Za-z0-9._:-]+$ ]]; then printf '%s\n' '중단: 승인된 context·Node·SSH host가 필요하다.' >&2; elif kubectl --context "$CTX" get node "$NODE" -o name >/dev/null 2>&1; then INPUT_READY=true; export CTX NODE NODE_HOST; printf 'INPUT_READY=true context=%s node=%s host=%s\n' "$CTX" "$NODE" "$NODE_HOST"; else printf '%s\n' '중단: exact-name Node API 조회 실패' >&2; fi`

### 2. host identity와 strict UTC 창 등록

host identity를 새로 읽을 때는 process start, mount/device, cgroup/netns, time 파생값과 모든 downstream gate를 먼저 버린다.

**host identity 함수 등록·실행**

`refresh_host(){ local j remote rc rh rm; invalidate_all; [ "$INPUT_READY" = true ] || { printf '%s\n' '중단: INPUT_READY가 아니다.' >&2; return 1; }; j="$(kubectl --context "$CTX" get node "$NODE" -o json)" || return 1; NODE_UID="$(jq -er '.metadata.uid' <<<"$j")" || return 1; API_MACHINE_ID="$(jq -er '.status.nodeInfo.machineID' <<<"$j")" || return 1; jq -e --arg h "$NODE_HOST" --arg n "$NODE" '(.metadata.name==$n) and (($h==$n) or any(.status.addresses[]?;.address==$h))' <<<"$j" >/dev/null || { invalidate_all; printf '%s\n' '중단: NODE_HOST가 Node name/address와 관계없다.' >&2; return 1; }; remote="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s <<< 'set -u; hostname -s; cat /etc/machine-id')"; rc=$?; [ "$rc" -eq 0 ] || { invalidate_all; printf '중단: SSH identity producer rc=%s\n' "$rc" >&2; return 1; }; rh="${remote%%$'\n'*}"; rm="${remote#*$'\n'}"; rm="${rm%%$'\n'*}"; jq -e --arg h "$rh" --arg n "$NODE" '($h==$n) or any(.status.addresses[]?;.address==$h)' <<<"$j" >/dev/null && [ "$rm" = "$API_MACHINE_ID" ] || { invalidate_all; printf '%s\n' '중단: API Node와 remote hostname/machine-id 불일치' >&2; return 1; }; REMOTE_HOSTNAME="$rh"; REMOTE_MACHINE_ID="$rm"; NODE_RE="$(printf '%s' "$NODE" | sed 's/[][\\.^$*+?(){}|]/\\&/g')"; HOST_READY=true; export NODE_UID API_MACHINE_ID REMOTE_HOSTNAME REMOTE_MACHINE_ID NODE_RE; printf 'HOST_READY=true node=%s uid=%s machine-id=%s remote=%s\n' "$NODE" "$NODE_UID" "$API_MACHINE_ID" "$REMOTE_HOSTNAME"; }; refresh_host`

**모든 후속 명령의 current host guard**

`guard_host(){ local j r rh rm; [ "$HOST_READY" = true ] || return 1; j="$(kubectl --context "$CTX" get node "$NODE" -o json)" || { invalidate_all; return 1; }; [ "$(jq -r '.metadata.uid // empty' <<<"$j")" = "$NODE_UID" ] && [ "$(jq -r '.status.nodeInfo.machineID // empty' <<<"$j")" = "$API_MACHINE_ID" ] || { invalidate_all; return 1; }; r="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s <<< 'set -u; hostname -s; cat /etc/machine-id')" || { invalidate_all; return 1; }; rh="${r%%$'\n'*}"; rm="${r#*$'\n'}"; rm="${rm%%$'\n'*}"; [ "$rh" = "$REMOTE_HOSTNAME" ] && [ "$rm" = "$REMOTE_MACHINE_ID" ] || { invalidate_all; return 1; }; }`

**UTC RFC3339 round-trip과 순서 확인**

`TIME_READY=false; EVIDENCE_READY=false; CHANGE_READY=false; unset START_CANONICAL END_CANONICAL START_EPOCH END_EPOCH EVIDENCE_AT EVIDENCE_EPOCH EVIDENCE_NODE_UID EVIDENCE_MAJOR_MINOR; if ! guard_host; then printf '%s\n' '중단: current host guard 실패' >&2; elif [[ ! "$START_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] || [[ ! "$END_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]]; then printf '%s\n' '중단: strict UTC RFC3339 Z 형식이 아니다.' >&2; elif START_CANONICAL="$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" && END_CANONICAL="$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" && [ "$START_CANONICAL" = "$START_RFC3339" ] && [ "$END_CANONICAL" = "$END_RFC3339" ] && START_EPOCH="$(date -u -d "$START_RFC3339" +%s)" && END_EPOCH="$(date -u -d "$END_RFC3339" +%s)" && [ "$START_EPOCH" -lt "$END_EPOCH" ]; then TIME_READY=true; export START_RFC3339 END_RFC3339 START_EPOCH END_EPOCH; printf 'TIME_READY=true start=%s end=%s\n' "$START_RFC3339" "$END_RFC3339"; else printf '%s\n' '중단: calendar round-trip 실패 또는 start>=end' >&2; fi`

## device·mount identity와 bounded evidence

**관측된 device 선택·등록**

`DEVICE_PATH='PASTE_DEVICE_PATH_HERE'; DEVICE_READY=false; PROCESS_READY=false; EVIDENCE_READY=false; CHANGE_READY=false; RECOVERY_READY=false; unset KNAME MAJOR_MINOR DEVICE_TYPE DEVICE_SYSFS DEVICE_SIZE MOUNT_ID MOUNT_SOURCE MOUNT_TARGET PID PID_START CGROUP_NS NET_NS MNT_NS ARGV_RE EVIDENCE_AT EVIDENCE_EPOCH EVIDENCE_NODE_UID EVIDENCE_MAJOR_MINOR; if [ "$TIME_READY" != true ] || ! guard_host; then printf '%s\n' '중단: current host/time guard 실패' >&2; elif [[ ! "$DEVICE_PATH" =~ ^/dev/[A-Za-z0-9._/+:-]+$ ]]; then printf '%s\n' '중단: 실제 /dev 경로가 필요하다.' >&2; else FACTS="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s -- "$DEVICE_PATH" <<< 'set -u; d="$1"; test -b "$d" || exit 3; command -v lsblk >/dev/null || exit 127; k="$(lsblk -dn -o KNAME "$d")" || exit 4; mm="$(lsblk -dn -o MAJ:MIN "$d")" || exit 5; ty="$(lsblk -dn -o TYPE "$d")" || exit 6; sy="$(readlink -f "/sys/class/block/$k")" || exit 7; sz="$(cat "/sys/class/block/$k/size")" || exit 8; printf "%s %s %s %s %s\n" "$k" "$mm" "$ty" "$sy" "$sz"')"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ] && read -r KNAME MAJOR_MINOR DEVICE_TYPE DEVICE_SYSFS DEVICE_SIZE <<<"$FACTS" && [ -n "$DEVICE_SYSFS" ]; then DEVICE_READY=true; export DEVICE_PATH KNAME MAJOR_MINOR DEVICE_TYPE DEVICE_SYSFS DEVICE_SIZE; printf 'DEVICE_READY=true path=%s kname=%s major-minor=%s type=%s sysfs=%s sectors=%s\n' "$DEVICE_PATH" "$KNAME" "$MAJOR_MINOR" "$DEVICE_TYPE" "$DEVICE_SYSFS" "$DEVICE_SIZE"; else printf '중단: device producer 실패 rc=%s\n' "$PRODUCER_RC" >&2; fi; unset FACTS; fi`

**current device guard 등록**

`guard_device(){ local f k mm ty sy sz; guard_host || return 1; [ "$DEVICE_READY" = true ] || return 1; f="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s -- "$DEVICE_PATH" <<< 'set -u; d="$1"; test -b "$d" || exit 3; k="$(lsblk -dn -o KNAME "$d")" || exit 4; printf "%s %s %s %s %s\n" "$k" "$(lsblk -dn -o MAJ:MIN "$d")" "$(lsblk -dn -o TYPE "$d")" "$(readlink -f "/sys/class/block/$k")" "$(cat "/sys/class/block/$k/size")"')" || { DEVICE_READY=false; PROCESS_READY=false; EVIDENCE_READY=false; CHANGE_READY=false; return 1; }; read -r k mm ty sy sz <<<"$f"; [ "$k" = "$KNAME" ] && [ "$mm" = "$MAJOR_MINOR" ] && [ "$ty" = "$DEVICE_TYPE" ] && [ "$sy" = "$DEVICE_SYSFS" ] && [ "$sz" = "$DEVICE_SIZE" ] || { DEVICE_READY=false; PROCESS_READY=false; EVIDENCE_READY=false; CHANGE_READY=false; return 1; }; }`

**device graph와 mount hop 확인**

`if guard_device; then ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s -- "$DEVICE_PATH" <<< 'set -u; d="$1"; command -v lsblk >/dev/null || exit 127; lsblk -p -o NAME,KNAME,TYPE,PKNAME,MAJ:MIN,FSTYPE,UUID,RO,SIZE,MOUNTPOINTS "$d"; command -v findmnt >/dev/null || exit 127; findmnt -rn -S "$d" -o TARGET,SOURCE,FSTYPE,OPTIONS'; else printf '%s\n' '중단: current device guard 실패' >&2; fi`

빈 mount 출력은 성공이 아니다. partition·mapper에 mount가 있을 수 있으므로 device graph에서 실제 하위/상위 관계를 확인하고, network source이면 이 런북을 중단한다.

**iostat producer와 filter를 분리한 bounded sample**

`EVIDENCE_READY=false; CHANGE_READY=false; unset SAMPLE FILTERED EVIDENCE_AT EVIDENCE_EPOCH EVIDENCE_NODE_UID EVIDENCE_MAJOR_MINOR; if [ "$TIME_READY" != true ] || ! guard_device; then printf '%s\n' '중단: time/device gate 실패' >&2; else SAMPLE="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s -- "$DEVICE_PATH" <<< 'set -u; command -v iostat >/dev/null || exit 127; iostat -xz -p "$1" 1 5')"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -ne 0 ]; then printf 'iostat producer 실패 rc=%s; 정상·빈 결과로 해석하지 않음\n' "$PRODUCER_RC" >&2; else FILTERED="$(grep -E -- "(^Device|^${KNAME}[[:space:]])" <<<"$SAMPLE")"; FILTER_RC=$?; if [ "$FILTER_RC" -eq 0 ]; then EVIDENCE_AT="$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; EVIDENCE_EPOCH="$(date -u +%s)"; EVIDENCE_NODE_UID="$NODE_UID"; EVIDENCE_MAJOR_MINOR="$MAJOR_MINOR"; EVIDENCE_READY=true; printf '%s\n' "$FILTERED"; printf 'EVIDENCE_READY=true at=%s device=%s\n' "$EVIDENCE_AT" "$MAJOR_MINOR"; elif [ "$FILTER_RC" -eq 1 ]; then printf '%s\n' 'filter 일치 없음; device 이름·sysstat field를 재확인' >&2; else printf 'filter 실패 rc=%s\n' "$FILTER_RC" >&2; fi; fi; unset SAMPLE FILTERED; fi`

`guard_evidence(){ local now; [ "$EVIDENCE_READY" = true ] || return 1; guard_device || return 1; now="$(date -u +%s)" || return 1; [ "$EVIDENCE_NODE_UID" = "$NODE_UID" ] && [ "$EVIDENCE_MAJOR_MINOR" = "$MAJOR_MINOR" ] && [[ "$EVIDENCE_EPOCH" =~ ^[0-9]+$ ]] && [ $((now-EVIDENCE_EPOCH)) -ge 0 ] && [ $((now-EVIDENCE_EPOCH)) -le 900 ] || { EVIDENCE_READY=false; CHANGE_READY=false; return 1; }; }`

`iostat`가 없으면 같은 KNAME의 `/proc/diskstats`를 정확히 두 시각에 수집한다. field layout·sector size·간격을 확인하지 못하면 latency를 계산하지 않는다.

`if guard_device; then ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s -- "$KNAME" <<< 'set -u; k="$1"; for n in 1 2; do date -u "+sample=%Y-%m-%dT%H:%M:%SZ"; awk -v k="$k" "$3==k{print; found=1} END{exit found?0:2}" /proc/diskstats || exit $?; [ "$n" -eq 1 ] && sleep 5; done'; else printf '%s\n' '중단: current device guard 실패' >&2; fi`

**kernel producer/filter 구분**

`if [ "$TIME_READY" != true ] || ! guard_device; then printf '%s\n' '중단: current time/device gate 실패' >&2; else RAW="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s -- "$START_RFC3339" "$END_RFC3339" <<< 'set -u; command -v journalctl >/dev/null || exit 127; sudo -n journalctl -k --since "$1" --until "$2" -o short-iso-precise --no-pager')"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -ne 0 ]; then printf 'journal producer 실패 rc=%s\n' "$PRODUCER_RC" >&2; else MATCH="$(grep -E -i -C 3 -- "${KNAME}|${MAJOR_MINOR}|I/O error|timeout|read-only|blocked for more than" <<<"$RAW")"; FILTER_RC=$?; if [ "$FILTER_RC" -eq 0 ]; then printf '%s\n' "$MATCH" | sed -n '1,80p'; elif [ "$FILTER_RC" -eq 1 ]; then printf '%s\n' '결정적 행 없음; 오류 부재가 아니라 이 evidence window의 filter 불일치' >&2; else printf 'journal filter 실패 rc=%s\n' "$FILTER_RC" >&2; fi; fi; unset RAW MATCH; fi`

출력에 tenant 경로, argv, credential, token 또는 개인정보가 보이면 표시·저장을 중단하고 시각·device·오류 종류만 남긴다. raw journal, 전체 process 목록과 파일 내용을 저장하지 않는다.

## process identity와 제한적 완화

process별 I/O가 같은 device와 시간상 연결된 경우에만 PID를 등록한다. PID 숫자만으로 동일 process라고 보지 않고 start tick·cgroup namespace·network namespace·mount namespace·argv regex를 함께 고정한다.

`PID='PASTE_PID_HERE'; ARGV_RE='PASTE_ARGV_REGEX_HERE'; PROCESS_READY=false; unset PID_START CGROUP_NS NET_NS MNT_NS; if ! guard_device || ! [[ "$PID" =~ ^[1-9][0-9]*$ ]] || [[ "$ARGV_RE" == PASTE_* ]]; then printf '%s\n' '중단: current device와 실제 PID·argv regex가 필요하다.' >&2; elif grep -E -- "$ARGV_RE" </dev/null >/dev/null 2>&1 || [ "$?" -eq 1 ]; then PF="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s -- "$PID" "$ARGV_RE" <<< 'set -u; p="$1"; re="$2"; test -r "/proc/$p/stat" || exit 3; s="$(cat "/proc/$p/stat")" || exit 4; r="${s##*) }"; set -- $r; start="${20}"; tr "\0" "\n" < "/proc/$p/cmdline" | grep -Eq -- "$re" || exit 5; printf "%s %s %s %s\n" "$start" "$(readlink "/proc/$p/ns/cgroup")" "$(readlink "/proc/$p/ns/net")" "$(readlink "/proc/$p/ns/mnt")"')"; PRODUCER_RC=$?; if [ "$PRODUCER_RC" -eq 0 ] && read -r PID_START CGROUP_NS NET_NS MNT_NS <<<"$PF"; then PROCESS_READY=true; export PID ARGV_RE PID_START CGROUP_NS NET_NS MNT_NS; printf 'PROCESS_READY=true pid=%s start=%s cgroup-ns=%s netns=%s mntns=%s\n' "$PID" "$PID_START" "$CGROUP_NS" "$NET_NS" "$MNT_NS"; else printf '중단: process identity producer rc=%s\n' "$PRODUCER_RC" >&2; fi; unset PF; else printf '%s\n' '중단: invalid argv regex' >&2; fi`

`guard_process(){ local f st cg nn mn; guard_device || return 1; [ "$PROCESS_READY" = true ] || return 1; f="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s -- "$PID" "$ARGV_RE" <<< 'set -u; p="$1"; re="$2"; test -r "/proc/$p/stat" || exit 3; s="$(cat "/proc/$p/stat")"; r="${s##*) }"; set -- $r; tr "\0" "\n" < "/proc/$p/cmdline" | grep -Eq -- "$re" || exit 5; printf "%s %s %s %s\n" "${20}" "$(readlink "/proc/$p/ns/cgroup")" "$(readlink "/proc/$p/ns/net")" "$(readlink "/proc/$p/ns/mnt")"')" || { PROCESS_READY=false; CHANGE_READY=false; return 1; }; read -r st cg nn mn <<<"$f"; [ "$st" = "$PID_START" ] && [ "$cg" = "$CGROUP_NS" ] && [ "$nn" = "$NET_NS" ] && [ "$mn" = "$MNT_NS" ] || { PROCESS_READY=false; CHANGE_READY=false; return 1; }; }`

### 변경 안전 경계

이 문서는 `ionice`, cgroup I/O 제한, process signal·restart, device scheduler 변경, filesystem remount, controller reset과 reboot 명령을 제공하지 않는다. 완화는 `guard_evidence`가 통과한 15분 이내의 same-device sample, current PID/start·namespace, 현재 I/O class·priority 원본, application quorum·availability, 승인 ID·window, idempotence, 중단 조건과 원래 값을 복원하는 rollback 명령을 별도 변경 절차에서 직접 재검증한 경우에만 수행한다.

외부 변경 뒤 process가 교체됐을 수 있으므로 PID·start tick·cgroup/netns/mntns·argv를 다시 등록하고, 동일 device의 새 bounded sample, kernel 오류 비재발, workload owner의 실제 read/write·latency·오류율을 확인한다. process가 사라졌으면 기존 PID를 성공으로 해석하지 않고 replacement PID를 새로 선택한다.

## 정상화·완료 보고

정상화 순서는 `backend/provider → device identity·kernel → latency/queue sample → filesystem/mount → replacement process identity → workload 실제 기능`이다. 빈 `iostat`, journal 또는 metric 결과는 성공이 아니다. 같은 source·field·단위·sample interval의 충분한 evidence window에서 수렴하고 새 I/O/read-only 오류가 없으며 실제 기능이 회복돼야 한다.

보존 항목은 Node name·UID·machineID, remote hostname/machine-id, device path·KNAME·major:minor·sysfs·size, mount 관계, strict UTC 창, producer/filter rc, bounded 결정적 행, PID·start tick·namespace identity, 승인·변경·rollback·기능 검증이다. argv 원문, 파일 경로, token, credential, 개인정보와 전체 journal은 보존하지 않는다.
