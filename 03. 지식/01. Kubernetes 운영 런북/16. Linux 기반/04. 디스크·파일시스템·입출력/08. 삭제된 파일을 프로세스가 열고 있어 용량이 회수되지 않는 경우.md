# 삭제된 파일을 프로세스가 열고 있어 용량이 회수되지 않는 경우

## 입력값과 빠른 수집

실행 hop은 `승인된 작업 PC 또는 bastion의 같은 Bash → Kubernetes Node API → 승인된 SSH host → host mount namespace → /proc/PID/fd → process mount·cgroup·network namespace → application`이다. 파일 내용과 삭제된 경로 원문은 읽거나 저장하지 않는다.

| 구분 | 값 | 출처와 gate |
|---|---|---|
| 최초 입력 | `CTX`, `NODE`, `NODE_HOST`, `MOUNT_TARGET` | 승인 기록과 용량 차이가 관측된 exact mountpoint |
| API 자동값 | `NODE_UID`, `API_MACHINE_ID`, `API_BOOT_ID` | 현재 Node API와 원격 machine-id·boot ID 일치 |
| 계산 후보값 | mount ID·source·major:minor·`st_dev`, PID·FD 후보 | current mount와 `/proc` metadata에서 계산 |
| 운영자 선택값 | `PID`, `FD`, `ARGV_RE` | 삭제-open 후보 출력과 process owner 확인 뒤 선택 |

### 1. 실행 위치·도구·최초 입력

```bash
TOOLS_READY=false
INPUT_READY=false
HOST_READY=false
MOUNT_READY=false
EVIDENCE_READY=false
PROCESS_READY=false
NO_MATCH_READY=false
unset NODE_UID API_MACHINE_ID API_BOOT_ID REMOTE_HOSTNAME REMOTE_MACHINE_ID REMOTE_BOOT_ID MOUNT_ID MOUNT_SOURCE MOUNT_FSTYPE MOUNT_MAJOR_MINOR MOUNT_OPTIONS MOUNT_ST_DEV EVIDENCE_EPOCH EVIDENCE_NODE_UID EVIDENCE_MOUNT_ID PID FD PID_START FD_INODE FD_SIZE CGROUP_NS NET_NS MNT_NS ARGV_RE
CTX='PASTE_DELETED_CONTEXT_HERE'
NODE='PASTE_DELETED_NODE_HERE'
NODE_HOST='PASTE_DELETED_NODE_HOST_HERE'
MOUNT_TARGET='PASTE_DELETED_MOUNT_TARGET_HERE'
MISSING=''
for tool in kubectl jq ssh date grep; do command -v "$tool" >/dev/null 2>&1 || MISSING="$MISSING $tool"; done
if [ -z "${BASH_VERSION:-}" ] || [ -n "$MISSING" ]; then
  printf '중단: Bash 또는 도구 미충족:%s\n' "$MISSING" >&2
elif [[ ! "$NODE_HOST" =~ ^[A-Za-z0-9._:-]+$ ]] || [[ ! "$MOUNT_TARGET" =~ ^/[A-Za-z0-9._/-]+$ ]]; then
  printf '%s\n' '중단: 승인된 SSH host와 절대 mount target이 필요하다.' >&2
elif kubectl --context "$CTX" get node "$NODE" -o name >/dev/null 2>&1; then
  TOOLS_READY=true
  INPUT_READY=true
  printf 'INPUT_READY=true context=%s node=%s target=%s\n' "$CTX" "$NODE" "$MOUNT_TARGET"
else
  printf '%s\n' '중단: exact-name Node API 조회 실패' >&2
fi
```

### 2. current host·mount gate

```bash
invalidate_deleted() {
  HOST_READY=false; MOUNT_READY=false; EVIDENCE_READY=false; PROCESS_READY=false; NO_MATCH_READY=false
  unset NODE_UID API_MACHINE_ID API_BOOT_ID REMOTE_HOSTNAME REMOTE_MACHINE_ID REMOTE_BOOT_ID MOUNT_ID MOUNT_SOURCE MOUNT_FSTYPE MOUNT_MAJOR_MINOR MOUNT_OPTIONS MOUNT_ST_DEV EVIDENCE_EPOCH EVIDENCE_NODE_UID EVIDENCE_MOUNT_ID PID_START FD_INODE FD_SIZE CGROUP_NS NET_NS MNT_NS
}
refresh_deleted_host() {
  local node_json remote rc rh rm rb
  invalidate_deleted
  [ "$INPUT_READY" = true ] || { printf '%s\n' '중단: INPUT_READY가 아니다.' >&2; return 1; }
  node_json="$(kubectl --context "$CTX" get node "$NODE" -o json)" || return 1
  NODE_UID="$(jq -er '.metadata.uid' <<<"$node_json")" || return 1
  API_MACHINE_ID="$(jq -er '.status.nodeInfo.machineID' <<<"$node_json")" || return 1
  API_BOOT_ID="$(jq -er '.status.nodeInfo.bootID' <<<"$node_json")" || return 1
  jq -e --arg h "$NODE_HOST" --arg n "$NODE" '(.metadata.name==$n) and (($h==$n) or any(.status.addresses[]?; .address==$h))' <<<"$node_json" >/dev/null || { invalidate_deleted; return 1; }
  remote="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s <<'REMOTE'
set -eu
printf '%s\t%s\t%s\n' "$(hostname -s)" "$(cat /etc/machine-id)" "$(cat /proc/sys/kernel/random/boot_id)"
REMOTE
)"; rc=$?
  [ "$rc" -eq 0 ] || { invalidate_deleted; printf 'SSH identity producer 실패 rc=%s\n' "$rc" >&2; return 1; }
  IFS=$'\t' read -r rh rm rb <<<"$remote"
  jq -e --arg h "$rh" --arg n "$NODE" '($h==$n) or any(.status.addresses[]?; .address==$h)' <<<"$node_json" >/dev/null && [ "$rm" = "$API_MACHINE_ID" ] && [ "$rb" = "$API_BOOT_ID" ] || { invalidate_deleted; printf '%s\n' '중단: API와 remote host identity 불일치' >&2; return 1; }
  REMOTE_HOSTNAME="$rh"; REMOTE_MACHINE_ID="$rm"; REMOTE_BOOT_ID="$rb"; HOST_READY=true
  printf 'HOST_READY=true node=%s uid=%s machine-id=%s boot-id=%s remote=%s\n' "$NODE" "$NODE_UID" "$API_MACHINE_ID" "$API_BOOT_ID" "$REMOTE_HOSTNAME"
}
guard_deleted_host() {
  local j r rh rm rb
  [ "$HOST_READY" = true ] || return 1
  j="$(kubectl --context "$CTX" get node "$NODE" -o json)" || { invalidate_deleted; return 1; }
  [ "$(jq -r '.metadata.uid // empty' <<<"$j")" = "$NODE_UID" ] && [ "$(jq -r '.status.nodeInfo.machineID // empty' <<<"$j")" = "$API_MACHINE_ID" ] && [ "$(jq -r '.status.nodeInfo.bootID // empty' <<<"$j")" = "$API_BOOT_ID" ] || { invalidate_deleted; return 1; }
  r="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s <<'REMOTE'
set -eu
printf '%s\t%s\t%s\n' "$(hostname -s)" "$(cat /etc/machine-id)" "$(cat /proc/sys/kernel/random/boot_id)"
REMOTE
)" || { invalidate_deleted; return 1; }
  IFS=$'\t' read -r rh rm rb <<<"$r"
  [ "$rh" = "$REMOTE_HOSTNAME" ] && [ "$rm" = "$REMOTE_MACHINE_ID" ] && [ "$rb" = "$REMOTE_BOOT_ID" ] || { invalidate_deleted; return 1; }
}
refresh_deleted_host
```

```bash
refresh_deleted_mount() {
  local out rc
  MOUNT_READY=false; EVIDENCE_READY=false; PROCESS_READY=false; NO_MATCH_READY=false
  unset MOUNT_ID MOUNT_SOURCE MOUNT_FSTYPE MOUNT_MAJOR_MINOR MOUNT_OPTIONS MOUNT_ST_DEV EVIDENCE_EPOCH EVIDENCE_NODE_UID EVIDENCE_MOUNT_ID PID_START FD_INODE FD_SIZE CGROUP_NS NET_NS MNT_NS
  guard_deleted_host || { printf '%s\n' '중단: current host guard 실패' >&2; return 1; }
  out="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s -- "$MOUNT_TARGET" <<'REMOTE'
set -eu
command -v findmnt >/dev/null 2>&1 || exit 127
line=$(findmnt -rn -M "$1" -o ID,SOURCE,FSTYPE,MAJ:MIN,OPTIONS) || exit $?
[ -n "$line" ] || exit 3
printf '%s\t%s\n' "$line" "$(stat -c %d "$1")"
REMOTE
)"; rc=$?
  [ "$rc" -eq 0 ] || { printf 'exact mount producer 실패/부재 rc=%s\n' "$rc" >&2; return 1; }
  read -r MOUNT_ID MOUNT_SOURCE MOUNT_FSTYPE MOUNT_MAJOR_MINOR MOUNT_OPTIONS MOUNT_ST_DEV <<<"${out//$'\t'/ }"
  [[ "$MOUNT_ID" =~ ^[0-9]+$ ]] && [[ "$MOUNT_ST_DEV" =~ ^[0-9]+$ ]] || return 1
  MOUNT_READY=true
  printf 'MOUNT_READY=true id=%s source=%s fstype=%s major-minor=%s options=%s st_dev=%s\n' "$MOUNT_ID" "$MOUNT_SOURCE" "$MOUNT_FSTYPE" "$MOUNT_MAJOR_MINOR" "$MOUNT_OPTIONS" "$MOUNT_ST_DEV"
}
guard_deleted_mount() {
  local out id src fs mm opt dev
  guard_deleted_host || return 1
  [ "$MOUNT_READY" = true ] || return 1
  out="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s -- "$MOUNT_TARGET" <<'REMOTE'
set -eu
line=$(findmnt -rn -M "$1" -o ID,SOURCE,FSTYPE,MAJ:MIN,OPTIONS) || exit $?
printf '%s\t%s\n' "$line" "$(stat -c %d "$1")"
REMOTE
)" || { MOUNT_READY=false; EVIDENCE_READY=false; PROCESS_READY=false; return 1; }
  read -r id src fs mm opt dev <<<"${out//$'\t'/ }"
  [ "$id" = "$MOUNT_ID" ] && [ "$src" = "$MOUNT_SOURCE" ] && [ "$fs" = "$MOUNT_FSTYPE" ] && [ "$mm" = "$MOUNT_MAJOR_MINOR" ] && [ "$opt" = "$MOUNT_OPTIONS" ] && [ "$dev" = "$MOUNT_ST_DEV" ] || { MOUNT_READY=false; EVIDENCE_READY=false; PROCESS_READY=false; return 1; }
}
refresh_deleted_mount
```

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | `df` 사용량은 높지만 pathname 기준 합계가 작고, 같은 mount device에서 link count 0인 open FD가 확인됨 |
| 적용 | host boot·mount ID·`st_dev`·PID start·FD inode와 namespace를 현재값으로 고정 가능 |
| 제외 | inode 고갈은 [[02. filesystem inode가 고갈된 경우]], read-only·I/O 오류는 [[03. filesystem이 read-only로 remount된 경우]] 또는 [[04. block device I-O error가 반복되는 경우]] |

검색 문자열: `(deleted)`, `link count 0`, `open file`, `df`, `du`, `st_dev`, `inode`.

## 관측별 진단 경로

| 지금 보이는 것 | 먼저 볼 것 | 그다음 어디를 볼지 | 상세 절 |
|---|---|---|---|
| 삭제-open 후보가 하나 이상 | PID·FD·inode·size·start tick | process owner와 안전한 reopen/restart 판단 | [[#삭제-open 후보 수집\|후보 수집]] |
| 후보가 없지만 사용량 차이가 큼 | producer 성공·mount `st_dev`·namespace | snapshot·reserved block·backend accounting | [[#원인 분기와 안전 경계\|원인 분기]] |
| process namespace가 바뀜 | PID 재사용 또는 replacement process | 새 PID·FD 재등록 | [[#process·FD identity 등록\|process identity]] |
| FD 해소 후 용량이 그대로 | filesystem·backend accounting과 실제 기능 | filesystem/storage owner | [[#정상화·증거\|정상화]] |

## 삭제-open 후보 수집

후보 출력은 path와 파일 내용을 제외하고 PID·FD·inode·size·process start tick·command name만 보여 준다. scan의 exit `3`은 일치 없음이고, 그 밖의 nonzero는 producer 실패다.

```bash
scan_deleted_open() {
  local out rc
  EVIDENCE_READY=false; PROCESS_READY=false; NO_MATCH_READY=false
  unset EVIDENCE_EPOCH EVIDENCE_NODE_UID EVIDENCE_MOUNT_ID PID_START FD_INODE FD_SIZE CGROUP_NS NET_NS MNT_NS
  guard_deleted_mount || { printf '%s\n' '중단: current mount guard 실패' >&2; return 1; }
  out="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" sudo -n bash -s -- "$MOUNT_ST_DEV" <<'REMOTE'
set -u
expected_dev=$1
count=0
for proc in /proc/[0-9]*; do
  pid=${proc##*/}
  [ -r "$proc/stat" ] || continue
  statline=$(cat "$proc/stat") || continue
  rest=${statline##*) }
  set -- $rest
  start=${20}
  comm=$(cat "$proc/comm" 2>/dev/null) || comm='-'
  for fdpath in "$proc"/fd/[0-9]*; do
    [ -L "$fdpath" ] || continue
    meta=$(stat -Lc '%d %i %h %s' "$fdpath" 2>/dev/null) || continue
    read -r dev inode links size <<<"$meta"
    if [ "$dev" = "$expected_dev" ] && [ "$links" = 0 ]; then
      fd=${fdpath##*/}
      printf 'pid=%s fd=%s inode=%s size=%s start=%s comm=%s\n' "$pid" "$fd" "$inode" "$size" "$start" "$comm"
      count=$((count+1))
      [ "$count" -le 200 ] || exit 75
    fi
  done
done
[ "$count" -gt 0 ] || exit 3
REMOTE
)"; rc=$?
  case "$rc" in
    0) EVIDENCE_EPOCH="$(date -u +%s)"; EVIDENCE_NODE_UID="$NODE_UID"; EVIDENCE_MOUNT_ID="$MOUNT_ID"; EVIDENCE_READY=true; printf '%s\n' "$out"; printf 'EVIDENCE_READY=true observed=%s\n' "$(date -u '+%Y-%m-%dT%H:%M:%SZ')" ;;
    3) NO_MATCH_READY=true; printf '%s\n' 'producer 성공, 삭제-open 후보 없음; 정상화 단독 근거로 사용하지 않음' ;;
    75) printf '%s\n' '중단: 후보가 200개를 초과해 owner가 범위를 좁혀야 함' >&2 ;;
    *) printf '삭제-open producer 실패 rc=%s\n' "$rc" >&2 ;;
  esac
}
scan_deleted_open
```

## process·FD identity 등록

```bash
PID='PASTE_DELETED_PID_HERE'
FD='PASTE_DELETED_FD_HERE'
ARGV_RE='PASTE_DELETED_ARGV_REGEX_HERE'
```

```bash
register_deleted_process() {
  local now out rc
  PROCESS_READY=false
  unset PID_START FD_INODE FD_SIZE CGROUP_NS NET_NS MNT_NS
  guard_deleted_mount || return 1
  [ "$EVIDENCE_READY" = true ] || { printf '%s\n' '중단: EVIDENCE_READY가 아니다.' >&2; return 1; }
  now="$(date -u +%s)" || return 1
  [ "$EVIDENCE_NODE_UID" = "$NODE_UID" ] && [ "$EVIDENCE_MOUNT_ID" = "$MOUNT_ID" ] && [ $((now-EVIDENCE_EPOCH)) -ge 0 ] && [ $((now-EVIDENCE_EPOCH)) -le 900 ] || { EVIDENCE_READY=false; return 1; }
  [[ "$PID" =~ ^[1-9][0-9]*$ ]] && [[ "$FD" =~ ^[0-9]+$ ]] || { printf '%s\n' '중단: PID·FD 형식 오류' >&2; return 1; }
  grep -E -- "$ARGV_RE" </dev/null >/dev/null 2>&1; rc=$?; [ "$rc" -eq 1 ] || { printf '중단: argv regex 오류 rc=%s\n' "$rc" >&2; return 1; }
  out="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" sudo -n bash -s -- "$PID" "$FD" "$MOUNT_ST_DEV" "$ARGV_RE" <<'REMOTE'
set -eu
p=$1; fd=$2; expected_dev=$3; re=$4
[ -r "/proc/$p/stat" ] && [ -L "/proc/$p/fd/$fd" ] || exit 3
meta=$(stat -Lc '%d %i %h %s' "/proc/$p/fd/$fd")
read -r dev inode links size <<<"$meta"
[ "$dev" = "$expected_dev" ] && [ "$links" = 0 ] || exit 4
tr '\0' '\n' < "/proc/$p/cmdline" | grep -Eq -- "$re" || exit 5
s=$(cat "/proc/$p/stat"); r=${s##*) }; set -- $r
printf '%s %s %s %s %s %s %s\n' "${20}" "$inode" "$size" "$(readlink "/proc/$p/ns/cgroup")" "$(readlink "/proc/$p/ns/net")" "$(readlink "/proc/$p/ns/mnt")" "$links"
REMOTE
)"; rc=$?
  [ "$rc" -eq 0 ] || { printf 'process·FD identity producer 실패 rc=%s\n' "$rc" >&2; return 1; }
  read -r PID_START FD_INODE FD_SIZE CGROUP_NS NET_NS MNT_NS links <<<"$out"
  [ "$links" = 0 ] || return 1
  PROCESS_READY=true
  printf 'PROCESS_READY=true pid=%s fd=%s start=%s inode=%s size=%s cgroup-ns=%s netns=%s mntns=%s\n' "$PID" "$FD" "$PID_START" "$FD_INODE" "$FD_SIZE" "$CGROUP_NS" "$NET_NS" "$MNT_NS"
}
register_deleted_process
```

`PROCESS_READY=true`는 현재 process가 같은 FD·inode를 열고 있다는 증거일 뿐, process 종료나 restart가 안전하다는 승인이 아니다. PID가 사라지거나 start tick·namespace·inode가 바뀌면 과거 후보를 replacement process에 연결하지 않는다.

## 원인 분기와 안전 경계

- 같은 mount `st_dev`에서 link count 0인 FD size 합계가 `df`와 pathname 집계 차이를 설명하면 삭제-open file이 직접 원인이다.
- 후보가 없으면 snapshot, filesystem reserved space, reflink, sparse file, project quota, backend accounting 또는 다른 mount namespace를 각각 미확정으로 남긴다.
- `du`와 `df`의 단위·시각·mount namespace가 다르면 차이를 직접 비교하지 않는다.
- 이 문서는 FD 강제 close, `/proc/PID/fd` truncate, process kill·restart, 파일 삭제, unmount·remount 명령을 제공하지 않는다. database·log writer의 reopen/restart는 owner 승인, quorum·availability, 데이터 내구성, rollback과 실제 기능 검증이 있는 별도 변경으로 수행한다.

## 정상화·증거

승인된 외부 조치 뒤 `refresh_deleted_host → refresh_deleted_mount → scan_deleted_open`을 다시 실행한다. process가 교체됐으면 새 PID·start tick·FD·inode·namespace를 다시 등록한다. producer 성공과 두 번 이상의 bounded sample에서 후보가 재발하지 않고, 같은 mount의 free bytes가 기대 범위로 회수되며, filesystem 오류가 없고 application read/write·로그 기록·실제 요청이 성공해야 정상화한다.

보존 항목은 Node UID·machineID·boot ID, remote identity, mount ID·source·fstype·major:minor·options·`st_dev`, PID·FD·start tick·inode·size·namespace, 관측 시각, producer exit, 승인된 외부 변경과 실제 기능 결과다. 삭제 경로, argv 원문, 파일 내용, credential, token과 개인정보는 저장하지 않는다.
