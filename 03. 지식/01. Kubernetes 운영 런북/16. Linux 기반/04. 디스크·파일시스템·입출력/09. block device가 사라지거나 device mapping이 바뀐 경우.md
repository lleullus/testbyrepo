# block device가 사라지거나 device mapping이 바뀐 경우

## 입력값과 빠른 수집

실행 hop은 `승인된 작업 PC/Bash → Node API identity → 승인된 SSH host → kernel journal → udev stable property → sysfs block graph → filesystem·mount → consumer`다. `/dev/sdX`와 major:minor는 재등록 뒤 바뀔 수 있으므로 stable property와 sysfs 관계 없이 이름만 재사용하지 않는다.

| 구분 | 값 | 출처와 gate |
|---|---|---|
| 최초 입력 | context·Node·SSH host, actual journal literal, strict UTC 창, 기대 stable key/value, mount target | incident와 inventory |
| API 자동값 | Node UID·machineID·boot ID | API와 remote exact match |
| 계산 후보값 | KNAME·major:minor·sysfs·size·mount ID | 현재 device에서 계산 |
| 운영자 선택값 | current `DEVICE_PATH` | bounded graph에서 stable property가 일치한 후보 |

### 1. 입력·시간·host identity

```bash
TOOLS_READY=false; INPUT_READY=false; HOST_READY=false; TIME_READY=false; EVIDENCE_READY=false; DEVICE_READY=false; MOUNT_READY=false
unset NODE_UID API_MACHINE_ID API_BOOT_ID REMOTE_HOSTNAME REMOTE_MACHINE_ID REMOTE_BOOT_ID START_EPOCH END_EPOCH KNAME MAJOR_MINOR DEVICE_TYPE DEVICE_SYSFS DEVICE_SIZE DEVICE_STABLE_VALUE MOUNT_ID MOUNT_SOURCE MOUNT_FSTYPE MOUNT_MAJOR_MINOR MOUNT_OPTIONS
CTX='PASTE_MAPPING_CONTEXT_HERE'
NODE='PASTE_MAPPING_NODE_HERE'
NODE_HOST='PASTE_MAPPING_NODE_HOST_HERE'
OBSERVED_LITERAL='PASTE_MAPPING_KERNEL_LITERAL_HERE'
START_RFC3339='PASTE_MAPPING_START_RFC3339_HERE'
END_RFC3339='PASTE_MAPPING_END_RFC3339_HERE'
EXPECTED_ID_KEY='PASTE_MAPPING_STABLE_KEY_HERE'
EXPECTED_ID_VALUE='PASTE_MAPPING_STABLE_VALUE_HERE'
MOUNT_TARGET='PASTE_MAPPING_MOUNT_TARGET_HERE'
MISSING=''
for tool in kubectl jq ssh date grep sed; do command -v "$tool" >/dev/null 2>&1 || MISSING="$MISSING $tool"; done
if [ -z "${BASH_VERSION:-}" ] || [ -n "$MISSING" ]; then
  printf '중단: Bash 또는 도구 미충족:%s\n' "$MISSING" >&2
elif [[ ! "$NODE_HOST" =~ ^[A-Za-z0-9._:-]+$ ]] || [[ ! "$MOUNT_TARGET" =~ ^/[A-Za-z0-9._/-]+$ ]] || [[ ! "$EXPECTED_ID_KEY" =~ ^(ID_WWN|ID_SERIAL|DM_UUID|ID_FS_UUID)$ ]] || [ -z "$EXPECTED_ID_VALUE" ] || [ -z "$OBSERVED_LITERAL" ]; then
  printf '%s\n' '중단: 대상·literal·stable identity 형식 오류' >&2
elif [[ "$START_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && [[ "$END_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && [ "$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" = "$START_RFC3339" ] && [ "$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" = "$END_RFC3339" ]; then
  START_EPOCH="$(date -u -d "$START_RFC3339" +%s)"; END_EPOCH="$(date -u -d "$END_RFC3339" +%s)"
  if [ "$START_EPOCH" -lt "$END_EPOCH" ] && kubectl --context "$CTX" get node "$NODE" -o name >/dev/null 2>&1; then TOOLS_READY=true; INPUT_READY=true; TIME_READY=true; fi
else
  printf '%s\n' '중단: strict UTC round-trip 실패' >&2
fi
printf 'INPUT_READY=%s TIME_READY=%s\n' "$INPUT_READY" "$TIME_READY"
```

```bash
invalidate_mapping() {
  HOST_READY=false; TIME_READY=false; EVIDENCE_READY=false; DEVICE_READY=false; MOUNT_READY=false
  unset NODE_UID API_MACHINE_ID API_BOOT_ID REMOTE_HOSTNAME REMOTE_MACHINE_ID REMOTE_BOOT_ID START_EPOCH END_EPOCH KNAME MAJOR_MINOR DEVICE_TYPE DEVICE_SYSFS DEVICE_SIZE DEVICE_STABLE_VALUE MOUNT_ID MOUNT_SOURCE MOUNT_FSTYPE MOUNT_MAJOR_MINOR MOUNT_OPTIONS
}
refresh_mapping_host() {
  local j r rc rh rm rb
  invalidate_mapping
  [ "$INPUT_READY" = true ] || return 1
  [[ "$START_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && [[ "$END_RFC3339" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] || return 1
  [ "$(date -u -d "$START_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" = "$START_RFC3339" ] && [ "$(date -u -d "$END_RFC3339" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" = "$END_RFC3339" ] || return 1
  START_EPOCH="$(date -u -d "$START_RFC3339" +%s)"; END_EPOCH="$(date -u -d "$END_RFC3339" +%s)"; [ "$START_EPOCH" -lt "$END_EPOCH" ] || return 1
  TIME_READY=true
  j="$(kubectl --context "$CTX" get node "$NODE" -o json)" || return 1
  NODE_UID="$(jq -er '.metadata.uid' <<<"$j")" || return 1
  API_MACHINE_ID="$(jq -er '.status.nodeInfo.machineID' <<<"$j")" || return 1
  API_BOOT_ID="$(jq -er '.status.nodeInfo.bootID' <<<"$j")" || return 1
  jq -e --arg h "$NODE_HOST" --arg n "$NODE" '(.metadata.name==$n) and (($h==$n) or any(.status.addresses[]?; .address==$h))' <<<"$j" >/dev/null || return 1
  r="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s <<'REMOTE'
set -eu
printf '%s\t%s\t%s\n' "$(hostname -s)" "$(cat /etc/machine-id)" "$(cat /proc/sys/kernel/random/boot_id)"
REMOTE
)"; rc=$?
  [ "$rc" -eq 0 ] || { invalidate_mapping; return 1; }
  IFS=$'\t' read -r rh rm rb <<<"$r"
  jq -e --arg h "$rh" --arg n "$NODE" '($h==$n) or any(.status.addresses[]?; .address==$h)' <<<"$j" >/dev/null && [ "$rm" = "$API_MACHINE_ID" ] && [ "$rb" = "$API_BOOT_ID" ] || { invalidate_mapping; return 1; }
  REMOTE_HOSTNAME="$rh"; REMOTE_MACHINE_ID="$rm"; REMOTE_BOOT_ID="$rb"; HOST_READY=true
  printf 'HOST_READY=true node=%s uid=%s machine-id=%s boot-id=%s remote=%s\n' "$NODE" "$NODE_UID" "$API_MACHINE_ID" "$API_BOOT_ID" "$REMOTE_HOSTNAME"
}
guard_mapping_host() {
  local j r rh rm rb
  [ "$HOST_READY" = true ] || return 1
  j="$(kubectl --context "$CTX" get node "$NODE" -o json)" || { invalidate_mapping; return 1; }
  [ "$(jq -r '.metadata.uid // empty' <<<"$j")" = "$NODE_UID" ] && [ "$(jq -r '.status.nodeInfo.machineID // empty' <<<"$j")" = "$API_MACHINE_ID" ] && [ "$(jq -r '.status.nodeInfo.bootID // empty' <<<"$j")" = "$API_BOOT_ID" ] || { invalidate_mapping; return 1; }
  r="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s <<'REMOTE'
set -eu
printf '%s\t%s\t%s\n' "$(hostname -s)" "$(cat /etc/machine-id)" "$(cat /proc/sys/kernel/random/boot_id)"
REMOTE
)" || { invalidate_mapping; return 1; }
  IFS=$'\t' read -r rh rm rb <<<"$r"
  [ "$rh" = "$REMOTE_HOSTNAME" ] && [ "$rm" = "$REMOTE_MACHINE_ID" ] && [ "$rb" = "$REMOTE_BOOT_ID" ] || { invalidate_mapping; return 1; }
}
refresh_mapping_host
```

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | 기존 path가 사라졌거나 KNAME·major:minor가 바뀌었고 stable identity 관계를 다시 확인해야 함 |
| 적용 | same host boot의 journal과 current stable property·sysfs·mount를 분리해 기록 가능 |
| 제외 | 단순 I/O error는 [[04. block device I-O error가 반복되는 경우]], filesystem corruption은 [[07. filesystem corruption 오류가 관측된 경우]], CSI attach lifecycle은 storage 런북 |

검색 문자열: `device offline`, `removed`, `renamed`, `change`, `rescan`, `multipath`, `no such device`, `I/O error`, `UUID`.

## bounded kernel evidence

```bash
collect_mapping_journal() {
  local raw rc matched frc
  EVIDENCE_READY=false
  [ "$TIME_READY" = true ] && guard_mapping_host || { printf '%s\n' '중단: current host/time guard 실패' >&2; return 1; }
  raw="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s -- "$REMOTE_MACHINE_ID" "$REMOTE_BOOT_ID" "$START_RFC3339" "$END_RFC3339" <<'REMOTE'
set -eu
[ "$(cat /etc/machine-id)" = "$1" ] || exit 42
[ "$(cat /proc/sys/kernel/random/boot_id)" = "$2" ] || exit 43
command -v journalctl >/dev/null 2>&1 || exit 127
sudo -n journalctl -k --utc --since "$3" --until "$4" -o short-iso-precise --no-pager --lines=1000
REMOTE
)"; rc=$?
  [ "$rc" -eq 0 ] || { printf 'journal producer 실패 rc=%s\n' "$rc" >&2; return 1; }
  matched="$(grep -F -C 4 -- "$OBSERVED_LITERAL" <<<"$raw")"; frc=$?
  case "$frc" in
    0) EVIDENCE_READY=true; printf '%s\n' "$matched" | sed -n '1,80p'; printf 'EVIDENCE_READY=true node-uid=%s boot-id=%s\n' "$NODE_UID" "$REMOTE_BOOT_ID" ;;
    1) printf '%s\n' 'literal filter 일치 없음; 오류 부재로 해석하지 않음' >&2 ;;
    *) printf 'filter 오류 rc=%s\n' "$frc" >&2; return "$frc" ;;
  esac
}
collect_mapping_journal
```

## current replacement device 등록

먼저 current graph를 bounded 출력으로 확인한다. serial·WWN은 운영 증거로 필요한 범위에서만 보존한다.

```bash
if guard_mapping_host; then ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s <<'REMOTE'
set -eu
command -v lsblk >/dev/null 2>&1 || exit 127
lsblk -p -o NAME,KNAME,TYPE,PKNAME,MAJ:MIN,FSTYPE,UUID,RO,SIZE,MOUNTPOINTS
REMOTE
else printf '%s\n' '중단: current host guard 실패' >&2; fi
```

```bash
register_mapping_device() {
  local out rc stable
  DEVICE_READY=false; MOUNT_READY=false
  unset KNAME MAJOR_MINOR DEVICE_TYPE DEVICE_SYSFS DEVICE_SIZE DEVICE_STABLE_VALUE MOUNT_ID MOUNT_SOURCE MOUNT_FSTYPE MOUNT_MAJOR_MINOR MOUNT_OPTIONS
  [ "$EVIDENCE_READY" = true ] || { printf '%s\n' '중단: EVIDENCE_READY가 아니다.' >&2; return 1; }
  guard_mapping_host || return 1
  [[ "$DEVICE_PATH" =~ ^/dev/[A-Za-z0-9._/+:-]+$ ]] || return 1
  out="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s -- "$DEVICE_PATH" "$EXPECTED_ID_KEY" <<'REMOTE'
set -eu
d=$1; key=$2
[ -b "$d" ] || exit 3
command -v lsblk >/dev/null 2>&1 || exit 127
command -v udevadm >/dev/null 2>&1 || exit 127
command -v awk >/dev/null 2>&1 || exit 127
k=$(lsblk -dn -o KNAME "$d")
value=$(udevadm info --query=property --name "$d" | awk -F= -v key="$key" '$1==key{n++; v=substr($0,index($0,"=")+1)} END{if(n==1 && length(v)>0) print v; else exit 4}')
printf '%s\t%s\t%s\t%s\t%s\t%s\n' "$k" "$(lsblk -dn -o MAJ:MIN "$d")" "$(lsblk -dn -o TYPE "$d")" "$(readlink -f "/sys/class/block/$k")" "$(cat "/sys/class/block/$k/size")" "$value"
REMOTE
)"; rc=$?
  [ "$rc" -eq 0 ] || { printf 'device producer 실패 rc=%s\n' "$rc" >&2; return 1; }
  IFS=$'\t' read -r KNAME MAJOR_MINOR DEVICE_TYPE DEVICE_SYSFS DEVICE_SIZE stable <<<"$out"
  [ "$stable" = "$EXPECTED_ID_VALUE" ] || { printf '%s\n' '중단: stable identity 불일치' >&2; return 1; }
  DEVICE_STABLE_VALUE="$stable"; DEVICE_READY=true
  printf 'DEVICE_READY=true path=%s kname=%s major-minor=%s type=%s sysfs=%s sectors=%s stable-key=%s\n' "$DEVICE_PATH" "$KNAME" "$MAJOR_MINOR" "$DEVICE_TYPE" "$DEVICE_SYSFS" "$DEVICE_SIZE" "$EXPECTED_ID_KEY"
}
guard_mapping_device() {
  local out k mm ty sy sz stable
  guard_mapping_host || return 1
  [ "$DEVICE_READY" = true ] || return 1
  out="$(ssh -o BatchMode=yes -o ConnectTimeout=10 -- "$NODE_HOST" bash -s -- "$DEVICE_PATH" "$EXPECTED_ID_KEY" <<'REMOTE'
set -eu
d=$1; key=$2; [ -b "$d" ] || exit 3
command -v awk >/dev/null 2>&1 || exit 127
k=$(lsblk -dn -o KNAME "$d")
stable=$(udevadm info --query=property --name "$d" | awk -F= -v key="$key" '$1==key{n++; v=substr($0,index($0,"=")+1)} END{if(n==1 && length(v)>0) print v; else exit 4}')
printf '%s\t%s\t%s\t%s\t%s\t%s\n' "$k" "$(lsblk -dn -o MAJ:MIN "$d")" "$(lsblk -dn -o TYPE "$d")" "$(readlink -f "/sys/class/block/$k")" "$(cat "/sys/class/block/$k/size")" "$stable"
REMOTE
)" || { DEVICE_READY=false; MOUNT_READY=false; return 1; }
  IFS=$'\t' read -r k mm ty sy sz stable <<<"$out"
  [ "$k" = "$KNAME" ] && [ "$mm" = "$MAJOR_MINOR" ] && [ "$ty" = "$DEVICE_TYPE" ] && [ "$sy" = "$DEVICE_SYSFS" ] && [ "$sz" = "$DEVICE_SIZE" ] && [ "$stable" = "$DEVICE_STABLE_VALUE" ] || { DEVICE_READY=false; MOUNT_READY=false; return 1; }
}
```

## 안전 경계·정상화

이 문서는 rescan, controller reset, multipath reload, device delete, detach·attach, force detach, mount·unmount, filesystem repair, kubelet·runtime·CSI restart와 reboot를 실행하지 않는다. stable identity, provider volume, filesystem state, workload availability·quorum, 승인·중단·rollback이 확정된 별도 변경으로 이관한다.

외부 조치 뒤 host identity부터 모두 재등록한다. 같은 stable key/value가 새 path의 current udev property와 exact match하고, sysfs graph·size가 승인 inventory와 맞으며, exact mount source·fstype·options가 수렴하고, 새 kernel 오류가 재발하지 않으며, consumer가 replacement device를 발견해 실제 read/write와 application 요청을 성공해야 정상화한다. Kubernetes consumer면 새 Pod UID·node·containerID와 PVC→PV→CSI driver→VolumeAttachment/node 관계를 storage 런북에서 별도로 재등록한다.

Node UID·machineID·boot ID, remote identity, strict UTC 창, literal의 비민감 결정적 행, producer/filter rc, old/new path·KNAME·major:minor·sysfs·size·stable key, mount 관계, provider 작업 ID와 실제 기능 결과를 보존한다. 전체 udev property·journal, token, credential, private key와 개인정보는 저장하지 않는다.
