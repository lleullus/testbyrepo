# ontap-nas-economy PVC 확장이 qtree quota·ONTAP API 단계에서 실패하는 경우

## 입력값과 빠른 수집

실행 위치는 승인된 `kubectl` context를 가진 작업 PC 또는 bastion의 같은 Bash다. 경로는 `작업 host → API server → PVC/PV/StorageClass → external-resizer Pod/container → Trident → 승인된 ONTAP 관리면 → SVM/qtree quota rule`이다.

| 구분 | 값 | 확정 기준 |
|---|---|---|
| 최초 입력 | `NS`, `PVC`, `CSI_NS` | 승인된 namespace와 exact name |
| API·시스템 자동값 | PVC/PV/StorageClass UID·RV, CSI driver/handle, resizer Pod·owner·containerID | current API exact relation |
| 계산 후보 | `EVIDENCE_RE` | PVC/PV/handle을 ERE literal로 escape |
| 운영자 선택값 | resizer Pod/container, UTC 증거 창 | 현재 Pod 목록과 최초 resize 실패를 보고 선택 |

다음 스크립트는 입력, 자동 수집, Event·로그와 query 생성을 한 shell에 연결한다. `refresh`는 실행할 때마다 identity, selection, authorization, evidence, mutation, rollback, normalization READY를 먼저 폐기한다.

```bash
NS='PASTE_NAMESPACE_HERE'
PVC='PASTE_PVC_HERE'
CSI_NS='PASTE_TRIDENT_NAMESPACE_HERE'
RESIZER_POD='PASTE_RESIZER_POD_HERE'
RESIZER_CONTAINER='PASTE_RESIZER_CONTAINER_HERE'
START_RFC3339='PASTE_START_RFC3339_HERE'
END_RFC3339='PASTE_END_RFC3339_HERE'

invalidate(){ unset PVC_JSON PV_JSON SC_JSON POD_JSON PVC_UID PVC_RV PV PV_UID PV_RV SC SC_UID SC_RV DRIVER HANDLE POD_UID POD_RV OWNER_KIND OWNER_NAME OWNER_UID CONTROLLER_KIND CONTROLLER_NAME CONTROLLER_UID CONTAINER_ID IMAGE_ID EVIDENCE_RE OBSERVED_AT; ID_READY=false; TIME_READY=false; SELECTION_READY=false; AUTH_READY=false; EVIDENCE_READY=false; MUTATION_READY=false; ROLLBACK_READY=false; NORMALIZATION_READY=false; }
fail(){ local message="$1"; invalidate; printf '중단: %s\n' "$message" >&2; return 1; }
canonical(){ local value="$1" round; [[ "$value" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] || return 1; round="$(date -u -d "$value" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" || return 1; [ "$round" = "$value" ]; }
time_gate(){ TIME_READY=false; canonical "$START_RFC3339" || { fail 'START_RFC3339가 canonical UTC가 아니다.'; return 1; }; canonical "$END_RFC3339" || { fail 'END_RFC3339가 canonical UTC가 아니다.'; return 1; }; [ "$(date -u -d "$START_RFC3339" +%s)" -lt "$(date -u -d "$END_RFC3339" +%s)" ] || { fail 'start는 end보다 빨라야 한다.'; return 1; }; TIME_READY=true; }
ere(){ printf '%s' "$1" | sed 's/[][\\.^$*+?(){}|]/\\&/g'; }
register_owner(){ local pod="$1" ref parent object; OWNER_KIND="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].kind // empty' <<<"$pod")"; OWNER_NAME="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].name // empty' <<<"$pod")"; OWNER_UID="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid // empty' <<<"$pod")"; [ -n "$OWNER_UID" ] || return 1; if [ "$OWNER_KIND" = ReplicaSet ]; then object="$(kubectl get replicaset "$OWNER_NAME" -n "$CSI_NS" -o json)" || return 1; [ "$(jq -r '.metadata.uid' <<<"$object")" = "$OWNER_UID" ] || return 1; CONTROLLER_KIND="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].kind // empty' <<<"$object")"; CONTROLLER_NAME="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].name // empty' <<<"$object")"; CONTROLLER_UID="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid // empty' <<<"$object")"; [ "$CONTROLLER_KIND" = Deployment ] || return 1; ref="$(kubectl get deployment "$CONTROLLER_NAME" -n "$CSI_NS" -o json)" || return 1; [ "$(jq -r '.metadata.uid' <<<"$ref")" = "$CONTROLLER_UID" ]; elif [ "$OWNER_KIND" = StatefulSet ] || [ "$OWNER_KIND" = DaemonSet ]; then CONTROLLER_KIND="$OWNER_KIND"; CONTROLLER_NAME="$OWNER_NAME"; CONTROLLER_UID="$OWNER_UID"; ref="$(kubectl get "${OWNER_KIND,,}" "$OWNER_NAME" -n "$CSI_NS" -o json)" || return 1; [ "$(jq -r '.metadata.uid' <<<"$ref")" = "$OWNER_UID" ]; else return 1; fi; }
refresh(){ invalidate; local pvc pv sc pod evidence; pvc="$(kubectl get pvc "$PVC" -n "$NS" -o json)" || { fail 'PVC 조회 실패'; return 1; }; PV="$(jq -r '.spec.volumeName // empty' <<<"$pvc")"; SC="$(jq -r '.spec.storageClassName // empty' <<<"$pvc")"; [ -n "$PV" ] && [ -n "$SC" ] || { fail 'PVC의 PV/StorageClass 관계가 없다.'; return 1; }; pv="$(kubectl get pv "$PV" -o json)" || { fail 'PV 조회 실패'; return 1; }; sc="$(kubectl get storageclass "$SC" -o json)" || { fail 'StorageClass 조회 실패'; return 1; }; PVC_UID="$(jq -r '.metadata.uid' <<<"$pvc")"; jq -e --arg ns "$NS" --arg name "$PVC" --arg uid "$PVC_UID" '.spec.claimRef.namespace==$ns and .spec.claimRef.name==$name and .spec.claimRef.uid==$uid and (.spec.csi.driver|length)>0 and (.spec.csi.volumeHandle|length)>0' <<<"$pv" >/dev/null || { fail 'PVC→PV claimRef/CSI 관계 불일치'; return 1; }; DRIVER="$(jq -r '.spec.csi.driver' <<<"$pv")"; HANDLE="$(jq -r '.spec.csi.volumeHandle' <<<"$pv")"; jq -e --arg driver "$DRIVER" '.provisioner==$driver and .allowVolumeExpansion==true' <<<"$sc" >/dev/null || { fail 'StorageClass expansion/provisioner gate 실패'; return 1; }; pod="$(kubectl get pod "$RESIZER_POD" -n "$CSI_NS" -o json)" || { fail 'resizer Pod 조회 실패'; return 1; }; jq -e --arg c "$RESIZER_CONTAINER" '([.spec.containers[]|select(.name==$c)]|length)==1 and ([.status.containerStatuses[]|select(.name==$c and (.containerID|length)>0)]|length)==1' <<<"$pod" >/dev/null || { fail 'regular container/current containerID 검증 실패'; return 1; }; register_owner "$pod" || { fail 'resizer current owner 검증 실패'; return 1; }; PVC_JSON="$pvc"; PV_JSON="$pv"; SC_JSON="$sc"; POD_JSON="$pod"; PVC_RV="$(jq -r '.metadata.resourceVersion' <<<"$pvc")"; PV_UID="$(jq -r '.metadata.uid' <<<"$pv")"; PV_RV="$(jq -r '.metadata.resourceVersion' <<<"$pv")"; SC_UID="$(jq -r '.metadata.uid' <<<"$sc")"; SC_RV="$(jq -r '.metadata.resourceVersion' <<<"$sc")"; POD_UID="$(jq -r '.metadata.uid' <<<"$pod")"; POD_RV="$(jq -r '.metadata.resourceVersion' <<<"$pod")"; CONTAINER_ID="$(jq -r --arg c "$RESIZER_CONTAINER" '.status.containerStatuses[]|select(.name==$c)|.containerID' <<<"$pod")"; IMAGE_ID="$(jq -r --arg c "$RESIZER_CONTAINER" '.status.containerStatuses[]|select(.name==$c)|.imageID' <<<"$pod")"; evidence="$(printf '%s\n%s\n%s' "$PVC" "$PV" "$HANDLE" | while IFS= read -r value; do ere "$value"; printf '\n'; done)" || { fail 'regex 생성 실패'; return 1; }; EVIDENCE_RE="$(paste -sd '|' <<<"$evidence")"; OBSERVED_AT="$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; ID_READY=true; SELECTION_READY=true; printf 'ID_READY=true PVC=%s uid=%s rv=%s PV=%s uid=%s driver=%s\nresizer=%s uid=%s owner=%s/%s:%s regular-container=%s containerID=%s observed=%s\n' "$PVC" "$PVC_UID" "$PVC_RV" "$PV" "$PV_UID" "$DRIVER" "$RESIZER_POD" "$POD_UID" "$CONTROLLER_KIND" "$CONTROLLER_NAME" "$CONTROLLER_UID" "$RESIZER_CONTAINER" "$CONTAINER_ID" "$OBSERVED_AT"; }
auth_gate(){ AUTH_READY=false; [ "$ID_READY" = true ] || return 1; local allowed; allowed="$(kubectl auth can-i get "pods/$RESIZER_POD" --subresource=log -n "$CSI_NS")" || { fail 'SelfSubjectAccessReview 실패'; return 1; }; [ "$allowed" = yes ] || { fail "현재 주체의 get pods/$RESIZER_POD/log 권한 없음 namespace=$CSI_NS"; return 1; }; AUTH_READY=true; }
window_log(){ local source="$1" output="$2" start end line stamp epoch; start="$(date -u -d "$START_RFC3339" +%s)" || return 2; end="$(date -u -d "$END_RFC3339" +%s)" || return 2; : >"$output" || return 2; while IFS= read -r line; do stamp="${line%% *}"; epoch="$(date -u -d "$stamp" +%s 2>/dev/null)" || return 2; if [ "$epoch" -ge "$start" ] && [ "$epoch" -le "$end" ]; then printf '%s\n' "$line" >>"$output" || return 2; fi; done <"$source"; }
snapshot(){ refresh && time_gate || return 1; jq -r '"=== PVC ===\nUID: \(.metadata.uid)\nRV: \(.metadata.resourceVersion)\nRequest: \(.spec.resources.requests.storage)\nCapacity: \(.status.capacity.storage // "-")\nConditions: \([.status.conditions[]?|.type+":"+.status]|join(","))"' <<<"$PVC_JSON"; jq -r '"=== PV/SC ===\nPV UID: \(.metadata.uid)\nPV RV: \(.metadata.resourceVersion)\nDriver: \(.spec.csi.driver)\nPV capacity: \(.spec.capacity.storage)"' <<<"$PV_JSON"; }
resize_evidence(){ refresh && time_gate && auth_gate || return 1; local before_uid="$POD_UID" before_cid="$CONTAINER_ID" raw window filtered rc after; raw="$(mktemp)" || { fail 'mktemp 실패'; return 1; }; window="$(mktemp)" || { rm -f "$raw"; fail 'mktemp 실패'; return 1; }; filtered="$(mktemp)" || { rm -f "$raw" "$window"; fail 'mktemp 실패'; return 1; }; if ! kubectl logs "$RESIZER_POD" -n "$CSI_NS" -c "$RESIZER_CONTAINER" --timestamps --since-time="$START_RFC3339" --tail=800 >"$raw"; then rm -f "$raw" "$window" "$filtered"; fail '로그 producer 실패'; return 1; fi; window_log "$raw" "$window"; rc=$?; if [ "$rc" -ne 0 ]; then rm -f "$raw" "$window" "$filtered"; fail "시간 filter 오류 rc=$rc"; return 1; fi; grep -E -i -C 3 "$EVIDENCE_RE|ControllerExpandVolume|qtree|quota|HTTP 400|resize|failed|error" "$window" >"$filtered"; rc=$?; if [ "$rc" -eq 0 ]; then cat "$filtered"; elif [ "$rc" -eq 1 ]; then printf '%s\n' '필터 일치 없음: producer 성공이며 오류 부재는 아니다.' >&2; else rm -f "$raw" "$window" "$filtered"; fail "grep 오류 rc=$rc"; return 1; fi; after="$(kubectl get pod "$RESIZER_POD" -n "$CSI_NS" -o json)" || { rm -f "$raw" "$window" "$filtered"; fail '로그 후 Pod 조회 실패'; return 1; }; [ "$(jq -r '.metadata.uid' <<<"$after")" = "$before_uid" ] && [ "$(jq -r --arg c "$RESIZER_CONTAINER" '.status.containerStatuses[]|select(.name==$c)|.containerID' <<<"$after")" = "$before_cid" ] || { rm -f "$raw" "$window" "$filtered"; fail '로그 수집 중 Pod/container lifecycle 변경'; return 1; }; rm -f "$raw" "$window" "$filtered"; EVIDENCE_READY=true; }
prom_queries(){ refresh || return 1; local ns pvc; ns="${NS//\\/\\\\}"; ns="${ns//\"/\\\"}"; pvc="${PVC//\\/\\\\}"; pvc="${pvc//\"/\\\"}"; printf 'kube_persistentvolumeclaim_resource_requests_storage_bytes{namespace="%s",persistentvolumeclaim="%s"}\n' "$ns" "$pvc"; printf 'kubelet_volume_stats_capacity_bytes{namespace="%s",persistentvolumeclaim="%s"}\n' "$ns" "$pvc"; }
tools_gate(){ local tool missing=''; for tool in kubectl jq sed paste date mktemp grep; do command -v "$tool" >/dev/null 2>&1 || missing="$missing $tool"; done; [ -z "$missing" ] || { printf '중단: 도구 없음=%s\n' "$missing" >&2; return 1; }; }
tools_gate && snapshot
```

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | 같은 PVC UID에서 request와 status/PV capacity가 다르고 `ControllerExpandVolume`, qtree/quota 또는 ONTAP HTTP 400이 관측됨 |
| 적용 | PVC→PV→driver→resizer current container lifecycle과 증거 창이 모두 연결됨 |
| 제외 | `allowVolumeExpansion=false`, driver 불일치, 미결합 PVC는 [[10. 스토리지/01. Kubernetes·CSI 공통/05. PVC 확장 후 PV·filesystem 용량이 수렴하지 않는 경우]] |

검색 문자열: `VolumeResizeFailed`, `ControllerExpandVolume`, `quota_rule_modify`, `Qtree quota update failed`, `HTTP 400`.

`resize_evidence`를 실행한다. filter no-match는 producer 성공과 구분되며 정상 판정이 아니다. Event는 같은 PVC UID와 canonical 창을 API/UI에서 제한해 `Event → resizer → Trident request → ONTAP request ID → PV/PVC RV` 순서를 확인한다.

## 판정과 변경 경계

| 관측 | 다음 확인 | 비보장 |
|---|---|---|
| resizer 호출 없음 | PVC condition와 sidecar leader/current owner | qtree 오류를 부정하지 않지만 provider 단계로 내려갈 근거가 없음 |
| qtree/quota 오류 | redacted handle→Trident volume→backend/SVM→qtree→quota exact mapping | 이름 유사성은 mapping 증거가 아님 |
| HTTP 400 | 비식별 request ID/status와 version-matched read-only rule 조회 | schema·권한·제품 결함 중 하나를 단독 확정하지 못함 |

Secret data, token, endpoint, Authorization header와 request body는 출력하지 않는다. Kubernetes `pods/log` 권한과 Trident/ONTAP authorization·audit을 분리한다.

이 문서는 상태 변경 명령을 제공하지 않는다. backend quota/config 변경은 fresh `snapshot`과 `resize_evidence`, 승인 ID, 이전 원본, 영향 consumer, idempotence, 중단 조건과 제품 owner rollback 절차가 있을 때만 승인된 관리면에서 수행한다. 변경 또는 rollback 직전에 다시 refresh하며 UID/RV/owner/containerID가 달라지면 중단한다.

정상화는 `ONTAP rule/API → Trident → resizer → PV/PVC capacity → workload filesystem → 승인된 read/write와 핵심 기능 → 재발 관찰` 순서다. 교체된 resizer는 새 이름을 자동 재사용하지 말고 현재 owner의 단일 후보를 확인해 `RESIZER_POD`를 갱신한 뒤 전체 등록을 다시 실행한다. `Bound`, HTTP 200 또는 capacity 한 번 증가만으로 닫지 않는다. `prom_queries` 출력은 완성 질의이며, 빈 vector는 metric 부재·label/cluster/time/API 오류 후보이지 성공이 아니다.
