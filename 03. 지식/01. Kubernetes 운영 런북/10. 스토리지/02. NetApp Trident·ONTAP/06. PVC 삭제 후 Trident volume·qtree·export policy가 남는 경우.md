# PVC 삭제 후 Trident volume·qtree·export policy가 남는 경우

## 입력값과 빠른 수집

연결 경로는 `승인된 작업 host → API server → 마지막 PVC UID → PV UID/claimRef/reclaimPolicy → CSI handle → Trident volume/backend → ONTAP qtree/quota → export policy/rule`이다. Kubernetes `NotFound`는 backend 삭제 성공을 보장하지 않는다.

| 구분 | 값 | 확정 기준 |
|---|---|---|
| 최초 입력 | namespace, 마지막 PVC 이름·UID, PV 이름 | 삭제 승인·감사 handoff |
| API 자동값 | 현재 PVC 존재 여부, PV/driver/handle/reclaim, provisioner lifecycle | current API와 마지막 UID exact match |
| 계산 후보 | evidence regex | PVC/PV/handle literal escape |
| 운영자 선택 | provisioner Pod/container, UTC 창 | 실제 inventory와 최초 delete 실패 |

```bash
NS='PASTE_NAMESPACE_HERE'
PVC='PASTE_LAST_PVC_NAME_HERE'
EXPECTED_PVC_UID='PASTE_LAST_PVC_UID_HERE'
PV='PASTE_PV_NAME_HERE'
CSI_NS='PASTE_TRIDENT_NAMESPACE_HERE'
PROVISIONER_POD='PASTE_PROVISIONER_POD_HERE'
PROVISIONER_CONTAINER='PASTE_PROVISIONER_CONTAINER_HERE'
START_RFC3339='PASTE_START_RFC3339_HERE'
END_RFC3339='PASTE_END_RFC3339_HERE'

invalidate(){ unset PVC_JSON PV_JSON POD_JSON PVC_STATE PVC_UID PVC_RV PV_UID PV_RV DRIVER HANDLE RECLAIM POD_UID POD_RV OWNER_KIND OWNER_NAME OWNER_UID CONTAINER_ID EVIDENCE_RE OBSERVED_AT; ID_READY=false; TIME_READY=false; SELECTION_READY=false; AUTH_READY=false; EVIDENCE_READY=false; REFERENCE_READY=false; MUTATION_READY=false; ROLLBACK_READY=false; NORMALIZATION_READY=false; }
fail(){ local message="$1"; invalidate; printf '중단: %s\n' "$message" >&2; return 1; }
canonical(){ local value="$1" round; [[ "$value" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] || return 1; round="$(date -u -d "$value" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" || return 1; [ "$round" = "$value" ]; }
time_gate(){ TIME_READY=false; canonical "$START_RFC3339" || { fail 'start 형식 오류'; return 1; }; canonical "$END_RFC3339" || { fail 'end 형식 오류'; return 1; }; [ "$(date -u -d "$START_RFC3339" +%s)" -lt "$(date -u -d "$END_RFC3339" +%s)" ] || { fail '시간 순서 오류'; return 1; }; TIME_READY=true; }
ere(){ printf '%s' "$1" | sed 's/[][\\.^$*+?(){}|]/\\&/g'; }
refresh(){ invalidate; local pvc='' pv pod state evidence; [[ "$EXPECTED_PVC_UID" =~ ^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$ ]] || { fail '마지막 PVC UID 형식 오류'; return 1; }; if pvc="$(kubectl get pvc "$PVC" -n "$NS" -o json 2>/dev/null)"; then state=present; [ "$(jq -r '.metadata.uid' <<<"$pvc")" = "$EXPECTED_PVC_UID" ] || { fail '같은 이름의 재생성 PVC'; return 1; }; [ "$(jq -r '.spec.volumeName // empty' <<<"$pvc")" = "$PV" ] || { fail '현재 PVC/PV handoff 불일치'; return 1; }; else state=absent; fi; pv="$(kubectl get pv "$PV" -o json 2>/dev/null)" || { fail 'PV가 없어 provider exact handoff를 복원할 수 없다.'; return 1; }; jq -e --arg ns "$NS" --arg name "$PVC" --arg uid "$EXPECTED_PVC_UID" '.spec.claimRef.namespace==$ns and .spec.claimRef.name==$name and .spec.claimRef.uid==$uid and (.spec.csi.driver|length)>0 and (.spec.csi.volumeHandle|length)>0' <<<"$pv" >/dev/null || { fail 'PV claimRef/CSI handoff 불일치'; return 1; }; pod="$(kubectl get pod "$PROVISIONER_POD" -n "$CSI_NS" -o json)" || { fail 'provisioner Pod 조회 실패'; return 1; }; jq -e --arg c "$PROVISIONER_CONTAINER" '([.spec.containers[]|select(.name==$c)]|length)==1 and ([.status.containerStatuses[]|select(.name==$c and (.containerID|length)>0)]|length)==1' <<<"$pod" >/dev/null || { fail 'regular container/current containerID 불일치'; return 1; }; OWNER_KIND="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].kind // empty' <<<"$pod")"; OWNER_NAME="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].name // empty' <<<"$pod")"; OWNER_UID="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid // empty' <<<"$pod")"; [ -n "$OWNER_UID" ] || { fail 'provisioner owner 없음'; return 1; }; if [ "$OWNER_KIND" = ReplicaSet ]; then local rs; rs="$(kubectl get replicaset "$OWNER_NAME" -n "$CSI_NS" -o json)" || { fail 'owner ReplicaSet 조회 실패'; return 1; }; [ "$(jq -r '.metadata.uid' <<<"$rs")" = "$OWNER_UID" ] || { fail 'owner UID 불일치'; return 1; }; elif [ "$OWNER_KIND" = StatefulSet ]; then [ "$(kubectl get statefulset "$OWNER_NAME" -n "$CSI_NS" -o jsonpath='{.metadata.uid}')" = "$OWNER_UID" ] || { fail 'owner UID 불일치'; return 1; }; else fail '지원하지 않는 provisioner owner kind'; return 1; fi; PVC_STATE="$state"; PVC_JSON="$pvc"; PV_JSON="$pv"; POD_JSON="$pod"; PVC_UID="$EXPECTED_PVC_UID"; PVC_RV="$(if [ "$state" = present ]; then jq -r '.metadata.resourceVersion' <<<"$pvc"; else printf '%s' '-'; fi)"; PV_UID="$(jq -r '.metadata.uid' <<<"$pv")"; PV_RV="$(jq -r '.metadata.resourceVersion' <<<"$pv")"; DRIVER="$(jq -r '.spec.csi.driver' <<<"$pv")"; HANDLE="$(jq -r '.spec.csi.volumeHandle' <<<"$pv")"; RECLAIM="$(jq -r '.spec.persistentVolumeReclaimPolicy' <<<"$pv")"; POD_UID="$(jq -r '.metadata.uid' <<<"$pod")"; POD_RV="$(jq -r '.metadata.resourceVersion' <<<"$pod")"; CONTAINER_ID="$(jq -r --arg c "$PROVISIONER_CONTAINER" '.status.containerStatuses[]|select(.name==$c)|.containerID' <<<"$pod")"; evidence="$(printf '%s\n%s\n%s' "$PVC" "$PV" "$HANDLE" | while IFS= read -r value; do ere "$value"; printf '\n'; done)" || { fail 'regex 생성 실패'; return 1; }; EVIDENCE_RE="$(paste -sd '|' <<<"$evidence")"; OBSERVED_AT="$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; ID_READY=true; SELECTION_READY=true; printf 'ID_READY=true pvc-state=%s last-pvc-uid=%s PV=%s uid=%s rv=%s reclaim=%s driver=%s\nprovisioner=%s uid=%s owner=%s/%s:%s regular-container=%s containerID=%s observed=%s\n' "$PVC_STATE" "$PVC_UID" "$PV" "$PV_UID" "$PV_RV" "$RECLAIM" "$DRIVER" "$PROVISIONER_POD" "$POD_UID" "$OWNER_KIND" "$OWNER_NAME" "$OWNER_UID" "$PROVISIONER_CONTAINER" "$CONTAINER_ID" "$OBSERVED_AT"; }
current_owner_gate(){ [ "$ID_READY" = true ] || return 1; local owner parent_name parent_uid parent; if [ "$OWNER_KIND" = ReplicaSet ]; then owner="$(kubectl get replicaset "$OWNER_NAME" -n "$CSI_NS" -o json)" || { fail 'current ReplicaSet 조회 실패'; return 1; }; [ "$(jq -r '.metadata.uid' <<<"$owner")" = "$OWNER_UID" ] || { fail 'current ReplicaSet UID 변경'; return 1; }; parent_name="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true and .kind=="Deployment")][0].name // empty' <<<"$owner")"; parent_uid="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true and .kind=="Deployment")][0].uid // empty' <<<"$owner")"; [ -n "$parent_uid" ] || { fail 'ReplicaSet의 Deployment owner 없음'; return 1; }; parent="$(kubectl get deployment "$parent_name" -n "$CSI_NS" -o json)" || { fail 'current Deployment 조회 실패'; return 1; }; [ "$(jq -r '.metadata.uid' <<<"$parent")" = "$parent_uid" ] || { fail 'current Deployment UID 변경'; return 1; }; elif [ "$OWNER_KIND" = StatefulSet ]; then [ "$(kubectl get statefulset "$OWNER_NAME" -n "$CSI_NS" -o jsonpath='{.metadata.uid}')" = "$OWNER_UID" ] || { fail 'current StatefulSet UID 변경'; return 1; }; else fail '지원하지 않는 provisioner current owner'; return 1; fi; }
auth_gate(){ AUTH_READY=false; [ "$ID_READY" = true ] || return 1; local allowed; allowed="$(kubectl auth can-i get "pods/$PROVISIONER_POD" --subresource=log -n "$CSI_NS")" || { fail 'authorization 조회 실패'; return 1; }; [ "$allowed" = yes ] || { fail "현재 주체 get pods/$PROVISIONER_POD/log 거부 namespace=$CSI_NS"; return 1; }; AUTH_READY=true; }
window_log(){ local source="$1" output="$2" start end line stamp epoch; start="$(date -u -d "$START_RFC3339" +%s)" || return 2; end="$(date -u -d "$END_RFC3339" +%s)" || return 2; : >"$output" || return 2; while IFS= read -r line; do stamp="${line%% *}"; epoch="$(date -u -d "$stamp" +%s 2>/dev/null)" || return 2; if [ "$epoch" -ge "$start" ] && [ "$epoch" -le "$end" ]; then printf '%s\n' "$line" >>"$output" || return 2; fi; done <"$source"; }
snapshot(){ refresh && current_owner_gate && time_gate || return 1; [ "$RECLAIM" != Retain ] || { fail 'Retain은 orphan cleanup 대상이 아니다.'; return 1; }; jq -r '"=== PV ===\nUID: \(.metadata.uid)\nRV: \(.metadata.resourceVersion)\nPhase: \(.status.phase)\nDeletion: \(.metadata.deletionTimestamp // "-")\nFinalizers: \(.metadata.finalizers // [])\nClaim UID: \(.spec.claimRef.uid)\nReclaim: \(.spec.persistentVolumeReclaimPolicy)\nDriver: \(.spec.csi.driver)"' <<<"$PV_JSON"; }
delete_evidence(){ snapshot >/dev/null && auth_gate || return 1; local before_uid="$POD_UID" before_cid="$CONTAINER_ID" raw window filtered rc after; raw="$(mktemp)" || return 1; window="$(mktemp)" || { rm -f "$raw"; return 1; }; filtered="$(mktemp)" || { rm -f "$raw" "$window"; return 1; }; if ! kubectl logs "$PROVISIONER_POD" -n "$CSI_NS" -c "$PROVISIONER_CONTAINER" --timestamps --since-time="$START_RFC3339" --tail=800 >"$raw"; then rm -f "$raw" "$window" "$filtered"; fail '로그 producer 실패'; return 1; fi; window_log "$raw" "$window"; rc=$?; if [ "$rc" -ne 0 ]; then rm -f "$raw" "$window" "$filtered"; fail "시간 filter 오류 rc=$rc"; return 1; fi; grep -E -i -C 3 "$EVIDENCE_RE|DeleteVolume|delete|finalizer|resource in use|volume not found|failed|error" "$window" >"$filtered"; rc=$?; if [ "$rc" -eq 0 ]; then cat "$filtered"; elif [ "$rc" -eq 1 ]; then printf '%s\n' 'filter no-match: producer 성공이며 cleanup 성공은 아니다.' >&2; else rm -f "$raw" "$window" "$filtered"; fail "grep 오류 rc=$rc"; return 1; fi; after="$(kubectl get pod "$PROVISIONER_POD" -n "$CSI_NS" -o json)" || { rm -f "$raw" "$window" "$filtered"; fail '수집 후 Pod 조회 실패'; return 1; }; [ "$(jq -r '.metadata.uid' <<<"$after")" = "$before_uid" ] && [ "$(jq -r --arg c "$PROVISIONER_CONTAINER" '.status.containerStatuses[]|select(.name==$c)|.containerID' <<<"$after")" = "$before_cid" ] || { rm -f "$raw" "$window" "$filtered"; fail '수집 중 provisioner lifecycle 변경'; return 1; }; rm -f "$raw" "$window" "$filtered"; EVIDENCE_READY=true; }
prom_queries(){ refresh || return 1; local ns pvc pv; ns="${NS//\"/\\\"}"; pvc="${PVC//\"/\\\"}"; pv="${PV//\"/\\\"}"; printf 'time()-kube_persistentvolumeclaim_deletion_timestamp{namespace="%s",persistentvolumeclaim="%s"}\n' "$ns" "$pvc"; printf 'kube_persistentvolume_status_phase{persistentvolume="%s",phase=~"Released|Failed"}==1\n' "$pv"; }
tools_gate(){ local tool missing=''; for tool in kubectl jq sed paste date mktemp grep; do command -v "$tool" >/dev/null 2>&1 || missing="$missing $tool"; done; [ -z "$missing" ] || { printf '중단: 도구 없음=%s\n' "$missing" >&2; return 1; }; }
tools_gate && snapshot
```

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | PVC 삭제 뒤 PV, Trident volume, qtree 또는 export policy가 예상 수렴 시간을 넘겨 남음 |
| 적용 | 마지막 PVC UID부터 provider object까지 exact mapping과 다른 참조 부재를 증명 가능 |
| 제외 | `Retain`, snapshot/clone/다른 consumer 참조, 마지막 UID handoff 부재 |

검색 문자열: `DeleteVolume`, `deletionTimestamp`, `finalizer`, `resource in use`, `volume not found`, `qtree`, `export policy`.

`delete_evidence`로 producer, window filter와 lifecycle을 한 번에 검증한다. provider owner는 version-matched read-only 경로에서 `HANDLE → Trident volume → backend/SVM → ONTAP qtree/quota → export policy/rule`을 확인한다. shared/auto-export policy는 남는 것이 정상일 수 있다. Secret·token·endpoint·client match 원문은 보존하지 않는다.

## 비가역 변경과 정상화

삭제, finalizer 제거와 reclaimPolicy 변경 명령은 제공하지 않는다. provider 삭제는 fresh `snapshot`/`delete_evidence`, 참조 0, 보존 종료, immutable backup·복구 계획, Kubernetes·Trident·NetApp owner 승인 ID와 중단 조건이 모두 있어야 한다. 삭제는 rollback할 수 없으며 `NotFound` 재시도는 rollback이 아니다.

정상화는 `남은 consumer 기능 → export 공유 관계 → qtree/quota → ONTAP volume → Trident volume → PV/PVC lifecycle → 재발 관찰` 순서다. controller 교체 시 current owner의 단일 Pod 후보를 다시 골라 새 Pod UID/containerID로 재등록한다. `prom_queries` 빈 vector는 backend cleanup 성공이 아니라 metric 미지원·label/cluster/time/API 오류 또는 객체 삭제 후보다.
