# PVC 확장 후 PV·filesystem 용량이 수렴하지 않는 경우

## 입력값과 빠른 수집

PVC request, PV capacity, PVC status capacity, block device와 filesystem capacity는 서로 다른 단계다. 한 단계가 새 값이라고 뒤 단계까지 성공한 것으로 해석하지 않는다.

| 입력 종류 | 변수 | 출처와 gate |
|---|---|---|
| 최초 입력 | `NS`, `PVC` | 장애 Event·대시보드에서 확인한 exact namespace/name |
| API 확인 자동값 | PVC/PV/StorageClass UID·RV, claimRef, driver, consumer Pod·node | 매 명령 직전에 새 API 응답으로 재등록하며 UID·관계 변경 시 중단 |
| 계산 후보값 | `PVC_RE` | PVC 이름을 regex escape한 값. bounded 출력 뒤 PromQL 생성에만 사용 |
| 운영자 선택값 | controller/node Pod·container, UTC 창, mount target/source, 승인 SSH host | 현재 관계와 remote identity 검증 후 사용 |

### 1. 실행 위치·도구·최초 입력

Kubernetes 명령은 승인된 context의 같은 Bash에서 실행한다. `kubectl config`의 user key는 실제 인증 주체를 보장하지 않으므로 RBAC와 audit gate를 별도로 확인한다.

`TOOLS_READY=false; MISSING=''; for T in kubectl jq sed grep awk date ssh; do command -v "$T" >/dev/null 2>&1 || MISSING="$MISSING $T"; done; if [ -n "${BASH_VERSION:-}" ] && [ -z "$MISSING" ]; then TOOLS_READY=true; else printf '중단: Bash 또는 도구 누락:%s\n' "$MISSING" >&2; fi`

`kubectl config view --minify -o json | jq -r '. as $x | "context=\($x["current-context"]|@sh) user-key=\($x.contexts[0].context.user|@sh) cluster-key=\($x.contexts[0].context.cluster|@sh)"'`

`unset INPUT_READY IDENTITY_READY RELATION_READY OWNER_READY CONTAINER_READY NODE_READY HOST_READY TIME_READY ARGUMENT_READY SELECTION_READY AUTH_READY EVIDENCE_READY MUTATION_READY ROLLBACK_READY NORMALIZATION_READY PVC_UID PVC_RV PV PV_UID PV_RV SC SC_UID SC_RV DRIVER PVC_RE; NS='PASTE_PVC_NAMESPACE_HERE'; PVC='PASTE_PVC_NAME_HERE'; INPUT_READY=false; if [ "$TOOLS_READY" != true ]; then printf '%s\n' '중단: TOOLS_READY가 아니다.' >&2; elif P0="$(kubectl get pvc "$PVC" -n "$NS" -o json)"; then EXPECTED_PVC_UID="$(jq -r '.metadata.uid' <<<"$P0")"; EXPECTED_PV="$(jq -r '.spec.volumeName // empty' <<<"$P0")"; EXPECTED_SC="$(jq -r '.spec.storageClassName // empty' <<<"$P0")"; [ -n "$EXPECTED_PV" ] && [ -n "$EXPECTED_SC" ] || { printf '%s\n' '중단: Bound PV 또는 StorageClass 없음' >&2; }; if [ -n "$EXPECTED_PV" ] && [ -n "$EXPECTED_SC" ]; then INPUT_READY=true; export NS PVC EXPECTED_PVC_UID EXPECTED_PV EXPECTED_SC; printf 'INPUT_READY=true ns=%s pvc=%s uid=%s pv=%s sc=%s\n' "$NS" "$PVC" "$EXPECTED_PVC_UID" "$EXPECTED_PV" "$EXPECTED_SC"; fi; else printf '%s\n' '중단: PVC exact 조회 실패' >&2; fi`

### 2. current identity·capacity relation refresh

`refresh_resize(){ unset IDENTITY_READY RELATION_READY OWNER_READY CONTAINER_READY NODE_READY HOST_READY TIME_READY ARGUMENT_READY SELECTION_READY AUTH_READY EVIDENCE_READY MUTATION_READY ROLLBACK_READY NORMALIZATION_READY PVC_UID PVC_RV PV PV_UID PV_RV SC SC_UID SC_RV DRIVER PVC_RE REQUESTED PVC_CAPACITY PV_CAPACITY; IDENTITY_READY=false; RELATION_READY=false; ARGUMENT_READY=false; [ "$INPUT_READY" = true ] || return 1; local pc pj sj; pc="$(kubectl get pvc "$PVC" -n "$NS" -o json)" || return 1; [ "$(jq -r '.metadata.uid' <<<"$pc")" = "$EXPECTED_PVC_UID" ] || { printf '%s\n' '중단: PVC UID 교체' >&2; return 1; }; [ "$(jq -r '.spec.volumeName // empty' <<<"$pc")" = "$EXPECTED_PV" ] && [ "$(jq -r '.spec.storageClassName // empty' <<<"$pc")" = "$EXPECTED_SC" ] || { printf '%s\n' '중단: PVC→PV/SC relation 변경' >&2; return 1; }; pj="$(kubectl get pv "$EXPECTED_PV" -o json)" || return 1; sj="$(kubectl get storageclass "$EXPECTED_SC" -o json)" || return 1; [ "$(jq -r '.spec.claimRef.uid // empty' <<<"$pj")" = "$EXPECTED_PVC_UID" ] && [ "$(jq -r '.spec.claimRef.namespace // empty' <<<"$pj")" = "$NS" ] && [ "$(jq -r '.spec.claimRef.name // empty' <<<"$pj")" = "$PVC" ] || { printf '%s\n' '중단: PV claimRef 변경' >&2; return 1; }; PVC_UID="$EXPECTED_PVC_UID"; PVC_RV="$(jq -r '.metadata.resourceVersion' <<<"$pc")"; PV="$EXPECTED_PV"; PV_UID="$(jq -r '.metadata.uid' <<<"$pj")"; PV_RV="$(jq -r '.metadata.resourceVersion' <<<"$pj")"; SC="$EXPECTED_SC"; SC_UID="$(jq -r '.metadata.uid' <<<"$sj")"; SC_RV="$(jq -r '.metadata.resourceVersion' <<<"$sj")"; DRIVER="$(jq -r '.spec.csi.driver // empty' <<<"$pj")"; REQUESTED="$(jq -r '.spec.resources.requests.storage // empty' <<<"$pc")"; PVC_CAPACITY="$(jq -r '.status.capacity.storage // empty' <<<"$pc")"; PV_CAPACITY="$(jq -r '.spec.capacity.storage // empty' <<<"$pj")"; PVC_RE="$(printf '%s' "$PVC" | sed 's/[][\\.^$*+?(){}|]/\\&/g')"; [ -n "$DRIVER" ] && [ "$(jq -r '.provisioner' <<<"$sj")" = "$DRIVER" ] || { printf '%s\n' '중단: SC provisioner/PV driver 불일치' >&2; return 1; }; IDENTITY_READY=true; RELATION_READY=true; ARGUMENT_READY=true; export PVC_UID PVC_RV PV PV_UID PV_RV SC SC_UID SC_RV DRIVER REQUESTED PVC_CAPACITY PV_CAPACITY PVC_RE; printf 'IDENTITY_READY=true RELATION_READY=true pvc=%s/%s pv=%s/%s sc=%s/%s driver=%s requested=%s pvc_capacity=%s pv_capacity=%s regex=%s\n' "$PVC_UID" "$PVC_RV" "$PV_UID" "$PV_RV" "$SC_UID" "$SC_RV" "$DRIVER" "$REQUESTED" "${PVC_CAPACITY:--}" "${PV_CAPACITY:--}" "$PVC_RE"; }`

`if refresh_resize; then kubectl get pvc "$PVC" -n "$NS" -o custom-columns='PVC:.metadata.name,UID:.metadata.uid,RV:.metadata.resourceVersion,PHASE:.status.phase,REQUESTED:.spec.resources.requests.storage,STATUS-CAPACITY:.status.capacity.storage,CONDITIONS:.status.conditions[*].type,MESSAGES:.status.conditions[*].message,PV:.spec.volumeName,SC:.spec.storageClassName' && kubectl get pv "$PV" -o custom-columns='PV:.metadata.name,UID:.metadata.uid,RV:.metadata.resourceVersion,SPEC-CAPACITY:.spec.capacity.storage,CLAIM-NS:.spec.claimRef.namespace,CLAIM:.spec.claimRef.name,CLAIM-UID:.spec.claimRef.uid,DRIVER:.spec.csi.driver' && kubectl get storageclass "$SC" -o custom-columns='SC:.metadata.name,UID:.metadata.uid,RV:.metadata.resourceVersion,PROVISIONER:.provisioner,ALLOW-EXPANSION:.allowVolumeExpansion'; fi`

성공 gate는 current PVC UID와 PV claimRef UID가 같고 SC provisioner와 PV CSI driver가 같은 것이다. Kubernetes quantity 문자열은 단순 사전식·정수 비교하지 않는다. 승인 도구로 bytes 정규화가 검증되지 않았다면 request/capacity 원문과 단위만 보존하고 크기 순서를 추정하지 않는다.

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | request를 늘린 뒤 PV capacity, PVC status capacity, `FileSystemResizePending`, 실제 filesystem 중 하나가 수렴하지 않는다. |
| 적용 | PVC/PV/SC identity와 controller/node/filesystem 단계를 연결할 수 있다. |
| 제외 | PVC가 Pending이고 PV가 없으면 [[01. PVC가 Pending에서 Bound로 수렴하지 않는 경우]], StatefulSet claim identity가 다르면 [[04. StatefulSet Pod 이름·ordinal과 PVC identity가 맞지 않는 경우]]로 이동한다. |

검색 문자열: `Resizing`, `FileSystemResizePending`, `ControllerExpandVolume`, `NodeExpandVolume`, `VolumeResizeFailed`, `operation not supported`, `quota exceeded`, `device is too small`.

## 관측별 진단 경로

| 지금 보이는 것 | 먼저 볼 것 | 그다음 어디를 볼지 | 상세 절 |
|---|---|---|---|
| request만 새 값 | `allowVolumeExpansion`, controller 로그 | resizer·backend | [[#1. controller expansion]] |
| PV만 새 값 | PVC UID·condition, resizer reconcile | API/controller owner | [[#1. controller expansion]] |
| PVC/PV 새 값과 `FileSystemResizePending` | current consumer Pod/node | CSI node·kubelet | [[#2. node expansion]] |
| API 새 값, filesystem 옛 값 | current mount source/device/fstype | Linux filesystem | [[#3. Linux filesystem]] |

### 1. controller expansion

실제 inventory에서 controller Pod와 resizer container를 선택한다. first container나 `external-resizer`라는 이름을 추측하지 않는다.

`CTRL_NS='PASTE_RESIZER_NAMESPACE_HERE'; CTRL_POD='PASTE_RESIZER_POD_HERE'; CTRL_CONTAINER='PASTE_RESIZER_CONTAINER_HERE'; START_INPUT='PASTE_RESIZE_START_RFC3339_HERE'; END_INPUT='PASTE_RESIZE_END_RFC3339_HERE'`

`refresh_time(){ unset TIME_READY START_RFC3339 END_RFC3339 COLLECTED_AT; TIME_READY=false; [[ "$START_INPUT" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] && [[ "$END_INPUT" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] || return 1; local se ee; se="$(date -u -d "$START_INPUT" +%s 2>/dev/null)" || return 1; ee="$(date -u -d "$END_INPUT" +%s 2>/dev/null)" || return 1; START_RFC3339="$(date -u -d "@$se" '+%Y-%m-%dT%H:%M:%SZ')"; END_RFC3339="$(date -u -d "@$ee" '+%Y-%m-%dT%H:%M:%SZ')"; [ "$START_RFC3339" = "$START_INPUT" ] && [ "$END_RFC3339" = "$END_INPUT" ] && [ "$se" -lt "$ee" ] || return 1; COLLECTED_AT="$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; TIME_READY=true; export START_RFC3339 END_RFC3339 COLLECTED_AT; }`

`refresh_controller(){ unset OWNER_READY CONTAINER_READY SELECTION_READY CTRL_UID CTRL_RV CTRL_CONTAINER_ID CTRL_NODE; OWNER_READY=false; CONTAINER_READY=false; SELECTION_READY=false; [ "$RELATION_READY" = true ] || return 1; local j; j="$(kubectl get pod "$CTRL_POD" -n "$CTRL_NS" -o json)" || return 1; jq -e --arg c "$CTRL_CONTAINER" 'any(.spec.containers[]?; .name==$c)' <<<"$j" >/dev/null || return 1; CTRL_UID="$(jq -r '.metadata.uid' <<<"$j")"; CTRL_RV="$(jq -r '.metadata.resourceVersion' <<<"$j")"; CTRL_CONTAINER_ID="$(jq -r --arg c "$CTRL_CONTAINER" '[.status.containerStatuses[]? | select(.name==$c)][0].containerID // empty' <<<"$j")"; CTRL_NODE="$(jq -r '.spec.nodeName // empty' <<<"$j")"; [ -n "$CTRL_CONTAINER_ID" ] || return 1; OWNER_READY=true; CONTAINER_READY=true; SELECTION_READY=true; export CTRL_UID CTRL_RV CTRL_CONTAINER_ID CTRL_NODE; printf 'SELECTION_READY=true pod=%s/%s uid=%s rv=%s node=%s container=%s containerID=%s\n' "$CTRL_NS" "$CTRL_POD" "$CTRL_UID" "$CTRL_RV" "$CTRL_NODE" "$CTRL_CONTAINER" "$CTRL_CONTAINER_ID"; }`

`refresh_pod_owner(){ local ns="$1" pod="$2" expected_uid="$3" prefix="$4" j ref kind name uid owner; j="$(kubectl get pod "$pod" -n "$ns" -o json)" || return 1; [ "$(jq -r '.metadata.uid' <<<"$j")" = "$expected_uid" ] || return 1; ref="$(jq -c '[.metadata.ownerReferences[]? | select(.controller==true)][0] // empty' <<<"$j")"; [ -n "$ref" ] || return 1; kind="$(jq -r '.kind' <<<"$ref")"; name="$(jq -r '.name' <<<"$ref")"; uid="$(jq -r '.uid' <<<"$ref")"; owner="$(kubectl get "$kind" "$name" -n "$ns" -o json)" || return 1; [ "$(jq -r '.metadata.uid' <<<"$owner")" = "$uid" ] || return 1; printf -v "${prefix}_OWNER_KIND" '%s' "$kind"; printf -v "${prefix}_OWNER_NAME" '%s' "$name"; printf -v "${prefix}_OWNER_UID" '%s' "$uid"; OWNER_READY=true; printf 'OWNER_READY=true pod=%s/%s owner=%s/%s/%s\n' "$ns" "$pod" "$kind" "$name" "$uid"; }`

`refresh_auth(){ AUTH_READY=false; [ "$(kubectl auth can-i get "persistentvolumeclaims/$PVC" -n "$NS")" = yes ] && [ "$(kubectl auth can-i get "persistentvolumes/$PV")" = yes ] && [ "$(kubectl auth can-i get "storageclasses.storage.k8s.io/$SC")" = yes ] && [ "$(kubectl auth can-i get "pods/$CTRL_POD" -n "$CTRL_NS")" = yes ] && [ "$(kubectl auth can-i get "pods/$CTRL_POD" --subresource=log -n "$CTRL_NS")" = yes ] || return 1; AUTH_READY=true; }`

`if refresh_resize && refresh_controller && refresh_pod_owner "$CTRL_NS" "$CTRL_POD" "$CTRL_UID" CTRL && refresh_time && refresh_auth; then kubectl logs "$CTRL_POD" -n "$CTRL_NS" -c "$CTRL_CONTAINER" --timestamps --since-time="$START_RFC3339" --tail=300 | awk -v end="$END_RFC3339" '$1 < end' | grep -E -i -C 3 'ControllerExpandVolume|resize|expand|FileSystemResizePending|permission|quota|timeout|unsupported|error'; RC=("${PIPESTATUS[@]}"); if [ "${RC[0]}" -ne 0 ]; then EVIDENCE_READY=false; printf '중단: producer rc=%s\n' "${RC[0]}" >&2; elif [ "${RC[1]}" -ne 0 ]; then EVIDENCE_READY=false; printf '중단: 시간 filter rc=%s\n' "${RC[1]}" >&2; elif [ "${RC[2]}" -eq 1 ]; then EVIDENCE_READY=false; printf '%s\n' '미확정: filter 일치 없음' >&2; elif [ "${RC[2]}" -ne 0 ]; then EVIDENCE_READY=false; printf '중단: 증상 filter rc=%s\n' "${RC[2]}" >&2; else EVIDENCE_READY=true; printf 'EVIDENCE_READY=true collected=%s pvc_uid=%s controller_uid=%s owner_uid=%s containerID=%s\n' "$COLLECTED_AT" "$PVC_UID" "$CTRL_UID" "$CTRL_OWNER_UID" "$CTRL_CONTAINER_ID"; fi; fi`

`allowVolumeExpansion=true`는 backend 확장 성공을 보장하지 않는다. `unsupported`, quota, permission, timeout은 각각 기능 계약, storage capacity/policy, 권한, endpoint 경계로 이관한다. filter 없음은 정상 증거가 아니며 raw 전체 로그는 자동 출력하지 않는다.

### 2. node expansion

현재 namespace에서 PVC를 참조하는 Pod 후보를 관계가 보존된 형태로 출력한다. 목록 producer 실패와 filter 불일치를 구분한다.

`if refresh_resize; then kubectl get pods -n "$NS" -o json | jq -r --arg pvc "$PVC" '.items[] | select(any(.spec.volumes[]?; .persistentVolumeClaim.claimName==$pvc)) | "pod=\(.metadata.name|@sh) uid=\(.metadata.uid|@sh) rv=\(.metadata.resourceVersion|@sh) node=\((.spec.nodeName // "")|@sh) owner=\([.metadata.ownerReferences[]? | select(.controller==true) | "\(.kind)/\(.name)/\(.uid)"] | join(",")|@sh) phase=\((.status.phase // "")|@sh)"'; RC=("${PIPESTATUS[@]}"); [ "${RC[0]}" -eq 0 ] && [ "${RC[1]}" -eq 0 ] || printf '중단: pod producer/filter 실패 producer=%s filter=%s\n' "${RC[0]}" "${RC[1]}" >&2; fi`

복수 consumer 중 API만으로 resize target을 확정할 수 없으면 임의 첫 Pod를 고르지 않는다. Event·CSI log의 node와 같은 current Pod를 선택한다.

`POD='PASTE_RESIZE_CONSUMER_POD_HERE'; NODE_NS='PASTE_RESIZE_NODE_PLUGIN_NAMESPACE_HERE'; NODE_POD='PASTE_RESIZE_NODE_PLUGIN_POD_HERE'; NODE_CONTAINER='PASTE_RESIZE_NODE_PLUGIN_CONTAINER_HERE'`

`register_consumer_identity(){ unset EXPECTED_CONSUMER_UID EXPECTED_CONSUMER_CREATED; SELECTION_READY=false; [ "$RELATION_READY" = true ] || return 1; local p; p="$(kubectl get pod "$POD" -n "$NS" -o json)" || return 1; jq -e --arg pvc "$PVC" 'any(.spec.volumes[]?; .persistentVolumeClaim.claimName==$pvc)' <<<"$p" >/dev/null || return 1; EXPECTED_CONSUMER_UID="$(jq -r '.metadata.uid' <<<"$p")"; EXPECTED_CONSUMER_CREATED="$(jq -r '.metadata.creationTimestamp' <<<"$p")"; [ -n "$EXPECTED_CONSUMER_UID" ] && [ -n "$EXPECTED_CONSUMER_CREATED" ] || return 1; export EXPECTED_CONSUMER_UID EXPECTED_CONSUMER_CREATED; printf 'consumer registered pod=%s uid=%s created=%s\n' "$POD" "$EXPECTED_CONSUMER_UID" "$EXPECTED_CONSUMER_CREATED"; }`

`refresh_node(){ unset OWNER_READY CONTAINER_READY NODE_READY SELECTION_READY POD_UID POD_RV NODE NODE_POD_UID NODE_POD_RV NODE_CONTAINER_ID; OWNER_READY=false; CONTAINER_READY=false; NODE_READY=false; SELECTION_READY=false; [ "$RELATION_READY" = true ] || return 1; local p n; p="$(kubectl get pod "$POD" -n "$NS" -o json)" || return 1; jq -e --arg pvc "$PVC" 'any(.spec.volumes[]?; .persistentVolumeClaim.claimName==$pvc)' <<<"$p" >/dev/null || return 1; NODE="$(jq -r '.spec.nodeName // empty' <<<"$p")"; [ -n "$NODE" ] || return 1; n="$(kubectl get pod "$NODE_POD" -n "$NODE_NS" -o json)" || return 1; [ "$(jq -r '.spec.nodeName // empty' <<<"$n")" = "$NODE" ] || return 1; jq -e --arg c "$NODE_CONTAINER" 'any(.spec.containers[]?; .name==$c)' <<<"$n" >/dev/null || return 1; POD_UID="$(jq -r '.metadata.uid' <<<"$p")"; POD_RV="$(jq -r '.metadata.resourceVersion' <<<"$p")"; NODE_POD_UID="$(jq -r '.metadata.uid' <<<"$n")"; NODE_POD_RV="$(jq -r '.metadata.resourceVersion' <<<"$n")"; NODE_CONTAINER_ID="$(jq -r --arg c "$NODE_CONTAINER" '[.status.containerStatuses[]? | select(.name==$c)][0].containerID // empty' <<<"$n")"; [ -n "$NODE_CONTAINER_ID" ] || return 1; OWNER_READY=true; CONTAINER_READY=true; NODE_READY=true; SELECTION_READY=true; export POD_UID POD_RV NODE NODE_POD_UID NODE_POD_RV NODE_CONTAINER_ID; printf 'SELECTION_READY=true consumer=%s uid=%s rv=%s node=%s nodePod=%s/%s uid=%s rv=%s container=%s containerID=%s\n' "$POD" "$POD_UID" "$POD_RV" "$NODE" "$NODE_NS" "$NODE_POD" "$NODE_POD_UID" "$NODE_POD_RV" "$NODE_CONTAINER" "$NODE_CONTAINER_ID"; }`

`refresh_registered_node(){ refresh_node || return 1; [ "$POD_UID" = "$EXPECTED_CONSUMER_UID" ] && [ "$(kubectl get pod "$POD" -n "$NS" -o jsonpath='{.metadata.creationTimestamp}')" = "$EXPECTED_CONSUMER_CREATED" ] || { unset OWNER_READY CONTAINER_READY NODE_READY SELECTION_READY POD_UID POD_RV NODE NODE_POD_UID NODE_POD_RV NODE_CONTAINER_ID; printf '%s\n' '중단: consumer Pod lifecycle 교체. PVC relation에서 replacement를 다시 선택·등록한다.' >&2; return 1; }; }`

`refresh_node_machine(){ unset NODE_UID NODE_MACHINE_ID; NODE_READY=false; local n; n="$(kubectl get node "$NODE" -o json)" || return 1; NODE_UID="$(jq -r '.metadata.uid' <<<"$n")"; NODE_MACHINE_ID="$(jq -r '.status.nodeInfo.machineID // empty' <<<"$n")"; [ -n "$NODE_UID" ] && [ -n "$NODE_MACHINE_ID" ] || return 1; NODE_READY=true; export NODE_UID NODE_MACHINE_ID; printf 'NODE_READY=true node=%s uid=%s machineID=%s\n' "$NODE" "$NODE_UID" "$NODE_MACHINE_ID"; }`

`if refresh_resize && refresh_registered_node && refresh_pod_owner "$NS" "$POD" "$POD_UID" CONSUMER && refresh_pod_owner "$NODE_NS" "$NODE_POD" "$NODE_POD_UID" NODE_PLUGIN && refresh_node_machine && refresh_time && [ "$(kubectl auth can-i get "pods/$NODE_POD" --subresource=log -n "$NODE_NS")" = yes ]; then kubectl logs "$NODE_POD" -n "$NODE_NS" -c "$NODE_CONTAINER" --timestamps --since-time="$START_RFC3339" --tail=300 | awk -v end="$END_RFC3339" '$1 < end' | grep -E -i -C 3 'NodeExpandVolume|resize|expand|filesystem|device|mount|permission|timeout|error'; RC=("${PIPESTATUS[@]}"); if [ "${RC[0]}" -ne 0 ]; then EVIDENCE_READY=false; printf '중단: producer rc=%s\n' "${RC[0]}" >&2; elif [ "${RC[1]}" -ne 0 ]; then EVIDENCE_READY=false; printf '중단: 시간 filter rc=%s\n' "${RC[1]}" >&2; elif [ "${RC[2]}" -eq 1 ]; then EVIDENCE_READY=false; printf '%s\n' '미확정: filter 일치 없음' >&2; elif [ "${RC[2]}" -ne 0 ]; then EVIDENCE_READY=false; printf '중단: 증상 filter rc=%s\n' "${RC[2]}" >&2; else EVIDENCE_READY=true; printf 'EVIDENCE_READY=true pod_uid=%s consumer_owner_uid=%s node_uid=%s node_pod_uid=%s node_owner_uid=%s collected=%s\n' "$POD_UID" "$CONSUMER_OWNER_UID" "$NODE_UID" "$NODE_POD_UID" "$NODE_PLUGIN_OWNER_UID" "$COLLECTED_AT"; fi; fi`

### 3. Linux filesystem

mount target과 source는 CSI/kubelet 증거에서 확인한 값만 사용한다. host path를 추측하지 않는다. remote hostname과 target node mapping은 승인 inventory에서 대조한다.

`MOUNT_TARGET='PASTE_CONFIRMED_MOUNT_TARGET_HERE'; DEVICE_SOURCE='PASTE_CONFIRMED_DEVICE_SOURCE_HERE'; APPROVED_HOST='PASTE_RESIZE_NODE_SSH_HOST_HERE'; EXPECTED_REMOTE_HOST='PASTE_RESIZE_REMOTE_HOSTNAME_HERE'; validate_host_args(){ unset HOST_READY; HOST_READY=false; [ "$NODE_READY" = true ] || return 1; [[ "$APPROVED_HOST" =~ ^[A-Za-z0-9._:@-]+$ ]] && [[ "$EXPECTED_REMOTE_HOST" =~ ^[A-Za-z0-9._-]+$ ]] && [[ "$MOUNT_TARGET" == /* ]] && [[ "$DEVICE_SOURCE" == /* ]] || return 1; HOST_READY=true; }`

`if refresh_resize && refresh_registered_node && refresh_pod_owner "$NS" "$POD" "$POD_UID" CONSUMER && refresh_pod_owner "$NODE_NS" "$NODE_POD" "$NODE_POD_UID" NODE_PLUGIN && refresh_node_machine && validate_host_args; then ssh -- "$APPROVED_HOST" bash -c 'set -u; expected=$1; machine=$2; target=$3; source=$4; actual=$(hostname -f); actual_machine=$(cat /etc/machine-id); [ "$actual" = "$expected" ] && [ "$actual_machine" = "$machine" ] || { printf "중단: remote identity mismatch expected=%s actual=%s\n" "$expected" "$actual" >&2; exit 40; }; findmnt -T "$target" -o TARGET,SOURCE,FSTYPE,OPTIONS || exit 41; findmnt -T "$target" -n -o SOURCE | grep -Fx -- "$source" >/dev/null || exit 42; df -B1 --output=source,fstype,size,used,avail,pcent,target "$target" || exit 43; df -i --output=source,itotal,iused,iavail,ipcent,target "$target" || exit 44' bash "$EXPECTED_REMOTE_HOST" "$NODE_MACHINE_ID" "$MOUNT_TARGET" "$DEVICE_SOURCE"; RC=$?; case "$RC" in 0) EVIDENCE_READY=true; printf 'filesystem evidence collected node=%s node_uid=%s pod_uid=%s at=%s\n' "$NODE" "$NODE_UID" "$POD_UID" "$(date -u '+%Y-%m-%dT%H:%M:%SZ')";; 40) EVIDENCE_READY=false; printf '%s\n' '중단: remote identity 불일치' >&2;; 42) EVIDENCE_READY=false; printf '%s\n' '중단: current mount source 불일치' >&2;; *) EVIDENCE_READY=false; printf '중단: filesystem producer 실패 rc=%s\n' "$RC" >&2;; esac; fi`

API capacity가 새 값이어도 device/filesystem이 옛 값이면 정상화가 아니다. `resize2fs`, `xfs_growfs`, `fsck`, `mkfs`, 수동 mount/umount, device 삭제는 실행하지 않는다.

## Prometheus 쿼리 생성

metric과 label이 실제 존재할 때만 출력된 한 줄을 사용한다. PVC 이름은 escape된 `$PVC_RE`에서 생성한다.

`if refresh_resize; then printf 'kube_persistentvolumeclaim_resource_requests_storage_bytes{namespace="%s",persistentvolumeclaim=~"^%s$"}\n' "$NS" "$PVC_RE"; printf 'kubelet_volume_stats_capacity_bytes{namespace="%s",persistentvolumeclaim=~"^%s$"}\n' "$NS" "$PVC_RE"; fi`

빈 결과는 정상이나 0이 아니다. metric 부재, CSI/kubelet 미노출, label 불일치, 조회 시간·cluster 오류, 권한/API 실패를 구분한다.

## 상태 변경 분리

PVC expansion은 일반적으로 축소 rollback이 불가능하다. 이 런북에는 PVC patch, Pod 삭제, CSI restart, VolumeAttachment 삭제, filesystem grow/repair 명령을 두지 않는다. 별도 절차는 fresh PVC/PV/SC/component/node identity와 evidence, exact RBAC/audit, workload·storage·data owner 승인, 변경 창, RWO/quorum, 중복 API 요청의 멱등성, backend 여유·중단 조건, 실제 검증된 rollback 또는 비가역 승인까지 갖춰야 한다. rollback이 없으면 이를 명시하고 `ROLLBACK_READY=false` 상태에서 일반 실행을 금지한다.

## 정상화 검증

`NORMALIZATION_READY=false; if refresh_resize; then kubectl get pvc "$PVC" -n "$NS" -o custom-columns='PVC:.metadata.name,UID:.metadata.uid,RV:.metadata.resourceVersion,REQUESTED:.spec.resources.requests.storage,STATUS-CAPACITY:.status.capacity.storage,CONDITIONS:.status.conditions[*].type,PV:.spec.volumeName' && kubectl get pv "$PV" -o custom-columns='PV:.metadata.name,UID:.metadata.uid,RV:.metadata.resourceVersion,SPEC-CAPACITY:.spec.capacity.storage,CLAIM-UID:.spec.claimRef.uid'; printf 'object capacity gate passed at=%s; filesystem and workload gates are still required\n' "$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; fi`

위 명령은 객체 관계 gate만 재확인하며 `NORMALIZATION_READY=false`를 유지한다. node 단계가 있었으면 replacement Pod를 이름으로 재사용하지 말고 PVC relation으로 current consumer를 다시 선택한 뒤 node plugin UID/containerID, mount source, device와 filesystem bytes를 재검증한다. request·PV·PVC·filesystem이 승인 목표와 일치하고 `FileSystemResizePending`·새 CSI/kubelet 오류가 관찰 창 동안 없으며 실제 workload read/write가 성공해야 최종 정상화한다.
