# system·protection domain·storage pool·capacity 문제로 provisioning이 실패하는 경우

## 입력값과 빠른 수집

경로는 `승인된 작업 host → API server → PVC UID → StorageClass UID/provisioner → external-provisioner current container → PowerFlex system → protection domain → storage pool → usable capacity/authorization`이다.

| 구분 | 값 | gate |
|---|---|---|
| 최초 입력 | namespace와 PVC | 승인된 request exact name |
| API 자동값 | PVC/StorageClass UID·RV/request/binding, controller owner/containerID | current API exact relation |
| 계산 후보 | escaped PVC identity regex | literal escape |
| 운영자 선택 | controller Pod/container, canonical UTC 창 | current inventory와 최초 Event |

```bash
NS='PASTE_NAMESPACE_HERE'
PVC='PASTE_PVC_HERE'
CSI_NS='PASTE_CSI_NAMESPACE_HERE'
CONTROLLER_POD='PASTE_CONTROLLER_POD_HERE'
CONTROLLER_CONTAINER='PASTE_CONTROLLER_CONTAINER_HERE'
START_RFC3339='PASTE_START_RFC3339_HERE'
END_RFC3339='PASTE_END_RFC3339_HERE'

invalidate(){ unset PVC_JSON SC_JSON POD_JSON PVC_UID PVC_RV SC SC_UID SC_RV PROVISIONER REQUEST BINDING POD_UID POD_RV OWNER_KIND OWNER_NAME OWNER_UID CONTAINER_ID IMAGE_ID EVIDENCE_RE OBSERVED_AT; ID_READY=false; TIME_READY=false; SELECTION_READY=false; AUTH_READY=false; EVIDENCE_READY=false; BACKEND_READY=false; MUTATION_READY=false; ROLLBACK_READY=false; NORMALIZATION_READY=false; }
fail(){ local message="$1"; invalidate; printf '중단: %s\n' "$message" >&2; return 1; }
canonical(){ local value="$1" round; [[ "$value" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] || return 1; round="$(date -u -d "$value" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" || return 1; [ "$round" = "$value" ]; }
time_gate(){ TIME_READY=false; canonical "$START_RFC3339" || { fail 'start 형식 오류'; return 1; }; canonical "$END_RFC3339" || { fail 'end 형식 오류'; return 1; }; [ "$(date -u -d "$START_RFC3339" +%s)" -lt "$(date -u -d "$END_RFC3339" +%s)" ] || { fail '시간 순서 오류'; return 1; }; TIME_READY=true; }
ere(){ printf '%s' "$1" | sed 's/[][\\.^$*+?(){}|]/\\&/g'; }
refresh(){ invalidate; local pvc sc pod; pvc="$(kubectl get pvc "$PVC" -n "$NS" -o json)" || { fail 'PVC 조회 실패'; return 1; }; [ "$(jq -r '.status.phase' <<<"$pvc")" = Pending ] && [ -z "$(jq -r '.spec.volumeName // empty' <<<"$pvc")" ] || { fail 'Pending/unbound PVC가 아니다.'; return 1; }; SC="$(jq -r '.spec.storageClassName // empty' <<<"$pvc")"; [ -n "$SC" ] || { fail 'StorageClass 없음'; return 1; }; sc="$(kubectl get storageclass "$SC" -o json)" || { fail 'StorageClass 조회 실패'; return 1; }; PROVISIONER="$(jq -r '.provisioner // empty' <<<"$sc")"; [ -n "$PROVISIONER" ] || { fail 'provisioner 없음'; return 1; }; pod="$(kubectl get pod "$CONTROLLER_POD" -n "$CSI_NS" -o json)" || { fail 'controller Pod 조회 실패'; return 1; }; jq -e --arg c "$CONTROLLER_CONTAINER" '([.spec.containers[]|select(.name==$c)]|length)==1 and ([.status.containerStatuses[]|select(.name==$c and (.containerID|length)>0)]|length)==1' <<<"$pod" >/dev/null || { fail 'regular container/current containerID 불일치'; return 1; }; OWNER_KIND="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].kind // empty' <<<"$pod")"; OWNER_NAME="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].name // empty' <<<"$pod")"; OWNER_UID="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid // empty' <<<"$pod")"; [ -n "$OWNER_UID" ] || { fail 'controller owner 없음'; return 1; }; if [ "$OWNER_KIND" = ReplicaSet ]; then [ "$(kubectl get replicaset "$OWNER_NAME" -n "$CSI_NS" -o jsonpath='{.metadata.uid}')" = "$OWNER_UID" ] || { fail 'current ReplicaSet UID 불일치'; return 1; }; elif [ "$OWNER_KIND" = StatefulSet ]; then [ "$(kubectl get statefulset "$OWNER_NAME" -n "$CSI_NS" -o jsonpath='{.metadata.uid}')" = "$OWNER_UID" ] || { fail 'current StatefulSet UID 불일치'; return 1; }; else fail '지원하지 않는 controller owner kind'; return 1; fi; PVC_JSON="$pvc"; SC_JSON="$sc"; POD_JSON="$pod"; PVC_UID="$(jq -r '.metadata.uid' <<<"$pvc")"; PVC_RV="$(jq -r '.metadata.resourceVersion' <<<"$pvc")"; SC_UID="$(jq -r '.metadata.uid' <<<"$sc")"; SC_RV="$(jq -r '.metadata.resourceVersion' <<<"$sc")"; REQUEST="$(jq -r '.spec.resources.requests.storage' <<<"$pvc")"; BINDING="$(jq -r '.volumeBindingMode // "Immediate"' <<<"$sc")"; POD_UID="$(jq -r '.metadata.uid' <<<"$pod")"; POD_RV="$(jq -r '.metadata.resourceVersion' <<<"$pod")"; CONTAINER_ID="$(jq -r --arg c "$CONTROLLER_CONTAINER" '.status.containerStatuses[]|select(.name==$c)|.containerID' <<<"$pod")"; IMAGE_ID="$(jq -r --arg c "$CONTROLLER_CONTAINER" '.status.containerStatuses[]|select(.name==$c)|.imageID' <<<"$pod")"; EVIDENCE_RE="$(ere "$PVC")|$(ere "$PVC_UID")"; OBSERVED_AT="$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; ID_READY=true; SELECTION_READY=true; printf 'ID_READY=true PVC=%s uid=%s rv=%s request=%s SC=%s uid=%s provisioner=%s binding=%s\ncontroller=%s uid=%s owner=%s/%s:%s regular-container=%s containerID=%s observed=%s\n' "$PVC" "$PVC_UID" "$PVC_RV" "$REQUEST" "$SC" "$SC_UID" "$PROVISIONER" "$BINDING" "$CONTROLLER_POD" "$POD_UID" "$OWNER_KIND" "$OWNER_NAME" "$OWNER_UID" "$CONTROLLER_CONTAINER" "$CONTAINER_ID" "$OBSERVED_AT"; }
current_owner_gate(){ [ "$ID_READY" = true ] || return 1; local owner parent_name parent_uid parent; if [ "$OWNER_KIND" = ReplicaSet ]; then owner="$(kubectl get replicaset "$OWNER_NAME" -n "$CSI_NS" -o json)" || { fail 'current ReplicaSet 조회 실패'; return 1; }; [ "$(jq -r '.metadata.uid' <<<"$owner")" = "$OWNER_UID" ] || { fail 'current ReplicaSet UID 변경'; return 1; }; parent_name="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true and .kind=="Deployment")][0].name // empty' <<<"$owner")"; parent_uid="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true and .kind=="Deployment")][0].uid // empty' <<<"$owner")"; [ -n "$parent_uid" ] || { fail 'ReplicaSet의 Deployment owner 없음'; return 1; }; parent="$(kubectl get deployment "$parent_name" -n "$CSI_NS" -o json)" || { fail 'current Deployment 조회 실패'; return 1; }; [ "$(jq -r '.metadata.uid' <<<"$parent")" = "$parent_uid" ] || { fail 'current Deployment UID 변경'; return 1; }; elif [ "$OWNER_KIND" = StatefulSet ]; then [ "$(kubectl get statefulset "$OWNER_NAME" -n "$CSI_NS" -o jsonpath='{.metadata.uid}')" = "$OWNER_UID" ] || { fail 'current StatefulSet UID 변경'; return 1; }; else fail '지원하지 않는 controller current owner'; return 1; fi; }
auth_gate(){ AUTH_READY=false; [ "$ID_READY" = true ] || return 1; [ "$(kubectl auth can-i get "pods/$CONTROLLER_POD" --subresource=log -n "$CSI_NS")" = yes ] || { fail "현재 주체 get pods/$CONTROLLER_POD/log 거부 namespace=$CSI_NS"; return 1; }; AUTH_READY=true; }
window_log(){ local source="$1" output="$2" start end line stamp epoch; start="$(date -u -d "$START_RFC3339" +%s)" || return 2; end="$(date -u -d "$END_RFC3339" +%s)" || return 2; : >"$output" || return 2; while IFS= read -r line; do stamp="${line%% *}"; epoch="$(date -u -d "$stamp" +%s 2>/dev/null)" || return 2; if [ "$epoch" -ge "$start" ] && [ "$epoch" -le "$end" ]; then printf '%s\n' "$line" >>"$output" || return 2; fi; done <"$source"; }
snapshot(){ refresh && current_owner_gate && time_gate || return 1; jq -r '"=== PVC ===\nUID: \(.metadata.uid) RV: \(.metadata.resourceVersion)\nPhase: \(.status.phase) Request: \(.spec.resources.requests.storage) Created: \(.metadata.creationTimestamp)"' <<<"$PVC_JSON"; jq -r '"=== StorageClass ===\nUID: \(.metadata.uid) RV: \(.metadata.resourceVersion)\nProvisioner: \(.provisioner) Binding: \(.volumeBindingMode // "Immediate")"' <<<"$SC_JSON"; if [ "$BINDING" = WaitForFirstConsumer ]; then jq -e '.metadata.annotations["volume.kubernetes.io/selected-node"]|length>0' <<<"$PVC_JSON" >/dev/null || { fail 'WaitForFirstConsumer selected-node가 없어 backend 분기가 아니다.'; return 1; }; fi; }
provision_evidence(){ snapshot >/dev/null && auth_gate || return 1; local before_uid="$POD_UID" before_cid="$CONTAINER_ID" raw window filtered rc after; raw="$(mktemp)" || return 1; window="$(mktemp)" || { rm -f "$raw"; return 1; }; filtered="$(mktemp)" || { rm -f "$raw" "$window"; return 1; }; if ! kubectl logs "$CONTROLLER_POD" -n "$CSI_NS" -c "$CONTROLLER_CONTAINER" --timestamps --since-time="$START_RFC3339" --tail=800 >"$raw"; then rm -f "$raw" "$window" "$filtered"; fail '로그 producer 실패'; return 1; fi; window_log "$raw" "$window"; rc=$?; if [ "$rc" -ne 0 ]; then rm -f "$raw" "$window" "$filtered"; fail "시간 filter 오류 rc=$rc"; return 1; fi; grep -E -i -C 3 "$EVIDENCE_RE|CreateVolume|provision|capacity|system|protection|domain|pool|unauthorized|forbidden|timeout|refused|not found|error" "$window" >"$filtered"; rc=$?; if [ "$rc" -eq 0 ]; then cat "$filtered"; elif [ "$rc" -eq 1 ]; then printf '%s\n' 'filter no-match: producer 성공이며 capacity 정상은 아니다.' >&2; else rm -f "$raw" "$window" "$filtered"; fail "grep 오류 rc=$rc"; return 1; fi; after="$(kubectl get pod "$CONTROLLER_POD" -n "$CSI_NS" -o json)" || { rm -f "$raw" "$window" "$filtered"; fail '수집 후 Pod 조회 실패'; return 1; }; [ "$(jq -r '.metadata.uid' <<<"$after")" = "$before_uid" ] && [ "$(jq -r --arg c "$CONTROLLER_CONTAINER" '.status.containerStatuses[]|select(.name==$c)|.containerID' <<<"$after")" = "$before_cid" ] || { rm -f "$raw" "$window" "$filtered"; fail '수집 중 controller lifecycle 변경'; return 1; }; rm -f "$raw" "$window" "$filtered"; EVIDENCE_READY=true; }
prom_queries(){ refresh || return 1; local ns pvc sc provisioner; ns="${NS//\"/\\\"}"; pvc="${PVC//\"/\\\"}"; sc="${SC//\"/\\\"}"; provisioner="${PROVISIONER//\"/\\\"}"; printf 'kube_storageclass_info{storageclass="%s",provisioner="%s"}\n' "$sc" "$provisioner"; printf 'kube_persistentvolumeclaim_status_phase{namespace="%s",persistentvolumeclaim="%s",phase="Pending"}==1\n' "$ns" "$pvc"; printf 'kube_persistentvolumeclaim_resource_requests_storage_bytes{namespace="%s",persistentvolumeclaim="%s"}\n' "$ns" "$pvc"; }
tools_gate(){ local tool missing=''; for tool in kubectl jq sed date mktemp grep; do command -v "$tool" >/dev/null 2>&1 || missing="$missing $tool"; done; [ -z "$missing" ] || { printf '중단: 도구 없음=%s\n' "$missing" >&2; return 1; }; }
tools_gate && snapshot
```

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | 같은 PVC UID의 CreateVolume에서 system/PD/pool/capacity/API 오류 발생 |
| 적용 | Pending/unbound PVC, StorageClass와 current controller container가 연결됨 |
| 제외 | `WaitForFirstConsumer` selected-node 없음, topology/binding 문제는 [[10. 스토리지/01. Kubernetes·CSI 공통/01. PVC가 Pending에서 Bound로 수렴하지 않는 경우]] |

검색 문자열: `ProvisioningFailed`, `CreateVolume`, `insufficient capacity`, `system not found`, `protection domain`, `storage pool`, `unauthorized`, `timeout`.

`provision_evidence` 뒤 provider owner가 `StorageClass revision → redacted config reference → system → PD → pool → usable/free/reserved capacity와 tenant authorization`을 같은 request ID·시간으로 확인한다. StorageClass parameters 전체, Secret data, endpoint와 Authorization header는 출력하지 않는다.

## 변경과 정상화

Secret/StorageClass/pool/PD 변경, driver restart와 PVC 삭제 명령은 제공하지 않는다. 변경은 fresh identity/evidence, 승인 ID, 이전 선택·capacity/authorization 원본, 다른 pool consumer, idempotence와 rollback 절차가 있을 때 provider owner가 수행한다. PVC UID가 바뀌면 과거 증거와 연결하지 않는다.

정상화는 `provider capacity/auth → CreateVolume → provider volume → PV claimRef/driver → Bound → attach/mount → 실제 consumer 기능 → 재발 관찰` 순서다. 교체 controller는 current owner에서 단일 후보를 다시 선택해 새 UID/containerID로 등록한다. PromQL 빈 결과는 정상이나 capacity 충분을 뜻하지 않는다.
