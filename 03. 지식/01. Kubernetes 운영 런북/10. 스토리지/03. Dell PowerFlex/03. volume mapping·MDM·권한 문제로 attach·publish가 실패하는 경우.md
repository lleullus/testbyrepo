# volume mapping·MDM·권한 문제로 attach·publish가 실패하는 경우

## 입력값과 빠른 수집

경로는 `승인된 작업 host → API server → Pod UID/node → PVC UID → PV UID/CSI handle → VolumeAttachment UID/node/attacher → CSI controller current container → PowerFlex mapping/MDM/authorization`이다.

| 구분 | 값 | gate |
|---|---|---|
| 최초 입력 | namespace와 Pod | 승인된 consumer exact name |
| API 자동값 | Pod/PVC/PV/SC/VA/CSIDriver UID·RV, node/driver/owner/containerID | current API exact one |
| 계산 후보 | escaped volume evidence regex | PVC/PV/handle/VA literal escape |
| 운영자 선택 | Pod의 PVC, controller Pod/container, canonical UTC 창 | volume mapping과 최초 attach Event |

```bash
NS='PASTE_NAMESPACE_HERE'
POD='PASTE_POD_HERE'
PVC='PASTE_TARGET_PVC_HERE'
CSI_NS='PASTE_CSI_NAMESPACE_HERE'
CONTROLLER_POD='PASTE_CONTROLLER_POD_HERE'
CONTROLLER_CONTAINER='PASTE_CONTROLLER_CONTAINER_HERE'
START_RFC3339='PASTE_START_RFC3339_HERE'
END_RFC3339='PASTE_END_RFC3339_HERE'

invalidate(){ unset POD_JSON PVC_JSON PV_JSON SC_JSON VA_JSON DRIVER_JSON CONTROLLER_JSON POD_UID POD_RV POD_OWNER_KIND POD_OWNER_NAME POD_OWNER_UID NODE NODE_UID PVC_UID PVC_RV PV PV_UID PV_RV SC SC_UID SC_RV DRIVER DRIVER_UID DRIVER_RV HANDLE VA VA_UID VA_RV VA_NODE VA_ATTACHER CONTROLLER_UID CONTROLLER_RV OWNER_KIND OWNER_NAME OWNER_UID OWNER_CONTROLLER_UID CONTAINER_ID IMAGE_ID EVIDENCE_RE OBSERVED_AT; ID_READY=false; TIME_READY=false; SELECTION_READY=false; AUTH_READY=false; EVIDENCE_READY=false; MAPPING_READY=false; MUTATION_READY=false; ROLLBACK_READY=false; NORMALIZATION_READY=false; }
fail(){ local message="$1"; invalidate; printf '중단: %s\n' "$message" >&2; return 1; }
canonical(){ local value="$1" round; [[ "$value" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] || return 1; round="$(date -u -d "$value" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" || return 1; [ "$round" = "$value" ]; }
time_gate(){ TIME_READY=false; canonical "$START_RFC3339" || { fail 'start 형식 오류'; return 1; }; canonical "$END_RFC3339" || { fail 'end 형식 오류'; return 1; }; [ "$(date -u -d "$START_RFC3339" +%s)" -lt "$(date -u -d "$END_RFC3339" +%s)" ] || { fail '시간 순서 오류'; return 1; }; TIME_READY=true; }
ere(){ printf '%s' "$1" | sed 's/[][\\.^$*+?(){}|]/\\&/g'; }
refresh(){ invalidate; local pod pvc pv sc va_list va driver controller node object rs parent count evidence; pod="$(kubectl get pod "$POD" -n "$NS" -o json)" || { fail 'Pod 조회 실패'; return 1; }; NODE="$(jq -r '.spec.nodeName // empty' <<<"$pod")"; [ -n "$NODE" ] || { fail 'Pod node 없음'; return 1; }; jq -e --arg pvc "$PVC" 'any(.spec.volumes[]?;.persistentVolumeClaim.claimName==$pvc)' <<<"$pod" >/dev/null || { fail '선택 PVC가 Pod volume이 아니다.'; return 1; }; pvc="$(kubectl get pvc "$PVC" -n "$NS" -o json)" || { fail 'PVC 조회 실패'; return 1; }; [ "$(jq -r '.status.phase' <<<"$pvc")" = Bound ] || { fail 'PVC가 Bound가 아니다.'; return 1; }; PV="$(jq -r '.spec.volumeName // empty' <<<"$pvc")"; SC="$(jq -r '.spec.storageClassName // empty' <<<"$pvc")"; pv="$(kubectl get pv "$PV" -o json)" || { fail 'PV 조회 실패'; return 1; }; PVC_UID="$(jq -r '.metadata.uid' <<<"$pvc")"; jq -e --arg ns "$NS" --arg name "$PVC" --arg uid "$PVC_UID" '.spec.claimRef.namespace==$ns and .spec.claimRef.name==$name and .spec.claimRef.uid==$uid and (.spec.csi.driver|length)>0 and (.spec.csi.volumeHandle|length)>0' <<<"$pv" >/dev/null || { fail 'PV claimRef/CSI 불일치'; return 1; }; sc="$(kubectl get storageclass "$SC" -o json)" || { fail 'StorageClass 조회 실패'; return 1; }; DRIVER="$(jq -r '.spec.csi.driver' <<<"$pv")"; [ "$(jq -r '.provisioner' <<<"$sc")" = "$DRIVER" ] || { fail 'StorageClass/PV driver 불일치'; return 1; }; driver="$(kubectl get csidriver "$DRIVER" -o json)" || { fail 'CSIDriver 조회 실패'; return 1; }; node="$(kubectl get node "$NODE" -o json)" || { fail 'Node 조회 실패'; return 1; }; va_list="$(kubectl get volumeattachments -o json)" || { fail 'VolumeAttachment 조회 실패'; return 1; }; count="$(jq -r --arg pv "$PV" --arg node "$NODE" --arg d "$DRIVER" '[.items[]|select(.spec.source.persistentVolumeName==$pv and .spec.nodeName==$node and .spec.attacher==$d)]|length' <<<"$va_list")"; [ "$count" -eq 1 ] || { fail "PV/node/attacher VolumeAttachment count=$count"; return 1; }; va="$(jq -c --arg pv "$PV" --arg node "$NODE" --arg d "$DRIVER" '.items[]|select(.spec.source.persistentVolumeName==$pv and .spec.nodeName==$node and .spec.attacher==$d)' <<<"$va_list")"; controller="$(kubectl get pod "$CONTROLLER_POD" -n "$CSI_NS" -o json)" || { fail 'controller Pod 조회 실패'; return 1; }; jq -e --arg c "$CONTROLLER_CONTAINER" '([.spec.containers[]|select(.name==$c)]|length)==1 and ([.status.containerStatuses[]|select(.name==$c and (.containerID|length)>0)]|length)==1' <<<"$controller" >/dev/null || { fail 'regular container/current containerID 불일치'; return 1; }; OWNER_KIND="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].kind // empty' <<<"$controller")"; OWNER_NAME="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].name // empty' <<<"$controller")"; OWNER_UID="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid // empty' <<<"$controller")"; if [ "$OWNER_KIND" = ReplicaSet ]; then rs="$(kubectl get replicaset "$OWNER_NAME" -n "$CSI_NS" -o json)" || { fail 'ReplicaSet 조회 실패'; return 1; }; [ "$(jq -r '.metadata.uid' <<<"$rs")" = "$OWNER_UID" ] || { fail 'ReplicaSet UID 불일치'; return 1; }; parent="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true and .kind=="Deployment")][0].uid // empty' <<<"$rs")"; object="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true and .kind=="Deployment")][0].name // empty' <<<"$rs")"; [ -n "$parent" ] && [ "$(kubectl get deployment "$object" -n "$CSI_NS" -o jsonpath='{.metadata.uid}')" = "$parent" ] || { fail 'Deployment current UID 불일치'; return 1; }; OWNER_CONTROLLER_UID="$parent"; elif [ "$OWNER_KIND" = StatefulSet ]; then [ "$(kubectl get statefulset "$OWNER_NAME" -n "$CSI_NS" -o jsonpath='{.metadata.uid}')" = "$OWNER_UID" ] || { fail 'StatefulSet current UID 불일치'; return 1; }; OWNER_CONTROLLER_UID="$OWNER_UID"; else fail '지원하지 않는 controller owner'; return 1; fi; POD_JSON="$pod"; PVC_JSON="$pvc"; PV_JSON="$pv"; SC_JSON="$sc"; VA_JSON="$va"; DRIVER_JSON="$driver"; CONTROLLER_JSON="$controller"; POD_UID="$(jq -r '.metadata.uid' <<<"$pod")"; POD_RV="$(jq -r '.metadata.resourceVersion' <<<"$pod")"; POD_OWNER_KIND="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].kind // empty' <<<"$pod")"; POD_OWNER_NAME="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].name // empty' <<<"$pod")"; POD_OWNER_UID="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid // empty' <<<"$pod")"; NODE_UID="$(jq -r '.metadata.uid' <<<"$node")"; PVC_RV="$(jq -r '.metadata.resourceVersion' <<<"$pvc")"; PV_UID="$(jq -r '.metadata.uid' <<<"$pv")"; PV_RV="$(jq -r '.metadata.resourceVersion' <<<"$pv")"; SC_UID="$(jq -r '.metadata.uid' <<<"$sc")"; SC_RV="$(jq -r '.metadata.resourceVersion' <<<"$sc")"; DRIVER_UID="$(jq -r '.metadata.uid' <<<"$driver")"; DRIVER_RV="$(jq -r '.metadata.resourceVersion' <<<"$driver")"; HANDLE="$(jq -r '.spec.csi.volumeHandle' <<<"$pv")"; VA="$(jq -r '.metadata.name' <<<"$va")"; VA_UID="$(jq -r '.metadata.uid' <<<"$va")"; VA_RV="$(jq -r '.metadata.resourceVersion' <<<"$va")"; VA_NODE="$(jq -r '.spec.nodeName' <<<"$va")"; VA_ATTACHER="$(jq -r '.spec.attacher' <<<"$va")"; CONTROLLER_UID="$(jq -r '.metadata.uid' <<<"$controller")"; CONTROLLER_RV="$(jq -r '.metadata.resourceVersion' <<<"$controller")"; CONTAINER_ID="$(jq -r --arg c "$CONTROLLER_CONTAINER" '.status.containerStatuses[]|select(.name==$c)|.containerID' <<<"$controller")"; IMAGE_ID="$(jq -r --arg c "$CONTROLLER_CONTAINER" '.status.containerStatuses[]|select(.name==$c)|.imageID' <<<"$controller")"; evidence="$(printf '%s\n%s\n%s\n%s' "$PVC" "$PV" "$HANDLE" "$VA" | while IFS= read -r value; do ere "$value"; printf '\n'; done)" || { fail 'regex 생성 실패'; return 1; }; EVIDENCE_RE="$(paste -sd '|' <<<"$evidence")"; OBSERVED_AT="$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; ID_READY=true; SELECTION_READY=true; printf 'ID_READY=true Pod=%s uid=%s owner=%s/%s:%s node=%s nodeUID=%s\nPVC=%s uid=%s PV=%s uid=%s VA=%s uid=%s driver=%s uid=%s\ncontroller=%s uid=%s owner=%s/%s:%s regular-container=%s containerID=%s observed=%s\n' "$POD" "$POD_UID" "$POD_OWNER_KIND" "$POD_OWNER_NAME" "$POD_OWNER_UID" "$NODE" "$NODE_UID" "$PVC" "$PVC_UID" "$PV" "$PV_UID" "$VA" "$VA_UID" "$DRIVER" "$DRIVER_UID" "$CONTROLLER_POD" "$CONTROLLER_UID" "$OWNER_KIND" "$OWNER_NAME" "$OWNER_UID" "$CONTROLLER_CONTAINER" "$CONTAINER_ID" "$OBSERVED_AT"; }
auth_gate(){ AUTH_READY=false; [ "$ID_READY" = true ] || return 1; [ "$(kubectl auth can-i get "pods/$CONTROLLER_POD" --subresource=log -n "$CSI_NS")" = yes ] || { fail "현재 주체 get pods/$CONTROLLER_POD/log 거부 namespace=$CSI_NS"; return 1; }; AUTH_READY=true; }
window_log(){ local source="$1" output="$2" start end line stamp epoch; start="$(date -u -d "$START_RFC3339" +%s)" || return 2; end="$(date -u -d "$END_RFC3339" +%s)" || return 2; : >"$output" || return 2; while IFS= read -r line; do stamp="${line%% *}"; epoch="$(date -u -d "$stamp" +%s 2>/dev/null)" || return 2; if [ "$epoch" -ge "$start" ] && [ "$epoch" -le "$end" ]; then printf '%s\n' "$line" >>"$output" || return 2; fi; done <"$source"; }
snapshot(){ refresh && time_gate || return 1; jq -r '"=== Consumer ===\nPod UID: \(.metadata.uid) RV: \(.metadata.resourceVersion) Node: \(.spec.nodeName)"' <<<"$POD_JSON"; jq -r '"=== VolumeAttachment ===\nUID: \(.metadata.uid) RV: \(.metadata.resourceVersion)\nPV: \(.spec.source.persistentVolumeName) Node: \(.spec.nodeName) Attacher: \(.spec.attacher)\nAttached: \(.status.attached // false) Error: \(.status.attachError.message // "-")"' <<<"$VA_JSON"; }
mapping_evidence(){ snapshot >/dev/null && auth_gate || return 1; local before_uid="$CONTROLLER_UID" before_cid="$CONTAINER_ID" raw window filtered rc after; raw="$(mktemp)" || return 1; window="$(mktemp)" || { rm -f "$raw"; return 1; }; filtered="$(mktemp)" || { rm -f "$raw" "$window"; return 1; }; if ! kubectl logs "$CONTROLLER_POD" -n "$CSI_NS" -c "$CONTROLLER_CONTAINER" --timestamps --since-time="$START_RFC3339" --tail=800 >"$raw"; then rm -f "$raw" "$window" "$filtered"; fail '로그 producer 실패'; return 1; fi; window_log "$raw" "$window"; rc=$?; if [ "$rc" -ne 0 ]; then rm -f "$raw" "$window" "$filtered"; fail "시간 filter 오류 rc=$rc"; return 1; fi; grep -E -i -C 3 "$EVIDENCE_RE|ControllerPublishVolume|attach|publish|mapping|MDM|permission|denied|unauthorized|timeout|refused|not found|error" "$window" >"$filtered"; rc=$?; if [ "$rc" -eq 0 ]; then cat "$filtered"; elif [ "$rc" -eq 1 ]; then printf '%s\n' 'filter no-match: producer 성공이며 mapping 성공은 아니다.' >&2; else rm -f "$raw" "$window" "$filtered"; fail "grep 오류 rc=$rc"; return 1; fi; after="$(kubectl get pod "$CONTROLLER_POD" -n "$CSI_NS" -o json)" || { rm -f "$raw" "$window" "$filtered"; fail '수집 후 controller 조회 실패'; return 1; }; [ "$(jq -r '.metadata.uid' <<<"$after")" = "$before_uid" ] && [ "$(jq -r --arg c "$CONTROLLER_CONTAINER" '.status.containerStatuses[]|select(.name==$c)|.containerID' <<<"$after")" = "$before_cid" ] || { rm -f "$raw" "$window" "$filtered"; fail '수집 중 controller lifecycle 변경'; return 1; }; rm -f "$raw" "$window" "$filtered"; EVIDENCE_READY=true; }
prom_queries(){ refresh || return 1; local ns pod va; ns="${NS//\"/\\\"}"; pod="${POD//\"/\\\"}"; va="${VA//\"/\\\"}"; printf 'kube_volumeattachment_info{volumeattachment="%s"}\n' "$va"; printf 'kube_volumeattachment_status_attached{volumeattachment="%s"}==0\n' "$va"; printf 'kube_pod_container_status_waiting_reason{namespace="%s",pod="%s",reason="ContainerCreating"}==1\n' "$ns" "$pod"; }
tools_gate(){ local tool missing=''; for tool in kubectl jq sed paste date mktemp grep; do command -v "$tool" >/dev/null 2>&1 || missing="$missing $tool"; done; [ -z "$missing" ] || { printf '중단: 도구 없음=%s\n' "$missing" >&2; return 1; }; }
tools_gate && snapshot
```

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | exact VA가 `attached=false`이고 같은 volume의 `ControllerPublishVolume`, mapping, MDM 또는 권한 오류가 있음 |
| 적용 | Pod→PVC→PV→VA→node/driver→controller 관계가 fresh gate를 통과함 |
| 제외 | VA 없음/Multi-Attach는 공통 VolumeAttachment 런북, attached 뒤 device 없음은 [[04. SDC·scini·CSI node가 device를 발견하지 못하는 경우]] |

검색 문자열: `FailedAttachVolume`, `ControllerPublishVolume`, `mapping`, `MDM`, `permission denied`, `unauthorized`, `timeout`.

`mapping_evidence` 뒤 provider owner가 redacted handle, volume scope, consumer node/SDC identity와 mapping request ID를 대조한다. endpoint, Secret data, SDC GUID 원문과 Authorization header는 출력하지 않는다. 다른 consumer/RWO/Multi-Attach/stale mapping이 하나라도 불명확하면 변경하지 않는다.

## 변경과 정상화

force detach, VA 삭제, unmap/remap과 MDM/SDC 재등록 명령은 제공하지 않는다. 변경은 fresh identity/evidence, 다른 consumer 부재, 이전 mapping 원본, 승인 ID, idempotence, rollback 담당자와 중단 조건이 필요하다. node/VA/UID가 바뀌면 중단한다.

정상화는 `provider mapping/MDM/auth → ControllerPublish → VA attached → node device/stage/publish → mount → consumer 기능 → 재발 관찰` 순서다. 교체 controller는 current owner의 단일 후보를 새 UID/containerID로 다시 등록한다. `attached=true`와 빈 PromQL vector만으로 닫지 않는다.
