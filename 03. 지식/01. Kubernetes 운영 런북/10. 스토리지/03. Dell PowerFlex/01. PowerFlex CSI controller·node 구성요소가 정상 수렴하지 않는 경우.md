# PowerFlex CSI controller·node 구성요소가 정상 수렴하지 않는 경우

## 입력값과 빠른 수집

실행 위치는 승인된 작업 PC/bastion의 같은 Bash다. 경로는 `작업 host → API server → controller workload/ReplicaSet/Pod → node DaemonSet/Pod → kubelet plugin registration → CSIDriver/CSINode → PowerFlex endpoint`다.

| 구분 | 값 | gate |
|---|---|---|
| 최초 입력 | CSI namespace, controller kind/name, node DaemonSet, driver | 승인된 설치 inventory |
| API 자동값 | workload/Pod/owner UID·RV, node, regular containerID/imageID, CSIDriver/CSINode | current exact relation |
| 계산 후보 | workload selector, escaped target regex | selector/API와 literal escape 검증 |
| 운영자 선택 | 실패 Pod/container, canonical UTC 창 | current Pod 세로 출력과 최초 오류 |

```bash
CSI_NS='PASTE_CSI_NAMESPACE_HERE'
CONTROLLER_REF='PASTE_CONTROLLER_TYPE_SLASH_NAME_HERE'
NODE_DS='PASTE_NODE_DAEMONSET_HERE'
DRIVER='PASTE_POWERFLEX_DRIVER_HERE'
TARGET_POD='PASTE_FAILED_CSI_POD_HERE'
TARGET_CONTAINER='PASTE_FAILED_CONTAINER_HERE'
START_RFC3339='PASTE_START_RFC3339_HERE'
END_RFC3339='PASTE_END_RFC3339_HERE'

invalidate(){ unset CONTROLLER_JSON DS_JSON DRIVER_JSON POD_JSON PODS_JSON CSINODES_JSON CONTROLLER_KIND CONTROLLER_NAME CONTROLLER_UID CONTROLLER_RV DS_UID DS_RV DRIVER_UID DRIVER_RV POD_UID POD_RV POD_NODE OWNER_KIND OWNER_NAME OWNER_UID CURRENT_CONTROLLER_UID CONTAINER_ID IMAGE_ID TARGET_RE OBSERVED_AT; ID_READY=false; TIME_READY=false; SELECTION_READY=false; AUTH_READY=false; EVIDENCE_READY=false; REGISTRATION_READY=false; MUTATION_READY=false; ROLLBACK_READY=false; NORMALIZATION_READY=false; }
fail(){ local message="$1"; invalidate; printf '중단: %s\n' "$message" >&2; return 1; }
canonical(){ local value="$1" round; [[ "$value" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$ ]] || return 1; round="$(date -u -d "$value" '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null)" || return 1; [ "$round" = "$value" ]; }
time_gate(){ TIME_READY=false; canonical "$START_RFC3339" || { fail 'start 형식 오류'; return 1; }; canonical "$END_RFC3339" || { fail 'end 형식 오류'; return 1; }; [ "$(date -u -d "$START_RFC3339" +%s)" -lt "$(date -u -d "$END_RFC3339" +%s)" ] || { fail '시간 순서 오류'; return 1; }; TIME_READY=true; }
ere(){ printf '%s' "$1" | sed 's/[][\\.^$*+?(){}|]/\\&/g'; }
refresh(){ invalidate; local kind name controller ds driver pod pods csinodes owner rs parent; [[ "$CONTROLLER_REF" =~ ^(deployment|statefulset)/[a-z0-9]([-a-z0-9]*[a-z0-9])?$ ]] || { fail 'controller ref 형식 오류'; return 1; }; kind="${CONTROLLER_REF%%/*}"; name="${CONTROLLER_REF#*/}"; controller="$(kubectl get "$kind" "$name" -n "$CSI_NS" -o json)" || { fail 'controller 조회 실패'; return 1; }; ds="$(kubectl get daemonset "$NODE_DS" -n "$CSI_NS" -o json)" || { fail 'DaemonSet 조회 실패'; return 1; }; driver="$(kubectl get csidriver "$DRIVER" -o json)" || { fail 'CSIDriver 조회 실패'; return 1; }; pod="$(kubectl get pod "$TARGET_POD" -n "$CSI_NS" -o json)" || { fail 'target Pod 조회 실패'; return 1; }; jq -e --arg c "$TARGET_CONTAINER" '([.spec.containers[]|select(.name==$c)]|length)==1 and ([.status.containerStatuses[]|select(.name==$c and (.containerID|length)>0)]|length)==1' <<<"$pod" >/dev/null || { fail 'regular container/current containerID 불일치'; return 1; }; OWNER_KIND="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].kind // empty' <<<"$pod")"; OWNER_NAME="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].name // empty' <<<"$pod")"; OWNER_UID="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true)][0].uid // empty' <<<"$pod")"; if [ "$OWNER_KIND" = ReplicaSet ] && [ "$kind" = deployment ]; then rs="$(kubectl get replicaset "$OWNER_NAME" -n "$CSI_NS" -o json)" || { fail 'ReplicaSet 조회 실패'; return 1; }; [ "$(jq -r '.metadata.uid' <<<"$rs")" = "$OWNER_UID" ] || { fail 'ReplicaSet UID 불일치'; return 1; }; parent="$(jq -r '[.metadata.ownerReferences[]?|select(.controller==true and .kind=="Deployment")][0].uid // empty' <<<"$rs")"; [ "$parent" = "$(jq -r '.metadata.uid' <<<"$controller")" ] || { fail 'ReplicaSet→Deployment current UID 불일치'; return 1; }; CURRENT_CONTROLLER_UID="$parent"; elif [ "$OWNER_KIND" = StatefulSet ] && [ "$kind" = statefulset ] && [ "$OWNER_UID" = "$(jq -r '.metadata.uid' <<<"$controller")" ]; then CURRENT_CONTROLLER_UID="$OWNER_UID"; elif [ "$OWNER_KIND" = DaemonSet ] && [ "$OWNER_UID" = "$(jq -r '.metadata.uid' <<<"$ds")" ]; then CURRENT_CONTROLLER_UID="$OWNER_UID"; else fail 'target Pod의 current controller 관계 불일치'; return 1; fi; pods="$(kubectl get pods -n "$CSI_NS" -o json)" || { fail 'Pod 목록 조회 실패'; return 1; }; csinodes="$(kubectl get csinodes -o json)" || { fail 'CSINode 목록 조회 실패'; return 1; }; CONTROLLER_JSON="$controller"; DS_JSON="$ds"; DRIVER_JSON="$driver"; POD_JSON="$pod"; PODS_JSON="$pods"; CSINODES_JSON="$csinodes"; CONTROLLER_KIND="$kind"; CONTROLLER_NAME="$name"; CONTROLLER_UID="$(jq -r '.metadata.uid' <<<"$controller")"; CONTROLLER_RV="$(jq -r '.metadata.resourceVersion' <<<"$controller")"; DS_UID="$(jq -r '.metadata.uid' <<<"$ds")"; DS_RV="$(jq -r '.metadata.resourceVersion' <<<"$ds")"; DRIVER_UID="$(jq -r '.metadata.uid' <<<"$driver")"; DRIVER_RV="$(jq -r '.metadata.resourceVersion' <<<"$driver")"; POD_UID="$(jq -r '.metadata.uid' <<<"$pod")"; POD_RV="$(jq -r '.metadata.resourceVersion' <<<"$pod")"; POD_NODE="$(jq -r '.spec.nodeName // empty' <<<"$pod")"; CONTAINER_ID="$(jq -r --arg c "$TARGET_CONTAINER" '.status.containerStatuses[]|select(.name==$c)|.containerID' <<<"$pod")"; IMAGE_ID="$(jq -r --arg c "$TARGET_CONTAINER" '.status.containerStatuses[]|select(.name==$c)|.imageID' <<<"$pod")"; TARGET_RE="$(ere "$TARGET_POD")" || { fail 'regex 생성 실패'; return 1; }; OBSERVED_AT="$(date -u '+%Y-%m-%dT%H:%M:%SZ')"; ID_READY=true; SELECTION_READY=true; printf 'ID_READY=true controller=%s/%s uid=%s rv=%s DS=%s uid=%s driver=%s uid=%s\ntarget=%s uid=%s owner=%s/%s:%s node=%s regular-container=%s containerID=%s observed=%s\n' "$CONTROLLER_KIND" "$CONTROLLER_NAME" "$CONTROLLER_UID" "$CONTROLLER_RV" "$NODE_DS" "$DS_UID" "$DRIVER" "$DRIVER_UID" "$TARGET_POD" "$POD_UID" "$OWNER_KIND" "$OWNER_NAME" "$OWNER_UID" "$POD_NODE" "$TARGET_CONTAINER" "$CONTAINER_ID" "$OBSERVED_AT"; }
auth_gate(){ AUTH_READY=false; [ "$ID_READY" = true ] || return 1; local allowed; allowed="$(kubectl auth can-i get "pods/$TARGET_POD" --subresource=log -n "$CSI_NS")" || { fail 'authorization 조회 실패'; return 1; }; [ "$allowed" = yes ] || { fail "현재 주체 get pods/$TARGET_POD/log 거부 namespace=$CSI_NS"; return 1; }; AUTH_READY=true; }
window_log(){ local source="$1" output="$2" start end line stamp epoch; start="$(date -u -d "$START_RFC3339" +%s)" || return 2; end="$(date -u -d "$END_RFC3339" +%s)" || return 2; : >"$output" || return 2; while IFS= read -r line; do stamp="${line%% *}"; epoch="$(date -u -d "$stamp" +%s 2>/dev/null)" || return 2; if [ "$epoch" -ge "$start" ] && [ "$epoch" -le "$end" ]; then printf '%s\n' "$line" >>"$output" || return 2; fi; done <"$source"; }
snapshot(){ refresh && time_gate || return 1; jq -r '"=== Controller ===\nUID: \(.metadata.uid) RV: \(.metadata.resourceVersion)\nGeneration: \(.metadata.generation)/\(.status.observedGeneration // 0)\nDesired/Updated/Ready/Available: \(.spec.replicas // 1)/\(.status.updatedReplicas // 0)/\(.status.readyReplicas // 0)/\(.status.availableReplicas // 0)"' <<<"$CONTROLLER_JSON"; jq -r '"=== Node DS ===\nUID: \(.metadata.uid) RV: \(.metadata.resourceVersion)\nDesired/Current/Updated/Ready/Misscheduled: \(.status.desiredNumberScheduled // 0)/\(.status.currentNumberScheduled // 0)/\(.status.updatedNumberScheduled // 0)/\(.status.numberReady // 0)/\(.status.numberMisscheduled // 0)"' <<<"$DS_JSON"; jq -r '"=== Selected Pod ===\nName: \(.metadata.name)\nUID: \(.metadata.uid) RV: \(.metadata.resourceVersion)\nOwner: \([.metadata.ownerReferences[]?|select(.controller==true)][0].kind)/\([.metadata.ownerReferences[]?|select(.controller==true)][0].name):\([.metadata.ownerReferences[]?|select(.controller==true)][0].uid)\nNode: \(.spec.nodeName // "-") Phase: \(.status.phase)\nContainers: \([.status.containerStatuses[]?|.name+" ready="+(.ready|tostring)+" restart="+(.restartCount|tostring)]|join("; "))"' <<<"$POD_JSON"; }
registration_gate(){ refresh || return 1; local desired ready ds_nodes registered_nodes; desired="$(jq -r '.status.desiredNumberScheduled // 0' <<<"$DS_JSON")"; ready="$(jq -r '.status.numberReady // 0' <<<"$DS_JSON")"; ds_nodes="$(jq -r --arg uid "$DS_UID" '[.items[]|select(any(.metadata.ownerReferences[]?;.uid==$uid))|.spec.nodeName]|sort|join("\n")' <<<"$PODS_JSON")"; registered_nodes="$(jq -r --arg d "$DRIVER" '[.items[]|select(any(.spec.drivers[]?;.name==$d))|.metadata.name]|sort|join("\n")' <<<"$CSINODES_JSON")"; [ "$desired" -gt 0 ] && [ "$ready" -eq "$desired" ] && [ "$ds_nodes" = "$registered_nodes" ] || { fail 'DaemonSet 대상 node와 CSINode driver exact set 불일치'; return 1; }; REGISTRATION_READY=true; printf 'REGISTRATION_READY=true nodes=%s\n' "$desired"; }
component_evidence(){ refresh && time_gate && auth_gate || return 1; local before_uid="$POD_UID" before_cid="$CONTAINER_ID" raw window filtered rc after; raw="$(mktemp)" || return 1; window="$(mktemp)" || { rm -f "$raw"; return 1; }; filtered="$(mktemp)" || { rm -f "$raw" "$window"; return 1; }; if ! kubectl logs "$TARGET_POD" -n "$CSI_NS" -c "$TARGET_CONTAINER" --timestamps --since-time="$START_RFC3339" --tail=800 >"$raw"; then rm -f "$raw" "$window" "$filtered"; fail '로그 producer 실패'; return 1; fi; window_log "$raw" "$window"; rc=$?; if [ "$rc" -ne 0 ]; then rm -f "$raw" "$window" "$filtered"; fail "시간 filter 오류 rc=$rc"; return 1; fi; grep -E -i -C 3 "$TARGET_RE|GetPluginInfo|registration|CreateVolume|ControllerPublishVolume|NodeStageVolume|NodePublishVolume|socket|grpc|timeout|refused|denied|error|fatal" "$window" >"$filtered"; rc=$?; if [ "$rc" -eq 0 ]; then cat "$filtered"; elif [ "$rc" -eq 1 ]; then printf '%s\n' 'filter no-match: producer 성공이며 구성요소 정상은 아니다.' >&2; else rm -f "$raw" "$window" "$filtered"; fail "grep 오류 rc=$rc"; return 1; fi; after="$(kubectl get pod "$TARGET_POD" -n "$CSI_NS" -o json)" || { rm -f "$raw" "$window" "$filtered"; fail '수집 후 Pod 조회 실패'; return 1; }; [ "$(jq -r '.metadata.uid' <<<"$after")" = "$before_uid" ] && [ "$(jq -r --arg c "$TARGET_CONTAINER" '.status.containerStatuses[]|select(.name==$c)|.containerID' <<<"$after")" = "$before_cid" ] || { rm -f "$raw" "$window" "$filtered"; fail '수집 중 Pod/container lifecycle 변경'; return 1; }; rm -f "$raw" "$window" "$filtered"; EVIDENCE_READY=true; }
prom_queries(){ refresh || return 1; local ns ds pod; ns="${CSI_NS//\"/\\\"}"; ds="${NODE_DS//\"/\\\"}"; pod="${TARGET_RE//\\/\\\\}"; pod="${pod//\"/\\\"}"; printf 'kube_daemonset_status_desired_number_scheduled{namespace="%s",daemonset="%s"}-kube_daemonset_status_number_ready{namespace="%s",daemonset="%s"}\n' "$ns" "$ds" "$ns" "$ds"; printf 'sum by(namespace,pod,uid,container)(increase(kube_pod_container_status_restarts_total{namespace="%s",pod=~"^%s$"}[15m]))>0\n' "$ns" "$pod"; }
tools_gate(){ local tool missing=''; for tool in kubectl jq sed date mktemp grep; do command -v "$tool" >/dev/null 2>&1 || missing="$missing $tool"; done; [ -z "$missing" ] || { printf '중단: 도구 없음=%s\n' "$missing" >&2; return 1; }; }
tools_gate && snapshot
```

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | controller/node desired·updated·ready 또는 CSIDriver/CSINode 등록이 수렴하지 않음 |
| 적용 | workload→ReplicaSet/Pod와 DaemonSet→Pod→node→CSINode를 current UID로 연결 가능 |
| 제외 | CreateVolume은 [[02. system·protection domain·storage pool·capacity 문제로 provisioning이 실패하는 경우]], attach는 [[03. volume mapping·MDM·권한 문제로 attach·publish가 실패하는 경우]], device는 [[04. SDC·scini·CSI node가 device를 발견하지 못하는 경우]] |

검색 문자열: `GetPluginInfo`, `registration`, `CSIDriver`, `CSINode`, `grpc`, `permission denied`, `timeout`.

`component_evidence`와 설치 형태에 맞는 `registration_gate`를 실행한다. count가 아닌 node 이름 exact set을 사용한다. Secret data, endpoint, nodeID와 token은 출력하지 않고 Kubernetes RBAC와 PowerFlex authorization을 분리한다.

## 변경과 정상화

재시작·재설치·manifest apply 명령은 제공하지 않는다. 승인된 선언 변경은 fresh workload UID/RV, selector, revision/image digest, 전체 replica availability, 승인 ID, 이전 원본과 rollback 절차를 변경 시스템에서 다시 확인해야 한다. 교체 후 이전 Pod 이름/UID를 재사용하지 않는다. current owner selector에서 단일 후보를 고르고 `TARGET_POD`를 갱신해 전체 등록을 다시 실행한다.

정상화는 `PowerFlex API → CSI controller → node plugin/socket → kubelet registration → CSINode exact set → 기존 PVC provision/attach/mount → consumer 기능 → 재발 관찰` 순서다. `Running`, socket 존재, CSINode 등록 하나만으로 닫지 않는다. PromQL 빈 vector는 metric 미지원·label/cluster/time/API 오류 후보이며 storage 기능 성공이 아니다.
