# 특정 노드에서 Pod 네트워크 생성이 반복 실패하는 경우

## 입력값과 빠른 수집

경로는 `대표 Pod → exact Node → 다른 owner 또는 다른 Pod UID의 같은 오류 → node의 exact CNI current container → active kubelet CRI endpoint → 두 sandbox → Linux host`다. 한 Pod 실패와 여러 node의 공통 pool 실패를 node 장애로 합치지 않는다.

| 종류 | 변수 | 출처·gate |
|---|---|---|
| 최초 입력 | `NS`, `POD` | 대표 실패 Pod |
| API 자동값 | 대표 Pod UID/RV/owner, Node UID/machineID/runtime | current API 전체 성공 |
| 운영자 선택 | 두 번째 Pod, CNI Pod/container/DS, UTC 창, SSH·unit | current 후보와 직접 오류 |
| 계산 후보 | active endpoint, 각 sandbox ID·PID/netns | strict node/CRI relation |

**도구·모든 downstream 폐기:** `unset INPUT_READY AUTO_READY SCOPE_READY CNI_READY WINDOW_READY REMOTE_READY POD_UID POD_RV OWNER_REL NODE NODE_UID NODE_MACHINE_ID RUNTIME_PROVIDER SECOND_NS SECOND_POD SECOND_UID SECOND_RV SECOND_OWNER CNI_NS CNI_POD CNI_UID CNI_RV CNI_CONTAINER CNI_CONTAINER_ID CNI_DS CNI_DS_UID START_RFC3339 END_RFC3339 EVIDENCE_RFC3339 START_EPOCH END_EPOCH EVIDENCE_EPOCH SSH_TARGET EXPECTED_REMOTE_HOST KUBELET_UNIT RUNTIME_UNIT POD_JSON NODE_JSON SECOND_JSON; TOOLS_READY=false; MISSING=''; for T in kubectl jq date grep ssh; do command -v "$T" >/dev/null 2>&1 || MISSING="$MISSING $T"; done; if [ -z "${BASH_VERSION:-}" ] || [ -n "$MISSING" ]; then printf '중단: Bash/도구 gate 실패:%s\n' "$MISSING" >&2; else TOOLS_READY=true; fi`

**최초 입력:** `NS='PASTE_NAMESPACE_HERE'; POD='PASTE_REPRESENTATIVE_POD_HERE'; INPUT_READY=false; [ "$TOOLS_READY" = true ] && [ -n "$NS" ] && [ -n "$POD" ] && INPUT_READY=true || printf '%s\n' '중단: 최초 입력 불완전' >&2`

**대표 Pod·Node 자동 등록:** `AUTO_READY=false; unset POD_UID POD_RV OWNER_REL NODE NODE_UID NODE_MACHINE_ID RUNTIME_PROVIDER POD_JSON NODE_JSON; J=$(kubectl get pod "$POD" -n "$NS" -o json) && U=$(jq -er '.metadata.uid' <<<"$J") && R=$(jq -er '.metadata.resourceVersion' <<<"$J") && N=$(jq -er '.spec.nodeName|select(length>0)' <<<"$J") && O=$(jq -er '[.metadata.ownerReferences[]?|select(.controller==true)|[.kind,.name,.uid]|join("/")]|if length==0 then "none" elif length==1 then .[0] else error("owner") end' <<<"$J") && NJ=$(kubectl get node "$N" -o json) && NU=$(jq -er '.metadata.uid' <<<"$NJ") && NM=$(jq -er '.status.nodeInfo.machineID|select(length>0)' <<<"$NJ") && RP=$(jq -er '.status.nodeInfo.containerRuntimeVersion|split(":")[0]|select(length>0)' <<<"$NJ") || { printf '%s\n' '중단: 대표 Pod/Node 수집 실패' >&2; unset J U R N O NJ NU NM RP; }; if [ -n "${NM:-}" ]; then POD_JSON=$J; POD_UID=$U; POD_RV=$R; OWNER_REL=$O; NODE=$N; NODE_JSON=$NJ; NODE_UID=$NU; NODE_MACHINE_ID=$NM; RUNTIME_PROVIDER=$RP; AUTO_READY=true; printf 'AUTO_READY=true pod=%s/%s uid=%s rv=%s owner=%s node=%s uid=%s machineID=%s runtime=%s\n' "$NS" "$POD" "$POD_UID" "$POD_RV" "$OWNER_REL" "$NODE" "$NODE_UID" "$NODE_MACHINE_ID" "$RUNTIME_PROVIDER"; fi; unset J U R N O NJ NU NM RP`

**direct current scope gate:** `refresh_scope(){ local j u r n o nj nu nm rp sj su sr sn so; AUTO_READY=false; SCOPE_READY=false; j=$(kubectl get pod "$POD" -n "$NS" -o json) || return 1; u=$(jq -er '.metadata.uid' <<<"$j") || return 1; r=$(jq -er '.metadata.resourceVersion' <<<"$j") || return 1; n=$(jq -er '.spec.nodeName' <<<"$j") || return 1; o=$(jq -er '[.metadata.ownerReferences[]?|select(.controller==true)|[.kind,.name,.uid]|join("/")]|if length==0 then "none" elif length==1 then .[0] else error("owner") end' <<<"$j") || return 1; nj=$(kubectl get node "$n" -o json) || return 1; nu=$(jq -er '.metadata.uid' <<<"$nj") || return 1; nm=$(jq -er '.status.nodeInfo.machineID|select(length>0)' <<<"$nj") || return 1; rp=$(jq -er '.status.nodeInfo.containerRuntimeVersion|split(":")[0]' <<<"$nj") || return 1; [ "$u" = "$POD_UID" ] && [ "$n" = "$NODE" ] && [ "$o" = "$OWNER_REL" ] && [ "$nu" = "$NODE_UID" ] && [ "$nm" = "$NODE_MACHINE_ID" ] && [ "$rp" = "$RUNTIME_PROVIDER" ] || return 1; POD_JSON=$j; POD_RV=$r; NODE_JSON=$nj; AUTO_READY=true; if [ -n "${SECOND_UID:-}" ]; then sj=$(kubectl get pod "$SECOND_POD" -n "$SECOND_NS" -o json) || return 1; su=$(jq -er '.metadata.uid' <<<"$sj") || return 1; sr=$(jq -er '.metadata.resourceVersion' <<<"$sj") || return 1; sn=$(jq -er '.spec.nodeName' <<<"$sj") || return 1; so=$(jq -er '[.metadata.ownerReferences[]?|select(.controller==true)|[.kind,.name,.uid]|join("/")]|if length==0 then "none" elif length==1 then .[0] else error("owner") end' <<<"$sj") || return 1; [ "$su" = "$SECOND_UID" ] && [ "$sn" = "$NODE" ] && [ "$so" = "$SECOND_OWNER" ] || return 1; SECOND_JSON=$sj; SECOND_RV=$sr; SCOPE_READY=true; fi; }`

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | 같은 current Node의 서로 다른 두 Pod UID에서 같은 CNI/sandbox 오류가 같은 창에 반복됨 |
| 적용 | 두 lifecycle·Node UID와 plugin 또는 kubelet/runtime 직접 증거를 연결 가능 |
| 제외 | 한 Pod는 [[01. Pod IP가 할당되지 않거나 CNI 설정이 실패하는 경우]], 여러 node의 같은 pool 소진은 [[03. Pod IP가 중복되거나 IPAM 주소 범위가 소진된 경우]] |

검색 문자열: `FailedCreatePodSandBox`, `failed to setup network`, `CNI ADD`, `context deadline exceeded`, `IPAM`, `veth`, `network namespace`, `rpc error`.

## node 범위·시간 확정

**node 후보:** `refresh_scope || exit $?; kubectl get pods -A --field-selector "spec.nodeName=$NODE" -o custom-columns='NS:.metadata.namespace,POD:.metadata.name,UID:.metadata.uid,CREATED:.metadata.creationTimestamp,PHASE:.status.phase,POD-IP:.status.podIP,OWNER-KIND:.metadata.ownerReferences[?(@.controller==true)].kind,OWNER-NAME:.metadata.ownerReferences[?(@.controller==true)].name,OWNER-UID:.metadata.ownerReferences[?(@.controller==true)].uid'`

**두 번째 Pod 선택·검증:** `unset SCOPE_READY SECOND_UID SECOND_RV SECOND_OWNER SECOND_JSON; SECOND_NS='PASTE_SECOND_NAMESPACE_HERE'; SECOND_POD='PASTE_SECOND_POD_HERE'; refresh_scope || exit $?; SJ=$(kubectl get pod "$SECOND_POD" -n "$SECOND_NS" -o json) && SU=$(jq -er '.metadata.uid' <<<"$SJ") && SR=$(jq -er '.metadata.resourceVersion' <<<"$SJ") && SN=$(jq -er '.spec.nodeName' <<<"$SJ") && SO=$(jq -er '[.metadata.ownerReferences[]?|select(.controller==true)|[.kind,.name,.uid]|join("/")]|if length==0 then "none" elif length==1 then .[0] else error("owner") end' <<<"$SJ") || { printf '%s\n' '중단: 두 번째 Pod 조회 실패' >&2; unset SJ SU SR SN SO; }; if [ -n "${SU:-}" ] && [ "$SU" != "$POD_UID" ] && [ "$SN" = "$NODE" ]; then SECOND_JSON=$SJ; SECOND_UID=$SU; SECOND_RV=$SR; SECOND_OWNER=$SO; SCOPE_READY=true; printf 'SCOPE_READY=true first=%s/%s/%s second=%s/%s/%s node=%s/%s\n' "$NS" "$POD" "$POD_UID" "$SECOND_NS" "$SECOND_POD" "$SECOND_UID" "$NODE" "$NODE_UID"; else printf '%s\n' '중단: 서로 다른 UID·같은 node 조건 불충족' >&2; fi; unset SJ SU SR SN SO`

**두 exact UID Event:** `refresh_scope || exit $?; [ "$SCOPE_READY" = true ] || exit 1; for SPEC in "$NS|$POD|$POD_UID" "$SECOND_NS|$SECOND_POD|$SECOND_UID"; do IFS='|' read -r ENS EP EU <<<"$SPEC"; E=$(kubectl get events -n "$ENS" --field-selector "involvedObject.kind=Pod,involvedObject.name=$EP,involvedObject.uid=$EU" --sort-by=.metadata.creationTimestamp -o json); RC=$?; if [ "$RC" -ne 0 ]; then printf 'Event 실패 pod=%s/%s rc=%s\n' "$ENS" "$EP" "$RC" >&2; elif [ "$(jq '.items|length' <<<"$E")" -eq 0 ]; then printf 'Event 0건 pod=%s/%s: 성공 아님\n' "$ENS" "$EP"; else jq -r --arg p "$ENS/$EP/$EU" '.items[]|[$p,.metadata.creationTimestamp,.eventTime,.firstTimestamp,.lastTimestamp,.series.lastObservedTime,.reason,.message]|@tsv' <<<"$E"; fi; done; unset SPEC ENS EP EU E RC`

**엄격한 UTC 창:** `unset WINDOW_READY START_EPOCH END_EPOCH EVIDENCE_EPOCH; START_RFC3339='PASTE_START_RFC3339_HERE'; END_RFC3339='PASTE_END_RFC3339_HERE'; EVIDENCE_RFC3339='PASTE_EVIDENCE_RFC3339_HERE'; RFC3339_RE='^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$'; WINDOW_READY=false; if [[ "$START_RFC3339" =~ $RFC3339_RE ]] && [[ "$END_RFC3339" =~ $RFC3339_RE ]] && [[ "$EVIDENCE_RFC3339" =~ $RFC3339_RE ]]; then START_EPOCH=$(date -u -d "$START_RFC3339" +%s 2>/dev/null) && END_EPOCH=$(date -u -d "$END_RFC3339" +%s 2>/dev/null) && EVIDENCE_EPOCH=$(date -u -d "$EVIDENCE_RFC3339" +%s 2>/dev/null) || true; fi; if [ -n "${START_EPOCH:-}" ] && [ "$(date -u -d "@$START_EPOCH" '+%Y-%m-%dT%H:%M:%SZ')" = "$START_RFC3339" ] && [ "$(date -u -d "@$END_EPOCH" '+%Y-%m-%dT%H:%M:%SZ')" = "$END_RFC3339" ] && [ "$(date -u -d "@$EVIDENCE_EPOCH" '+%Y-%m-%dT%H:%M:%SZ')" = "$EVIDENCE_RFC3339" ] && [ "$START_EPOCH" -lt "$END_EPOCH" ] && [ "$START_EPOCH" -le "$EVIDENCE_EPOCH" ] && [ "$EVIDENCE_EPOCH" -lt "$END_EPOCH" ]; then WINDOW_READY=true; else unset START_EPOCH END_EPOCH EVIDENCE_EPOCH; printf '%s\n' '중단: canonical UTC/relation 위반' >&2; fi`

## 관측별 진단 경로

| 관측 | 직접 증거 | 다음 owner |
|---|---|---|
| plugin에서 두 UID의 ADD 실패 | CNI Pod/current container/DS | node plugin 또는 IPAM |
| plugin 요청 없음, runtime timeout | active endpoint와 두 sandbox | kubelet/runtime |
| ADD 뒤 여러 netns/veth 실패 | host interface·route·kernel | Linux network |
| 다른 node도 같은 pool 오류 | pool ID·lease 범위 | cluster IPAM |

### 1. CNI relation·로그

**CNI 선택·등록:** `unset CNI_READY CNI_UID CNI_RV CNI_CONTAINER_ID CNI_DS_UID; CNI_NS='PASTE_CNI_NAMESPACE_HERE'; CNI_POD='PASTE_CNI_POD_HERE'; CNI_CONTAINER='PASTE_CNI_CONTAINER_HERE'; CNI_DS='PASTE_CNI_DAEMONSET_HERE'; refresh_scope || exit $?; CJ=$(kubectl get pod "$CNI_POD" -n "$CNI_NS" -o json) && CU=$(jq -er '.metadata.uid' <<<"$CJ") && CR=$(jq -er '.metadata.resourceVersion' <<<"$CJ") && CN=$(jq -er '.spec.nodeName' <<<"$CJ") && DU=$(jq -er --arg ds "$CNI_DS" '[.metadata.ownerReferences[]?|select(.controller==true and .kind=="DaemonSet" and .name==$ds)|.uid]|if length==1 then .[0] else error("owner") end' <<<"$CJ") && CID=$(jq -er --arg c "$CNI_CONTAINER" '[.status.containerStatuses[]?|select(.name==$c)|.containerID|select(length>0)]|if length==1 then .[0] else error("container") end' <<<"$CJ") && DSU=$(kubectl get ds "$CNI_DS" -n "$CNI_NS" -o jsonpath='{.metadata.uid}') || { printf '%s\n' '중단: CNI relation 실패' >&2; unset CJ CU CR CN DU CID DSU; }; if [ -n "${DSU:-}" ] && [ "$CN" = "$NODE" ] && [ "$DU" = "$DSU" ]; then CNI_UID=$CU; CNI_RV=$CR; CNI_CONTAINER_ID=$CID; CNI_DS_UID=$DSU; CNI_READY=true; printf 'CNI_READY=true uid=%s rv=%s container=%s id=%s owner=DaemonSet/%s/%s/%s\n' "$CNI_UID" "$CNI_RV" "$CNI_CONTAINER" "$CNI_CONTAINER_ID" "$CNI_NS" "$CNI_DS" "$CNI_DS_UID"; fi; unset CJ CU CR CN DU CID DSU`

**두 UID plugin 로그·RC 분리:** `refresh_scope || exit $?; [ "$CNI_READY" = true ] && [ "$WINDOW_READY" = true ] || exit 1; CJ=$(kubectl get pod "$CNI_POD" -n "$CNI_NS" -o json) && CU=$(jq -er '.metadata.uid' <<<"$CJ") && CN=$(jq -er '.spec.nodeName' <<<"$CJ") && CID=$(jq -er --arg c "$CNI_CONTAINER" '[.status.containerStatuses[]?|select(.name==$c)|.containerID][0]' <<<"$CJ") && DU=$(jq -er --arg ds "$CNI_DS" '[.metadata.ownerReferences[]?|select(.controller==true and .kind=="DaemonSet" and .name==$ds)|.uid]|if length==1 then .[0] else error("owner") end' <<<"$CJ") && DSU=$(kubectl get ds "$CNI_DS" -n "$CNI_NS" -o jsonpath='{.metadata.uid}') || { CNI_READY=false; exit 1; }; [ "$CU" = "$CNI_UID" ] && [ "$CN" = "$NODE" ] && [ "$CID" = "$CNI_CONTAINER_ID" ] && [ "$DU" = "$CNI_DS_UID" ] && [ "$DSU" = "$CNI_DS_UID" ] || { CNI_READY=false; exit 1; }; RAW=$(kubectl logs "$CNI_POD" -n "$CNI_NS" -c "$CNI_CONTAINER" --timestamps --since-time="$START_RFC3339" --tail=500 2>&1); RC=$?; if [ "$RC" -ne 0 ]; then printf '수집 실패 rc=%s\n' "$RC" >&2; elif MATCH=$(grep -E -i -C 3 "$POD_UID|$SECOND_UID|CNI[ _-]*ADD|IPAM|veth|route|namespace|timeout|error" <<<"$RAW"); then printf '%s\n' "$MATCH"; else printf '%s\n' '필터 불일치: 정상 아님'; tail -n 50 <<<"$RAW"; fi; unset CJ CU CN CID DU DSU RAW RC MATCH`

### 2. node host·active CRI

**SSH/unit 선택:** `unset REMOTE_READY SSH_TARGET EXPECTED_REMOTE_HOST KUBELET_UNIT RUNTIME_UNIT; SSH_TARGET='PASTE_SSH_TARGET_HERE'; EXPECTED_REMOTE_HOST='PASTE_EXPECTED_REMOTE_HOST_HERE'; KUBELET_UNIT='PASTE_KUBELET_UNIT_HERE'; RUNTIME_UNIT='PASTE_RUNTIME_UNIT_HERE'; run_remote(){ local script=$1 arg q cmd='bash -s --'; shift; for arg in "$@"; do printf -v q '%q' "$arg"; cmd+=" $q"; done; ssh -- "$SSH_TARGET" "$cmd" <<<"$script"; }; REMOTE_READY=false; [ -n "$SSH_TARGET" ] && [ -n "$EXPECTED_REMOTE_HOST" ] && [ -n "$KUBELET_UNIT" ] && [ -n "$RUNTIME_UNIT" ] && REMOTE_READY=true || printf '%s\n' '중단: SSH/unit 불완전' >&2`

**journal·active endpoint·두 sandbox·host network:** `refresh_scope || exit $?; [ "$REMOTE_READY" = true ] && [ "$WINDOW_READY" = true ] || exit 1; run_remote 'expected=$1; expected_mid=$2; kubelet=$3; runtime=$4; start=$5; end=$6; uid1=$7; uid2=$8; actual=$(hostname -f 2>/dev/null || hostname) || exit 20; mid=$(cat /etc/machine-id) || exit 21; [ "$actual" = "$expected" ] && [ "$mid" = "$expected_mid" ] || exit 22; kpid=$(systemctl show "$kubelet" -p MainPID --value) || exit 23; [[ "$kpid" =~ ^[1-9][0-9]*$ ]] && [ -r "/proc/$kpid/cmdline" ] || exit 24; mapfile -d "" -t a < "/proc/$kpid/cmdline"; ep=""; cfg=""; for ((i=0;i<${#a[@]};i++)); rawcfg=$(grep -E "^[[:space:]]*containerRuntimeEndpoint:[[:space:]]*[^[:space:]#]+[[:space:]]*(#.*)?$" "$cfg"); rc=$?; [ "$rc" -eq 0 ] && [ "$(grep -Ec "^[[:space:]]*containerRuntimeEndpoint:[[:space:]]*[^[:space:]#]+[[:space:]]*(#.*)?$" "$cfg")" -eq 1 ] || exit 28; then printf "%s\n" "$match"; else printf "%s\n" "journal 필터 불일치: 정상 아님"; tail -n 50 <<<"$raw"; fi; for uid in "$uid1" "$uid2"; do pods=$(sudo crictl --runtime-endpoint "$ep" pods --label "io.kubernetes.pod.uid=$uid" -o json 2>&1); rc=$?; [ "$rc" -eq 0 ] || exit 36; count=$(jq ".items|length" <<<"$pods") || exit 37; [ "$count" -le 1 ] || exit 38; if [ "$count" -eq 1 ]; then got_uid=$(jq -er ".items[0].metadata.uid" <<<"$pods") || exit 39; 두 sandbox가 모두 없으면 생성 전 공통 실패 후보이지 정상은 아니다. 한 UID만 실패하면 node 공통 원인을 확정하지 않는다. journal producer 실패, filter 불일치, CRI 실패는 별도 종료로 보존한다.

## 변경·rollback·정상화

CNI/kubelet/runtime restart, 일괄 Pod 삭제, route·pool·lease 변경 명령은 제공하지 않는다. 변경 직전 fresh 두 Pod UID/RV·owner·Node UID/machineID, CNI UID/RV/current containerID/DS UID, UTC 증거, active endpoint·provider·정상 node 비교, 승인 ID·영향·idempotence를 확인한다. 검증된 rollback 원본·명령·trigger가 없으면 중단한다.

replacement 뒤 모든 UID/RV/containerID/sandbox/PID/time/READY를 폐기한다. 두 개 이상의 새 workload Pod가 **같은 원래 node**에서 CNI ADD, sandbox/netns, Ready와 실제 기능까지 정상이어야 한다. 다른 node로 이동한 성공은 원인 해소가 아니다.

## 증거·미검증·상위 문서

보존: 두 Pod UID/RV·owner, Node UID/machineID, Event UTC·오류, CNI UID/current containerID/DS UID, remote hostname·unit·active endpoint/provider, sandbox/PID start/netns, host link/route, 승인·rollback·기능 결과. credential과 전체 로그는 저장하지 않는다. CNI/IPAM 제품과 unit은 실제 환경 미검증이다.
