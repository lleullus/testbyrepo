# Pod IP가 할당되지 않거나 CNI 설정이 실패하는 경우

## 입력값과 빠른 수집

경로는 `작업 PC → API server → exact Pod/Event → 같은 node의 CNI Pod/current container → SSH 승인 node → active kubelet CRI endpoint → exact PodSandbox → Linux network namespace`다.

| 종류 | 변수 | 출처·gate |
|---|---|---|
| 최초 입력 | `NS`, `POD` | 장애 화면의 exact namespace/name |
| API 자동값 | Pod UID/RV·owner, Node UID/machineID/runtime, Pod IP·hostNetwork | current API 전체 성공 시만 등록 |
| 계산 후보 | CNI DaemonSet 관계, active CRI endpoint, sandbox ID·netns PID | current API·node argv/config·CRI exact match |
| 운영자 선택 | CNI Pod/container/DaemonSet, UTC 창, SSH host·unit | 후보 출력과 직접 오류 |

**도구·stale 상태 초기화:** `unset INPUT_READY AUTO_READY CNI_READY WINDOW_READY REMOTE_READY POD_UID POD_RV OWNER_REL NODE NODE_UID NODE_MACHINE_ID RUNTIME_PROVIDER POD_IP HOST_NETWORK CNI_NS CNI_POD CNI_UID CNI_RV CNI_CONTAINER CNI_CONTAINER_ID CNI_DS CNI_DS_UID START_RFC3339 END_RFC3339 EVIDENCE_RFC3339 START_EPOCH END_EPOCH EVIDENCE_EPOCH SSH_TARGET EXPECTED_REMOTE_HOST KUBELET_UNIT RUNTIME_UNIT POD_JSON NODE_JSON; TOOLS_READY=false; MISSING=''; for T in kubectl jq date grep ssh; do command -v "$T" >/dev/null 2>&1 || MISSING="$MISSING $T"; done; if [ -z "${BASH_VERSION:-}" ] || [ -n "$MISSING" ]; then printf '중단: Bash/도구 gate 실패:%s\n' "$MISSING" >&2; else TOOLS_READY=true; fi`

**최초 입력:** `NS='PASTE_NAMESPACE_HERE'; POD='PASTE_POD_HERE'; INPUT_READY=false; [ "$TOOLS_READY" = true ] && [ -n "$NS" ] && [ -n "$POD" ] && INPUT_READY=true || printf '%s\n' '중단: 최초 입력 불완전' >&2`

**Pod·Node 자동 등록:** `AUTO_READY=false; unset POD_UID POD_RV OWNER_REL NODE NODE_UID NODE_MACHINE_ID RUNTIME_PROVIDER POD_IP HOST_NETWORK POD_JSON NODE_JSON; if [ "$INPUT_READY" != true ]; then printf '%s\n' '중단: INPUT_READY=false' >&2; else J=$(kubectl get pod "$POD" -n "$NS" -o json) && U=$(jq -er '.metadata.uid' <<<"$J") && R=$(jq -er '.metadata.resourceVersion' <<<"$J") && N=$(jq -er '.spec.nodeName|select(length>0)' <<<"$J") && O=$(jq -er '[.metadata.ownerReferences[]?|select(.controller==true)|[.kind,.name,.uid]|join("/")]|if length==0 then "none" elif length==1 then .[0] else error("owner") end' <<<"$J") && IP=$(jq -r '.status.podIP//""' <<<"$J") && HN=$(jq -r '.spec.hostNetwork//false' <<<"$J") && NJ=$(kubectl get node "$N" -o json) && NU=$(jq -er '.metadata.uid' <<<"$NJ") && NM=$(jq -er '.status.nodeInfo.machineID|select(length>0)' <<<"$NJ") && RP=$(jq -er '.status.nodeInfo.containerRuntimeVersion|split(":")[0]|select(length>0)' <<<"$NJ") || { printf '%s\n' '중단: Pod/Node API 수집 실패' >&2; unset J U R N O IP HN NJ NU NM RP; }; if [ -n "${NM:-}" ]; then POD_JSON=$J; POD_UID=$U; POD_RV=$R; OWNER_REL=$O; NODE=$N; NODE_JSON=$NJ; NODE_UID=$NU; NODE_MACHINE_ID=$NM; RUNTIME_PROVIDER=$RP; POD_IP=$IP; HOST_NETWORK=$HN; AUTO_READY=true; printf 'AUTO_READY=true pod=%s/%s uid=%s rv=%s owner=%s node=%s uid=%s machineID=%s runtime=%s podIP=%s hostNetwork=%s\n' "$NS" "$POD" "$POD_UID" "$POD_RV" "$OWNER_REL" "$NODE" "$NODE_UID" "$NODE_MACHINE_ID" "$RUNTIME_PROVIDER" "${POD_IP:-empty}" "$HOST_NETWORK"; fi; unset J U R N O IP HN NJ NU NM RP; fi`

**direct current Pod gate:** `refresh_pod(){ local j u r n o ip hn nj nu nm rp; AUTO_READY=false; j=$(kubectl get pod "$POD" -n "$NS" -o json) || return 1; u=$(jq -er '.metadata.uid' <<<"$j") || return 1; r=$(jq -er '.metadata.resourceVersion' <<<"$j") || return 1; n=$(jq -er '.spec.nodeName|select(length>0)' <<<"$j") || return 1; o=$(jq -er '[.metadata.ownerReferences[]?|select(.controller==true)|[.kind,.name,.uid]|join("/")]|if length==0 then "none" elif length==1 then .[0] else error("owner") end' <<<"$j") || return 1; hn=$(jq -r '.spec.hostNetwork//false' <<<"$j"); nj=$(kubectl get node "$n" -o json) || return 1; nu=$(jq -er '.metadata.uid' <<<"$nj") || return 1; nm=$(jq -er '.status.nodeInfo.machineID|select(length>0)' <<<"$nj") || return 1; rp=$(jq -er '.status.nodeInfo.containerRuntimeVersion|split(":")[0]' <<<"$nj") || return 1; [ "$u" = "$POD_UID" ] && [ "$n" = "$NODE" ] && [ "$o" = "$OWNER_REL" ] && [ "$hn" = "$HOST_NETWORK" ] && [ "$nu" = "$NODE_UID" ] && [ "$nm" = "$NODE_MACHINE_ID" ] && [ "$rp" = "$RUNTIME_PROVIDER" ] || { printf '%s\n' '중단: Pod/owner/node/runtime identity 변경; 전체 재등록' >&2; return 1; }; ip=$(jq -r '.status.podIP//""' <<<"$j"); POD_JSON=$j; POD_RV=$r; POD_IP=$ip; NODE_JSON=$nj; AUTO_READY=true; }`

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | scheduled Pod의 `podIP`가 비었거나 CNI ADD/IPAM 오류가 직접 관측됨 |
| 적용 | Pod UID·node와 같은 UTC 창의 Event/plugin/kubelet 증거를 연결 가능 |
| 제외 | `hostNetwork=true`, scheduling 전 Pod, node 공통 실패는 [[02. 특정 노드에서 Pod 네트워크 생성이 반복 실패하는 경우]], pool 소진은 [[03. Pod IP가 중복되거나 IPAM 주소 범위가 소진된 경우]] |

검색 문자열: `FailedCreatePodSandBox`, `failed to setup network`, `CNI ADD`, `IPAM`, `failed to allocate`, `no available IP`, `network namespace`, `veth`, `route`.

## 최초 확인과 시간 gate

**current Pod·Event:** `refresh_pod || exit $?; [ "$HOST_NETWORK" = false ] || exit 1; jq -r '"UID/RV: \(.metadata.uid)/\(.metadata.resourceVersion)\nNode: \(.spec.nodeName)\nCreated: \(.metadata.creationTimestamp)\nPhase: \(.status.phase)\nPodIP: \(.status.podIP//"empty")\nScheduled: \([.status.conditions[]?|select(.type=="PodScheduled")][0].status//"missing")"' <<<"$POD_JSON"; E=$(kubectl get events -n "$NS" --field-selector "involvedObject.kind=Pod,involvedObject.name=$POD,involvedObject.uid=$POD_UID" --sort-by=.metadata.creationTimestamp -o json); RC=$?; if [ "$RC" -ne 0 ]; then printf 'Event 수집 실패 rc=%s\n' "$RC" >&2; elif [ "$(jq '.items|length' <<<"$E")" -eq 0 ]; then printf '%s\n' 'Event 0건: 성공 아님'; else jq -r '.items[]|[.metadata.creationTimestamp,.eventTime,.firstTimestamp,.lastTimestamp,.series.lastObservedTime,.reason,.message]|@tsv' <<<"$E"; fi; unset E RC`

**엄격한 UTC 창:** `unset WINDOW_READY START_EPOCH END_EPOCH EVIDENCE_EPOCH; START_RFC3339='PASTE_START_RFC3339_HERE'; END_RFC3339='PASTE_END_RFC3339_HERE'; EVIDENCE_RFC3339='PASTE_EVIDENCE_RFC3339_HERE'; RFC3339_RE='^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$'; WINDOW_READY=false; if [[ "$START_RFC3339" =~ $RFC3339_RE ]] && [[ "$END_RFC3339" =~ $RFC3339_RE ]] && [[ "$EVIDENCE_RFC3339" =~ $RFC3339_RE ]]; then START_EPOCH=$(date -u -d "$START_RFC3339" +%s 2>/dev/null) && END_EPOCH=$(date -u -d "$END_RFC3339" +%s 2>/dev/null) && EVIDENCE_EPOCH=$(date -u -d "$EVIDENCE_RFC3339" +%s 2>/dev/null) || true; fi; if [ -n "${START_EPOCH:-}" ] && [ "$(date -u -d "@$START_EPOCH" '+%Y-%m-%dT%H:%M:%SZ')" = "$START_RFC3339" ] && [ "$(date -u -d "@$END_EPOCH" '+%Y-%m-%dT%H:%M:%SZ')" = "$END_RFC3339" ] && [ "$(date -u -d "@$EVIDENCE_EPOCH" '+%Y-%m-%dT%H:%M:%SZ')" = "$EVIDENCE_RFC3339" ] && [ "$START_EPOCH" -lt "$END_EPOCH" ] && [ "$START_EPOCH" -le "$EVIDENCE_EPOCH" ] && [ "$EVIDENCE_EPOCH" -lt "$END_EPOCH" ]; then WINDOW_READY=true; else unset START_EPOCH END_EPOCH EVIDENCE_EPOCH; printf '%s\n' '중단: canonical UTC 또는 start<=evidence<end 위반' >&2; fi`

## 관측별 진단 경로

| 지금 보이는 것 | 먼저 볼 것 | 다음 계층 |
|---|---|---|
| plugin에 exact UID의 ADD 실패 | CNI current container·DS owner | plugin config/IPAM owner |
| plugin 요청 없음, sandbox 실패 | active kubelet endpoint | runtime/CRI |
| ADD 성공, API IP/interface 없음 | exact sandbox netns | Linux veth/route |
| allocation/pool 오류 | pool·node 범위 | [[03. Pod IP가 중복되거나 IPAM 주소 범위가 소진된 경우]] |

### 1. exact CNI Pod·container

**node Pod 후보:** `refresh_pod || exit $?; kubectl get pods -A --field-selector "spec.nodeName=$NODE" -o custom-columns='NS:.metadata.namespace,POD:.metadata.name,UID:.metadata.uid,OWNER-KIND:.metadata.ownerReferences[?(@.controller==true)].kind,OWNER-NAME:.metadata.ownerReferences[?(@.controller==true)].name,CONTAINERS:.spec.containers[*].name,IMAGES:.spec.containers[*].image,CONTAINER-IDS:.status.containerStatuses[*].containerID,PHASE:.status.phase'`

**CNI 선택·관계 등록:** `unset CNI_READY CNI_UID CNI_RV CNI_CONTAINER_ID CNI_DS_UID; CNI_NS='PASTE_CNI_NAMESPACE_HERE'; CNI_POD='PASTE_CNI_POD_HERE'; CNI_CONTAINER='PASTE_CNI_CONTAINER_HERE'; CNI_DS='PASTE_CNI_DAEMONSET_HERE'; refresh_pod || exit $?; CJ=$(kubectl get pod "$CNI_POD" -n "$CNI_NS" -o json) && CU=$(jq -er '.metadata.uid' <<<"$CJ") && CR=$(jq -er '.metadata.resourceVersion' <<<"$CJ") && CN=$(jq -er '.spec.nodeName' <<<"$CJ") && DU=$(jq -er --arg ds "$CNI_DS" '[.metadata.ownerReferences[]?|select(.controller==true and .kind=="DaemonSet" and .name==$ds)|.uid]|if length==1 then .[0] else error("owner") end' <<<"$CJ") && CID=$(jq -er --arg c "$CNI_CONTAINER" '[.status.containerStatuses[]?|select(.name==$c)|.containerID|select(length>0)]|if length==1 then .[0] else error("container") end' <<<"$CJ") && DSU=$(kubectl get ds "$CNI_DS" -n "$CNI_NS" -o jsonpath='{.metadata.uid}') || { printf '%s\n' '중단: CNI relation 수집 실패' >&2; unset CJ CU CR CN DU CID DSU; }; if [ -n "${DSU:-}" ] && [ "$CN" = "$NODE" ] && [ "$DU" = "$DSU" ]; then CNI_UID=$CU; CNI_RV=$CR; CNI_CONTAINER_ID=$CID; CNI_DS_UID=$DSU; CNI_READY=true; printf 'CNI_READY=true pod=%s/%s uid=%s rv=%s container=%s id=%s owner=DaemonSet/%s/%s/%s\n' "$CNI_NS" "$CNI_POD" "$CNI_UID" "$CNI_RV" "$CNI_CONTAINER" "$CNI_CONTAINER_ID" "$CNI_NS" "$CNI_DS" "$CNI_DS_UID"; else printf '%s\n' '중단: CNI node/owner mismatch' >&2; fi; unset CJ CU CR CN DU CID DSU`

**plugin log, direct current gate와 pipeline RC:** `refresh_pod || exit $?; [ "$CNI_READY" = true ] && [ "$WINDOW_READY" = true ] || exit 1; CJ=$(kubectl get pod "$CNI_POD" -n "$CNI_NS" -o json) && CU=$(jq -er '.metadata.uid' <<<"$CJ") && CN=$(jq -er '.spec.nodeName' <<<"$CJ") && CID=$(jq -er --arg c "$CNI_CONTAINER" '[.status.containerStatuses[]?|select(.name==$c)|.containerID][0]' <<<"$CJ") && DU=$(jq -er --arg ds "$CNI_DS" '[.metadata.ownerReferences[]?|select(.controller==true and .kind=="DaemonSet" and .name==$ds)|.uid]|if length==1 then .[0] else error("owner") end' <<<"$CJ") && DSU=$(kubectl get ds "$CNI_DS" -n "$CNI_NS" -o jsonpath='{.metadata.uid}') || { CNI_READY=false; exit 1; }; [ "$CU" = "$CNI_UID" ] && [ "$CN" = "$NODE" ] && [ "$CID" = "$CNI_CONTAINER_ID" ] && [ "$DU" = "$CNI_DS_UID" ] && [ "$DSU" = "$CNI_DS_UID" ] || { CNI_READY=false; printf '%s\n' '중단: CNI lifecycle/containerID/owner 변경' >&2; exit 1; }; RAW=$(kubectl logs "$CNI_POD" -n "$CNI_NS" -c "$CNI_CONTAINER" --timestamps --since-time="$START_RFC3339" --tail=300 2>&1); RC=$?; if [ "$RC" -ne 0 ]; then printf '로그 수집 실패 rc=%s\n' "$RC" >&2; elif MATCH=$(grep -E -i -C 3 "$POD_UID|CNI[ _-]*ADD|IPAM|allocate|pool|veth|interface|route|timeout|denied|error" <<<"$RAW"); then printf '%s\n' "$MATCH"; else printf '%s\n' '필터 불일치: 오류 없음이 아님'; tail -n 50 <<<"$RAW"; fi; unset CJ CU CN CID DU DSU RAW RC MATCH`

### 2. active CRI·exact sandbox·network namespace

**SSH·unit 선택과 CRI 함수 등록:** `unset REMOTE_READY SSH_TARGET EXPECTED_REMOTE_HOST KUBELET_UNIT RUNTIME_UNIT; SSH_TARGET='PASTE_SSH_TARGET_HERE'; EXPECTED_REMOTE_HOST='PASTE_EXPECTED_REMOTE_HOST_HERE'; KUBELET_UNIT='PASTE_KUBELET_UNIT_HERE'; RUNTIME_UNIT='PASTE_RUNTIME_UNIT_HERE'; run_remote(){ local script=$1 arg q cmd='bash -s --'; shift; for arg in "$@"; do printf -v q '%q' "$arg"; cmd+=" $q"; done; ssh -- "$SSH_TARGET" "$cmd" <<<"$script"; }; REMOTE_READY=false; [ -n "$SSH_TARGET" ] && [ -n "$EXPECTED_REMOTE_HOST" ] && [ -n "$KUBELET_UNIT" ] && [ -n "$RUNTIME_UNIT" ] && REMOTE_READY=true || printf '%s\n' '중단: SSH/unit 선택 불완전' >&2`

빈 CRI 결과는 정상이나 IP 미할당의 증거가 아니다. endpoint·socket provider·sandbox·PID start/netns identity 중 하나라도 달라지면 출력 전체를 버리고 처음부터 재등록한다.

## 변경·rollback·정상화

Pod 삭제, CNI DaemonSet·kubelet·runtime restart, lease/pool·route·netns 변경 명령은 제공하지 않는다. 변경 owner는 fresh Pod UID/RV·owner/node, CNI UID/RV/current containerID/DS UID, UTC 직접 증거, active endpoint·socket provider·sandbox/netns, 승인 ID·영향·idempotence를 실행 직전에 확인한다. 검증된 rollback 명령·원본·trigger가 없으면 중단한다.

replacement 뒤에는 이전 UID/RV/node/containerID/sandbox/PID/time/READY를 모두 폐기한다. 정상화는 `실제 서비스 요청 → Pod Ready 안정 → API Pod IP와 netns 주소·route 일치 → CNI ADD/IPAM 성공 → CRI sandbox → kubelet Event 중단` 역순이며, 다른 node로 이동한 Pod만 정상인 것으로 원래 node 원인 해소를 선언하지 않는다.

## 증거·미검증·상위 문서

보존: Pod UID/RV·owner·node UID/machineID, Event 시각/message, CNI UID/current containerID/DS UID, UTC 창, active endpoint와 provider PID, sandbox·PID start·netns·주소/route, 승인·rollback·기능 결과. CNI config 원문, credential, 전체 로그는 저장하지 않는다. CNI/IPAM 제품·unit·netns PID 제공 여부는 실제 환경 미검증이다.
