# Pod IP가 중복되거나 IPAM 주소 범위가 소진된 경우

## 입력값과 빠른 수집

진단 경로는 `current Pod UID·Pod IP → 같은 IP의 current API 객체 → 각 Node identity → CNI current container 로그 → 구현체가 확인된 IPAM pool/lease evidence → 각 sandbox network namespace`다. API에 같은 문자열이 남은 것과 실제 두 namespace가 동시에 같은 주소를 가진 것을 구분한다.

| 종류 | 변수 | 출처·gate |
|---|---|---|
| 최초 입력 | `NS`, `POD` | 대표 Pod exact name |
| API 자동값 | Pod UID/RV·owner·PodIPs, Node UID/machineID, 같은 IP 후보 | current API 성공 뒤 등록 |
| 운영자 선택 | 다른 Pod, CNI Pod/container/DS, UTC 창, pool ID | 후보·직접 오류·승인된 redacted IPAM evidence |
| 계산 후보 | `POD_IP_RE`, PromQL | 선택 IP를 escape해 생성, API/metric 대조 전 미확정 |

**도구·모든 downstream 초기화:** `unset INPUT_READY AUTO_READY DUP_READY CNI_READY WINDOW_READY POD_UID POD_RV OWNER_REL NODE NODE_UID NODE_MACHINE_ID POD_IP POD_IP_RE OTHER_NS OTHER_POD OTHER_UID OTHER_RV OTHER_OWNER OTHER_NODE OTHER_NODE_UID CNI_NS CNI_POD CNI_UID CNI_RV CNI_CONTAINER CNI_CONTAINER_ID CNI_DS CNI_DS_UID POOL_ID START_RFC3339 END_RFC3339 EVIDENCE_RFC3339 START_EPOCH END_EPOCH EVIDENCE_EPOCH POD_JSON NODE_JSON OTHER_JSON; TOOLS_READY=false; MISSING=''; for T in kubectl jq date grep sed; do command -v "$T" >/dev/null 2>&1 || MISSING="$MISSING $T"; done; if [ -z "${BASH_VERSION:-}" ] || [ -n "$MISSING" ]; then printf '중단: Bash/도구 gate 실패:%s\n' "$MISSING" >&2; else TOOLS_READY=true; fi`

**최초 입력:** `NS='PASTE_NAMESPACE_HERE'; POD='PASTE_POD_HERE'; INPUT_READY=false; [ "$TOOLS_READY" = true ] && [ -n "$NS" ] && [ -n "$POD" ] && INPUT_READY=true || printf '%s\n' '중단: 최초 입력 불완전' >&2`

**대표 Pod 자동 등록:** `AUTO_READY=false; unset POD_UID POD_RV OWNER_REL NODE NODE_UID NODE_MACHINE_ID POD_IP POD_IP_RE POD_JSON NODE_JSON; J=$(kubectl get pod "$POD" -n "$NS" -o json) && U=$(jq -er '.metadata.uid' <<<"$J") && R=$(jq -er '.metadata.resourceVersion' <<<"$J") && N=$(jq -er '.spec.nodeName|select(length>0)' <<<"$J") && O=$(jq -er '[.metadata.ownerReferences[]?|select(.controller==true)|[.kind,.name,.uid]|join("/")]|if length==0 then "none" elif length==1 then .[0] else error("owner") end' <<<"$J") && IP=$(jq -r '.status.podIPs[0].ip // .status.podIP // ""' <<<"$J") && NJ=$(kubectl get node "$N" -o json) && NU=$(jq -er '.metadata.uid' <<<"$NJ") && NM=$(jq -er '.status.nodeInfo.machineID|select(length>0)' <<<"$NJ") || { printf '%s\n' '중단: Pod/Node 수집 실패' >&2; unset J U R N O IP NJ NU NM; }; if [ -n "${NM:-}" ]; then POD_JSON=$J; POD_UID=$U; POD_RV=$R; OWNER_REL=$O; NODE=$N; NODE_JSON=$NJ; NODE_UID=$NU; NODE_MACHINE_ID=$NM; POD_IP=$IP; if [ -n "$POD_IP" ]; then POD_IP_RE="^$(printf '%s' "$POD_IP" | sed 's/[][\\.^$*+?(){}|]/\\&/g')$"; fi; AUTO_READY=true; printf 'AUTO_READY=true pod=%s/%s uid=%s rv=%s owner=%s node=%s uid=%s machineID=%s podIP=%s regex=%s\n' "$NS" "$POD" "$POD_UID" "$POD_RV" "$OWNER_REL" "$NODE" "$NODE_UID" "$NODE_MACHINE_ID" "${POD_IP:-empty}" "${POD_IP_RE:-none}"; fi; unset J U R N O IP NJ NU NM`

**direct current pair gate:** `refresh_pair(){ local j u r n o ip nj nu nm oj ou or on oo oip onj onu; AUTO_READY=false; DUP_READY=false; j=$(kubectl get pod "$POD" -n "$NS" -o json) || return 1; u=$(jq -er '.metadata.uid' <<<"$j") || return 1; r=$(jq -er '.metadata.resourceVersion' <<<"$j") || return 1; n=$(jq -er '.spec.nodeName' <<<"$j") || return 1; o=$(jq -er '[.metadata.ownerReferences[]?|select(.controller==true)|[.kind,.name,.uid]|join("/")]|if length==0 then "none" elif length==1 then .[0] else error("owner") end' <<<"$j") || return 1; ip=$(jq -r '.status.podIPs[0].ip // .status.podIP // ""' <<<"$j"); nj=$(kubectl get node "$n" -o json) || return 1; nu=$(jq -er '.metadata.uid' <<<"$nj") || return 1; nm=$(jq -er '.status.nodeInfo.machineID|select(length>0)' <<<"$nj") || return 1; [ "$u" = "$POD_UID" ] && [ "$n" = "$NODE" ] && [ "$o" = "$OWNER_REL" ] && [ "$ip" = "$POD_IP" ] && [ "$nu" = "$NODE_UID" ] && [ "$nm" = "$NODE_MACHINE_ID" ] || return 1; POD_JSON=$j; POD_RV=$r; NODE_JSON=$nj; AUTO_READY=true; if [ -n "${OTHER_UID:-}" ]; then oj=$(kubectl get pod "$OTHER_POD" -n "$OTHER_NS" -o json) || return 1; ou=$(jq -er '.metadata.uid' <<<"$oj") || return 1; or=$(jq -er '.metadata.resourceVersion' <<<"$oj") || return 1; on=$(jq -er '.spec.nodeName' <<<"$oj") || return 1; oo=$(jq -er '[.metadata.ownerReferences[]?|select(.controller==true)|[.kind,.name,.uid]|join("/")]|if length==0 then "none" elif length==1 then .[0] else error("owner") end' <<<"$oj") || return 1; oip=$(jq -r '.status.podIPs[0].ip // .status.podIP // ""' <<<"$oj"); onj=$(kubectl get node "$on" -o json) || return 1; onu=$(jq -er '.metadata.uid' <<<"$onj") || return 1; [ "$ou" = "$OTHER_UID" ] && [ "$oo" = "$OTHER_OWNER" ] && [ "$on" = "$OTHER_NODE" ] && [ "$onu" = "$OTHER_NODE_UID" ] && [ "$oip" = "$POD_IP" ] || return 1; OTHER_JSON=$oj; OTHER_RV=$or; DUP_READY=true; fi; }`

## 진입 판단

| 구분 | 판단 기준 |
|---|---|
| 사용 | 동일 Pod IP 후보가 둘 이상이거나 `failed to allocate`·`address pool exhausted`가 직접 관측됨 |
| 적용 | Pod UID·node·시각과 중복 대상 또는 pool/lease 오류를 대조 가능 |
| 제외 | 단순 미할당은 [[01. Pod IP가 할당되지 않거나 CNI 설정이 실패하는 경우]], 한 node 공통 실패는 [[02. 특정 노드에서 Pod 네트워크 생성이 반복 실패하는 경우]] |

검색 문자열: `duplicate IP`, `IPAM`, `failed to allocate`, `no available IP`, `address pool exhausted`, `lease`, `IP conflict`.

## 최초 확인·시간 gate

**current API 동일 IP 후보:** `refresh_pair || exit $?; if [ -z "$POD_IP" ]; then printf '%s\n' 'Pod IP가 비어 있음: allocation 오류 분기로 이동'; else ALL=$(kubectl get pods -A -o json); RC=$?; if [ "$RC" -ne 0 ]; then printf '전체 Pod API 수집 실패 rc=%s\n' "$RC" >&2; else jq -r --arg ip "$POD_IP" '.items[]|select(any(.status.podIPs[]?;.ip==$ip) or .status.podIP==$ip)|[.metadata.namespace,.metadata.name,.metadata.uid,.metadata.creationTimestamp,.spec.nodeName,.status.phase,([.metadata.ownerReferences[]?|select(.controller==true)|[.kind,.name,.uid]|join("/")]|join(","))]|@tsv' <<<"$ALL"; fi; unset ALL RC; fi`

**엄격한 UTC 창:** `unset WINDOW_READY START_EPOCH END_EPOCH EVIDENCE_EPOCH; START_RFC3339='PASTE_START_RFC3339_HERE'; END_RFC3339='PASTE_END_RFC3339_HERE'; EVIDENCE_RFC3339='PASTE_EVIDENCE_RFC3339_HERE'; RFC3339_RE='^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}Z$'; WINDOW_READY=false; if [[ "$START_RFC3339" =~ $RFC3339_RE ]] && [[ "$END_RFC3339" =~ $RFC3339_RE ]] && [[ "$EVIDENCE_RFC3339" =~ $RFC3339_RE ]]; then START_EPOCH=$(date -u -d "$START_RFC3339" +%s 2>/dev/null) && END_EPOCH=$(date -u -d "$END_RFC3339" +%s 2>/dev/null) && EVIDENCE_EPOCH=$(date -u -d "$EVIDENCE_RFC3339" +%s 2>/dev/null) || true; fi; if [ -n "${START_EPOCH:-}" ] && [ "$(date -u -d "@$START_EPOCH" '+%Y-%m-%dT%H:%M:%SZ')" = "$START_RFC3339" ] && [ "$(date -u -d "@$END_EPOCH" '+%Y-%m-%dT%H:%M:%SZ')" = "$END_RFC3339" ] && [ "$(date -u -d "@$EVIDENCE_EPOCH" '+%Y-%m-%dT%H:%M:%SZ')" = "$EVIDENCE_RFC3339" ] && [ "$START_EPOCH" -lt "$END_EPOCH" ] && [ "$START_EPOCH" -le "$EVIDENCE_EPOCH" ] && [ "$EVIDENCE_EPOCH" -lt "$END_EPOCH" ]; then WINDOW_READY=true; else unset START_EPOCH END_EPOCH EVIDENCE_EPOCH; printf '%s\n' '중단: canonical UTC/relation 위반' >&2; fi`

## 관측별 진단 경로

| 지금 보이는 것 | 직접 증거 | 다음 경로 |
|---|---|---|
| 같은 IP의 current Pod 둘 | 두 UID·owner·node·creation time | 각 sandbox/netns와 IPAM lease |
| 종료 Pod와 current Pod | 실제 namespace 존재 여부 | stale API/metric과 정상 재사용 분리 |
| allocation/pool 오류 | CNI current container와 pool ID | IPAM owner의 read-only capacity/lease evidence |
| 한 node에만 집중 | 같은 node의 두 Pod 오류 | [[02. 특정 노드에서 Pod 네트워크 생성이 반복 실패하는 경우]] |

### 1. 중복 current 객체 등록

**다른 Pod 선택·검증:** `unset DUP_READY OTHER_UID OTHER_RV OTHER_OWNER OTHER_NODE OTHER_NODE_UID OTHER_JSON; OTHER_NS='PASTE_OTHER_NAMESPACE_HERE'; OTHER_POD='PASTE_OTHER_POD_HERE'; refresh_pair || exit $?; OJ=$(kubectl get pod "$OTHER_POD" -n "$OTHER_NS" -o json) && OU=$(jq -er '.metadata.uid' <<<"$OJ") && ORV=$(jq -er '.metadata.resourceVersion' <<<"$OJ") && ON=$(jq -er '.spec.nodeName|select(length>0)' <<<"$OJ") && OO=$(jq -er '[.metadata.ownerReferences[]?|select(.controller==true)|[.kind,.name,.uid]|join("/")]|if length==0 then "none" elif length==1 then .[0] else error("owner") end' <<<"$OJ") && OIP=$(jq -r '.status.podIPs[0].ip // .status.podIP // ""' <<<"$OJ") && ONJ=$(kubectl get node "$ON" -o json) && ONU=$(jq -er '.metadata.uid' <<<"$ONJ") || { printf '%s\n' '중단: other relation 실패' >&2; unset OJ OU ORV ON OO OIP ONJ ONU; }; if [ -n "${ONU:-}" ] && [ "$OU" != "$POD_UID" ] && [ -n "$POD_IP" ] && [ "$OIP" = "$POD_IP" ]; then OTHER_JSON=$OJ; OTHER_UID=$OU; OTHER_RV=$ORV; OTHER_OWNER=$OO; OTHER_NODE=$ON; OTHER_NODE_UID=$ONU; DUP_READY=true; printf 'DUP_READY=true first=%s/%s/%s/%s second=%s/%s/%s/%s\n' "$NS" "$POD" "$POD_UID" "$NODE" "$OTHER_NS" "$OTHER_POD" "$OTHER_UID" "$OTHER_NODE"; else printf '%s\n' '중단: current duplicate relation 불충족' >&2; fi; unset OJ OU ORV ON OO OIP ONJ ONU`

각 Pod의 actual sandbox/netns 검증은 current UID별로 [[01. Pod IP가 할당되지 않거나 CNI 설정이 실패하는 경우#2. active CRI·exact sandbox·network namespace]]를 독립 실행한다. 한 namespace가 이미 사라졌으면 과거 PID를 재사용하지 않는다.

### 2. CNI/IPAM evidence

**CNI 선택·등록:** `unset CNI_READY CNI_UID CNI_RV CNI_CONTAINER_ID CNI_DS_UID; CNI_NS='PASTE_CNI_NAMESPACE_HERE'; CNI_POD='PASTE_CNI_POD_HERE'; CNI_CONTAINER='PASTE_CNI_CONTAINER_HERE'; CNI_DS='PASTE_CNI_DAEMONSET_HERE'; POOL_ID='PASTE_REDACTED_POOL_ID_HERE'; refresh_pair || exit $?; CJ=$(kubectl get pod "$CNI_POD" -n "$CNI_NS" -o json) && CU=$(jq -er '.metadata.uid' <<<"$CJ") && CR=$(jq -er '.metadata.resourceVersion' <<<"$CJ") && CN=$(jq -er '.spec.nodeName' <<<"$CJ") && DU=$(jq -er --arg ds "$CNI_DS" '[.metadata.ownerReferences[]?|select(.controller==true and .kind=="DaemonSet" and .name==$ds)|.uid]|if length==1 then .[0] else error("owner") end' <<<"$CJ") && CID=$(jq -er --arg c "$CNI_CONTAINER" '[.status.containerStatuses[]?|select(.name==$c)|.containerID|select(length>0)]|if length==1 then .[0] else error("container") end' <<<"$CJ") && DSU=$(kubectl get ds "$CNI_DS" -n "$CNI_NS" -o jsonpath='{.metadata.uid}') || { printf '%s\n' '중단: CNI relation 실패' >&2; unset CJ CU CR CN DU CID DSU; }; if [ -n "${DSU:-}" ] && [ "$CN" = "$NODE" ] && [ "$DU" = "$DSU" ] && [ -n "$POOL_ID" ]; then CNI_UID=$CU; CNI_RV=$CR; CNI_CONTAINER_ID=$CID; CNI_DS_UID=$DSU; CNI_READY=true; printf 'CNI_READY=true uid=%s rv=%s containerID=%s DS_UID=%s pool=%s\n' "$CNI_UID" "$CNI_RV" "$CNI_CONTAINER_ID" "$CNI_DS_UID" "$POOL_ID"; fi; unset CJ CU CR CN DU CID DSU`

**CNI log, producer/filter RC 분리:** `refresh_pair || exit $?; [ "$CNI_READY" = true ] && [ "$WINDOW_READY" = true ] || exit 1; CJ=$(kubectl get pod "$CNI_POD" -n "$CNI_NS" -o json) && CU=$(jq -er '.metadata.uid' <<<"$CJ") && CN=$(jq -er '.spec.nodeName' <<<"$CJ") && CID=$(jq -er --arg c "$CNI_CONTAINER" '[.status.containerStatuses[]?|select(.name==$c)|.containerID][0]' <<<"$CJ") && DU=$(jq -er --arg ds "$CNI_DS" '[.metadata.ownerReferences[]?|select(.controller==true and .kind=="DaemonSet" and .name==$ds)|.uid]|if length==1 then .[0] else error("owner") end' <<<"$CJ") && DSU=$(kubectl get ds "$CNI_DS" -n "$CNI_NS" -o jsonpath='{.metadata.uid}') || { CNI_READY=false; exit 1; }; [ "$CU" = "$CNI_UID" ] && [ "$CN" = "$NODE" ] && [ "$CID" = "$CNI_CONTAINER_ID" ] && [ "$DU" = "$CNI_DS_UID" ] && [ "$DSU" = "$CNI_DS_UID" ] || { CNI_READY=false; exit 1; }; RAW=$(kubectl logs "$CNI_POD" -n "$CNI_NS" -c "$CNI_CONTAINER" --timestamps --since-time="$START_RFC3339" --tail=400 2>&1); RC=$?; if [ "$RC" -ne 0 ]; then printf '로그 수집 실패 rc=%s\n' "$RC" >&2; elif MATCH=$(grep -E -i -C 3 "$POD_UID|${OTHER_UID:-no-other-uid}|$POOL_ID|ipam|allocate|pool|lease|exhaust|duplicate|conflict|timeout|error" <<<"$RAW"); then printf '%s\n' "$MATCH"; else printf '%s\n' '필터 불일치: 오류 없음이 아님'; tail -n 50 <<<"$RAW"; fi; unset CJ CU CN CID DU DSU RAW RC MATCH`

제품별 pool/lease API는 CNI image/version과 승인된 read-only 절차가 확인된 owner만 실행한다. 총 주소, 사용·예약·제외, 만료 lease, node별 allocation과 중복 lease를 요청하되 credential·lease payload 원문은 저장하지 않는다.

### 3. 완성 PromQL

Prometheus와 kube-state-metrics가 현재 cluster의 `kube_pod_info`와 `pod_ip` label을 실제 노출할 때만 쓴다.

**현재 IP exact query 생성:** `refresh_pair || exit $?; [ -n "$POD_IP" ] && printf 'count by (pod_ip) (kube_pod_info{pod_ip="%s"})\n' "$POD_IP" || printf '%s\n' '중단: Pod IP 없음' >&2`

결과 없음은 metric 미지원·scrape/RBAC·stale 범위 후보이며 중복 해소가 아니다. 값이 2 이상이어도 현재 API UID와 각 netns를 대조하기 전에는 실제 중복으로 확정하지 않는다.

## 변경·rollback·정상화

Pod 삭제, lease 강제 회수, pool 변경, IP 재사용, plugin restart 명령은 제공하지 않는다. 변경 직전 fresh 두 Pod UID/RV·owner/node, CNI UID/RV/current containerID/DS UID, UTC 오류, 실제 netns와 IPAM lease/pool, 영향·승인 ID·idempotence를 확인한다. 검증된 rollback 원본·명령·trigger가 없으면 중단한다.

replacement 뒤 모든 UID/IP/READY를 폐기하고 새 Pod UID·IP·owner·node를 재등록한다. 정상화는 `실제 서비스 요청 → Ready 안정 → API IP와 sandbox netns 일치·유일 → CNI allocation 성공 → pool 여유·lease 정합 → Event 재발 없음`이며 pool 여유나 `Running` 하나로 닫지 않는다.

## 증거·미검증·상위 문서

보존: Pod 쌍 UID/RV·owner·node·IP·creation time, UTC Event/plugin 오류, CNI current containerID/DS UID, 비민감 pool/lease 요약, netns identity, PromQL/API 대조, 승인·rollback·기능 결과. IPAM 제품·lease 정책·metric 지원은 실제 환경 미검증이다.
