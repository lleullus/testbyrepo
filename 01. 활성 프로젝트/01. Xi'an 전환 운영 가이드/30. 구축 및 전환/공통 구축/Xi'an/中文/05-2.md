填入从 Secret 确认的实际 key 名称，完成 `3.servicemonitor-metrics.yaml`。

```yaml
apiVersion: monitoring.coreos.com/v1
kind: ServiceMonitor
metadata:
  name: etcd-k8s
  namespace: monitoring
  labels:
    k8s-app: etcd-k8s
spec:
  jobLabel: k8s-app
  endpoints:
    - port: metrics
      interval: 30s
      scheme: https
      tlsConfig:
        caFile: /etc/prometheus/secrets/etcd-certs/ca.pem
        certFile: /etc/prometheus/secrets/etcd-certs/<已确认的-dev-node-cert>.pem
        keyFile: /etc/prometheus/secrets/etcd-certs/<已确认的-dev-node-cert>-key.pem

    matchLabels:
      k8s-app: etcd-k8s
```
